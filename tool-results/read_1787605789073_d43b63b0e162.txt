    75→function parseBR(s: string): number {
    76→  const neg = s.startsWith('-');
    77→  const clean = neg ? s.slice(1) : s;
    78→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    79→}
    80→
    81→function findLine(lines: string[], pattern: RegExp): number {
    82→  return lines.findIndex(l => pattern.test(l));
    83→}
    84→
    85→function findAllMoney(text: string): string[] {
    86→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
    87→  if (!matches) return [];
    88→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
    89→}

    91→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    92→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    93→    if (Math.abs(i - startIdx) > maxDist) break;
    94→    const line = lines[i];
    95→    if (mustBeOnlyNum) {
    96→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    97→      if (m) return m[1];
    98→    } else {
    99→      const m = line.match(/([\d.]+,\d{2})/);
   100→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   101→    }
   102→  }
   103→  return EMPTY;
   104→}

   106→// ===================== MAIN EXTRACTOR =====================
   107→
   108→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   109→  const tableData = tableExtraction?.items;
   110→  const tributosFromTable = tableExtraction
   111→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   112→    : null;
   113→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   114→  const flat = lines.join(' ');
   115→
   116→  // ===== CONCESSIONÁRIA =====
   117→  const Concessionaria = (() => {
   118→    for (const line of lines.slice(0, 20)) {
   119→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
   120→        return line.replace(/\s+/g, ' ');
   121→      }
   122→    }
   123→    return EMPTY;
   124→  })();
   125→
   126→  // ===== TIPO =====
   127→  const Tipo = (() => {
   128→    const u = flat.toUpperCase();
   129→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   130→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   131→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   132→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   133→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   134→    if (/RURAL/.test(u)) return 'RURAL';
   135→    return EMPTY;
   136→  })();
   137→
   138→  // ===== MERCADO =====
   139→  const Mercado = (() => {
   140→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   141→    if (m) {
   142→      const val = m[1].trim().replace(/\s+/g, ' ');
   143→      if (val.length < 50) return val.toUpperCase();
   144→    }
   145→    return EMPTY;
   146→  })();
   147→
   148→  // ===== CIDADE =====
   149→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   150→  const Cidade = (() => {
   151→    for (const line of lines.slice(0, 60)) {
   152→      // Match: CityName [StateAbbr] (AG: XX)
   153→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   154→      if (m) {
   155→        const name = m[1].trim().toUpperCase();
   156→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   157→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   158→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   159→      }
   160→    }
   161→    return EMPTY;
   162→  })();
   163→
   164→  // ===== UC =====
   165→  const UC = (() => {
   166→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   167→    for (const line of lines) {
   168→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   169→      if (m) {
   170→        const num = m[1].replace(/^0+/, '') || '0';
   171→        if (num.length >= 3) return num;
   172→      }
   173→    }
   174→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   175→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   176→    if (slashCtx) {
   177→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   178→      if (num.length >= 3) return num;
   179→    }
   180→    // Strategy 3: Standard dotted format X.XXX.XXX-XX on its own line
   181→    for (const line of lines) {
   182→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   183→      if (m) return m[1];
   184→    }
   185→    // Strategy 4: Dotted format on a short line (not CNPJ/CPF/CEP)
   186→    for (const line of lines) {
   187→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   188→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   189→    }
   190→    // Strategy 5: MATRÍCULA field (e.g., "MATRÍCULA:0000033836-2025-08-6")
   191→    // Extracts the UC number from the matrícula, stripping leading zeros
   192→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,10})(?:-\d{4}-\d{2}-\d)?/i);
   193→    if (ucMat) return ucMat[1];
   194→    // Strategy 6: 13-digit UC (Energisa MS format) - look in address lines
   195→    const uc13 = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{13})\s/);
   196→    if (uc13) return uc13[1];
   197→    // Strategy 7: 13-digit standalone or near "Instalação:"
   198→    const uc13b = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   199→    if (uc13b) return uc13b[1];
   200→    // Strategy 8: UC label with number
   201→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   202→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   203→    // Strategy 9: broad search for dotted format
   204→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   205→    return broad ? broad[1] : EMPTY;
   206→  })();
   207→
   208→  // ===== MÊS REFERÊNCIA =====
   209→  const MesReferencia = (() => {
   210→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   211→    const numToMonth: Record<string, string> = {
   212→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   213→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   214→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   215→    };
   216→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   217→    if (nameMatch) {
   218→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   219→      return `${month} / ${nameMatch[2]}`;
   220→    }
   221→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   222→    if (numMatch && numToMonth[numMatch[1]]) {
   223→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   224→    }
   225→    return EMPTY;
   226→  })();
   227→
   228→  // ===== ANO LEITURA =====
   229→  const AnoLeitura = (() => {
   230→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   231→    if (m) return m[1];
   232→    if (MesReferencia) {
   233→      const ym = MesReferencia.match(/(\d{4})/);
   234→      if (ym) return ym[1];
   235→    }
   236→    return EMPTY;
   237→  })();
   238→
   239→  // ===== VENCIMENTO =====
   240→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   241→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   242→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   243→  const Vencimento = (() => {
   244→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   245→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   246→    if (venMoney) return venMoney[1];
   247→
   248→    // Strategy 2: R$ + money + date on same line (reversed order)
   249→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   250→    if (moneyVen) return moneyVen[1];
   251→
   252→    // Strategy 3: VENCIMENTO label nearby (original approach)
   253→    const venIdx = findLine(lines, /VENCIMENTO/i);
   254→    if (venIdx >= 0) {
   255→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   256→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   257→        if (m) return m[1];
   258→      }
   259→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   260→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   261→        if (m) return m[1];
   262→      }
   263→    }
   264→
   265→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   266→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   267→    if (apresMatch) return apresMatch[1];
   268→
   269→    return EMPTY;
   270→  })();
   271→
   272→  // ===== NOTA FISCAL =====
   273→  const NotaFiscal = (() => {
   274→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   275→    return m ? m[1] : EMPTY;
   276→  })();
   277→
   278→  // ===== MODALIDADE TARIFÁRIA =====
   279→  const ModalidadeTarifaria = (() => {
   280→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   281→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   282→    if (/AZUL/i.test(flat)) return 'AZUL';
   283→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   284→    if (/VERDE/i.test(flat)) return 'VERDE';
   285→    return EMPTY;
   286→  })();
   287→
   288→  // ===== SUBGRUPO =====
   289→  const Subgrupo = (() => {
   290→    // Look for "/A4" or "/B3" etc. in the classification line
   291→    // Must be a letter + single digit, right after "/"
   292→    const m = flat.match(/\/\s*([AB]\d)\s/);
   293→    if (m) return m[1].toUpperCase();
   294→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   295→    if (classIdx >= 0) {
   296→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   297→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   298→      if (m2) return m2[1].toUpperCase();
   299→    }
   300→    return EMPTY;
   301→  })();
   302→
   303→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   304→
   305→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   306→  let ConsumoPonta = EMPTY;
   307→  let ConsumoForaPonta = EMPTY;
   308→  let ConsumoTotal = EMPTY;
   309→  let ValorConsumoPonta = EMPTY;
   310→  let ValorConsumoForaPonta = EMPTY;
   311→
   312→  if (isSplitTariff) {
   313→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   314→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   315→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   316→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   317→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   318→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   319→  } else {
   320→    for (const line of lines) {
   321→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   322→      if (km) { ConsumoTotal = km[1]; break; }
   323→    }
   324→    if (!ConsumoTotal) {
   325→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   326→      if (consIdx >= 0) {
   327→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   328→          const line = lines[i];
   329→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   330→          if (/[A-Za-z]{4,}/.test(line)) continue;
   331→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   332→          if (m) { ConsumoTotal = m[1]; break; }
   333→        }
   334→      }
   335→    }
   336→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   337→  }
   338→
   339→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   340→  let DemandaContratadaPonta = EMPTY;
   341→  let DemandaContratadaForaPonta = EMPTY;
   342→  let DemandaMedidaPonta = EMPTY;
   343→  let DemandaMedidaForaPonta = EMPTY;
   344→  let DemandaFaturadaPonta = EMPTY;
   345→  let DemandaFaturadaForaPonta = EMPTY;
   346→  let ValorDemandaPonta = EMPTY;
   347→  let ValorDemandaForaPonta = EMPTY;
   348→
   349→  if (isSplitTariff) {
   350→    // Try to extract demanda contratada from flat text: "KW Ponta:33" or "KW Ponta: 33"
   351→    const dcMatch = flat.match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   352→    if (dcMatch) DemandaContratadaPonta = dcMatch[1].replace('.', ',');
   353→    const dcFpMatch = flat.match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   354→    if (dcFpMatch) DemandaContratadaForaPonta = dcFpMatch[1].replace('.', ',');
   355→
   356→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   357→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   358→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   359→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   360→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   361→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   362→  }
   363→
   364→  // ===== ULTRAPASSAGEM =====
   365→  let UltrapassagemPonta = EMPTY;
   366→  let UltrapassagemForaPonta = EMPTY;
   367→  let ValorUltrapassagemPonta = EMPTY;
   368→  let ValorUltrapassagemForaPonta = EMPTY;
   369→  // ===== DEMANDA NÃO CONSUMIDA =====
   370→  let DemandaNaoConsumidaPonta = EMPTY;
   371→  let DemandaNaoConsumidaForaPonta = EMPTY;
   372→  let ValorNaoConsumidaPonta = EMPTY;
   373→  let ValorNaoConsumidaForaPonta = EMPTY;
   374→  if (isSplitTariff) {
   375→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   376→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   377→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   378→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   379→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   380→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   381→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   382→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   383→  }
   384→
   385→  // ===== REATIVA EXCEDENTE =====
   386→  let ReativaExcedentePonta = EMPTY;
   387→  let ReativaExcedenteForaPonta = EMPTY;
   388→  let EnergiaReativaTotal = EMPTY;
   389→  let ValorReativaExcedentePonta = EMPTY;
   390→  let ValorReativaExcedenteForaPonta = EMPTY;
   391→  if (isSplitTariff) {
   392→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   393→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   394→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   395→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   396→  } else {
   397→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   398→  }
   399→
   400→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   401→  // Priority: table extraction > text-based extraction
   402→  let PIS = EMPTY;
   403→  let COFINS = EMPTY;
   404→  let ICMS = EMPTY;
   405→
   406→  // Use tributos from table extraction if available (most reliable)
   407→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   408→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   409→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   410→
   411→  // Fallback: text-based extraction for any missing tributo
   412→  if (!PIS || !COFINS || !ICMS) {
   413→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   414→    if (pisIdx >= 0) {
   415→      const nums: string[] = [];
   416→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   417→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   418→        if (lineNums) {
   419→          for (const n of lineNums) {
   420→            nums.push(n);
   421→            if (nums.length >= 3) break;
   422→          }
   423→        }
   424→      }
   425→      if (nums.length >= 3) {
   426→        if (!PIS) PIS = nums[0];
   427→        if (!COFINS) COFINS = nums[1];
   428→        if (!ICMS) ICMS = nums[2];
   429→      }
   430→    }
   431→    if (!PIS && !COFINS) {
   432→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   433→      if (combinedIdx >= 0) {
   434→        const nums: string[] = [];
   435→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   436→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   437→          if (m) nums.push(m[1]);
   438→        }
   439→        if (nums.length >= 3) {
   440→          if (!PIS) PIS = nums[0];
   441→          if (!COFINS) COFINS = nums[1];
   442→          if (!ICMS) ICMS = nums[2];
   443→        }
   444→      }
   445→    }
   446→    if (!PIS || !COFINS) {
   447→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   448→      if (flatMatch) {
   449→        if (!PIS) PIS = flatMatch[1];
   450→        if (!COFINS) COFINS = flatMatch[2];
   451→      }
   452→    }
   453→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   454→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   455→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
   456→  }
   457→
   458→  // ===== GDI =====
   459→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   460→
   461→  // ===== CIP (text-based) =====
   462→  const CIP_text = (() => {
   463→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   464→    if (cipIdx >= 0) {
   465→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   466→      const money = findAllMoney(region);
   467→      if (money.length > 0) return money[money.length - 1];
   468→    }
   469→    return findTributoValue(lines, flat, /\bCIP\b/i);
   470→  })();
   471→
   472→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   473→  let {
   474→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   475→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   476→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   477→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   478→  } = extractGDI(lines, flat);
   479→
   480→  // ===== VALOR TOTAL =====
   481→  // Best pattern: R$ value on same line as vencimento date + mes referência
   482→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   483→  const ValorTotal = (() => {
   484→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
   485→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
   486→    if (venTotal) return venTotal[1];
   487→
   488→    // Strategy 2: R$ + value + mes referência on same line
   489→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
   490→    if (totalMes) return totalMes[1];
   491→
   492→    // Strategy 3: "TOTAL A PAGAR" with R$
   493→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   494→    if (totalIdx >= 0) {
   495→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   496→      if (m1) return m1[1];
   497→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   498→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   499→        if (m2) return m2[1];
   500→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   501→        if (m3) return m3[1];
   502→      }
   503→    }
   504→
   505→    // Strategy 4: Last resort - largest R$ value
   506→    const lastMoney = findAllMoney(flat);
   507→    if (lastMoney.length > 0) {
   508→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   509→      return lastMoney[0];
   510→    }
   511→    return EMPTY;
   512→  })();
   513→
   514→  // ===== TABLE DATA OVERRIDES =====
   515→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
   516→  let CIP = CIP_text;
   517→
   518→  if (tableData && tableData.length > 0) {
   519→    let cipFromTable = EMPTY;
   520→    // Reset GDI accumulators for clean table-based extraction
   521→    let tbl_oUC_Qtd = '' as string;
   522→    let tbl_oUC_Valor = '' as string;
   523→    let tbl_mUC_Qtd = '' as string;
   524→    let tbl_mUC_Valor = '' as string;
   525→    // Track GDI section in table for rows with truncated/missing labels
   526→    let inGDISection = false;
   527→    let gdiSectionType = '' as string; // 'oUC' or 'mUC'
   528→    let lastGDILabel = '' as string;
   529→
   530→    for (const item of tableData) {
   531→      const lbl = item.label;
   532→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
   533→      const isPonta = /Ponta/i.test(lbl) && !isFora;
   534→
   535→      // Consumo - check Convencional (no Ponta/Fora) first, then split
   536→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
   537→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
   538→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
   539→          if (item.quantidade) ConsumoTotal = item.quantidade;
   540→          if (item.valor) ValorConsumoForaPonta = item.valor;
   541→        }
   542→        // VERDE Intermediária
   543→        else if (/Intermedi[áa]ria/i.test(lbl)) {
   544→          if (item.quantidade && !ConsumoForaPonta) ConsumoForaPonta = item.quantidade;
   545→        }
   546→        // Split tariff (AZUL/VERDE)
   547→        else if (isPonta) {
   548→          if (item.quantidade) ConsumoPonta = item.quantidade;
   549→          if (item.valor) ValorConsumoPonta = item.valor;
   550→        } else if (isFora) {
   551→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
   552→          if (item.valor) ValorConsumoForaPonta = item.valor;
   553→        }
   554→      }
   555→      // Demanda Medida
   556→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
   557→        if (!isFora) {
   558→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
   559→          if (item.valor) ValorDemandaPonta = item.valor;
   560→        } else {
   561→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
   562→          if (item.valor) ValorDemandaForaPonta = item.valor;
   563→        }
   564→      }
   565→      // Ultrapassagem
   566→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
   567→        if (!isFora) {
   568→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
   569→          if (item.valor) ValorUltrapassagemPonta = item.valor;
   570→        } else {
   571→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
   572→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
   573→        }
   574→      }
   575→      // Demanda Potência Não Consumida
   576→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
   577→        if (!isFora) {
   578→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
   579→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
   580→        } else {
   581→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
   582→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
   583→        }
   584→      }
   585→      // Reativa Excedente
   586→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
   587→        if (!isFora && /Ponta/i.test(lbl)) {
   588→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
   589→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   590→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
   591→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
   592→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
   593→        } else if (!isFora && !/Ponta/i.test(lbl)) {
   594→          // Generic reativa (Convencional)
   595→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
   596→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   597→        }
   598→      }
   599→      // CIP Municipal
   600→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
   601→        if (item.valor) cipFromTable = item.valor;
   602→      }
   603→      // GDI items
   604→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
   605→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
   606→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
   607→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
   608→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
   609→        // Multiple mUC entries: accumulate
   610→        if (item.quantidade) {
   611→          tbl_mUC_Qtd = tbl_mUC_Qtd
   612→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   613→            : item.quantidade;
   614→        }
   615→        if (item.valor) {
   616→          tbl_mUC_Valor = tbl_mUC_Valor
   617→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
   618→            : item.valor;
   619→        }
   620→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
   621→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
   622→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
   623→        // Accumulate as the same GDI type as the last full-label row
   624→        if (gdiSectionType === 'mUC') {
   625→          tbl_mUC_Qtd = tbl_mUC_Qtd
   626→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   627→            : item.quantidade;
   628→          if (item.valor) {
   629→            tbl_mUC_Valor = tbl_mUC_Valor
   630→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
   631→              : item.valor;
   632→          }
   633→        } else if (gdiSectionType === 'oUC') {
   634→          tbl_oUC_Qtd = tbl_oUC_Qtd
   635→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
   636→            : item.quantidade;
   637→          if (item.valor) {
   638→            tbl_oUC_Valor = tbl_oUC_Valor
   639→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
   640→              : item.valor;
   641→          }
   642→        }
   643→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
   644→        // Non-GDI, non-truncated row ends the GDI section
   645→        inGDISection = false;
   646→      }
   647→    }
   648→
   649→    // Override CIP from table
   650→    if (cipFromTable) CIP = cipFromTable;
   651→
   652→    // Use table GDI values if found, otherwise keep text-based fallback
   653→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
   654→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
   655→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
   656→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
   657→  }
   658→
   659→  return {
   660→    UC, Cidade, Tipo, Mercado, Concessionaria,
   661→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   662→    ModalidadeTarifaria, Subgrupo,
   663→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   664→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   665→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   666→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   667→    UltrapassagemPonta, UltrapassagemForaPonta,
   668→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
   669→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
   670→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   671→    ValorConsumoPonta, ValorConsumoForaPonta,
   672→    ValorDemandaPonta, ValorDemandaForaPonta,
   673→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   674→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   675→    PIS, COFINS, ICMS, GDI, CIP,
   676→    EnergiaReativaTotal,
   677→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   678→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   679→    ValorTotal,
   680→  };
   681→}

   683→// ===================== GDI EXTRACTION =====================
   684→
   685→function extractGDI(lines: string[], flat: string) {
   686→  const EMPTY = '';
   687→  let nOUC = 0;
   688→  let nMUC = 0;
   689→  let firstGDI = -1;
   690→
   691→  for (let i = 0; i < lines.length; i++) {
   692→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   693→      nOUC++;
   694→      if (firstGDI < 0) firstGDI = i;
   695→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   696→      nMUC++;
   697→      if (firstGDI < 0) firstGDI = i;
   698→    }
   699→  }
   700→
   701→  if (nOUC === 0 && nMUC === 0) {
   702→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   703→  }
   704→
   705→  const chaveIdx = (() => {
   706→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   707→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   708→    }
   709→    return -1;
   710→  })();
   711→
   712→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
   713→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   714→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
   715→      sectionEnd = i; break;
   716→    }
   717→  }
   718→  if (chaveIdx >= 0) {
   719→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   720→  }
   721→
   722→  const allQtds: number[] = [];
   723→  if (chaveIdx >= 0) {
   724→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   725→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   726→      if (m) allQtds.push(parseBR(m[1]));
   727→    }
   728→  }
   729→  if (allQtds.length === 0) {
   730→    for (let i = firstGDI; i < sectionEnd; i++) {
   731→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   732→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   733→      if (allQtds.length >= nOUC + nMUC) break;
   734→    }
   735→  }
   736→
   737→  const allRVals: number[] = [];
   738→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   739→  const totalNeeded = nOUC + nMUC;
   740→
   741→  for (let i = firstGDI; i < rSearchEnd; i++) {
   742→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   743→    if (m && allRVals.length < totalNeeded) {
   744→      allRVals.push(parseBR('-' + m[2]));
   745→    }
   746→  }
   747→  if (allRVals.length < totalNeeded) {
   748→    for (let i = firstGDI; i < rSearchEnd; i++) {
   749→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   750→      if (m && allRVals.length < totalNeeded) {
   751→        allRVals.push(parseBR(m[1]));
   752→      }
   753→    }
   754→  }
   755→
   756→  const oUCQtds = allQtds.splice(0, nOUC);
   757→  const oUCRVals = allRVals.splice(0, nOUC);
   758→
   759→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   760→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   761→
   762→  return {
   763→    oUC_Qtd: sumQtd(oUCQtds),
   764→    oUC_Valor: sumR(oUCRVals),
   765→    mUC_Qtd: sumQtd(allQtds),
   766→    mUC_Valor: sumR(allRVals),
   767→  };
   768→}
   769→
   770→// ===================== NUMBER FORMATTING =====================
   771→
   772→function fmtBRNumber(n: number): string {
   773→  const abs = Math.abs(n);
   774→  const intPart = Math.floor(abs).toString();
   775→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   776→  let formatted = '';
   777→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   778→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   779→    formatted = intPart[i] + formatted;
   780→  }
   781→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   782→}
   783→
   784→function fmtSumBR(a: string, b: string): string {
   785→  return fmtBRNumber(parseBR(a) + parseBR(b));
   786→}

   788→// ===================== ITEM HELPERS =====================
   789→
   790→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   791→  const idx = findLine(lines, labelPattern);
   792→  if (idx < 0) return EMPTY;
   793→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   794→    const line = lines[i];
   795→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   796→    const m = line.match(/([\d.]+,\d{2})/);
   797→    if (m) return m[1];
   798→  }
   799→  return EMPTY;
   800→}
   801→
   802→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   803→  const idx = findLine(lines, labelPattern);
   804→  if (idx < 0) return EMPTY;
   805→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   806→  const moneyVals = findAllMoney(region);
   807→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   808→  return EMPTY;
   809→}
   810→
   811→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   812→  const idx = findLine(lines, labelPattern);
   813→  if (idx >= 0) {
   814→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   815→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   816→      if (m) return m[1];
   817→    }
   818→  }
   819→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   820→  return flatMatch ? flatMatch[1] : EMPTY;
   821→}
