    76→function parseBR(s: string): number {
    77→  const neg = s.startsWith('-');
    78→  const clean = neg ? s.slice(1) : s;
    79→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    80→}
    81→
    82→function findLine(lines: string[], pattern: RegExp): number {
    83→  return lines.findIndex(l => pattern.test(l));
    84→}
    85→
    86→function findAllMoney(text: string): string[] {
    87→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
    88→  if (!matches) return [];
    89→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
    90→}

    92→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    93→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    94→    if (Math.abs(i - startIdx) > maxDist) break;
    95→    const line = lines[i];
    96→    if (mustBeOnlyNum) {
    97→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    98→      if (m) return m[1];
    99→    } else {
   100→      const m = line.match(/([\d.]+,\d{2})/);
   101→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   102→    }
   103→  }
   104→  return EMPTY;
   105→}

   107→// ===================== MAIN EXTRACTOR =====================
   108→
   109→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   110→  const tableData = tableExtraction?.items;
   111→  const tributosFromTable = tableExtraction
   112→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   113→    : null;
   114→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   115→  const flat = lines.join(' ');
   116→
   117→  // ===== CONCESSIONÁRIA =====
   118→  const Concessionaria = (() => {
   119→    // Strategy 1: line starting with ENERGISA (most common)
   120→    for (const line of lines.slice(0, 25)) {
   121→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
   122→        return line.replace(/\s+/g, ' ');
   123→      }
   124→    }
   125→    // Strategy 2: ENERGISA anywhere in first 25 lines
   126→    for (const line of lines.slice(0, 25)) {
   127→      if (/ENERGISA/i.test(line) && line.length > 20 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
   128→        const m = line.match(/ENERGISA[\s\S]{10,}/i);
   129→        if (m) return m[0].replace(/\s+/g, ' ').trim();
   130→      }
   131→    }
   132→    // Strategy 3: find in flat text
   133→    const flatM = flat.match(/(ENERGISA\s+[\w\s.\-]+S\.?A\.?)/i);
   134→    if (flatM) return flatM[1].replace(/\s+/g, ' ').trim();
   135→    return EMPTY;
   136→  })();
   137→
   138→  // ===== TIPO =====
   139→  const Tipo = (() => {
   140→    const u = flat.toUpperCase();
   141→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   142→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   143→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   144→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   145→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   146→    if (/RURAL/.test(u)) return 'RURAL';
   147→    return EMPTY;
   148→  })();
   149→
   150→  // ===== MERCADO =====
   151→  const Mercado = (() => {
   152→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   153→    if (m) {
   154→      const val = m[1].trim().replace(/\s+/g, ' ');
   155→      if (val.length < 50) return val.toUpperCase();
   156→    }
   157→    return EMPTY;
   158→  })();
   159→
   160→  // ===== CIDADE =====
   161→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   162→  const Cidade = (() => {
   163→    for (const line of lines.slice(0, 60)) {
   164→      // Match: CityName [StateAbbr] (AG: XX)
   165→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   166→      if (m) {
   167→        const name = m[1].trim().toUpperCase();
   168→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   169→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   170→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   171→      }
   172→    }
   173→    return EMPTY;
   174→  })();
   175→
   176→  // ===== UC =====
   177→  const UC = (() => {
   178→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   179→    for (const line of lines) {
   180→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   181→      if (m) {
   182→        const num = m[1].replace(/^0+/, '') || '0';
   183→        if (num.length >= 3) return num;
   184→      }
   185→    }
   186→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   187→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   188→    if (slashCtx) {
   189→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   190→      if (num.length >= 3) return num;
   191→    }
   192→    // Strategy 3: Standard dotted format X.XXX.XXX-XX on its own line
   193→    for (const line of lines) {
   194→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   195→      if (m) return m[1];
   196→    }
   197→    // Strategy 4: Dotted format on a short line (not CNPJ/CPF/CEP)
   198→    for (const line of lines) {
   199→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   200→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   201→    }
   202→    // Strategy 5: MATRÍCULA field (e.g., "MATRÍCULA:0000033836-2025-08-6")
   203→    // Extracts the UC number from the matrícula, stripping leading zeros
   204→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,10})(?:-\d{4}-\d{2}-\d)?/i);
   205→    if (ucMat) return ucMat[1];
   206→    // Strategy 6: 13-digit UC (Energisa MS format) - look in address lines
   207→    const uc13 = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{13})\s/);
   208→    if (uc13) return uc13[1];
   209→    // Strategy 7: 13-digit standalone or near "Instalação:"
   210→    const uc13b = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   211→    if (uc13b) return uc13b[1];
   212→    // Strategy 8: UC label with number
   213→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   214→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   215→    // Strategy 9: broad search for dotted format
   216→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   217→    return broad ? broad[1] : EMPTY;
   218→  })();
   219→
   220→  // Clean UC: remove dots and dashes (e.g., "43.210.051-68" → "4321005168")
   221→  const cleanUC = UC.replace(/[.\-]/g, '');
   222→
   223→  // ===== MÊS REFERÊNCIA =====
   224→  const MesReferencia = (() => {
   225→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   226→    const numToMonth: Record<string, string> = {
   227→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   228→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   229→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   230→    };
   231→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   232→    if (nameMatch) {
   233→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   234→      return `${month} / ${nameMatch[2]}`;
   235→    }
   236→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   237→    if (numMatch && numToMonth[numMatch[1]]) {
   238→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   239→    }
   240→    return EMPTY;
   241→  })();
   242→
   243→  // ===== ANO LEITURA =====
   244→  const AnoLeitura = (() => {
   245→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   246→    if (m) return m[1];
   247→    if (MesReferencia) {
   248→      const ym = MesReferencia.match(/(\d{4})/);
   249→      if (ym) return ym[1];
   250→    }
   251→    return EMPTY;
   252→  })();
   253→
   254→  // ===== VENCIMENTO =====
   255→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   256→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   257→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   258→  const Vencimento = (() => {
   259→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   260→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   261→    if (venMoney) return venMoney[1];
   262→
   263→    // Strategy 2: R$ + money + date on same line (reversed order)
   264→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   265→    if (moneyVen) return moneyVen[1];
   266→
   267→    // Strategy 3: VENCIMENTO label nearby (original approach)
   268→    const venIdx = findLine(lines, /VENCIMENTO/i);
   269→    if (venIdx >= 0) {
   270→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   271→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   272→        if (m) return m[1];
   273→      }
   274→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   275→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   276→        if (m) return m[1];
   277→      }
   278→    }
   279→
   280→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   281→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   282→    if (apresMatch) return apresMatch[1];
   283→
   284→    return EMPTY;
   285→  })();
   286→
   287→  // ===== NOTA FISCAL =====
   288→  const NotaFiscal = (() => {
   289→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   290→    return m ? m[1] : EMPTY;
   291→  })();
   292→
   293→  // ===== MODALIDADE TARIFÁRIA =====
   294→  const ModalidadeTarifaria = (() => {
   295→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   296→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   297→    if (/AZUL/i.test(flat)) return 'AZUL';
   298→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   299→    if (/VERDE/i.test(flat)) return 'VERDE';
   300→    return EMPTY;
   301→  })();
   302→
   303→  // ===== SUBGRUPO =====
   304→  const Subgrupo = (() => {
   305→    // Look for "/A4" or "/B3" etc. in the classification line
   306→    // Must be a letter + single digit, right after "/"
   307→    const m = flat.match(/\/\s*([AB]\d)\s/);
   308→    if (m) return m[1].toUpperCase();
   309→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   310→    if (classIdx >= 0) {
   311→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   312→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   313→      if (m2) return m2[1].toUpperCase();
   314→    }
   315→    return EMPTY;
   316→  })();
   317→
   318→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   319→
   320→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   321→  let ConsumoPonta = EMPTY;
   322→  let ConsumoForaPonta = EMPTY;
   323→  let ConsumoTotal = EMPTY;
   324→  let ValorConsumoPonta = EMPTY;
   325→  let ValorConsumoForaPonta = EMPTY;
   326→
   327→  if (isSplitTariff) {
   328→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   329→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   330→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   331→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   332→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   333→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   334→  } else {
   335→    for (const line of lines) {
   336→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   337→      if (km) { ConsumoTotal = km[1]; break; }
   338→    }
   339→    if (!ConsumoTotal) {
   340→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   341→      if (consIdx >= 0) {
   342→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   343→          const line = lines[i];
   344→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   345→          if (/[A-Za-z]{4,}/.test(line)) continue;
   346→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   347→          if (m) { ConsumoTotal = m[1]; break; }
   348→        }
   349→      }
   350→    }
   351→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   352→  }
   353→
   354→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   355→  let DemandaContratadaPonta = EMPTY;
   356→  let DemandaContratadaForaPonta = EMPTY;
   357→  let DemandaMedidaPonta = EMPTY;
   358→  let DemandaMedidaForaPonta = EMPTY;
   359→  let DemandaFaturadaPonta = EMPTY;
   360→  let DemandaFaturadaForaPonta = EMPTY;
   361→  let ValorDemandaPonta = EMPTY;
   362→  let ValorDemandaForaPonta = EMPTY;
   363→
   364→  if (isSplitTariff) {
   365→    // Try to extract demanda contratada from flat text: "KW Ponta:33" or "KW Ponta: 33"
   366→    const dcMatch = flat.match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   367→    if (dcMatch) DemandaContratadaPonta = dcMatch[1].replace('.', ',');
   368→    const dcFpMatch = flat.match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   369→    if (dcFpMatch) DemandaContratadaForaPonta = dcFpMatch[1].replace('.', ',');
   370→
   371→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   372→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   373→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   374→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   375→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   376→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   377→  }
   378→
   379→  // ===== ULTRAPASSAGEM =====
   380→  let UltrapassagemPonta = EMPTY;
   381→  let UltrapassagemForaPonta = EMPTY;
   382→  let ValorUltrapassagemPonta = EMPTY;
   383→  let ValorUltrapassagemForaPonta = EMPTY;
   384→  // ===== DEMANDA NÃO CONSUMIDA =====
   385→  let DemandaNaoConsumidaPonta = EMPTY;
   386→  let DemandaNaoConsumidaForaPonta = EMPTY;
   387→  let ValorNaoConsumidaPonta = EMPTY;
   388→  let ValorNaoConsumidaForaPonta = EMPTY;
   389→  if (isSplitTariff) {
   390→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   391→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   392→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   393→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   394→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   395→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   396→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   397→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   398→  }
   399→
   400→  // ===== REATIVA EXCEDENTE =====
   401→  let ReativaExcedentePonta = EMPTY;
   402→  let ReativaExcedenteForaPonta = EMPTY;
   403→  let EnergiaReativaTotal = EMPTY;
   404→  let ValorReativaExcedentePonta = EMPTY;
   405→  let ValorReativaExcedenteForaPonta = EMPTY;
   406→  if (isSplitTariff) {
   407→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   408→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   409→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   410→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   411→  } else {
   412→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   413→  }
   414→
   415→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   416→  // Priority: table extraction > text-based extraction
   417→  let PIS = EMPTY;
   418→  let COFINS = EMPTY;
   419→  let ICMS = EMPTY;
   420→
   421→  // Use tributos from table extraction if available (most reliable)
   422→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   423→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   424→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   425→
   426→  // Fallback: text-based extraction for any missing tributo
   427→  if (!PIS || !COFINS || !ICMS) {
   428→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   429→    if (pisIdx >= 0) {
   430→      const nums: string[] = [];
   431→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   432→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   433→        if (lineNums) {
   434→          for (const n of lineNums) {
   435→            nums.push(n);
   436→            if (nums.length >= 3) break;
   437→          }
   438→        }
   439→      }
   440→      if (nums.length >= 3) {
   441→        if (!PIS) PIS = nums[0];
   442→        if (!COFINS) COFINS = nums[1];
   443→        if (!ICMS) ICMS = nums[2];
   444→      }
   445→    }
   446→    if (!PIS && !COFINS) {
   447→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   448→      if (combinedIdx >= 0) {
   449→        const nums: string[] = [];
   450→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   451→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   452→          if (m) nums.push(m[1]);
   453→        }
   454→        if (nums.length >= 3) {
   455→          if (!PIS) PIS = nums[0];
   456→          if (!COFINS) COFINS = nums[1];
   457→          if (!ICMS) ICMS = nums[2];
   458→        }
   459→      }
   460→    }
   461→    if (!PIS || !COFINS) {
   462→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   463→      if (flatMatch) {
   464→        if (!PIS) PIS = flatMatch[1];
   465→        if (!COFINS) COFINS = flatMatch[2];
   466→      }
   467→    }
   468→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   469→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   470→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
   471→  }
   472→
   473→  // ===== GDI =====
   474→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   475→
   476→  // ===== CIP (text-based) =====
   477→  const CIP_text = (() => {
   478→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   479→    if (cipIdx >= 0) {
   480→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   481→      const money = findAllMoney(region);
   482→      if (money.length > 0) return money[money.length - 1];
   483→    }
   484→    return findTributoValue(lines, flat, /\bCIP\b/i);
   485→  })();
   486→
   487→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   488→  let {
   489→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   490→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   491→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   492→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   493→  } = extractGDI(lines, flat);
   494→
   495→  // ===== SALDO ACUMULADO GD =====
   496→  const SaldoAcumuladoGD = (() => {
   497→    const m = flat.match(/Saldo\s*Acumulado\s*:\s*([\d.]+,\d{2})/i);
   498→    if (m) return m[1];
   499→    // Fallback: integer format like "14.100"
   500→    const m2 = flat.match(/Saldo\s*Acumulado\s*:\s*(\d[\d.]*)/i);
   501→    if (m2) return m2[1];
   502→    return EMPTY;
   503→  })();
   504→
   505→  // ===== VALOR TOTAL =====
   506→  // Best pattern: R$ value on same line as vencimento date + mes referência
   507→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   508→  const ValorTotal = (() => {
   509→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
   510→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
   511→    if (venTotal) return venTotal[1];
   512→
   513→    // Strategy 2: R$ + value + mes referência on same line
   514→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
   515→    if (totalMes) return totalMes[1];
   516→
   517→    // Strategy 3: "TOTAL A PAGAR" with R$
   518→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   519→    if (totalIdx >= 0) {
   520→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   521→      if (m1) return m1[1];
   522→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   523→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   524→        if (m2) return m2[1];
   525→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   526→        if (m3) return m3[1];
   527→      }
   528→    }
   529→
   530→    // Strategy 4: Last resort - largest R$ value
   531→    const lastMoney = findAllMoney(flat);
   532→    if (lastMoney.length > 0) {
   533→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   534→      return lastMoney[0];
   535→    }
   536→    return EMPTY;
   537→  })();
   538→
   539→  // ===== TABLE DATA OVERRIDES =====
   540→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
   541→  let CIP = CIP_text;
   542→
   543→  if (tableData && tableData.length > 0) {
   544→    let cipFromTable = EMPTY;
   545→    // Reset GDI accumulators for clean table-based extraction
   546→    let tbl_oUC_Qtd = '' as string;
   547→    let tbl_oUC_Valor = '' as string;
   548→    let tbl_mUC_Qtd = '' as string;
   549→    let tbl_mUC_Valor = '' as string;
   550→    // Track GDI section in table for rows with truncated/missing labels
   551→    let inGDISection = false;
   552→    let gdiSectionType = '' as string; // 'oUC' or 'mUC'
   553→    let lastGDILabel = '' as string;
   554→
   555→    for (const item of tableData) {
   556→      const lbl = item.label;
   557→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
   558→      const isPonta = /Ponta/i.test(lbl) && !isFora;
   559→
   560→      // Consumo - check Convencional (no Ponta/Fora) first, then split
   561→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
   562→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
   563→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
   564→          if (item.quantidade) ConsumoTotal = item.quantidade;
   565→          if (item.valor) ValorConsumoForaPonta = item.valor;
   566→        }
   567→        // VERDE Intermediária
   568→        else if (/Intermedi[áa]ria/i.test(lbl)) {
   569→          if (item.quantidade && !ConsumoForaPonta) ConsumoForaPonta = item.quantidade;
   570→        }
   571→        // Split tariff (AZUL/VERDE)
   572→        else if (isPonta) {
   573→          if (item.quantidade) ConsumoPonta = item.quantidade;
   574→          if (item.valor) ValorConsumoPonta = item.valor;
   575→        } else if (isFora) {
   576→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
   577→          if (item.valor) ValorConsumoForaPonta = item.valor;
   578→        }
   579→      }
   580→      // Demanda Medida
   581→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
   582→        if (!isFora) {
   583→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
   584→          if (item.valor) ValorDemandaPonta = item.valor;
   585→        } else {
   586→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
   587→          if (item.valor) ValorDemandaForaPonta = item.valor;
   588→        }
   589→      }
   590→      // Ultrapassagem
   591→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
   592→        if (!isFora) {
   593→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
   594→          if (item.valor) ValorUltrapassagemPonta = item.valor;
   595→        } else {
   596→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
   597→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
   598→        }
   599→      }
   600→      // Demanda Potência Não Consumida
   601→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
   602→        if (!isFora) {
   603→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
   604→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
   605→        } else {
   606→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
   607→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
   608→        }
   609→      }
   610→      // Reativa Excedente
   611→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
   612→        if (!isFora && /Ponta/i.test(lbl)) {
   613→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
   614→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   615→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
   616→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
   617→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
   618→        } else if (!isFora && !/Ponta/i.test(lbl)) {
   619→          // Generic reativa (Convencional)
   620→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
   621→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   622→        }
   623→      }
   624→      // CIP Municipal
   625→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
   626→        if (item.valor) cipFromTable = item.valor;
   627→      }
   628→      // GDI items
   629→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
   630→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
   631→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
   632→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
   633→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
   634→        // Multiple mUC entries: accumulate
   635→        if (item.quantidade) {
   636→          tbl_mUC_Qtd = tbl_mUC_Qtd
   637→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   638→            : item.quantidade;
   639→        }
   640→        if (item.valor) {
   641→          tbl_mUC_Valor = tbl_mUC_Valor
   642→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
   643→            : item.valor;
   644→        }
   645→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
   646→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
   647→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
   648→        // Accumulate as the same GDI type as the last full-label row
   649→        if (gdiSectionType === 'mUC') {
   650→          tbl_mUC_Qtd = tbl_mUC_Qtd
   651→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   652→            : item.quantidade;
   653→          if (item.valor) {
   654→            tbl_mUC_Valor = tbl_mUC_Valor
   655→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
   656→              : item.valor;
   657→          }
   658→        } else if (gdiSectionType === 'oUC') {
   659→          tbl_oUC_Qtd = tbl_oUC_Qtd
   660→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
   661→            : item.quantidade;
   662→          if (item.valor) {
   663→            tbl_oUC_Valor = tbl_oUC_Valor
   664→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
   665→              : item.valor;
   666→          }
   667→        }
   668→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
   669→        // Non-GDI, non-truncated row ends the GDI section
   670→        inGDISection = false;
   671→      }
   672→    }
   673→
   674→    // Override CIP from table
   675→    if (cipFromTable) CIP = cipFromTable;
   676→
   677→    // Use table GDI values if found, otherwise keep text-based fallback
   678→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
   679→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
   680→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
   681→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
   682→  }
   683→
   684→  return {
   685→    UC: cleanUC, Cidade, Tipo, Mercado, Concessionaria,
   686→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   687→    ModalidadeTarifaria, Subgrupo,
   688→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   689→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   690→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   691→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   692→    UltrapassagemPonta, UltrapassagemForaPonta,
   693→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
   694→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
   695→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   696→    ValorConsumoPonta, ValorConsumoForaPonta,
   697→    ValorDemandaPonta, ValorDemandaForaPonta,
   698→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   699→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   700→    PIS, COFINS, ICMS, GDI, CIP,
   701→    EnergiaReativaTotal,
   702→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   703→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   704→    SaldoAcumuladoGD,
   705→    ValorTotal,
   706→  };
   707→}
   708→
   709→// ===================== GDI EXTRACTION =====================
   710→
   711→function extractGDI(lines: string[], flat: string) {
   712→  const EMPTY = '';
   713→  let nOUC = 0;
   714→  let nMUC = 0;
   715→  let firstGDI = -1;
   716→
   717→  for (let i = 0; i < lines.length; i++) {
   718→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   719→      nOUC++;
   720→      if (firstGDI < 0) firstGDI = i;
   721→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   722→      nMUC++;
   723→      if (firstGDI < 0) firstGDI = i;
   724→    }
   725→  }
   726→
   727→  if (nOUC === 0 && nMUC === 0) {
   728→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   729→  }
   730→
   731→  const chaveIdx = (() => {
   732→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   733→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   734→    }
   735→    return -1;
   736→  })();
   737→
   738→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
   739→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   740→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
   741→      sectionEnd = i; break;
   742→    }
   743→  }
   744→  if (chaveIdx >= 0) {
   745→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   746→  }
   747→
   748→  const allQtds: number[] = [];
   749→  if (chaveIdx >= 0) {
   750→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   751→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   752→      if (m) allQtds.push(parseBR(m[1]));
   753→    }
   754→  }
   755→  if (allQtds.length === 0) {
   756→    for (let i = firstGDI; i < sectionEnd; i++) {
   757→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   758→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   759→      if (allQtds.length >= nOUC + nMUC) break;
   760→    }
   761→  }
   762→
   763→  const allRVals: number[] = [];
   764→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   765→  const totalNeeded = nOUC + nMUC;
   766→
   767→  for (let i = firstGDI; i < rSearchEnd; i++) {
   768→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   769→    if (m && allRVals.length < totalNeeded) {
   770→      allRVals.push(parseBR('-' + m[2]));
   771→    }
   772→  }
   773→  if (allRVals.length < totalNeeded) {
   774→    for (let i = firstGDI; i < rSearchEnd; i++) {
   775→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   776→      if (m && allRVals.length < totalNeeded) {
   777→        allRVals.push(parseBR(m[1]));
   778→      }
   779→    }
   780→  }
   781→
   782→  const oUCQtds = allQtds.splice(0, nOUC);
   783→  const oUCRVals = allRVals.splice(0, nOUC);
   784→
   785→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   786→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   787→
   788→  return {
   789→    oUC_Qtd: sumQtd(oUCQtds),
   790→    oUC_Valor: sumR(oUCRVals),
   791→    mUC_Qtd: sumQtd(allQtds),
   792→    mUC_Valor: sumR(allRVals),
   793→  };
   794→}
   795→
   796→// ===================== NUMBER FORMATTING =====================
   797→
   798→function fmtBRNumber(n: number): string {
   799→  const abs = Math.abs(n);
   800→  const intPart = Math.floor(abs).toString();
   801→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   802→  let formatted = '';
   803→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   804→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   805→    formatted = intPart[i] + formatted;
   806→  }
   807→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   808→}

   810→function fmtSumBR(a: string, b: string): string {
   811→  return fmtBRNumber(parseBR(a) + parseBR(b));
   812→}

   816→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   817→  const idx = findLine(lines, labelPattern);
   818→  if (idx < 0) return EMPTY;
   819→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   820→    const line = lines[i];
   821→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   822→    const m = line.match(/([\d.]+,\d{2})/);
   823→    if (m) return m[1];
   824→  }
   825→  return EMPTY;
   826→}

   828→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   829→  const idx = findLine(lines, labelPattern);
   830→  if (idx < 0) return EMPTY;
   831→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   832→  const moneyVals = findAllMoney(region);
   833→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   834→  return EMPTY;
   835→}

   837→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   838→  const idx = findLine(lines, labelPattern);
   839→  if (idx >= 0) {
   840→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   841→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   842→      if (m) return m[1];
   843→    }
   844→  }
   845→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   846→  return flatMatch ? flatMatch[1] : EMPTY;
   847→}
