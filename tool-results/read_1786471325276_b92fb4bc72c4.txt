    63→    63→// Parse Brazilian number format: "1.135,00" → 1135.00, "-110,49" → -110.49
    64→    64→function parseBR(s: string): number {
    65→    65→  const neg = s.startsWith('-');
    66→    66→  const clean = neg ? s.slice(1) : s;
    67→    67→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    68→    68→}
    69→    69→
    70→    70→function findLine(lines: string[], pattern: RegExp): number {
    71→    71→  return lines.findIndex(l => pattern.test(l));
    72→    72→}
    73→    73→
    74→    74→function findAllMoney(text: string): string[] {
    75→    75→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
    76→    76→  if (!matches) return [];
    77→    77→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
    78→    78→}
    79→    79→
    80→    80→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    81→    81→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    82→    82→    if (Math.abs(i - startIdx) > maxDist) break;
    83→    83→    const line = lines[i];
    84→    84→    if (mustBeOnlyNum) {
    85→    85→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    86→    86→      if (m) return m[1];
    87→    87→    } else {
    88→    88→      const m = line.match(/([\d.]+,\d{2})/);
    89→    89→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
    90→    90→    }
    91→    91→  }
    92→    92→  return EMPTY;
    93→    93→}

    97→    97→export function extractEnergisaData(rawText: string): EnergisaInvoiceData {
    98→    98→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
    99→    99→  const flat = lines.join(' ');
   100→   100→
   101→   101→  // ===== CONCESSIONÁRIA =====
   102→   102→  // Full company name on its own line near the top
   103→   103→  const Concessionaria = (() => {
   104→   104→    for (const line of lines.slice(0, 20)) {
   105→   105→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
   106→   106→        return line.replace(/\s+/g, ' ');
   107→   107→      }
   108→   108→    }
   109→   109→    return EMPTY;
   110→   110→  })();
   111→   111→
   112→   112→  // ===== TIPO =====
   113→   113→  const Tipo = (() => {
   114→   114→    const u = flat.toUpperCase();
   115→   115→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   116→   116→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   117→   117→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   118→   118→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   119→   119→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   120→   120→    if (/RURAL/.test(u)) return 'RURAL';
   121→   121→    return EMPTY;
   122→   122→  })();
   123→   123→
   124→   124→  // ===== MERCADO =====
   125→   125→  // Look for classification line: "SERVIÇO PÚBLICO / ÁGUA, ESGOTO E SANEAMENTO"
   126→   126→  const Mercado = (() => {
   127→   127→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   128→   128→    if (m) {
   129→   129→      const val = m[1].trim().replace(/\s+/g, ' ');
   130→   130→      if (val.length < 50) return val.toUpperCase();
   131→   131→    }
   132→   132→    return EMPTY;
   133→   133→  })();
   134→   134→
   135→   135→  // ===== CIDADE =====
   136→   136→  // Pattern: "DOURADOS (AG: 10)"
   137→   137→  const Cidade = (() => {
   138→   138→    for (const line of lines.slice(0, 60)) {
   139→   139→      const m = line.match(/([A-ZÀ-Ú]{4,25})\s*\(\s*AG\s*:\s*\d+\s*\)/);
   140→   140→      if (m) {
   141→   141→        const name = m[1].toUpperCase();
   142→   142→        // Skip company names
   143→   143→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA'];
   144→   144→        if (!skip.includes(name)) return name;
   145→   145→      }
   146→   146→    }
   147→   147→    return EMPTY;
   148→   148→  })();
   149→   149→
   150→   150→  // ===== UC =====
   151→   151→  // Pattern: dotted format X.XXX.XXX-XX on its own line or with "KWH" prefix
   152→   152→  const UC = (() => {
   153→   153→    // Strategy 1: line with only UC (e.g., "3.562.051-59", "43.210.051-68", "131.752.051-07")
   154→   154→    for (const line of lines) {
   155→   155→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   156→   156→      if (m) return m[1];
   157→   157→    }
   158→   158→    // Strategy 2: UC embedded in short line
   159→   159→    for (const line of lines) {
   160→   160→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   161→   161→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   162→   162→    }
   163→   163→    // Strategy 3: broad search
   164→   164→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   165→   165→    return broad ? broad[1] : EMPTY;
   166→   166→  })();
   167→   167→
   168→   168→  // ===== MÊS REFERÊNCIA =====
   169→   169→  // Pattern: "Junho / 2026" or "JUNHO / 2026"
   170→   170→  const MesReferencia = (() => {
   171→   171→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   172→   172→    const numToMonth: Record<string, string> = {
   173→   173→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   174→   174→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   175→   175→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   176→   176→    };
   177→   177→    // Named month pattern
   178→   178→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   179→   179→    if (nameMatch) {
   180→   180→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   181→   181→      return `${month} / ${nameMatch[2]}`;
   182→   182→    }
   183→   183→    // Numeric month pattern (e.g., "06/2026")
   184→   184→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   185→   185→    if (numMatch && numToMonth[numMatch[1]]) {
   186→   186→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   187→   187→    }
   188→   188→    return EMPTY;
   189→   189→  })();
   190→   190→
   191→   191→  // ===== ANO LEITURA =====
   192→   192→  const AnoLeitura = (() => {
   193→   193→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   194→   194→    if (m) return m[1];
   195→   195→    if (MesReferencia) {
   196→   196→      const ym = MesReferencia.match(/(\d{4})/);
   197→   197→      if (ym) return ym[1];
   198→   198→    }
   199→   199→    return EMPTY;
   200→   200→  })();
   201→   201→
   202→   202→  // ===== VENCIMENTO =====
   203→   203→  // ENERGISA PDFs often concatenate "VENCIMENTO" with other labels like "MATRÍCULA"
   204→   204→  // The date may appear BEFORE or AFTER the label
   205→   205→  const Vencimento = (() => {
   206→   206→    // Strategy 1: find line containing VENCIMENTO (even concatenated), check nearby for date
   207→   207→    const venIdx = findLine(lines, /VENCIMENTO/i);
   208→   208→    if (venIdx >= 0) {
   209→   209→      // Check lines BEFORE the label (ENERGISA often puts date before)
   210→   210→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   211→   211→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   212→   212→        if (m) return m[1];
   213→   213→      }
   214→   214→      // Check lines AFTER the label
   215→   215→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   216→   216→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   217→   217→        if (m) return m[1];
   218→   218→      }
   219→   219→      // Check same line
   220→   220→      const sameLine = lines[venIdx].match(/(\d{2}\/\d{2}\/\d{4})/);
   221→   221→      if (sameLine) return sameLine[1];
   222→   222→    }
   223→   223→    // Strategy 2: "TOTAL A PAGAR" area - date often nearby
   224→   224→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   225→   225→    if (totalIdx >= 0) {
   226→   226→      for (let i = totalIdx - 3; i <= Math.min(totalIdx + 3, lines.length - 1); i++) {
   227→   227→        if (i < 0) continue;
   228→   228→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   229→   229→        if (m) return m[1];
   230→   230→      }
   231→   231→    }
   232→   232→    return EMPTY;
   233→   233→  })();
   234→   234→
   235→   235→  // ===== NOTA FISCAL =====
   236→   236→  const NotaFiscal = (() => {
   237→   237→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   238→   238→    return m ? m[1] : EMPTY;
   239→   239→  })();
   240→   240→
   241→   241→  // ===== MODALIDADE TARIFÁRIA =====
   242→   242→  const ModalidadeTarifaria = (() => {
   243→   243→    // In the classification line: "MTC-CONVENCIONAL BAIXA TENSÃO / B3"
   244→   244→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   245→   245→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   246→   246→    if (/AZUL/i.test(flat)) return 'AZUL';
   247→   247→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   248→   248→    if (/VERDE/i.test(flat)) return 'VERDE';
   249→   249→    return EMPTY;
   250→   250→  })();
   251→   251→
   252→   252→  // ===== SUBGRUPO =====
   253→   253→  // Pattern: "/ B3" in classification line
   254→   254→  const Subgrupo = (() => {
   255→   255→    const m = flat.match(/\/(\s*[AB]\d?\s*)(?:\s|\/|$)/);
   256→   256→    if (m) return m[1].trim().toUpperCase();
   257→   257→    // Fallback: search in classification area
   258→   258→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   259→   259→    if (classIdx >= 0) {
   260→   260→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   261→   261→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   262→   262→      if (m2) return m2[1].toUpperCase();
   263→   263→    }
   264→   264→    return EMPTY;
   265→   265→  })();
   266→   266→
   267→   267→  const hasTableData = tableData && tableData.length > 0;
   268→   268→  let tablePIS = EMPTY, tableCOFINS = EMPTY, tableICMS = EMPTY;
   269→   269→  if (hasTableData) {
   270→   270→    for (const item of tableData) {
   271→   271→      if (/Consumo em kWh.*Ponta(?!.*Fora)/i.test(item.label) && !tableConsumoPonta) { tableConsumoPonta = item.quantidade; tableValorConsumoPonta = item.valor; }
   272→   272→      else if (/Consumo em kWh.*Fora\s*Ponta/i.test(item.label) && !tableConsumoFP) { tableConsumoFP = item.quantidade; tableValorConsumoFP = item.valor; }
   273→   273→      else if (/Demanda.*Medida.*Ponta(?!.*Fora)/i.test(item.label) && !tableDemMedPonta) { tableDemMedPonta = item.quantidade; tableValorDemandaPonta = item.valor; }
   274→   274→      else if (/Demanda.*Medida.*Fora\s*Ponta/i.test(item.label) && !tableDemMedFP) { tableDemMedFP = item.quantidade; tableValorDemFP = item.valor; }
   275→   275→      else if (/Ultrapassagem|NÃO.*Consumida/i.test(item.label)) {
   276→   276→        if (/Ponta(?!.*Fora)/i.test(item.label) && !tableUltrapPonta) { tableUltrapPonta = item.quantidade; tableValorUltraponta = item.valor; }
   277→   277→        else if (/Fora\s*Ponta/i.test(item.label) && !tableUltrapFP) { tableUltrapFP = item.quantidade; tableValorUltrapFP = item.valor; }
   278→   278→      else if (/Reactiv?a.*Excedente.*Ponta/i.test(item.label)) {
   279→   279→        if (/Ponta(?!.*Fora)/i.test(item.label) && !tableReatExFP) { tableReatExFP = item.quantidade; tableValorReatFPonta = item.valor; }
   280→   280→        else if (/Reactiv?a.*Excedente.*Fora\s*Ponta/i.test(item.label) && !tableReatEFP) { tableReatEFP = item.quantidade; tableValorReatEFP = item.valor; }
   281→   281→      if (item.pisCofins && !tablePIS) tablePIS = item.pisCofins;
   282→   282→      if (item.baseCalcICMS && !tableCOFINS) tableCOFINS = item.baseCalcICMS;
   283→   283→      if (item.icms && !tableICMS) tableICMS = item.icms;
   284→   284→    }
   285→   285→  }
   286→   286→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   287→   287→
   288→   288→  // ===== CONSUMO =====
   289→   289→  let ConsumoPonta = EMPTY;
   290→   290→  let ConsumoForaPonta = EMPTY;
   291→   291→  let ConsumoTotal = EMPTY;
   292→   292→  let ValorConsumoPonta = EMPTY;
   293→   293→  let ValorConsumoForaPonta = EMPTY;
   294→   294→
   295→   295→  if (isSplitTariff) {
   296→   296→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   297→   297→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   298→   298→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   299→   299→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   300→   300→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   301→   301→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   302→   302→  } else {
   303→   303→    // Convencional: look for "KWH1.044,00" or "KWH 1.044,00"
   304→   304→    for (const line of lines) {
   305→   305→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   306→   306→      if (km) { ConsumoTotal = km[1]; break; }
   307→   307→    }
   308→   308→    // Fallback: "CONSUMO FATURADO" area
   309→   309→    if (!ConsumoTotal) {
   310→   310→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   311→   311→      if (consIdx >= 0) {
   312→   312→        // Look for numbers near CONSUMO FATURADO, but skip the month reference line
   313→   313→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   314→   314→          const line = lines[i];
   315→   315→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   316→   316→          if (/[A-Za-z]{4,}/.test(line)) continue;
   317→   317→          const m = line.match(/^([\d.]+,\d{2})$/);
   318→   318→          if (m) { ConsumoTotal = m[1]; break; }
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→    // R$ value for consumo (Convencional)
   323→   323→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   324→   324→  }
   325→   325→
   326→   326→  // ===== DEMANDA (only split tariffs) =====
   327→   327→  let DemandaContratadaPonta = EMPTY;
   328→   328→  let DemandaContratadaForaPonta = EMPTY;
   329→   329→  let DemandaMedidaPonta = EMPTY;
   330→   330→  let DemandaMedidaForaPonta = EMPTY;
   331→   331→  let DemandaFaturadaPonta = EMPTY;
   332→   332→  let DemandaFaturadaForaPonta = EMPTY;
   333→   333→  let ValorDemandaPonta = EMPTY;
   334→   334→  let ValorDemandaForaPonta = EMPTY;
   335→   335→  if (isSplitTariff) {
   336→   336→    DemandaContratadaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Ponta(?![^\n]*Fora)/i);
   337→   337→    DemandaContratadaForaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Fora\s*Ponta/i);
   338→   338→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?![^\n]*Fora)/i);
   339→   339→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   340→   340→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?![^\n]*Fora)/i) || DemandaMedidaPonta;
   341→   341→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   342→   342→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?![^\n]*Fora)/i);
   343→   343→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   344→   344→  }
   345→   345→
   346→   346→  // ===== ULTRAPASSAGEM (only split tariffs) =====
   347→   347→  let UltrapassagemPonta = EMPTY;
   348→   348→  let UltrapassagemForaPonta = EMPTY;
   349→   349→  let ValorUltrapassagemPonta = EMPTY;
   350→   350→  let ValorUltrapassagemForaPonta = EMPTY;
   351→   351→  if (isSplitTariff) {
   352→   352→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   353→   353→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   354→   354→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   355→   355→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   356→   356→  }
   357→   357→
   358→   358→  // ===== REATIVA EXCEDENTE =====
   359→   359→  let ReativaExcedentePonta = EMPTY;
   360→   360→  let ReativaExcedenteForaPonta = EMPTY;
   361→   361→  let EnergiaReativaTotal = EMPTY;
   362→   362→  let ValorReativaExcedentePonta = EMPTY;
   363→   363→  let ValorReativaExcedenteForaPonta = EMPTY;
   364→   364→  if (isSplitTariff) {
   365→   365→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   366→   366→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   367→   367→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   368→   368→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   369→   369→  } else {
   370→   370→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   371→   371→  }
   372→   372→
   373→   373→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   374→   374→  // ENERGISA format (from real PDF):
   375→   375→  // Line N: "PIS/"  Line N+1: "COFINS (R$)"  Line N+2: "18,47 85,09 236,20"
   376→   376→  // Or on the same flat section: "PIS/COFINS (R$) 18,47 85,09 236,20"
   377→   377→  let PIS = EMPTY;
   378→   378→  let COFINS = EMPTY;
   379→   379→  let ICMS = EMPTY;
   380→   380→
   381→   381→  (() => {
   382→   382→    // Strategy 1: Find "PIS/" line (the label is split)
   383→   383→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   384→   384→    if (pisIdx >= 0) {
   385→   385→      // Next line might be "COFINS (R$)" or values
   386→   386→      // Check next few lines for the 3 consecutive numbers
   387→   387→      const nums: string[] = [];
   388→   388→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   389→   389→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   390→   390→        if (lineNums) {
   391→   391→          for (const n of lineNums) {
   392→   392→            nums.push(n);
   393→   393→            if (nums.length >= 3) break;
   394→   394→          }
   395→   395→        }
   396→   396→      }
   397→   397→      if (nums.length >= 3) {
   398→   398→        PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   399→   399→      }
   400→   400→    }
   401→   401→    // Strategy 2: "PIS/COFINS" on same line
   402→   402→    if (!PIS && !COFINS) {
   403→   403→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   404→   404→      if (combinedIdx >= 0) {
   405→   405→        const nums: string[] = [];
   406→   406→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   407→   407→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   408→   408→          if (m) nums.push(m[1]);
   409→   409→        }
   410→   410→        if (nums.length >= 3) {
   411→   411→          PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   412→   412→        }
   413→   413→      }
   414→   414→    }
   415→   415→    // Strategy 3: flat text search
   416→   416→    if (!PIS || !COFINS) {
   417→   417→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   418→   418→      if (flatMatch) {
   419→   419→        if (!PIS) PIS = flatMatch[1];
   420→   420→        if (!COFINS) COFINS = flatMatch[2];
   421→   421→      }
   422→   422→    }
   423→   423→    // Strategy 4: Individual fallback
   424→   424→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   425→   425→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   426→   426→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%)/i);
   427→   427→  })();
   428→   428→
   429→   429→  // ===== GDI =====
   430→   430→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   431→   431→
   432→   432→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   433→   433→  // ENERGISA GDI tabular layout: quantities appear as standalone numbers
   434→   434→  // after "chave de acesso:", R$ values are negative numbers before it.
   435→   435→  // When multiple mUC entries exist, they must be SUMMED.
   436→   436→  const {
   437→   437→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   438→   438→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   439→   439→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   440→   440→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   441→   441→  } = (() => {
   442→   442→    // Count how many oUC and mUC label lines exist
   443→   443→    let nOUC = 0;
   444→   444→    let nMUC = 0;
   445→   445→    let firstGDI = -1;
   446→   446→    for (let i = 0; i < lines.length; i++) {
   447→   447→      if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   448→   448→        nOUC++;
   449→   449→        if (firstGDI < 0) firstGDI = i;
   450→   450→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   451→   451→        nMUC++;
   452→   452→        if (firstGDI < 0) firstGDI = i;
   453→   453→      }
   454→   454→    }
   455→   455→    if (nOUC === 0 && nMUC === 0) return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   456→   456→
   457→   457→    // Find "chave de acesso:" as section anchor (search up to 100 lines ahead)
   458→   458→    const chaveIdx = (() => {
   459→   459→      for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   460→   460→        if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   461→   461→      }
   462→   462→      return -1;
   463→   463→    })();
   464→   464→
   465→   465→    // Find section end boundary (search up to 120 lines ahead)
   466→   466→    let sectionEnd = Math.min(firstGDI + 100, lines.length);
   467→   467→    for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   468→   468→      if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) { sectionEnd = i; break; }
   469→   469→    }
   470→   470→    // Ensure sectionEnd extends past chaveIdx so quantities can be found
   471→   471→    if (chaveIdx >= 0) {
   472→   472→      sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   473→   473→    }
   474→   474→
   475→   475→    // --- Quantities (kWh) after "chave de acesso:" ---
   476→   476→    const allQtds: number[] = [];
   477→   477→    if (chaveIdx >= 0) {
   478→   478→      for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   479→   479→        const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   480→   480→        if (m) allQtds.push(parseBR(m[1]));
   481→   481→      }
   482→   482→    }
   483→   483→    // Fallback: large standalone numbers in GDI section
   484→   484→    if (allQtds.length === 0) {
   485→   485→      for (let i = firstGDI; i < sectionEnd; i++) {
   486→   486→        const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   487→   487→        if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   488→   488→        if (allQtds.length >= nOUC + nMUC) break;
   489→   489→      }
   490→   490→    }
   491→   491→
   492→   492→    // --- R$ Values (embedded negative numbers before "chave de acesso:") ---
   493→   493→    // ENERGISA PDFs put GDI R$ values embedded in composite lines like
   494→   494→    // "1,083930-17,34" (alíquota + credit value). Standalone negatives like
   495→   495→    // "-1,56" are tax adjustments, NOT the GDI credit amounts.
   496→   496→    const allRVals: number[] = [];
   497→   497→    const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   498→   498→    const totalNeeded = nOUC + nMUC;
   499→   499→
   500→   500→    // Strategy 1: Embedded negatives — lines ending with a negative number
   501→   501→    // but containing additional content before it (e.g. "1,083930-17,34")
   502→   502→    for (let i = firstGDI; i < rSearchEnd; i++) {
   503→   503→      const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   504→   504→      if (m && allRVals.length < totalNeeded) {
   505→   505→        allRVals.push(parseBR('-' + m[2]));
   506→   506→      }
   507→   507→    }
   508→   508→    // Strategy 2 fallback: standalone negative numbers (for PDFs without embedded format)
   509→   509→    if (allRVals.length < totalNeeded) {
   510→   510→      for (let i = firstGDI; i < rSearchEnd; i++) {
   511→   511→        const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   512→   512→        if (m && allRVals.length < totalNeeded) {
   513→   513→          allRVals.push(parseBR(m[1]));
   514→   514→        }
   515→   515→      }
   516→   516→    }
   517→   517→
   518→   518→    // Split: first nOUC values → oUC, next nMUC → mUC (sum each group)
   519→   519→    const oUCQtds = allQtds.splice(0, nOUC);
   520→   520→    const oUCRVals = allRVals.splice(0, nOUC);
   521→   521→    // Remaining are mUC
   522→   522→    const mUCQtds = allQtds; // already spliced
   523→   523→    const mUCRVals = allRVals;
   524→   524→
   525→   525→    const fmt = (n: number) => n.toFixed(2).replace('.', ',');
   526→   526→    const fmtInt = (n: number) => {
   527→   527→      const abs = Math.abs(n);
   528→   528→      const intPart = Math.floor(abs).toString();
   529→   529→      const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   530→   530→      let formatted = '';
   531→   531→      for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   532→   532→        if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   533→   533→        formatted = intPart[i] + formatted;
   534→   534→      }
   535→   535→      return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   536→   536→    };
   537→   537→
   538→   538→    const sumQtd = (arr: number[]) => arr.length > 0 ? fmtInt(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   539→   539→    const sumR = (arr: number[]) => arr.length > 0 ? fmtInt(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   540→   540→
   541→   541→    return {
   542→   542→      oUC_Qtd: sumQtd(oUCQtds),
   543→   543→      oUC_Valor: sumR(oUCRVals),
   544→   544→      mUC_Qtd: sumQtd(mUCQtds),
   545→   545→      mUC_Valor: sumR(mUCRVals),
   546→   546→    };
   547→   547→  })();
   548→   548→
   549→   549→  // ===== CIP =====
   550→   550→  const CIP = (() => {
   551→   551→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   552→   552→    if (cipIdx >= 0) {
   553→   553→      // Look on same line and nearby for R$ value
   554→   554→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   555→   555→      const money = findAllMoney(region);
   556→   556→      if (money.length > 0) return money[money.length - 1];
   557→   557→    }
   558→   558→    return findTributoValue(lines, flat, /\bCIP\b/i);
   559→   559→  })();
   560→   560→
   561→   561→  // ===== VALOR TOTAL =====
   562→   562→  const ValorTotal = (() => {
   563→   563→    // Strategy 1: "TOTAL A PAGAR" with R$ on same or next line
   564→   564→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   565→   565→    if (totalIdx >= 0) {
   566→   566→      // Check same line
   567→   567→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   568→   568→      if (m1) return m1[1];
   569→   569→      // Check next lines
   570→   570→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   571→   571→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   572→   572→        if (m2) return m2[1];
   573→   573→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   574→   574→        if (m3) return m3[1];
   575→   575→      }
   576→   576→    }
   577→   577→    // Strategy 2: flat search
   578→   578→    const flatMatch = flat.match(/TOTAL\s*A\s+PAGAR[^\d]*?R\$\s*([\d.]+,\d{2})/i);
   579→   579→    if (flatMatch) return flatMatch[1];
   580→   580→    // Strategy 3: "R$ X.XXX,XX" near end of document
   581→   581→    const lastMoney = findAllMoney(flat);
   582→   582→    if (lastMoney.length > 0) {
   583→   583→      // The total is typically one of the largest values
   584→   584→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   585→   585→      return lastMoney[0];
   586→   586→    }
   587→   587→    return EMPTY;
   588→   588→  })();
   589→   589→
   590→   590→  return {
   591→   591→    UC, Cidade, Tipo, Mercado, Concessionaria,
   592→   592→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   593→   593→    ModalidadeTarifaria, Subgrupo,
   594→   594→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   595→   595→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   596→   596→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   597→   597→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   598→   598→    UltrapassagemPonta, UltrapassagemForaPonta,
   599→   599→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   600→   600→    ValorConsumoPonta, ValorConsumoForaPonta,
   601→   601→    ValorDemandaPonta, ValorDemandaForaPonta,
   602→   602→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   603→   603→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   604→   604→    PIS, COFINS, ICMS, GDI, CIP,
   605→   605→    EnergiaReativaTotal,
   606→   606→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   607→   607→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   608→   608→    ValorTotal,
   609→   609→  };
   610→   610→}

   614→   614→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   615→   615→  const idx = findLine(lines, labelPattern);
   616→   616→  if (idx < 0) return EMPTY;
   617→   617→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   618→   618→    const line = lines[i];
   619→   619→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   620→   620→    const m = line.match(/([\d.]+,\d{2})/);
   621→   621→    if (m) return m[1];
   622→   622→  }
   623→   623→  return EMPTY;
   624→   624→}
   625→   625→
   626→   626→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   627→   627→  const idx = findLine(lines, labelPattern);
   628→   628→  if (idx < 0) return EMPTY;
   629→   629→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   630→   630→  const moneyVals = findAllMoney(region);
   631→   631→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   632→   632→  return EMPTY;
   633→   633→}
   634→   634→
   635→   635→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   636→   636→  const idx = findLine(lines, labelPattern);
   637→   637→  if (idx >= 0) {
   638→   638→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   639→   639→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   640→   640→      if (m) return m[1];
   641→   641→    }
   642→   642→  }
   643→   643→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   644→   644→  return flatMatch ? flatMatch[1] : EMPTY;
   645→   645→}
