    90→    90→  return label
    91→    91→    .replace(/^Demanda (Contratada|Medida|Faturada) /, '')

    93→    93→    .replace(/^Valor (Consumo|Demanda|Ultrapassagem|Reativa Excedente|Não Consumida) /, '')
    94→    94→    .replace(/^Energia Atv Injetada GDI /, '')
    95→    95→    .replace(/^Consumo em kWh \(Grupo B\)/, 'Consumo kWh')
    96→    96→    .replace(/^Consumo \(R\$\) \(Grupo B\)/, 'Consumo R$')
    97→    97→    .replace(/^Ultrapassagem /, '')
    98→    98→    .replace(/^Reativa Excedente /, '')
    99→    99→    .replace(/^Energia Reativa Total/, 'Total')
   100→   100→    .replace(/^Saldo Acumulado GD/, 'Saldo GD')
   101→   101→    .replace(/\s*\(kWh\)\s*/, '')
   102→   102→    .replace(/\s*\(kW\)\s*/, '')
   103→   103→    .replace(/\s*\(R\$\)\s*/, '')
   104→   104→    .trim();
   105→   105→}

   129→   129→      const file = pdfFiles[i];

   131→   131→      try {
   132→   132→        const fd = new FormData(); fd.append('file', file);
   133→   133→        const res = await fetch('/api/extract-pdf', { method: 'POST', body: fd });
   134→   134→        const json = await res.json();
   135→   135→        if (!res.ok) {
   136→   136→          setFileProgress((p) => p.map((x, idx) => idx === i ? { ...x, status: 'error' as const, errorMsg: json.error } : x));
   137→   137→          toast.error(`${file.name}: ${json.error}`);
   138→   138→        } else {
   139→   139→          const er: ExtractionResult = { fileName: file.name, data: json.data, stats: json.stats, rawText: json.rawText };
   140→   140→          newResults.push(er);
   141→   141→          setFileProgress((p) => p.map((x, idx) => idx === i ? { ...x, status: 'done' as const, result: er } : x));
   142→   142→          toast.success(`${file.name}: ${json.stats.extractedCount}/${json.stats.totalFields} campos (${json.stats.extractionRate}%)`);
   143→   143→        }
   144→   144→      } catch {
   145→   145→        setFileProgress((p) => p.map((x, idx) => idx === i ? { ...x, status: 'error' as const, errorMsg: 'Erro ao processar' } : x));
   146→   146→        toast.error(`${file.name}: Erro ao processar.`);
   147→   147→      }
   148→   148→    }
   149→   149→
   150→   150→    setResults((prev) => {
   151→   151→      const updated = [...newResults];
   152→   152→      for (const e of prev) {
   153→   153→        const uc = e.data.UC.replace(/\D/g, '');
   154→   154→        if (uc && !updated.some((r) => r.data.UC.replace(/\D/g, '') === uc)) updated.push(e);
   155→   155→      }
   156→   156→      for (const n of newResults) {
   157→   157→        const uc = n.data.UC.replace(/\D/g, '');
   158→   158→        if (uc && prev.some((r) => r.data.UC.replace(/\D/g, '') === uc)) toast.info(`UC ${n.data.UC} atualizada.`);
   159→   159→      }
   160→   160→      return updated;
   161→   161→    });
   162→   162→    setIsProcessing(false);
   163→   163→    setFileProgress([]);
   164→   164→    if (newResults.length > 0) setActiveTab('results');
   165→   165→  }, []);

   167→   167→  const handleDrop = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(false); if (e.dataTransfer.files.length > 0) processFiles(e.dataTransfer.files); }, [processFiles]);
   168→   168→  const handleDragOver = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(true); }, []);
   169→   169→  const handleDragLeave = useCallback(() => setIsDragging(false), []);
   170→   170→  const handleFileChange = useCallback((e: React.ChangeEvent<HTMLInputElement>) => { if (e.target.files?.length) { processFiles(e.target.files); e.target.value = ''; } }, [processFiles]);
