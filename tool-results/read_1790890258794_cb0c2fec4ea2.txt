    81→function parseBR(s: string): number {
    82→  const neg = s.startsWith('-');
    83→  const clean = neg ? s.slice(1) : s;
    84→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    85→}
    86→
    87→function formatBR(n: number): string {
    88→  if (n === 0) return '0,00';
    89→  const neg = n < 0;
    90→  const abs = Math.abs(n);
    91→  const rounded = Math.round(abs * 100) / 100;
    92→  const parts = rounded.toFixed(2).split('.');
    93→  const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    94→  const result = intPart + ',' + parts[1];
    95→  return neg ? '-' + result : result;
    96→}

    98→function findLine(lines: string[], pattern: RegExp): number {
    99→  return lines.findIndex(l => pattern.test(l));

   102→function findAllMoney(text: string): string[] {
   103→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
   104→  if (!matches) return [];
   105→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
   106→}

   108→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
   109→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
   110→    if (Math.abs(i - startIdx) > maxDist) break;
   111→    const line = lines[i];
   112→    if (mustBeOnlyNum) {
   113→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
   114→      if (m) return m[1];
   115→    } else {
   116→      const m = line.match(/([\d.]+,\d{2})/);
   117→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   118→    }
   119→  }
   120→  return EMPTY;
   121→}

   123→// ===================== MAIN EXTRACTOR =====================
   124→
   125→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   126→  const tableData = tableExtraction?.items;
   127→  const tributosFromTable = tableExtraction
   128→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   129→    : null;
   130→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   131→  const flat = lines.join(' ');
   132→
   133→  // ===== CONCESSIONÁRIA =====
   134→  // Priority: detect the ISSUING concessionária (not the customer).
   135→  // For electricity invoices (DANF3E / CONTA DE ENERGIA), ENERGISA/ELEKTRO is the issuer.
   136→  // For water/sewage invoices, SANESUL is the issuer.
   137→  const Concessionaria = (() => {
   138→    const u = flat.toUpperCase();
   139→    const isEnergyInvoice = /DANF3E|ENERGIA\s*EL[eÉ]TRICA|DISTR\.\s*DE\s*ENERGIA/i.test(u);
   140→    if (isEnergyInvoice) {
   141→      // For energy invoices, the issuing utility is ENERGISA/ELEKTRO/NEOENERGIA
   142→      if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   143→      if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   144→      if (/ENERGISA/i.test(u)) return 'ENERGISA';
   145→    }
   146→    // For non-energy invoices or when no energy indicator found
   147→    if (/SANESUL/i.test(u)) return 'SANESUL';
   148→    if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   149→    if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   150→    if (/ENERGISA/i.test(u)) return 'ENERGISA';
   151→    return EMPTY;
   152→  })();
   153→
   154→  // ===== TIPO =====
   155→  const Tipo = (() => {
   156→    const u = flat.toUpperCase();
   157→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   158→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   159→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   160→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   161→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   162→    if (/RURAL/.test(u)) return 'RURAL';
   163→    return EMPTY;
   164→  })();
   165→
   166→  // ===== MERCADO =====
   167→  const Mercado = (() => {
   168→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   169→    if (m) {
   170→      const val = m[1].trim().replace(/\s+/g, ' ');
   171→      if (val.length < 50) return val.toUpperCase();
   172→    }
   173→    return EMPTY;
   174→  })();
   175→
   176→  // ===== CIDADE =====
   177→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   178→  const Cidade = (() => {
   179→    for (const line of lines.slice(0, 60)) {
   180→      // Match: CityName [StateAbbr] (AG: XX)
   181→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   182→      if (m) {
   183→        const name = m[1].trim().toUpperCase();
   184→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   185→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   186→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   187→      }
   188→    }
   189→    return EMPTY;
   190→  })();
   191→
   192→  // ===== UC =====
   193→  const UC = (() => {
   194→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   195→    const isCnpjCpf = (s: string) => {
   196→      const raw = s.replace(/[.\-\/]/g, '');
   197→      return raw.length === 14 || raw.length === 11;
   198→    };
   199→
   200→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   201→    for (const line of lines) {
   202→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   203→      if (m) {
   204→        const num = m[1].replace(/^0+/, '') || '0';
   205→        if (num.length >= 3) return num;
   206→      }
   207→    }
   208→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   209→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   210→    if (slashCtx) {
   211→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   212→      if (num.length >= 3) return num;
   213→    }
   214→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   215→    // e.g., "1.409.407.051-47" → "140940705147"
   216→    for (const line of lines) {
   217→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   218→      if (m && !isCnpjCpf(m[1])) return m[1];
   219→    }
   220→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   221→    for (const line of lines) {
   222→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   223→      if (m) return m[1];
   224→    }
   225→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   226→    for (const line of lines) {
   227→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   228→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   229→    }
   230→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   231→    for (const line of lines) {
   232→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   233→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   234→    }
   235→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   236→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   237→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   238→    if (ucMat) return ucMat[1];
   239→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   240→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   241→    if (ucAddr) return ucAddr[1];
   242→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   243→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   244→    if (ucInst) return ucInst[1];
   245→    // Strategy 10: UC label with number — 4-13 digits
   246→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   247→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   248→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   249→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   250→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   251→    // Strategy 12: broad search for 3-group dotted format
   252→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   253→    if (broad3) return broad3[1];
   254→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   255→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   256→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   257→    return EMPTY;
   258→  })();
   259→
   260→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   261→  const cleanUC = UC.replace(/[.\-]/g, '');
   262→
   263→  // ===== ENDEREÇO =====
   264→  const Endereco = (() => {
   265→    // Address line typically follows the city (AG: XX) line and starts with RUA, AV, AVE, LOT, ESTRADA, etc.
   266→    // Examples from real PDFs:
   267→    //   "RUA SALDANHA DA GAMA, 896 - 21.004.65.029401 LAD RE RA - 422 - 79370000"
   268→    //   "LOT RECANTO NUARA, S/N - 24.416.10.154000 PAT 01/REL01 2289 - 79190000"
   269→    //   "AVE MATO GROSSO DO SUL, SN - 12.029.07.090000 COX EAT 006 2243 - 79400000"
   270→    const STREET_RE = /^(RUA|AV[. ]|AVE |AVENIDA|ESTRADA|LOT |RODOVIA|TRAVESSA|ALAMEDA|PCA[. ]|PRACA)/i;
   271→
   272→    // Clean an address line: keep street + number, strip internal codes, keep CEP
   273→    const extractCleanAddress = (line: string): string => {
   274→      // Split at the first " - " that precedes an internal code pattern (XX.XXX.XX.XXXXXX)
   275→      const beforeCodes = line.split(/\s*-\s*\d{2}\.\d{3}/)[0];
   276→      let addr = (beforeCodes || line).trim();
   277→      // Remove trailing dash
   278→      addr = addr.replace(/\s*-\s*$/, '').trim();
   279→      // Append CEP if found at end of original line
   280→      const cepMatch = line.match(/(\d{5}-?\d{3})\s*$/);
   281→      if (cepMatch) addr += ' - CEP ' + cepMatch[1];
   282→      return addr;
   283→    };
   284→
   285→    // Strategy 1: Find address line after city (AG: XX) pattern
   286→    for (let i = 0; i < Math.min(60, lines.length); i++) {
   287→      if (/\(\s*AG\s*:\s*\d+\s*\)/.test(lines[i])) {
   288→        // Check next 3 lines for an address pattern
   289→        for (let j = i + 1; j <= Math.min(i + 3, lines.length - 1); j++) {
   290→          if (STREET_RE.test(lines[j])) {
   291→            return extractCleanAddress(lines[j]);
   292→          }
   293→        }
   294→      }
   295→    }
   296→
   297→    // Strategy 2: Find address line near CNPJ/CPF line (appears before it)
   298→    for (let i = 0; i < lines.length; i++) {
   299→      if (/CNPJ\/CPF/i.test(lines[i]) && i > 0) {
   300→        if (STREET_RE.test(lines[i - 1])) {
   301→          return extractCleanAddress(lines[i - 1]);
   302→        }
   303→      }
   304→    }
   305→
   306→    // Strategy 3: Look for "ENDEREÇO" label followed by address
   307→    for (let i = 0; i < lines.length; i++) {
   308→      if (/^ENDERE[ÇC]O$/i.test(lines[i].trim()) && i + 1 < lines.length) {
   309→        const next = lines[i + 1];
   310→        const cleanAddr = next.replace(/\s*-\s*(SA[IÍ]DA|ENTRADA)\s+PARA\s+[^-]+/i, '')
   311→          .replace(/\s*-\s*[A-ZÀ-Ú]+\s*\/\s*[A-Z]{2}\s*-\s*CEP\s+\d{5}-\d{3}/i, '').trim();
   312→        if (cleanAddr.length > 5) return cleanAddr;
   313→      }
   314→    }
   315→
   316→    return EMPTY;
   317→  })();
   318→
   319→  // ===== MÊS REFERÊNCIA =====
   320→  const MesReferencia = (() => {
   321→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   322→    const numToMonth: Record<string, string> = {
   323→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   324→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   325→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   326→    };
   327→    const capitalize = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
   328→
   329→    // Helper: get Vencimento date components for sanity checks
   330→    const venMoneyMatch = flat.match(/(\d{2})\/(\d{2})\/(\d{4})\s*R\$/);
   331→    const venMonth = venMoneyMatch ? parseInt(venMoneyMatch[2], 10) : 0;
   332→    const venYear = venMoneyMatch ? parseInt(venMoneyMatch[3], 10) : 0;
   333→
   334→    // Helper: check if a month/year is consistent with the Vencimento date
   335→    // (reference month is typically 1-2 months before the vencimento month)
   336→    const isConsistentWithVencimento = (month: number, year: number): boolean => {
   337→      if (!venMonth || !venYear) return true; // no vencimento to compare
   338→      const venTotal = venYear * 12 + venMonth;
   339→      const refTotal = year * 12 + month;
   340→      const diff = venTotal - refTotal;
   341→      return diff >= 1 && diff <= 3; // reference is 1-3 months before vencimento
   342→    };
   343→
   344→    // Strategy 1a: "Mês / Ano" pattern BEFORE date + R$ on the SAME LINE (most reliable)
   345→    // Energisa format: "Fevereiro / 202621/03/2026R$ 538,49" or "Junho / 2026  21/07/2026  R$ 13.356,50"
   346→    for (const line of lines) {
   347→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})\\s*\\d{2}/\\d{2}/\\d{4}\\s*R\\$`, 'i'));
   348→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   349→    }
   350→    // Strategy 1b: date + R$ pattern BEFORE "Mês / Ano" on the SAME LINE
   351→    // pdf-parse reordering: "21/07/2026R$ 6.980,48Junho / 2026"
   352→    for (const line of lines) {
   353→      const m = line.match(new RegExp(`\\d{2}/\\d{2}/\\d{4}\\s*R\\$\\s*[\\d.,]+\\s*(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   354→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   355→    }
   356→    // Strategy 2: "Mês / Ano" pattern on the SAME LINE as VENCIMENTO or NOTA FISCAL
   357→    for (const line of lines) {
   358→      if (/VENCIMENTO|NOTA FISCAL|Nº FATURA/i.test(line)) {
   359→        const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   360→        if (m) return `${capitalize(m[1])} / ${m[2]}`;
   361→      }
   362→    }
   363→    // Strategy 3: "Mês / Ano" on a line that also contains a date DD/MM/YYYY (likely the billing line)
   364→    for (const line of lines) {
   365→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   366→      if (m && /\d{2}\/\d{2}\/\d{4}/.test(line)) {
   367→        const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   368→        const monthNum = monthIdx + 1;
   369→        if (isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) {
   370→          return `${capitalize(m[1])} / ${m[2]}`;
   371→        }
   372→      }
   373→    }
   374→    // Strategy 4: "Mês / Ano" anywhere in flat text — skip "Desde mês/ano", "Referência:" history, etc.
   375→    const allNameMatches = [...flat.matchAll(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'gi'))];
   376→    for (const m of allNameMatches) {
   377→      // Check context: skip if preceded by "Desde", "Referência:", "Ref", or other informative words
   378→      const start = Math.max(0, m.index! - 20);
   379→      const context = flat.slice(start, m.index! + m[0].length).toLowerCase();
   380→      if (/desde|até|a partir|inform|dispon|referência|ref\b/.test(context)) continue;
   381→      // Also validate against Vencimento date
   382→      const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   383→      const monthNum = monthIdx + 1;
   384→      if (!isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) continue;
   385→      return `${capitalize(m[1])} / ${m[2]}`;
   386→    }
   387→    // Strategy 5: Use Leitura Atual date to derive the reference month
   388→    // pdf-parse sometimes garbles text, so try multiple patterns
   389→    const leituraPatterns = [
   390→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*(\d{2})\s*\/\s*(\d{4})/i,
   391→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*(\d{2})\s*\/\s*(\d{4})/i,
   392→    ];
   393→    for (const pat of leituraPatterns) {
   394→      const lm = flat.match(pat);
   395→      if (lm && numToMonth[lm[1]]) {
   396→        const monthNum = parseInt(lm[1], 10);
   397→        const year = parseInt(lm[2], 10);
   398→        if (isConsistentWithVencimento(monthNum, year)) {
   399→          return `${numToMonth[lm[1]]} / ${lm[2]}`;
   400→        }
   401→      }
   402→    }
   403→    // Strategy 6: numeric MM/YYYY near Vencimento date — with consistency check
   404→    if (venMoneyMatch) {
   405→      const venIdx = flat.indexOf(venMoneyMatch[0]);
   406→      const region = flat.slice(Math.max(0, venIdx - 150), venIdx + 50);
   407→      const numMatches = [...region.matchAll(/(\d{2})\s*\/\s*(\d{4})/g)];
   408→      for (const nm of numMatches) {
   409→        if (numToMonth[nm[1]]) {
   410→          const monthNum = parseInt(nm[1], 10);
   411→          const year = parseInt(nm[2], 10);
   412→          if (isConsistentWithVencimento(monthNum, year)) {
   413→            return `${numToMonth[nm[1]]} / ${nm[2]}`;
   414→          }
   415→        }
   416→      }
   417→    }
   418→    // Strategy 7: first valid MM/YYYY that is a month and consistent with Vencimento
   419→    const allNumMatches = [...flat.matchAll(/(\d{2})\s*[\/]\s*(\d{4})/g)];
   420→    for (const nm of allNumMatches) {
   421→      if (numToMonth[nm[1]]) {
   422→        const monthNum = parseInt(nm[1], 10);
   423→        const year = parseInt(nm[2], 10);
   424→        if (isConsistentWithVencimento(monthNum, year)) {
   425→          return `${numToMonth[nm[1]]} / ${nm[2]}`;
   426→        }
   427→      }
   428→    }
   429→    return EMPTY;
   430→  })();
   431→
   432→  // ===== ANO LEITURA =====
   433→  const AnoLeitura = (() => {
   434→    // Try multiple patterns for Leitura Atual date (pdf-parse can garble text)
   435→    const leituraYearPatterns = [
   436→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i,
   437→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*\d{2}\s*\/\s*(\d{4})/i,
   438→    ];
   439→    for (const pat of leituraYearPatterns) {
   440→      const m = flat.match(pat);
   441→      if (m) return m[1];
   442→    }
   443→    // Fallback: extract year from MesReferencia
   444→    if (MesReferencia) {
   445→      const ym = MesReferencia.match(/(\d{4})/);
   446→      if (ym) return ym[1];
   447→    }
   448→    return EMPTY;
   449→  })();
   450→
   451→  // ===== VENCIMENTO =====
   452→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   453→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   454→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   455→  const Vencimento = (() => {
   456→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   457→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   458→    if (venMoney) return venMoney[1];
   459→
   460→    // Strategy 2: R$ + money + date on same line (reversed order)
   461→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   462→    if (moneyVen) return moneyVen[1];
   463→
   464→    // Strategy 3: VENCIMENTO label nearby (original approach)
   465→    const venIdx = findLine(lines, /VENCIMENTO/i);
   466→    if (venIdx >= 0) {
   467→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   468→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   469→        if (m) return m[1];
   470→      }
   471→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   472→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   473→        if (m) return m[1];
   474→      }
   475→    }
   476→
   477→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   478→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   479→    if (apresMatch) return apresMatch[1];
   480→
   481→    return EMPTY;
   482→  })();
   483→
   484→  // ===== NOTA FISCAL =====
   485→  const NotaFiscal = (() => {
   486→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   487→    return m ? m[1] : EMPTY;
   488→  })();
   489→
   490→  // ===== MODALIDADE TARIFÁRIA =====
   491→  const ModalidadeTarifaria = (() => {
   492→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   493→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   494→    if (/AZUL/i.test(flat)) return 'AZUL';
   495→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   496→    if (/VERDE/i.test(flat)) return 'VERDE';
   497→    return EMPTY;
   498→  })();
   499→
   500→  // ===== SUBGRUPO =====
   501→  const Subgrupo = (() => {
   502→    // Look for "/A4" or "/B3" etc. in the classification line
   503→    // Must be a letter + single digit, right after "/"
   504→    const m = flat.match(/\/\s*([AB]\d)\s/);
   505→    if (m) return m[1].toUpperCase();
   506→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   507→    if (classIdx >= 0) {
   508→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   509→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   510→      if (m2) return m2[1].toUpperCase();
   511→    }
   512→    return EMPTY;
   513→  })();
   514→
   515→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   516→
   517→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   518→  let ConsumoPonta = EMPTY;
   519→  let ConsumoIntermediaria = EMPTY;
   520→  let ConsumoForaPonta = EMPTY;
   521→  let ConsumoTotal = EMPTY;
   522→  let ConsumoGrupoB_kWh = EMPTY;
   523→  let ConsumoGrupoB_Valor = EMPTY;
   524→  let ValorConsumoPonta = EMPTY;
   525→  let ValorConsumoIntermediaria = EMPTY;
   526→  let ValorConsumoForaPonta = EMPTY;
   527→  // Accumulators for Grupo B consumption
   528→  let grupoB_consumoKWh_total = 0;
   529→  let grupoB_consumoValor_total = 0;
   530→
   531→  if (isSplitTariff) {
   532→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   533→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   534→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   535→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   536→    // Intermediária (Tarifa Branca / 3-period AZUL)
   537→    const interIdx = findLine(lines, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   538→    if (interIdx >= 0) ConsumoIntermediaria = findNumNearLine(lines, interIdx, 1, 8, false);
   539→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)(?!.*Intermedi)/i);
   540→    ValorConsumoIntermediaria = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   541→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   542→  } else {
   543→    for (const line of lines) {
   544→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   545→      if (km) { ConsumoTotal = km[1]; break; }
   546→    }
   547→    if (!ConsumoTotal) {
   548→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   549→      if (consIdx >= 0) {
   550→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   551→          const line = lines[i];
   552→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   553→          if (/[A-Za-z]{4,}/.test(line)) continue;
   554→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   555→          if (m) { ConsumoTotal = m[1]; break; }
   556→        }
   557→      }
   558→    }
   559→    // ELEKTRO/SANESUL pattern: "CONSUMO TE" / "CONSUMO TUSD" labels followed by kWh units and consumption value
   560→    // e.g., Line 22: "CONSUMO TE", Line 23: "CONSUMO TUSD", Line 25: "kWh", Line 28: "4.963"
   561→    if (!ConsumoTotal) {
   562→      for (let i = 0; i < lines.length - 3; i++) {
   563→        if (/^CONSUMO\s+(TE|TUSD)$/i.test(lines[i])) {
   564→          // Found CONSUMO TE/TUSD label — look for the first numeric value after kWh units
   565→          const afterLabels = lines.slice(i + 1, Math.min(i + 15, lines.length));
   566→          // Find the first numeric value that looks like a consumption number (e.g., "4.963" or "4963")
   567→          for (const al of afterLabels) {
   568→            // Skip unit labels like "kWh", "kW"
   569→            if (/^(kWh|kW)$/i.test(al)) continue;
   570→            // Match a number like "4.963" (with dot as thousands separator) or "4963"
   571→            const m = al.match(/^(\d{1,2}\.\d{3}|\d{3,6})$/);
   572→            if (m) {
   573→              ConsumoTotal = m[1];
   574→              break;
   575→            }
   576→          }
   577→          if (ConsumoTotal) break;
   578→        }
   579→      }
   580→    }
   581→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   582→  }
   583→
   584→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   585→  let DemandaContratadaPonta = EMPTY;
   586→  let DemandaContratadaForaPonta = EMPTY;
   587→  let DemandaMedidaPonta = EMPTY;
   588→  let DemandaMedidaForaPonta = EMPTY;
   589→  let DemandaFaturadaPonta = EMPTY;
   590→  let DemandaFaturadaForaPonta = EMPTY;
   591→  let ValorDemandaPonta = EMPTY;
   592→  let ValorDemandaForaPonta = EMPTY;
   593→
   594→  if (isSplitTariff) {
   595→    // ===== DEMANDA CONTRATADA EXTRACTION =====
   596→    // Energisa PDFs show contracted demand in two locations:
   597→    // 1) "Grandezas Contratadas" section: "Demanda ponta - kW  15,3" / "Demanda fora ponta - kW  30"
   598→    // 2) Bottom line: "KW Ponta: 15,3  Fora Ponta: 30  TUSDG:"
   599→    //
   600→    // pdf-parse often garbles the column order, producing:
   601→    //   "Demanda ponta - kW Demanda fora ponta - kW 30 15,3"  (values reversed)
   602→    //   "30 Fora Ponta: 15,3 KW Ponta:"  (values before labels, or between labels)
   603→    //
   604→    // We try multiple strategies and use the first one that gives both values.
   605→
   606→    // Strategy 1: "Demanda ponta - kW <value>" on the SAME LINE (pdftotext-like format)
   607→    const dpSameLine = flat.match(/Demanda\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   608→    const dfpSameLine = flat.match(/Demanda\s+fora\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   609→    if (dpSameLine && dfpSameLine) {
   610→      DemandaContratadaPonta = dpSameLine[1].replace('.', ',');
   611→      DemandaContratadaForaPonta = dfpSameLine[1].replace('.', ',');
   612→    }
   613→
   614→    // Strategy 2: Detect pdf-parse column reordering in bottom line
   615→    // When "Fora Ponta:" appears BEFORE "KW Ponta:" in the text, the values are
   616→    // rearranged: "<fora_val> Fora Ponta: <ponta_val> KW Ponta:"
   617→    // - Number BEFORE "Fora Ponta:" = Fora Ponta value
   618→    // - Number BETWEEN "Fora Ponta:" and "KW Ponta:" = Ponta value
   619→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   620→      const foraIdx = flat.search(/Fora\s+Ponta\s*:/i);
   621→      const kwIdx = flat.search(/KW\s+Ponta\s*:/i);
   622→      if (foraIdx >= 0 && kwIdx >= 0 && foraIdx < kwIdx) {
   623→        // "Fora Ponta:" appears before "KW Ponta:" — this is the reordered pattern
   624→        // Number between "Fora Ponta:" and "KW Ponta:" → Ponta value
   625→        const between = flat.slice(foraIdx, kwIdx);
   626→        const betweenMatch = between.match(/:\s*(\d+(?:[.,]\d+)?)/);
   627→        if (betweenMatch && !DemandaContratadaPonta) {
   628→          DemandaContratadaPonta = betweenMatch[1].replace('.', ',');
   629→        }
   630→        // Number before "Fora Ponta:" (within 25 chars) → Fora Ponta value
   631→        const before = flat.slice(Math.max(0, foraIdx - 25), foraIdx);
   632→        const beforeMatch = before.match(/(\d+(?:[.,]\d+)?)\s*$/);
   633→        if (beforeMatch && !DemandaContratadaForaPonta) {
   634→          DemandaContratadaForaPonta = beforeMatch[1].replace('.', ',');
   635→        }
   636→      } else if (kwIdx >= 0 && foraIdx >= 0 && kwIdx < foraIdx) {
   637→        // Normal order: "KW Ponta:" appears before "Fora Ponta:" — values follow labels
   638→        const kwAfter = flat.slice(kwIdx).match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   639→        const fpAfter = flat.slice(foraIdx).match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   640→        if (kwAfter && !DemandaContratadaPonta) DemandaContratadaPonta = kwAfter[1].replace('.', ',');
   641→        if (fpAfter && !DemandaContratadaForaPonta) DemandaContratadaForaPonta = fpAfter[1].replace('.', ',');
   642→      }
   643→    }
   644→
   645→    // Strategy 3: "Demanda ponta/fora ponta - kW" labels with values on nearby lines
   646→    // (pdf-parse puts labels and values on separate lines)
   647→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   648→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
   649→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
   650→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
   651→        // Collect numeric values from the lines after both labels
   652→        const startLine = Math.max(dpLineIdx, dfpLineIdx) + 1;
   653→        const numVal = /^(\d+(?:[.,]\d+)?)$/;
   654→        const values: string[] = [];
   655→        for (let i = startLine; i < Math.min(startLine + 5, lines.length); i++) {
   656→          const m = lines[i].match(numVal);
   657→          if (m) values.push(m[1].replace('.', ','));
   658→          if (values.length >= 2) break;
   659→        }
   660→        if (values.length === 2) {
   661→          // Use already-found values to disambiguate, or assume reverse order
   662→          if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
   663→            if (values[0] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[1];
   664→            else if (values[1] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[0];
   665→          } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
   666→            if (values[0] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[1];
   667→            else if (values[1] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[0];
   668→          } else {
   669→            // No disambiguation; assume pdf-parse reverse column order:
   670→            // values[0] = Fora Ponta, values[1] = Ponta
   671→            if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = values[0];
   672→            if (!DemandaContratadaPonta) DemandaContratadaPonta = values[1];
   673→          }
   674→        }
   675→      }
   676→    }
   677→
   678→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   679→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   680→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   681→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   682→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   683→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   684→  }
   685→
   686→  // ===== ULTRAPASSAGEM =====
   687→  let UltrapassagemPonta = EMPTY;
   688→  let UltrapassagemForaPonta = EMPTY;
   689→  let ValorUltrapassagemPonta = EMPTY;
   690→  let ValorUltrapassagemForaPonta = EMPTY;
   691→  // ===== DEMANDA NÃO CONSUMIDA =====
   692→  let DemandaNaoConsumidaPonta = EMPTY;
   693→  let DemandaNaoConsumidaForaPonta = EMPTY;
   694→  let ValorNaoConsumidaPonta = EMPTY;
   695→  let ValorNaoConsumidaForaPonta = EMPTY;
   696→  if (isSplitTariff) {
   697→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   698→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   699→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   700→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   701→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   702→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   703→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   704→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   705→  }
   706→
   707→  // ===== REATIVA EXCEDENTE =====
   708→  let ReativaExcedentePonta = EMPTY;
   709→  let ReativaExcedenteForaPonta = EMPTY;
   710→  let EnergiaReativaTotal = EMPTY;
   711→  let ValorReativaExcedentePonta = EMPTY;
   712→  let ValorReativaExcedenteForaPonta = EMPTY;
   713→  if (isSplitTariff) {
   714→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   715→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   716→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   717→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   718→  } else {
   719→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   720→  }
   721→
   722→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   723→  // Priority: table extraction > text-based extraction
   724→  let PIS = EMPTY;
   725→  let COFINS = EMPTY;
   726→  let ICMS = EMPTY;
   727→
   728→  // Use tributos from table extraction if available (most reliable)
   729→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   730→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   731→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   732→
   733→  // Fallback: ELEKTRO/SANESUL pattern — ICMS, COFINS, PIS on consecutive lines,
   734→  // followed by base values, aliquotas (%), then actual tributo values.
   735→  // Example:
   736→  //   L52: ICMS
   737→  //   L53: COFINS
   738→  //   L54: PIS
   739→  //   L55: 5087,88   ← Base de cálculo ICMS
   740→  //   L56: 4222,95   ← Base de cálculo COFINS
   741→  //   L57: 4222,95   ← Base de cálculo PIS
   742→  //   L58: 17%       ← Alíquota ICMS
   743→  //   L59: 3,48%     ← Alíquota COFINS
   744→  //   L60: 0,75%     ← Alíquota PIS
   745→  //   L61: 864,93    ← Valor ICMS ✓
   746→  //   L62: 146,96    ← Valor COFINS ✓
   747→  //   L63: 31,67     ← Valor PIS ✓
   748→  if (!PIS || !COFINS || !ICMS) {
   749→    for (let i = 0; i < lines.length - 2; i++) {
   750→      if (/^ICMS$/i.test(lines[i]) && /^COFINS$/i.test(lines[i + 1]) && /^PIS$/i.test(lines[i + 2])) {
   751→        // Found the ICMS/COFINS/PIS label block
   752→        // Look for 3 money values after the labels (base values), then skip aliquotas, then get 3 more money values (actual values)
   753→        const afterLabels = lines.slice(i + 3, Math.min(i + 20, lines.length));
   754→        const moneyValues: string[] = [];
   755→        for (const line of afterLabels) {
   756→          const m = line.match(/^(\d[\d.]*,\d{2})$/);
   757→          if (m) moneyValues.push(m[1]);
   758→          if (moneyValues.length >= 6) break;
   759→        }
   760→        // If we have at least 6 money values, the last 3 are the actual tributo values
   761→        // (first 3 = bases, then 3 = actual values)
   762→        if (moneyValues.length >= 6) {
   763→          if (!ICMS) ICMS = moneyValues[3]; // 4th value = ICMS valor
   764→          if (!COFINS) COFINS = moneyValues[4]; // 5th value = COFINS valor
   765→          if (!PIS) PIS = moneyValues[5]; // 6th value = PIS valor
   766→        } else if (moneyValues.length >= 3) {
   767→          // If only 3 values found, they might be the base values
   768→          // Check if there are percentage values between the base and actual values
   769→          const afterBases = afterLabels.slice(3);
   770→          const actualValues: string[] = [];
   771→          for (const line of afterBases) {
   772→            if (/^\d+([.,]\d+)?%$/.test(line)) continue; // skip aliquotas
   773→            const m = line.match(/^(\d[\d.]*,\d{2})$/);
   774→            if (m) actualValues.push(m[1]);
   775→            if (actualValues.length >= 3) break;
   776→          }
   777→          if (actualValues.length >= 3) {
   778→            if (!ICMS) ICMS = actualValues[0];
   779→            if (!COFINS) COFINS = actualValues[1];
   780→            if (!PIS) PIS = actualValues[2];
   781→          }
   782→        }
   783→        break; // Only process the first match
   784→      }
   785→    }
   786→  }
   787→
   788→  // Fallback: ENERGISA DANF3E pattern — "PIS / ICMS / COFINS" on consecutive lines,
   789→  // with PIS and COFINS total values above, and ICMS total embedded in the line before.
   790→  // Example from 1069.PDF:
   791→  //   L124: 18,47        ← PIS total ✓
   792→  //   L125: 85,06        ← COFINS total ✓
   793→  //   L126: 0,6757       ← (alíquota PIS)
   794→  //   L127: 3,1122       ← (alíquota COFINS)
   795→  //   L128: 17,00        ← (alíquota ICMS)
   796→  //   L129: 2.733,23     ← (base ICMS)
   797→  //   L130: 2.733,23     ← (base ICMS)
   798→  //   L131: 3.293,07559,82 ← "baseICMS,ICMS_total" concatenated
   799→  //   L132: PIS
   800→  //   L133: ICMS
   801→  //   L134: COFINS
   802→  if (!PIS || !COFINS || !ICMS) {
   803→    for (let i = 0; i < lines.length - 2; i++) {
   804→      if (/^PIS$/i.test(lines[i]) && /^ICMS$/i.test(lines[i + 1]) && /^COFINS$/i.test(lines[i + 2])) {
   805→        // Look backwards from the block. The PIS/COFINS values appear BEFORE the aliquotas.
   806→        // Pattern: <pis_val> <cofins_val> <aliq_pis> <aliq_cofins> <aliq_icms> <base_icms> <base_icms> <concat_base_icms,icms_total>
   807→        // The aliquotas look like "0,6757" or "3,1122" or "17,00"
   808→        // Find the first aliquota-looking value going backwards, then PIS/COFINS are the 2 money values before it
   809→        let firstAliquotIdx = -1;
   810→        for (let j = i - 1; j >= Math.max(0, i - 12); j--) {
   811→          if (/^\d+,\d{2,4}$/.test(lines[j]) && !/^\d+,\d{2}$/.test(lines[j])) {
   812→            // This looks like an aliquota (3+ decimal places) but NOT a money value (2 decimal places)
   813→            firstAliquotIdx = j;
   814→            break;
   815→          }
   816→        }
   817→        if (firstAliquotIdx >= 0) {
   818→          // PIS and COFINS are the 2 money values before the first aliquota
   819→          const pisCofinsValues: string[] = [];
   820→          for (let j = firstAliquotIdx - 1; j >= Math.max(0, firstAliquotIdx - 5); j--) {
   821→            const m = lines[j].match(/^(\d[\d.]*,\d{2})$/);
   822→            if (m) {
   823→              pisCofinsValues.unshift(m[1]);
   824→              if (pisCofinsValues.length >= 2) break;
   825→            }
   826→          }
   827→          if (pisCofinsValues.length >= 2) {
   828→            if (!PIS) PIS = pisCofinsValues[0];
   829→            if (!COFINS) COFINS = pisCofinsValues[1];
   830→          }
   831→        }
   832→        // ICMS total is in the concatenated line just before the block (e.g., "3.293,07559,82")
   833→        // The ICMS total is the LAST number in this concatenated string
   834→        if (!ICMS) {
   835→          for (let j = i - 1; j >= Math.max(0, i - 4); j--) {
   836→            // Look for a line with two concatenated money values (no space between them)
   837→            const catMatch = lines[j].match(/(\d[\d.]*,\d{2})(\d[\d.]*,\d{2})$/);
   838→            if (catMatch) {
   839→              ICMS = catMatch[2];
   840→              break;
   841→            }
   842→          }
   843→        }
   844→        break; // Only process the first match
   845→      }
   846→    }
   847→  }
   848→
   849→  // Fallback: text-based extraction for any missing tributo
   850→  if (!PIS || !COFINS || !ICMS) {
   851→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   852→    if (pisIdx >= 0) {
   853→      const nums: string[] = [];
   854→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   855→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   856→        if (lineNums) {
   857→          for (const n of lineNums) {
   858→            nums.push(n);
   859→            if (nums.length >= 3) break;
   860→          }
   861→        }
   862→      }
   863→      if (nums.length >= 3) {
   864→        if (!PIS) PIS = nums[0];
   865→        if (!COFINS) COFINS = nums[1];
   866→        if (!ICMS) ICMS = nums[2];
   867→      }
   868→    }
   869→    if (!PIS && !COFINS) {
   870→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   871→      if (combinedIdx >= 0) {
   872→        const nums: string[] = [];
   873→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   874→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   875→          if (m) nums.push(m[1]);
   876→        }
   877→        if (nums.length >= 3) {
   878→          if (!PIS) PIS = nums[0];
   879→          if (!COFINS) COFINS = nums[1];
   880→          if (!ICMS) ICMS = nums[2];
   881→        }
   882→      }
   883→    }
   884→    if (!PIS || !COFINS) {
   885→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   886→      if (flatMatch) {
   887→        if (!PIS) PIS = flatMatch[1];
   888→        if (!COFINS) COFINS = flatMatch[2];
   889→      }
   890→    }
   891→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   892→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   893→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
   894→  }
   895→
   896→  // ===== GDI =====
   897→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   898→
   899→  // ===== CIP (text-based) =====
   900→  const CIP_text = (() => {
   901→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   902→    if (cipIdx >= 0) {
   903→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   904→      const money = findAllMoney(region);
   905→      if (money.length > 0) return money[money.length - 1];
   906→    }
   907→    return findTributoValue(lines, flat, /\bCIP\b/i);
   908→  })();
   909→
   910→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   911→  let {
   912→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   913→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   914→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   915→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   916→  } = extractGDI(lines, flat);
   917→
   918→  // ===== SALDO ACUMULADO GD =====
   919→  const SaldoAcumuladoGD = (() => {
   920→    const m = flat.match(/Saldo\s*Acumulado\s*:\s*([\d.]+,\d{2})/i);
   921→    if (m) return m[1];
   922→    // Fallback: integer format like "14.100"
   923→    const m2 = flat.match(/Saldo\s*Acumulado\s*:\s*(\d[\d.]*)/i);
   924→    if (m2) return m2[1];
   925→    return EMPTY;
   926→  })();
   927→
   928→  // ===== VALOR TOTAL =====
   929→  // Best pattern: R$ value on same line as vencimento date + mes referência
   930→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   931→  const ValorTotal = (() => {
   932→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
   933→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
   934→    if (venTotal) return venTotal[1];
   935→
   936→    // Strategy 2: R$ + value + mes referência on same line
   937→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
   938→    if (totalMes) return totalMes[1];
   939→
   940→    // Strategy 3: "TOTAL A PAGAR" with R$
   941→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   942→    if (totalIdx >= 0) {
   943→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   944→      if (m1) return m1[1];
   945→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   946→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   947→        if (m2) return m2[1];
   948→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   949→        if (m3) return m3[1];
   950→      }
   951→    }
   952→
   953→    // Strategy 4: Last resort - largest R$ value
   954→    const lastMoney = findAllMoney(flat);
   955→    if (lastMoney.length > 0) {
   956→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   957→      return lastMoney[0];
   958→    }
   959→    return EMPTY;
   960→  })();
   961→
   962→  // ===== TABLE DATA OVERRIDES =====
   963→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
   964→  let CIP = CIP_text;
   965→
   966→  if (tableData && tableData.length > 0) {
   967→    let cipFromTable = EMPTY;
   968→    // Reset GDI accumulators for clean table-based extraction
   969→    let tbl_oUC_Qtd = '' as string;
   970→    let tbl_oUC_Valor = '' as string;
   971→    let tbl_mUC_Qtd = '' as string;
   972→    let tbl_mUC_Valor = '' as string;
   973→    // Track GDI section in table for rows with truncated/missing labels
   974→    let inGDISection = false;
   975→    let gdiSectionType = '' as string; // 'oUC' or 'mUC'
   976→    let lastGDILabel = '' as string;
   977→
   978→    for (const item of tableData) {
   979→      const lbl = item.label;
   980→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
   981→      const isPonta = /Ponta/i.test(lbl) && !isFora;
   982→
   983→      // Consumo - check Convencional (no Ponta/Fora) first, then split
   984→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
   985→        // Accumulate for Grupo B (regardless of Ponta/Fora/Intermediária)
   986→        if (item.quantidade) {
   987→          const qtd = parseBR(item.quantidade);
   988→          if (qtd > 0) grupoB_consumoKWh_total += qtd;
   989→        }
   990→        if (item.valor) {
   991→          const val = parseBR(item.valor);
   992→          if (val > 0) grupoB_consumoValor_total += val;
   993→        }
   994→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
   995→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
   996→          if (item.quantidade) ConsumoTotal = item.quantidade;
   997→          if (item.valor) ValorConsumoForaPonta = item.valor;
   998→        }
   999→        // BRANCA/AZUL Intermediária — dedicated field
  1000→        else if (/Intermedi[áa]ria/i.test(lbl)) {
  1001→          if (item.quantidade) ConsumoIntermediaria = item.quantidade;
  1002→          if (item.valor) ValorConsumoIntermediaria = item.valor;
  1003→        }
  1004→        // Split tariff (AZUL/VERDE)
  1005→        else if (isPonta) {
  1006→          if (item.quantidade) ConsumoPonta = item.quantidade;
  1007→          if (item.valor) ValorConsumoPonta = item.valor;
  1008→        } else if (isFora) {
  1009→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
  1010→          if (item.valor) ValorConsumoForaPonta = item.valor;
  1011→        }
  1012→      }
  1013→      // Demanda Medida
  1014→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
  1015→        if (!isFora) {
  1016→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
  1017→          if (item.valor) ValorDemandaPonta = item.valor;
  1018→        } else {
  1019→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
  1020→          if (item.valor) ValorDemandaForaPonta = item.valor;
  1021→        }
  1022→      }
  1023→      // Ultrapassagem
  1024→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
  1025→        if (!isFora) {
  1026→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
  1027→          if (item.valor) ValorUltrapassagemPonta = item.valor;
  1028→        } else {
  1029→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
  1030→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
  1031→        }
  1032→      }
  1033→      // Demanda Potência Não Consumida
  1034→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
  1035→        if (!isFora) {
  1036→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
  1037→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
  1038→        } else {
  1039→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
  1040→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
  1041→        }
  1042→      }
  1043→      // Reativa Excedente
  1044→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
  1045→        if (!isFora && /Ponta/i.test(lbl)) {
  1046→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
  1047→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1048→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
  1049→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
  1050→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
  1051→        } else if (!isFora && !/Ponta/i.test(lbl)) {
  1052→          // Generic reativa (Convencional)
  1053→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
  1054→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1055→        }
  1056→      }
  1057→      // CIP Municipal
  1058→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
  1059→        if (item.valor) cipFromTable = item.valor;
  1060→      }
  1061→      // GDI items
  1062→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
  1063→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
  1064→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
  1065→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
  1066→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
  1067→        // Multiple mUC entries: accumulate
  1068→        if (item.quantidade) {
  1069→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1070→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1071→            : item.quantidade;
  1072→        }
  1073→        if (item.valor) {
  1074→          tbl_mUC_Valor = tbl_mUC_Valor
  1075→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1076→            : item.valor;
  1077→        }
  1078→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
  1079→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
  1080→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
  1081→        // Accumulate as the same GDI type as the last full-label row
  1082→        if (gdiSectionType === 'mUC') {
  1083→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1084→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1085→            : item.quantidade;
  1086→          if (item.valor) {
  1087→            tbl_mUC_Valor = tbl_mUC_Valor
  1088→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1089→              : item.valor;
  1090→          }
  1091→        } else if (gdiSectionType === 'oUC') {
  1092→          tbl_oUC_Qtd = tbl_oUC_Qtd
  1093→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
  1094→            : item.quantidade;
  1095→          if (item.valor) {
  1096→            tbl_oUC_Valor = tbl_oUC_Valor
  1097→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
  1098→              : item.valor;
  1099→          }
  1100→        }
  1101→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
  1102→        // Non-GDI, non-truncated row ends the GDI section
  1103→        inGDISection = false;
  1104→      }
  1105→    }
  1106→
  1107→    // Override CIP from table
  1108→    if (cipFromTable) CIP = cipFromTable;
  1109→
  1110→    // Use table GDI values if found, otherwise keep text-based fallback
  1111→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
  1112→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
  1113→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
  1114→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
  1115→  }
  1116→
  1117→  // ===== Grupo B: calculate dedicated fields =====
  1118→  // For Subgrupo B (B1, B2, B3, B4), fill ConsumoGrupoB_kWh and ConsumoGrupoB_Valor
  1119→  const isGrupoB = /^B/i.test(Subgrupo);
  1120→  if (isGrupoB) {
  1121→    // ConsumoGrupoB_kWh: total consumption in kWh
  1122→    if (grupoB_consumoKWh_total > 0) {
  1123→      ConsumoGrupoB_kWh = formatBR(grupoB_consumoKWh_total);
  1124→    } else if (ConsumoTotal) {
  1125→      // Fallback: use ConsumoTotal if no table accumulation
  1126→      ConsumoGrupoB_kWh = ConsumoTotal;
  1127→    } else {
  1128→      // Fallback: sum ConsumoPonta + ConsumoForaPonta
  1129→      const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1130→      const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1131→      if (p + f > 0) ConsumoGrupoB_kWh = formatBR(p + f);
  1132→    }
  1133→    // ConsumoGrupoB_Valor: total consumption value in R$
  1134→    if (grupoB_consumoValor_total > 0) {
  1135→      ConsumoGrupoB_Valor = formatBR(grupoB_consumoValor_total);
  1136→    } else {
  1137→      // Fallback: sum ValorConsumoPonta + ValorConsumoForaPonta
  1138→      const vp = ValorConsumoPonta ? parseBR(ValorConsumoPonta) : 0;
  1139→      const vf = ValorConsumoForaPonta ? parseBR(ValorConsumoForaPonta) : 0;
  1140→      if (vp + vf > 0) ConsumoGrupoB_Valor = formatBR(vp + vf);
  1141→    }
  1142→  }
  1143→
  1144→  return {
  1145→    UC: cleanUC, Cidade, Endereco, Tipo, Mercado, Concessionaria,
  1146→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
  1147→    ModalidadeTarifaria, Subgrupo,
  1148→    ConsumoPonta, ConsumoIntermediaria, ConsumoForaPonta, ConsumoTotal, ConsumoGrupoB_kWh, ConsumoGrupoB_Valor,
  1149→    DemandaContratadaPonta, DemandaContratadaForaPonta,

  1169→// ===================== GDI EXTRACTION =====================
  1170→
  1171→function extractGDI(lines: string[], flat: string) {
  1172→  const EMPTY = '';
  1173→  let nOUC = 0;
  1174→  let nMUC = 0;
  1175→  let firstGDI = -1;
  1176→
  1177→  for (let i = 0; i < lines.length; i++) {
  1178→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
  1179→      nOUC++;
  1180→      if (firstGDI < 0) firstGDI = i;
  1181→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
  1182→      nMUC++;
  1183→      if (firstGDI < 0) firstGDI = i;
  1184→    }
  1185→  }
  1186→
  1187→  if (nOUC === 0 && nMUC === 0) {
  1188→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
  1189→  }
  1190→
  1191→  const chaveIdx = (() => {
  1192→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
  1193→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
  1194→    }
  1195→    return -1;
  1196→  })();
  1197→
  1198→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
  1199→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
  1200→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
  1201→      sectionEnd = i; break;
  1202→    }
  1203→  }
  1204→  if (chaveIdx >= 0) {
  1205→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
  1206→  }
  1207→
  1208→  const allQtds: number[] = [];
  1209→  if (chaveIdx >= 0) {
  1210→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
  1211→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1212→      if (m) allQtds.push(parseBR(m[1]));
  1213→    }
  1214→  }
  1215→  if (allQtds.length === 0) {
  1216→    for (let i = firstGDI; i < sectionEnd; i++) {
  1217→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1218→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
  1219→      if (allQtds.length >= nOUC + nMUC) break;
  1220→    }
  1221→  }
  1222→
  1223→  const allRVals: number[] = [];
  1224→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
  1225→  const totalNeeded = nOUC + nMUC;
  1226→
  1227→  for (let i = firstGDI; i < rSearchEnd; i++) {
  1228→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
  1229→    if (m && allRVals.length < totalNeeded) {
  1230→      allRVals.push(parseBR('-' + m[2]));
  1231→    }
  1232→  }
  1233→  if (allRVals.length < totalNeeded) {
  1234→    for (let i = firstGDI; i < rSearchEnd; i++) {
  1235→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
  1236→      if (m && allRVals.length < totalNeeded) {
  1237→        allRVals.push(parseBR(m[1]));
  1238→      }
  1239→    }
  1240→  }
  1241→
  1242→  const oUCQtds = allQtds.splice(0, nOUC);
  1243→  const oUCRVals = allRVals.splice(0, nOUC);
  1244→
  1245→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
  1246→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
  1247→
  1248→  return {
  1249→    oUC_Qtd: sumQtd(oUCQtds),
  1250→    oUC_Valor: sumR(oUCRVals),
  1251→    mUC_Qtd: sumQtd(allQtds),
  1252→    mUC_Valor: sumR(allRVals),
  1253→  };
  1254→}

  1256→// ===================== NUMBER FORMATTING =====================
  1257→
  1258→function fmtBRNumber(n: number): string {
  1259→  const abs = Math.abs(n);
  1260→  const intPart = Math.floor(abs).toString();
  1261→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
  1262→  let formatted = '';
  1263→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
  1264→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
  1265→    formatted = intPart[i] + formatted;
  1266→  }
  1267→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
  1268→}
  1269→
  1270→function fmtSumBR(a: string, b: string): string {
  1271→  return fmtBRNumber(parseBR(a) + parseBR(b));
  1272→}

  1276→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
  1277→  const idx = findLine(lines, labelPattern);
  1278→  if (idx < 0) return EMPTY;
  1279→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
  1280→    const line = lines[i];
  1281→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
  1282→    const m = line.match(/([\d.]+,\d{2})/);
  1283→    if (m) return m[1];
  1284→  }
  1285→  return EMPTY;
  1286→}

  1288→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
  1289→  const idx = findLine(lines, labelPattern);
  1290→  if (idx < 0) return EMPTY;
  1291→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
  1292→  const moneyVals = findAllMoney(region);
  1293→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
  1294→  return EMPTY;
  1295→}

  1297→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
  1298→  const idx = findLine(lines, labelPattern);
  1299→  if (idx >= 0) {
  1300→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
  1301→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1302→      if (m) return m[1];
  1303→    }
  1304→  }
  1305→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
  1306→  return flatMatch ? flatMatch[1] : EMPTY;
  1307→}
