    71→    71→function parseBR(s: string): number {
    72→    72→  const neg = s.startsWith('-');
    73→    73→  const clean = neg ? s.slice(1) : s;
    74→    74→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    75→    75→}

    87→    87→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    88→    88→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    89→    89→    if (Math.abs(i - startIdx) > maxDist) break;
    90→    90→    const line = lines[i];
    91→    91→    if (mustBeOnlyNum) {
    92→    92→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    93→    93→      if (m) return m[1];
    94→    94→    } else {
    95→    95→      const m = line.match(/([\d.]+,\d{2})/);
    96→    96→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
    97→    97→    }
    98→    98→  }
    99→    99→  return EMPTY;

   104→   104→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   105→   105→  const tableData = tableExtraction?.items;
   106→   106→  const tributosFromTable = tableExtraction
   107→   107→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   108→   108→    : null;
   109→   109→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   110→   110→  const flat = lines.join(' ');
   111→   111→
   112→   112→  // ===== CONCESSIONÁRIA =====
   113→   113→  const Concessionaria = (() => {
   114→   114→    for (const line of lines.slice(0, 20)) {
   115→   115→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
   116→   116→        return line.replace(/\s+/g, ' ');
   117→   117→      }
   118→   118→    }
   119→   119→    return EMPTY;
   120→   120→  })();
   121→   121→
   122→   122→  // ===== TIPO =====
   123→   123→  const Tipo = (() => {
   124→   124→    const u = flat.toUpperCase();
   125→   125→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   126→   126→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   127→   127→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   128→   128→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   129→   129→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   130→   130→    if (/RURAL/.test(u)) return 'RURAL';
   131→   131→    return EMPTY;
   132→   132→  })();
   133→   133→
   134→   134→  // ===== MERCADO =====
   135→   135→  const Mercado = (() => {
   136→   136→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   137→   137→    if (m) {
   138→   138→      const val = m[1].trim().replace(/\s+/g, ' ');
   139→   139→      if (val.length < 50) return val.toUpperCase();
   140→   140→    }
   141→   141→    return EMPTY;
   142→   142→  })();
   143→   143→
   144→   144→  // ===== CIDADE =====
   145→   145→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   146→   146→  const Cidade = (() => {
   147→   147→    for (const line of lines.slice(0, 60)) {
   148→   148→      // Match: CityName [StateAbbr] (AG: XX)
   149→   149→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   150→   150→      if (m) {
   151→   151→        const name = m[1].trim().toUpperCase();
   152→   152→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   153→   153→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   154→   154→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   155→   155→      }
   156→   156→    }
   157→   157→    return EMPTY;
   158→   158→  })();
   159→   159→
   160→   160→  // ===== UC =====
   161→   161→  // Strategy 1: Standard dotted format X.XXX.XXX-XX on its own line
   162→   162→  // Strategy 2: 13-digit format (e.g., 2103715020000) near address or standalone
   163→   163→  // Strategy 3: Number concatenated with text (e.g., "9993000001DOM.  ENT.:")
   164→   164→  const UC = (() => {
   165→   165→    // Strategy 1: Standard dotted format on its own line
   166→   166→    for (const line of lines) {
   167→   167→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   168→   168→      if (m) return m[1];
   169→   169→    }
   170→   170→    // Strategy 2: Dotted format on a short line (not CNPJ/CPF/CEP)
   171→   171→    for (const line of lines) {
   172→   172→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   173→   173→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   174→   174→    }
   175→   175→    // Strategy 3: Number (7-13 digits) concatenated with text at start of line
   176→   176→    // e.g., "9993000001DOM.  ENT.:" or "9993000001 INSTALAÇÃO"
   177→   177→    for (const line of lines) {
   178→   178→      const m = line.match(/^(\d{7,13})(?=[A-Za-zÀ-ÿ])/);
   179→   179→      if (m && !/^(\d{4})[-\/]/.test(line) && !/CEP|CNPJ|CPF|Insc/i.test(line)) return m[1];
   180→   180→    }
   181→   181→    // Strategy 4: 13-digit UC (Energisa MS format) - look in address lines
   182→   182→    const uc13 = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{13})\s/);
   183→   183→    if (uc13) return uc13[1];
   184→   184→    // Strategy 5: 13-digit standalone or near "Instalação:"
   185→   185→    const uc13b = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   186→   186→    if (uc13b) return uc13b[1];
   187→   187→    // Strategy 6: 7-13 digit number near "MATRÍCULA:" or "UC" or "Inst"
   188→   188→    const ucMat = flat.match(/(?:MATR[IÍ]CULA|UC\s*:?|Inst\.?\s*:?)[^\d]*(\d{7,13})/i);
   189→   189→    if (ucMat) return ucMat[1];
   190→   190→    // Strategy 7: broad search for dotted format
   191→   191→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   192→   192→    return broad ? broad[1] : EMPTY;
   193→   193→  })();
   194→   194→
   195→   195→  // ===== MÊS REFERÊNCIA =====
   196→   196→  const MesReferencia = (() => {
   197→   197→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   198→   198→    const numToMonth: Record<string, string> = {
   199→   199→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   200→   200→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   201→   201→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   202→   202→    };
   203→   203→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   204→   204→    if (nameMatch) {
   205→   205→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   206→   206→      return `${month} / ${nameMatch[2]}`;
   207→   207→    }
   208→   208→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   209→   209→    if (numMatch && numToMonth[numMatch[1]]) {
   210→   210→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   211→   211→    }
   212→   212→    return EMPTY;
   213→   213→  })();
   214→   214→
   215→   215→  // ===== ANO LEITURA =====
   216→   216→  const AnoLeitura = (() => {
   217→   217→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   218→   218→    if (m) return m[1];
   219→   219→    if (MesReferencia) {
   220→   220→      const ym = MesReferencia.match(/(\d{4})/);
   221→   221→      if (ym) return ym[1];
   222→   222→    }
   223→   223→    return EMPTY;
   224→   224→  })();
   225→   225→
   226→   226→  // ===== VENCIMENTO =====
   227→   227→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   228→   228→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   229→   229→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   230→   230→  const Vencimento = (() => {
   231→   231→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   232→   232→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   233→   233→    if (venMoney) return venMoney[1];
   234→   234→
   235→   235→    // Strategy 2: R$ + money + date on same line (reversed order)
   236→   236→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   237→   237→    if (moneyVen) return moneyVen[1];
   238→   238→
   239→   239→    // Strategy 3: VENCIMENTO label nearby (original approach)
   240→   240→    const venIdx = findLine(lines, /VENCIMENTO/i);
   241→   241→    if (venIdx >= 0) {
   242→   242→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   243→   243→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   244→   244→        if (m) return m[1];
   245→   245→      }
   246→   246→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   247→   247→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   248→   248→        if (m) return m[1];
   249→   249→      }
   250→   250→    }
   251→   251→
   252→   252→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   253→   253→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   254→   254→    if (apresMatch) return apresMatch[1];
   255→   255→
   256→   256→    return EMPTY;
   257→   257→  })();
   258→   258→
   259→   259→  // ===== NOTA FISCAL =====
   260→   260→  const NotaFiscal = (() => {
   261→   261→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   262→   262→    return m ? m[1] : EMPTY;
   263→   263→  })();
   264→   264→
   265→   265→  // ===== MODALIDADE TARIFÁRIA =====
   266→   266→  const ModalidadeTarifaria = (() => {
   267→   267→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   268→   268→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   269→   269→    if (/AZUL/i.test(flat)) return 'AZUL';
   270→   270→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   271→   271→    if (/VERDE/i.test(flat)) return 'VERDE';
   272→   272→    return EMPTY;
   273→   273→  })();
   274→   274→
   275→   275→  // ===== SUBGRUPO =====
   276→   276→  const Subgrupo = (() => {
   277→   277→    // Look for "/A4" or "/B3" etc. in the classification line
   278→   278→    // Must be a letter + single digit, right after "/"
   279→   279→    const m = flat.match(/\/\s*([AB]\d)\s/);
   280→   280→    if (m) return m[1].toUpperCase();
   281→   281→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   282→   282→    if (classIdx >= 0) {
   283→   283→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   284→   284→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   285→   285→      if (m2) return m2[1].toUpperCase();
   286→   286→    }
   287→   287→    return EMPTY;
   288→   288→  })();
   289→   289→
   290→   290→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   291→   291→
   292→   292→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   293→   293→  let ConsumoPonta = EMPTY;
   294→   294→  let ConsumoForaPonta = EMPTY;
   295→   295→  let ConsumoTotal = EMPTY;
   296→   296→  let ValorConsumoPonta = EMPTY;
   297→   297→  let ValorConsumoForaPonta = EMPTY;
   298→   298→
   299→   299→  if (isSplitTariff) {
   300→   300→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   301→   301→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   302→   302→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   303→   303→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   304→   304→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   305→   305→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   306→   306→  } else {
   307→   307→    for (const line of lines) {
   308→   308→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   309→   309→      if (km) { ConsumoTotal = km[1]; break; }
   310→   310→    }
   311→   311→    if (!ConsumoTotal) {
   312→   312→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   313→   313→      if (consIdx >= 0) {
   314→   314→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   315→   315→          const line = lines[i];
   316→   316→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   317→   317→          if (/[A-Za-z]{4,}/.test(line)) continue;
   318→   318→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   319→   319→          if (m) { ConsumoTotal = m[1]; break; }
   320→   320→        }
   321→   321→      }
   322→   322→    }
   323→   323→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   324→   324→  }
   325→   325→
   326→   326→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   327→   327→  let DemandaContratadaPonta = EMPTY;
   328→   328→  let DemandaContratadaForaPonta = EMPTY;
   329→   329→  let DemandaMedidaPonta = EMPTY;
   330→   330→  let DemandaMedidaForaPonta = EMPTY;
   331→   331→  let DemandaFaturadaPonta = EMPTY;
   332→   332→  let DemandaFaturadaForaPonta = EMPTY;
   333→   333→  let ValorDemandaPonta = EMPTY;
   334→   334→  let ValorDemandaForaPonta = EMPTY;
   335→   335→
   336→   336→  if (isSplitTariff) {
   337→   337→    // Try to extract demanda contratada from flat text: "KW Ponta:33" or "KW Ponta: 33"
   338→   338→    const dcMatch = flat.match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   339→   339→    if (dcMatch) DemandaContratadaPonta = dcMatch[1].replace('.', ',');
   340→   340→    const dcFpMatch = flat.match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   341→   341→    if (dcFpMatch) DemandaContratadaForaPonta = dcFpMatch[1].replace('.', ',');
   342→   342→
   343→   343→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   344→   344→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   345→   345→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   346→   346→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   347→   347→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   348→   348→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   349→   349→  }
   350→   350→
   351→   351→  // ===== ULTRAPASSAGEM =====
   352→   352→  let UltrapassagemPonta = EMPTY;
   353→   353→  let UltrapassagemForaPonta = EMPTY;
   354→   354→  let ValorUltrapassagemPonta = EMPTY;
   355→   355→  let ValorUltrapassagemForaPonta = EMPTY;
   356→   356→  if (isSplitTariff) {
   357→   357→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   358→   358→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   359→   359→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   360→   360→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   361→   361→  }
   362→   362→
   363→   363→  // ===== REATIVA EXCEDENTE =====
   364→   364→  let ReativaExcedentePonta = EMPTY;
   365→   365→  let ReativaExcedenteForaPonta = EMPTY;
   366→   366→  let EnergiaReativaTotal = EMPTY;
   367→   367→  let ValorReativaExcedentePonta = EMPTY;
   368→   368→  let ValorReativaExcedenteForaPonta = EMPTY;
   369→   369→  if (isSplitTariff) {
   370→   370→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   371→   371→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   372→   372→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   373→   373→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   374→   374→  } else {
   375→   375→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   376→   376→  }
   377→   377→
   378→   378→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   379→   379→  // Priority: table extraction > text-based extraction
   380→   380→  let PIS = EMPTY;
   381→   381→  let COFINS = EMPTY;
   382→   382→  let ICMS = EMPTY;
   383→   383→
   384→   384→  // Use tributos from table extraction if available (most reliable)
   385→   385→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   386→   386→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   387→   387→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   388→   388→
   389→   389→  // Fallback: text-based extraction for any missing tributo
   390→   390→  if (!PIS || !COFINS || !ICMS) {
   391→   391→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   392→   392→    if (pisIdx >= 0) {
   393→   393→      const nums: string[] = [];
   394→   394→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   395→   395→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   396→   396→        if (lineNums) {
   397→   397→          for (const n of lineNums) {
   398→   398→            nums.push(n);
   399→   399→            if (nums.length >= 3) break;
   400→   400→          }
   401→   401→        }
   402→   402→      }
   403→   403→      if (nums.length >= 3) {
   404→   404→        if (!PIS) PIS = nums[0];
   405→   405→        if (!COFINS) COFINS = nums[1];
   406→   406→        if (!ICMS) ICMS = nums[2];
   407→   407→      }
   408→   408→    }
   409→   409→    if (!PIS && !COFINS) {
   410→   410→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   411→   411→      if (combinedIdx >= 0) {
   412→   412→        const nums: string[] = [];
   413→   413→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   414→   414→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   415→   415→          if (m) nums.push(m[1]);
   416→   416→        }
   417→   417→        if (nums.length >= 3) {
   418→   418→          if (!PIS) PIS = nums[0];
   419→   419→          if (!COFINS) COFINS = nums[1];
   420→   420→          if (!ICMS) ICMS = nums[2];
   421→   421→        }
   422→   422→      }
   423→   423→    }
   424→   424→    if (!PIS || !COFINS) {
   425→   425→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   426→   426→      if (flatMatch) {
   427→   427→        if (!PIS) PIS = flatMatch[1];
   428→   428→        if (!COFINS) COFINS = flatMatch[2];
   429→   429→      }
   430→   430→    }
   431→   431→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   432→   432→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   433→   433→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
   434→   434→  }
   435→   435→
   436→   436→  // ===== GDI =====
   437→   437→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   438→   438→
   439→   439→  // ===== CIP (text-based) =====
   440→   440→  const CIP_text = (() => {
   441→   441→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   442→   442→    if (cipIdx >= 0) {
   443→   443→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   444→   444→      const money = findAllMoney(region);
   445→   445→      if (money.length > 0) return money[money.length - 1];
   446→   446→    }
   447→   447→    return findTributoValue(lines, flat, /\bCIP\b/i);
   448→   448→  })();
   449→   449→
   450→   450→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   451→   451→  let {
   452→   452→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   453→   453→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   454→   454→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   455→   455→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   456→   456→  } = extractGDI(lines, flat);
   457→   457→
   458→   458→  // ===== VALOR TOTAL =====
   459→   459→  // Best pattern: R$ value on same line as vencimento date + mes referência
   460→   460→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   461→   461→  const ValorTotal = (() => {
   462→   462→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
   463→   463→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
   464→   464→    if (venTotal) return venTotal[1];
   465→   465→
   466→   466→    // Strategy 2: R$ + value + mes referência on same line
   467→   467→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
   468→   468→    if (totalMes) return totalMes[1];
   469→   469→
   470→   470→    // Strategy 3: "TOTAL A PAGAR" with R$
   471→   471→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   472→   472→    if (totalIdx >= 0) {
   473→   473→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   474→   474→      if (m1) return m1[1];
   475→   475→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   476→   476→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   477→   477→        if (m2) return m2[1];
   478→   478→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   479→   479→        if (m3) return m3[1];
   480→   480→      }
   481→   481→    }
   482→   482→
   483→   483→    // Strategy 4: Last resort - largest R$ value
   484→   484→    const lastMoney = findAllMoney(flat);
   485→   485→    if (lastMoney.length > 0) {
   486→   486→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   487→   487→      return lastMoney[0];
   488→   488→    }
   489→   489→    return EMPTY;
   490→   490→  })();
   491→   491→
   492→   492→  // ===== TABLE DATA OVERRIDES =====
   493→   493→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
   494→   494→  let CIP = CIP_text;
   495→   495→
   496→   496→  if (tableData && tableData.length > 0) {
   497→   497→    let cipFromTable = EMPTY;
   498→   498→
   499→   499→    for (const item of tableData) {
   500→   500→      const lbl = item.label;
   501→   501→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
   502→   502→      const isPonta = /Ponta/i.test(lbl) && !isFora;
   503→   503→
   504→   504→      // Consumo - check Convencional (no Ponta/Fora) first, then split
   505→   505→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
   506→   506→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
   507→   507→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
   508→   508→          if (item.quantidade) ConsumoTotal = item.quantidade;
   509→   509→          if (item.valor) ValorConsumoForaPonta = item.valor;
   510→   510→        }
   511→   511→        // VERDE Intermediária
   512→   512→        else if (/Intermedi[áa]ria/i.test(lbl)) {
   513→   513→          if (item.quantidade && !ConsumoForaPonta) ConsumoForaPonta = item.quantidade;
   514→   514→        }
   515→   515→        // Split tariff (AZUL/VERDE)
   516→   516→        else if (isPonta) {
   517→   517→          if (item.quantidade) ConsumoPonta = item.quantidade;
   518→   518→          if (item.valor) ValorConsumoPonta = item.valor;
   519→   519→        } else if (isFora) {
   520→   520→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
   521→   521→          if (item.valor) ValorConsumoForaPonta = item.valor;
   522→   522→        }
   523→   523→      }
   524→   524→      // Demanda Medida
   525→   525→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
   526→   526→        if (!isFora) {
   527→   527→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
   528→   528→          if (item.valor) ValorDemandaPonta = item.valor;
   529→   529→        } else {
   530→   530→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
   531→   531→          if (item.valor) ValorDemandaForaPonta = item.valor;
   532→   532→        }
   533→   533→      }
   534→   534→      // Ultrapassagem / Não Consumida
   535→   535→      else if (/(?:Ultrapassagem|Ultrap|Não\s*Consumida)/i.test(lbl) && /Demanda/i.test(lbl)) {
   536→   536→        if (!isFora) {
   537→   537→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
   538→   538→          if (item.valor) ValorUltrapassagemPonta = item.valor;
   539→   539→        } else {
   540→   540→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
   541→   541→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
   542→   542→        }
   543→   543→      }
   544→   544→      // Reativa Excedente
   545→   545→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
   546→   546→        if (!isFora && /Ponta/i.test(lbl)) {
   547→   547→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
   548→   548→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   549→   549→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
   550→   550→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
   551→   551→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
   552→   552→        } else if (!isFora && !/Ponta/i.test(lbl)) {
   553→   553→          // Generic reativa (Convencional)
   554→   554→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
   555→   555→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   556→   556→        }
   557→   557→      }
   558→   558→      // CIP Municipal
   559→   559→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
   560→   560→        if (item.valor) cipFromTable = item.valor;
   561→   561→      }
   562→   562→      // GDI items
   563→   563→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
   564→   564→        if (item.quantidade && !EnergiaAtvInjetadaGDIoUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = item.quantidade;
   565→   565→        if (item.valor && !EnergiaAtvInjetadaGDIoUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = item.valor;
   566→   566→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
   567→   567→        // Multiple mUC entries: accumulate
   568→   568→        if (item.quantidade) {
   569→   569→          EnergiaAtvInjetadaGDImUC_Qtd = EnergiaAtvInjetadaGDImUC_Qtd
   570→   570→            ? fmtSumBR(EnergiaAtvInjetadaGDImUC_Qtd, item.quantidade)
   571→   571→            : item.quantidade;
   572→   572→        }
   573→   573→        if (item.valor) {
   574→   574→          EnergiaAtvInjetadaGDImUC_Valor = EnergiaAtvInjetadaGDImUC_Valor
   575→   575→            ? fmtSumBR(EnergiaAtvInjetadaGDImUC_Valor, item.valor)
   576→   576→            : item.valor;
   577→   577→        }
   578→   578→      }
   579→   579→    }
   580→   580→
   581→   581→    // Override CIP from table
   582→   582→    if (cipFromTable) CIP = cipFromTable;
   583→   583→  }
   584→   584→
   585→   585→  return {
   586→   586→    UC, Cidade, Tipo, Mercado, Concessionaria,
   587→   587→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   588→   588→    ModalidadeTarifaria, Subgrupo,
   589→   589→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   590→   590→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   591→   591→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   592→   592→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   593→   593→    UltrapassagemPonta, UltrapassagemForaPonta,
   594→   594→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   595→   595→    ValorConsumoPonta, ValorConsumoForaPonta,
   596→   596→    ValorDemandaPonta, ValorDemandaForaPonta,
   597→   597→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   598→   598→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   599→   599→    PIS, COFINS, ICMS, GDI, CIP,
   600→   600→    EnergiaReativaTotal,
   601→   601→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   602→   602→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   603→   603→    ValorTotal,
   604→   604→  };
   605→   605→}

   607→   607→// ===================== GDI EXTRACTION =====================
   608→   608→
   609→   609→function extractGDI(lines: string[], flat: string) {
   610→   610→  const EMPTY = '';
   611→   611→  let nOUC = 0;
   612→   612→  let nMUC = 0;
   613→   613→  let firstGDI = -1;
   614→   614→
   615→   615→  for (let i = 0; i < lines.length; i++) {
   616→   616→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   617→   617→      nOUC++;
   618→   618→      if (firstGDI < 0) firstGDI = i;
   619→   619→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   620→   620→      nMUC++;
   621→   621→      if (firstGDI < 0) firstGDI = i;
   622→   622→    }
   623→   623→  }
   624→   624→
   625→   625→  if (nOUC === 0 && nMUC === 0) {
   626→   626→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   627→   627→  }
   628→   628→
   629→   629→  const chaveIdx = (() => {
   630→   630→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   631→   631→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   632→   632→    }
   633→   633→    return -1;
   634→   634→  })();
   635→   635→
   636→   636→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
   637→   637→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   638→   638→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
   639→   639→      sectionEnd = i; break;
   640→   640→    }
   641→   641→  }
   642→   642→  if (chaveIdx >= 0) {
   643→   643→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   644→   644→  }
   645→   645→
   646→   646→  const allQtds: number[] = [];
   647→   647→  if (chaveIdx >= 0) {
   648→   648→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   649→   649→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   650→   650→      if (m) allQtds.push(parseBR(m[1]));
   651→   651→    }
   652→   652→  }
   653→   653→  if (allQtds.length === 0) {
   654→   654→    for (let i = firstGDI; i < sectionEnd; i++) {
   655→   655→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   656→   656→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   657→   657→      if (allQtds.length >= nOUC + nMUC) break;
   658→   658→    }
   659→   659→  }
   660→   660→
   661→   661→  const allRVals: number[] = [];
   662→   662→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   663→   663→  const totalNeeded = nOUC + nMUC;
   664→   664→
   665→   665→  for (let i = firstGDI; i < rSearchEnd; i++) {
   666→   666→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   667→   667→    if (m && allRVals.length < totalNeeded) {
   668→   668→      allRVals.push(parseBR('-' + m[2]));
   669→   669→    }
   670→   670→  }
   671→   671→  if (allRVals.length < totalNeeded) {
   672→   672→    for (let i = firstGDI; i < rSearchEnd; i++) {
   673→   673→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   674→   674→      if (m && allRVals.length < totalNeeded) {
   675→   675→        allRVals.push(parseBR(m[1]));
   676→   676→      }
   677→   677→    }
   678→   678→  }
   679→   679→
   680→   680→  const oUCQtds = allQtds.splice(0, nOUC);
   681→   681→  const oUCRVals = allRVals.splice(0, nOUC);
   682→   682→
   683→   683→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   684→   684→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   685→   685→
   686→   686→  return {
   687→   687→    oUC_Qtd: sumQtd(oUCQtds),
   688→   688→    oUC_Valor: sumR(oUCRVals),
   689→   689→    mUC_Qtd: sumQtd(allQtds),
   690→   690→    mUC_Valor: sumR(allRVals),
   691→   691→  };
   692→   692→}
   693→   693→
   694→   694→// ===================== NUMBER FORMATTING =====================
   695→   695→
   696→   696→function fmtBRNumber(n: number): string {
   697→   697→  const abs = Math.abs(n);
   698→   698→  const intPart = Math.floor(abs).toString();
   699→   699→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   700→   700→  let formatted = '';
   701→   701→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   702→   702→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   703→   703→    formatted = intPart[i] + formatted;
   704→   704→  }
   705→   705→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   706→   706→}
   707→   707→
   708→   708→function fmtSumBR(a: string, b: string): string {
   709→   709→  return fmtBRNumber(parseBR(a) + parseBR(b));
   710→   710→}

   714→   714→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   715→   715→  const idx = findLine(lines, labelPattern);
   716→   716→  if (idx < 0) return EMPTY;
   717→   717→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   718→   718→    const line = lines[i];
   719→   719→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   720→   720→    const m = line.match(/([\d.]+,\d{2})/);
   721→   721→    if (m) return m[1];
   722→   722→  }
   723→   723→  return EMPTY;
   724→   724→}

   726→   726→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   727→   727→  const idx = findLine(lines, labelPattern);
   728→   728→  if (idx < 0) return EMPTY;
   729→   729→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   730→   730→  const moneyVals = findAllMoney(region);
   731→   731→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   732→   732→  return EMPTY;
   733→   733→}

   735→   735→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   736→   736→  const idx = findLine(lines, labelPattern);
   737→   737→  if (idx >= 0) {
   738→   738→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   739→   739→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   740→   740→      if (m) return m[1];
   741→   741→    }
   742→   742→  }
   743→   743→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   744→   744→  return flatMatch ? flatMatch[1] : EMPTY;
   745→   745→}
