    81→function parseBR(s: string): number {
    82→  const neg = s.startsWith('-');
    83→  const clean = neg ? s.slice(1) : s;
    84→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    85→}
    86→
    87→function formatBR(n: number): string {
    88→  if (n === 0) return '0,00';
    89→  const neg = n < 0;
    90→  const abs = Math.abs(n);
    91→  const rounded = Math.round(abs * 100) / 100;
    92→  const parts = rounded.toFixed(2).split('.');
    93→  const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    94→  const result = intPart + ',' + parts[1];
    95→  return neg ? '-' + result : result;
    96→}

   102→function findAllMoney(text: string): string[] {
   103→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
   104→  if (!matches) return [];
   105→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
   106→}

   108→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
   109→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
   110→    if (Math.abs(i - startIdx) > maxDist) break;
   111→    const line = lines[i];
   112→    if (mustBeOnlyNum) {
   113→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
   114→      if (m) return m[1];
   115→    } else {
   116→      const m = line.match(/([\d.]+,\d{2})/);
   117→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   118→    }
   119→  }
   120→  return EMPTY;
   121→}

   123→// Find a BR-format integer number near a line (e.g., "1.346" or "12.929" without decimal places)
   124→// This is needed because consumption kWh values are often displayed as integers in PDFs.
   125→function findIntNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number): string {
   126→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
   127→    if (Math.abs(i - startIdx) > maxDist) break;
   128→    const line = lines[i];
   129→    // Match a BR-format integer: "1.346" or "12.929" or "346" (no decimal comma)
   130→    // Must be the whole line (or most of it) to avoid matching prices or other numbers
   131→    const m = line.match(/^(\d{1,2}(?:\.\d{3})+|\d{3,6})$/);
   132→    if (m) return m[1];
   133→  }
   134→  return EMPTY;
   135→}

   139→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   140→  const tableData = tableExtraction?.items;
   141→  const tributosFromTable = tableExtraction
   142→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   143→    : null;
   144→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   145→  const flat = lines.join(' ');
   146→
   147→  // ===== CONCESSIONÁRIA =====
   148→  // Priority: detect the ISSUING concessionária (not the customer).
   149→  // For electricity invoices (DANF3E / CONTA DE ENERGIA), ENERGISA/ELEKTRO is the issuer.
   150→  // For water/sewage invoices, SANESUL is the issuer.
   151→  const Concessionaria = (() => {
   152→    const u = flat.toUpperCase();
   153→    const isEnergyInvoice = /DANF3E|ENERGIA\s*EL[eÉ]TRICA|DISTR\.\s*DE\s*ENERGIA/i.test(u);
   154→    if (isEnergyInvoice) {
   155→      // For energy invoices, the issuing utility is ENERGISA/ELEKTRO/NEOENERGIA
   156→      if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   157→      if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   158→      if (/ENERGISA/i.test(u)) return 'ENERGISA';
   159→    }
   160→    // For non-energy invoices or when no energy indicator found
   161→    if (/SANESUL/i.test(u)) return 'SANESUL';
   162→    if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   163→    if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   164→    if (/ENERGISA/i.test(u)) return 'ENERGISA';
   165→    return EMPTY;
   166→  })();
   167→
   168→  // ===== TIPO =====
   169→  const Tipo = (() => {
   170→    const u = flat.toUpperCase();
   171→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   172→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   173→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   174→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   175→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   176→    if (/RURAL/.test(u)) return 'RURAL';
   177→    return EMPTY;
   178→  })();
   179→
   180→  // ===== MERCADO =====
   181→  const Mercado = (() => {
   182→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   183→    if (m) {
   184→      const val = m[1].trim().replace(/\s+/g, ' ');
   185→      if (val.length < 50) return val.toUpperCase();
   186→    }
   187→    return EMPTY;
   188→  })();
   189→
   190→  // ===== CIDADE =====
   191→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   192→  const Cidade = (() => {
   193→    for (const line of lines.slice(0, 60)) {
   194→      // Match: CityName [StateAbbr] (AG: XX)
   195→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   196→      if (m) {
   197→        const name = m[1].trim().toUpperCase();
   198→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   199→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   200→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   201→      }
   202→    }
   203→    return EMPTY;
   204→  })();
   205→
   206→  // ===== UC =====
   207→  const UC = (() => {
   208→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   209→    const isCnpjCpf = (s: string) => {
   210→      const raw = s.replace(/[.\-\/]/g, '');
   211→      return raw.length === 14 || raw.length === 11;
   212→    };
   213→
   214→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   215→    for (const line of lines) {
   216→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   217→      if (m) {
   218→        const num = m[1].replace(/^0+/, '') || '0';
   219→        if (num.length >= 3) return num;
   220→      }
   221→    }
   222→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   223→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   224→    if (slashCtx) {
   225→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   226→      if (num.length >= 3) return num;
   227→    }
   228→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   229→    // e.g., "1.409.407.051-47" → "140940705147"
   230→    for (const line of lines) {
   231→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   232→      if (m && !isCnpjCpf(m[1])) return m[1];
   233→    }
   234→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   235→    for (const line of lines) {
   236→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   237→      if (m) return m[1];
   238→    }
   239→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   240→    for (const line of lines) {
   241→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   242→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   243→    }
   244→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   245→    for (const line of lines) {
   246→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   247→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   248→    }
   249→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   250→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   251→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   252→    if (ucMat) return ucMat[1];
   253→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   254→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   255→    if (ucAddr) return ucAddr[1];
   256→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   257→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   258→    if (ucInst) return ucInst[1];
   259→    // Strategy 10: UC label with number — 4-13 digits
   260→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   261→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   262→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   263→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   264→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   265→    // Strategy 12: broad search for 3-group dotted format
   266→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   267→    if (broad3) return broad3[1];
   268→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   269→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   270→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   271→    return EMPTY;
   272→  })();
   273→
   274→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   275→  const cleanUC = UC.replace(/[.\-]/g, '');
   276→
   277→  // ===== ENDEREÇO =====
   278→  const Endereco = (() => {
   279→    // Address line typically follows the city (AG: XX) line and starts with RUA, AV, AVE, LOT, ESTRADA, etc.
   280→    // Examples from real PDFs:
   281→    //   "RUA SALDANHA DA GAMA, 896 - 21.004.65.029401 LAD RE RA - 422 - 79370000"
   282→    //   "LOT RECANTO NUARA, S/N - 24.416.10.154000 PAT 01/REL01 2289 - 79190000"
   283→    //   "AVE MATO GROSSO DO SUL, SN - 12.029.07.090000 COX EAT 006 2243 - 79400000"
   284→    const STREET_RE = /^(RUA|AV[. ]|AVE |AVENIDA|ESTRADA|LOT |RODOVIA|TRAVESSA|ALAMEDA|PCA[. ]|PRACA)/i;
   285→
   286→    // Clean an address line: keep street + number, strip internal codes, keep CEP
   287→    const extractCleanAddress = (line: string): string => {
   288→      // Split at the first " - " that precedes an internal code pattern (XX.XXX.XX.XXXXXX)
   289→      const beforeCodes = line.split(/\s*-\s*\d{2}\.\d{3}/)[0];
   290→      let addr = (beforeCodes || line).trim();
   291→      // Remove trailing dash
   292→      addr = addr.replace(/\s*-\s*$/, '').trim();
   293→      // Append CEP if found at end of original line
   294→      const cepMatch = line.match(/(\d{5}-?\d{3})\s*$/);
   295→      if (cepMatch) addr += ' - CEP ' + cepMatch[1];
   296→      return addr;
   297→    };
   298→
   299→    // Strategy 1: Find address line after city (AG: XX) pattern
   300→    for (let i = 0; i < Math.min(60, lines.length); i++) {
   301→      if (/\(\s*AG\s*:\s*\d+\s*\)/.test(lines[i])) {
   302→        // Check next 3 lines for an address pattern
   303→        for (let j = i + 1; j <= Math.min(i + 3, lines.length - 1); j++) {
   304→          if (STREET_RE.test(lines[j])) {
   305→            return extractCleanAddress(lines[j]);
   306→          }
   307→        }
   308→      }
   309→    }
   310→
   311→    // Strategy 2: Find address line near CNPJ/CPF line (appears before it)
   312→    for (let i = 0; i < lines.length; i++) {
   313→      if (/CNPJ\/CPF/i.test(lines[i]) && i > 0) {
   314→        if (STREET_RE.test(lines[i - 1])) {
   315→          return extractCleanAddress(lines[i - 1]);
   316→        }
   317→      }
   318→    }
   319→
   320→    // Strategy 3: Look for "ENDEREÇO" label followed by address
   321→    for (let i = 0; i < lines.length; i++) {
   322→      if (/^ENDERE[ÇC]O$/i.test(lines[i].trim()) && i + 1 < lines.length) {
   323→        const next = lines[i + 1];
   324→        const cleanAddr = next.replace(/\s*-\s*(SA[IÍ]DA|ENTRADA)\s+PARA\s+[^-]+/i, '')
   325→          .replace(/\s*-\s*[A-ZÀ-Ú]+\s*\/\s*[A-Z]{2}\s*-\s*CEP\s+\d{5}-\d{3}/i, '').trim();
   326→        if (cleanAddr.length > 5) return cleanAddr;
   327→      }
   328→    }
   329→
   330→    return EMPTY;
   331→  })();
   332→
   333→  // ===== MÊS REFERÊNCIA =====
   334→  const MesReferencia = (() => {
   335→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   336→    const numToMonth: Record<string, string> = {
   337→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   338→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   339→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   340→    };
   341→    const capitalize = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
   342→
   343→    // Helper: get Vencimento date components for sanity checks
   344→    const venMoneyMatch = flat.match(/(\d{2})\/(\d{2})\/(\d{4})\s*R\$/);
   345→    const venMonth = venMoneyMatch ? parseInt(venMoneyMatch[2], 10) : 0;
   346→    const venYear = venMoneyMatch ? parseInt(venMoneyMatch[3], 10) : 0;
   347→
   348→    // Helper: check if a month/year is consistent with the Vencimento date
   349→    // (reference month is typically 1-2 months before the vencimento month)
   350→    const isConsistentWithVencimento = (month: number, year: number): boolean => {
   351→      if (!venMonth || !venYear) return true; // no vencimento to compare
   352→      const venTotal = venYear * 12 + venMonth;
   353→      const refTotal = year * 12 + month;
   354→      const diff = venTotal - refTotal;
   355→      return diff >= 1 && diff <= 3; // reference is 1-3 months before vencimento
   356→    };
   357→
   358→    // Strategy 1a: "Mês / Ano" pattern BEFORE date + R$ on the SAME LINE (most reliable)
   359→    // Energisa format: "Fevereiro / 202621/03/2026R$ 538,49" or "Junho / 2026  21/07/2026  R$ 13.356,50"
   360→    for (const line of lines) {
   361→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})\\s*\\d{2}/\\d{2}/\\d{4}\\s*R\\$`, 'i'));
   362→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   363→    }
   364→    // Strategy 1b: date + R$ pattern BEFORE "Mês / Ano" on the SAME LINE
   365→    // pdf-parse reordering: "21/07/2026R$ 6.980,48Junho / 2026"
   366→    for (const line of lines) {
   367→      const m = line.match(new RegExp(`\\d{2}/\\d{2}/\\d{4}\\s*R\\$\\s*[\\d.,]+\\s*(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   368→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   369→    }
   370→    // Strategy 2: "Mês / Ano" pattern on the SAME LINE as VENCIMENTO or NOTA FISCAL
   371→    for (const line of lines) {
   372→      if (/VENCIMENTO|NOTA FISCAL|Nº FATURA/i.test(line)) {
   373→        const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   374→        if (m) return `${capitalize(m[1])} / ${m[2]}`;
   375→      }
   376→    }
   377→    // Strategy 3: "Mês / Ano" on a line that also contains a date DD/MM/YYYY (likely the billing line)
   378→    for (const line of lines) {
   379→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   380→      if (m && /\d{2}\/\d{2}\/\d{4}/.test(line)) {
   381→        const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   382→        const monthNum = monthIdx + 1;
   383→        if (isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) {
   384→          return `${capitalize(m[1])} / ${m[2]}`;
   385→        }
   386→      }
   387→    }
   388→    // Strategy 4: "Mês / Ano" anywhere in flat text — skip "Desde mês/ano", "Referência:" history, etc.
   389→    const allNameMatches = [...flat.matchAll(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'gi'))];
   390→    for (const m of allNameMatches) {
   391→      // Check context: skip if preceded by "Desde", "Referência:", "Ref", or other informative words
   392→      const start = Math.max(0, m.index! - 20);
   393→      const context = flat.slice(start, m.index! + m[0].length).toLowerCase();
   394→      if (/desde|até|a partir|inform|dispon|referência|ref\b/.test(context)) continue;
   395→      // Also validate against Vencimento date
   396→      const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   397→      const monthNum = monthIdx + 1;
   398→      if (!isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) continue;
   399→      return `${capitalize(m[1])} / ${m[2]}`;
   400→    }
   401→    // Strategy 5: Use Leitura Atual date to derive the reference month
   402→    // pdf-parse sometimes garbles text, so try multiple patterns
   403→    const leituraPatterns = [
   404→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*(\d{2})\s*\/\s*(\d{4})/i,
   405→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*(\d{2})\s*\/\s*(\d{4})/i,
   406→    ];
   407→    for (const pat of leituraPatterns) {
   408→      const lm = flat.match(pat);
   409→      if (lm && numToMonth[lm[1]]) {
   410→        const monthNum = parseInt(lm[1], 10);
   411→        const year = parseInt(lm[2], 10);
   412→        if (isConsistentWithVencimento(monthNum, year)) {
   413→          return `${numToMonth[lm[1]]} / ${lm[2]}`;
   414→        }
   415→      }
   416→    }
   417→    // Strategy 6: numeric MM/YYYY near Vencimento date — with consistency check
   418→    if (venMoneyMatch) {
   419→      const venIdx = flat.indexOf(venMoneyMatch[0]);
   420→      const region = flat.slice(Math.max(0, venIdx - 150), venIdx + 50);
   421→      const numMatches = [...region.matchAll(/(\d{2})\s*\/\s*(\d{4})/g)];
   422→      for (const nm of numMatches) {
   423→        if (numToMonth[nm[1]]) {
   424→          const monthNum = parseInt(nm[1], 10);
   425→          const year = parseInt(nm[2], 10);
   426→          if (isConsistentWithVencimento(monthNum, year)) {
   427→            return `${numToMonth[nm[1]]} / ${nm[2]}`;
   428→          }
   429→        }
   430→      }
   431→    }
   432→    // Strategy 7: first valid MM/YYYY that is a month and consistent with Vencimento
   433→    const allNumMatches = [...flat.matchAll(/(\d{2})\s*[\/]\s*(\d{4})/g)];
   434→    for (const nm of allNumMatches) {
   435→      if (numToMonth[nm[1]]) {
   436→        const monthNum = parseInt(nm[1], 10);
   437→        const year = parseInt(nm[2], 10);
   438→        if (isConsistentWithVencimento(monthNum, year)) {
   439→          return `${numToMonth[nm[1]]} / ${nm[2]}`;
   440→        }
   441→      }
   442→    }
   443→    return EMPTY;
   444→  })();
   445→
   446→  // ===== ANO LEITURA =====
   447→  const AnoLeitura = (() => {
   448→    // Try multiple patterns for Leitura Atual date (pdf-parse can garble text)
   449→    const leituraYearPatterns = [
   450→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i,
   451→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*\d{2}\s*\/\s*(\d{4})/i,
   452→    ];
   453→    for (const pat of leituraYearPatterns) {
   454→      const m = flat.match(pat);
   455→      if (m) return m[1];
   456→    }
   457→    // Fallback: extract year from MesReferencia
   458→    if (MesReferencia) {
   459→      const ym = MesReferencia.match(/(\d{4})/);
   460→      if (ym) return ym[1];
   461→    }
   462→    return EMPTY;
   463→  })();
   464→
   465→  // ===== VENCIMENTO =====
   466→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   467→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   468→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   469→  const Vencimento = (() => {
   470→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   471→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   472→    if (venMoney) return venMoney[1];
   473→
   474→    // Strategy 2: R$ + money + date on same line (reversed order)
   475→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   476→    if (moneyVen) return moneyVen[1];
   477→
   478→    // Strategy 3: VENCIMENTO label nearby (original approach)
   479→    const venIdx = findLine(lines, /VENCIMENTO/i);
   480→    if (venIdx >= 0) {
   481→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   482→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   483→        if (m) return m[1];
   484→      }
   485→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   486→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   487→        if (m) return m[1];
   488→      }
   489→    }
   490→
   491→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   492→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   493→    if (apresMatch) return apresMatch[1];
   494→
   495→    return EMPTY;
   496→  })();
   497→
   498→  // ===== NOTA FISCAL =====
   499→  const NotaFiscal = (() => {
   500→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   501→    return m ? m[1] : EMPTY;
   502→  })();
   503→
   504→  // ===== MODALIDADE TARIFÁRIA =====
   505→  const ModalidadeTarifaria = (() => {
   506→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   507→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   508→    if (/AZUL/i.test(flat)) return 'AZUL';
   509→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   510→    if (/VERDE/i.test(flat)) return 'VERDE';
   511→    return EMPTY;
   512→  })();
   513→
   514→  // ===== SUBGRUPO =====
   515→  const Subgrupo = (() => {
   516→    // Look for "/A4" or "/B3" etc. in the classification line
   517→    // Must be a letter + single digit, right after "/"
   518→    const m = flat.match(/\/\s*([AB]\d)\s/);
   519→    if (m) return m[1].toUpperCase();
   520→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   521→    if (classIdx >= 0) {
   522→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   523→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   524→      if (m2) return m2[1].toUpperCase();
   525→    }
   526→    return EMPTY;
   527→  })();
   528→
   529→  const isSplitTariff = (/AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat))
   530→    || /Consumo\s+em\s+kWh\s*-\s*Intermedi/i.test(flat);
   531→
   532→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   533→  let ConsumoPonta = EMPTY;
   534→  let ConsumoIntermediaria = EMPTY;
   535→  let ConsumoForaPonta = EMPTY;
   536→  let ConsumoTotal = EMPTY;
   537→  let ConsumoGrupoB_kWh = EMPTY;
   538→  let ConsumoGrupoB_Valor = EMPTY;
   539→  let ValorConsumoPonta = EMPTY;
   540→  let ValorConsumoIntermediaria = EMPTY;
   541→  let ValorConsumoForaPonta = EMPTY;
   542→  // Accumulators for Grupo B consumption
   543→  let grupoB_consumoKWh_total = 0;
   544→  let grupoB_consumoValor_total = 0;
   545→
   546→  if (isSplitTariff) {
   547→    // IMPORTANT: For split tariffs (VERDE/AZUL), the "Consumo em kWh - Ponta/Fora Ponta" label
   548→    // in the "Itens da Fatura" section is often followed by garbled text containing the unit price
   549→    // (e.g., "0,6178306.734,55") rather than the actual kWh consumption.
   550→    // The findNumNearLine function picks up the unit price (0,61) instead of the consumption.
   551→    // SOLUTION: First try the "KWH1.198,96" pattern and compressed "Energia ativa em kWh" data,
   552→    // which are more reliable. Only fall back to findNumNearLine if those don't work,
   553→    // AND validate that the found value is a reasonable consumption number (>= 10 kWh).
   554→
   555→    // Strategy 1: KWH + value pattern for split tariff (e.g., "KWH1.052,34")
   556→    for (const line of lines) {
   557→      const km = line.match(/^KWH(\d{1,2}(?:\.\d{3})*,\d{2})$/i);
   558→      if (km) { ConsumoPonta = km[1]; break; }
   559→    }
   560→
   561→    // Strategy 2: Parse compressed "Energia ativa em kWh" data (W90...Ponta/Fora Ponta sections)
   562→    if (!ConsumoPonta || !ConsumoForaPonta) {
   563→      const periodSections: { period: string; data: string }[] = [];
   564→      for (let i = 0; i < lines.length - 2; i++) {
   565→        if (/Ponta$/i.test(lines[i]) && !/Fora/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   566→          periodSections.push({ period: 'Ponta', data: lines[i + 2] });
   567→        }
   568→        if (/Fora\s*Ponta$/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   569→          periodSections.push({ period: 'ForaPonta', data: lines[i + 2] });
   570→        }
   571→      }
   572→      // If we have Ponta from KWH label, use it to determine prefix length for compressed data
   573→      const pontaSection = periodSections.find(s => s.period === 'Ponta');
   574→      const foraSection = periodSections.find(s => s.period === 'ForaPonta');
   575→      if (pontaSection && ConsumoPonta) {
   576→        const pontaNoDot = ConsumoPonta.replace(/\./g, '');
   577→        const prefixIdx = pontaSection.data.indexOf(pontaNoDot);
   578→        if (prefixIdx >= 0) {
   579→          const prefixLen = prefixIdx;
   580→          if (foraSection && prefixLen < foraSection.data.length) {
   581→            const foraRest = foraSection.data.substring(prefixLen);
   582→            const commaPos = foraRest.indexOf(',');
   583→            if (commaPos >= 2 && commaPos <= 7) {
   584→              const foraNum = foraRest.substring(0, commaPos + 3);
   585→              const parts = foraNum.split(',');
   586→              const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   587→              if (!ConsumoForaPonta) ConsumoForaPonta = intPart + ',' + parts[1];
   588→            }
   589→          }
   590→        }
   591→      }
   592→      // If no Ponta from KWH label, try to parse compressed data with prefix heuristics
   593→      // CRITICAL: The compressed data has a prefix (1-2 digits) that we must skip.
   594→      // We try ALL possible prefix lengths (0-3) for BOTH sections simultaneously,
   595→      // and pick the prefix that produces valid consumption values for BOTH periods.
   596→      if (!ConsumoPonta && periodSections.length >= 1) {
   597→        // Helper to try extracting a BR number from compressed data at a given prefix length
   598→        const tryExtract = (data: string, prefix: number): { value: string; numeric: number } | null => {
   599→          if (prefix >= data.length) return null;
   600→          const rest = data.substring(prefix);
   601→          const commaPos = rest.indexOf(',');
   602→          if (commaPos < 1 || commaPos > 6) return null;
   603→          const numStr = rest.substring(0, commaPos + 3);
   604→          const parts = numStr.split(',');
   605→          if (parts.length !== 2 || !/^\d+$/.test(parts[0]) || !/^\d{2}$/.test(parts[1])) return null;
   606→          const numeric = parseInt(parts[0]);
   607→          if (numeric < 10 || numeric > 999999) return null;
   608→          const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   609→          return { value: intPart + ',' + parts[1], numeric };
   610→        };
   611→
   612→        // Try each prefix length and find the one that gives valid values for all available sections.
   613→        // We try from SMALLEST prefix (0) up to 3, collecting all valid candidates.
   614→        // The prefix is typically 1-2 digits encoding meter reading info.
   615→        // We score candidates: prefer values in the typical Grupo A consumption range (100-50,000),
   616→        // with Fora Ponta > Ponta and reasonable ratio (1-15).
   617→        let found = false;
   618→        // Collect all valid prefix candidates, then pick the best one
   619→        const prefixCandidates: { prefix: number; results: Map<string, { value: string; numeric: number }>; score: number }[] = [];
   620→        for (let prefix = 0; prefix <= 3; prefix++) {
   621→          const results = new Map<string, { value: string; numeric: number }>();
   622→          let allValid = true;
   623→          for (const sec of periodSections) {
   624→            const result = tryExtract(sec.data, prefix);
   625→            if (!result) { allValid = false; break; }
   626→            if (result.numeric < 100) { allValid = false; break; }
   627→            // Reject values with leading zeros (e.g., "01346" from prefix=1 on "101346,00...")
   628→            if (result.value.startsWith('0') && result.value.length > 3) { allValid = false; break; }
   629→            results.set(sec.period, result);
   630→          }
   631→          if (allValid && results.size > 0) {
   632→            // Additional validation: Fora Ponta / Ponta ratio should be reasonable (0.1 - 20)
   633→            // Typical ratios in Grupo A are 3-15; ratios > 20 indicate wrong prefix
   634→            const pontaR = results.get('Ponta');
   635→            const foraR = results.get('ForaPonta');
   636→            if (pontaR && foraR) {
   637→              const ratio = foraR.numeric / pontaR.numeric;
   638→              if (ratio < 0.1 || ratio > 20) continue; // skip this prefix
   639→              // Score: prefer values in typical Grupo A range (100-50000) and reasonable ratio
   640→              // Penalize very large values (likely wrong prefix with too few digits stripped)
   641→              const inRangePonta = (pontaR.numeric >= 100 && pontaR.numeric <= 50000) ? 2000 : 0;
   642→              const inRangeFora = (foraR.numeric >= 100 && foraR.numeric <= 50000) ? 2000 : 0;
   643→              const ratioScore = (ratio >= 1 && ratio <= 15) ? 1000 : 0; // bonus for typical ratio
   644→              const rangeScore = inRangePonta + inRangeFora + ratioScore;
   645→              prefixCandidates.push({ prefix, results, score: rangeScore });
   646→            } else {
   647→              // Only one period found — score based on range
   648→              let rangeScore = 0;
   649→              for (const r of results.values()) {
   650→                if (r.numeric >= 100 && r.numeric <= 50000) rangeScore += 2000;
   651→              }
   652→              prefixCandidates.push({ prefix, results, score: rangeScore });
   653→            }
   654→          }
   655→        }
   656→        // Sort by score descending (prefer larger values with typical ratio)
   657→        prefixCandidates.sort((a, b) => b.score - a.score);
   658→        if (prefixCandidates.length > 0) {
   659→          const best = prefixCandidates[0];
   660→          if (!ConsumoPonta && best.results.has('Ponta')) ConsumoPonta = best.results.get('Ponta')!.value;
   661→          if (!ConsumoForaPonta && best.results.has('ForaPonta')) ConsumoForaPonta = best.results.get('ForaPonta')!.value;
   662→          found = true;
   663→        }
   664→        if (!found) {
   665→          // Fallback: try each section independently with per-section prefix
   666→          for (const sec of periodSections) {
   667→            for (let prefix = 0; prefix <= 3; prefix++) {
   668→              const result = tryExtract(sec.data, prefix);
   669→              if (result && result.numeric >= 100) {
   670→                if (sec.period === 'Ponta' && !ConsumoPonta) ConsumoPonta = result.value;
   671→                if (sec.period === 'ForaPonta' && !ConsumoForaPonta) ConsumoForaPonta = result.value;
   672→                break;
   673→              }
   674→            }
   675→          }
   676→        }
   677→      }
   678→    }
   679→
   680→    // Strategy 3: findNumNearLine with validation (value must be >= 10 kWh for consumption)
   681→    if (!ConsumoPonta) {
   682→      const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   683→      if (pontaIdx >= 0) {
   684→        const val = findNumNearLine(lines, pontaIdx, 1, 8, false);
   685→        if (val && parseBR(val) >= 10) ConsumoPonta = val;
   686→      }
   687→    }
   688→    if (!ConsumoForaPonta) {
   689→      const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   690→      if (fpIdx >= 0) {
   691→        const val = findNumNearLine(lines, fpIdx, 1, 8, false);
   692→        if (val && parseBR(val) >= 10) ConsumoForaPonta = val;
   693→      }
   694→    }
   695→    // Strategy 3b: findIntNearLine — for consumption displayed as integers (e.g., "1.346" or "12.929")
   696→    // Many PDFs (especially Elektro) show consumption kWh as integers without decimal places.
   697→    // The findNumNearLine above only matches BR numbers with ,XX decimal format.
   698→    if (!ConsumoPonta) {
   699→      const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   700→      if (pontaIdx >= 0) {
   701→        const val = findIntNearLine(lines, pontaIdx, 1, 8);
   702→        if (val && parseBR(val) >= 10) ConsumoPonta = val;
   703→      }
   704→    }
   705→    if (!ConsumoForaPonta) {
   706→      const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   707→      if (fpIdx >= 0) {
   708→        const val = findIntNearLine(lines, fpIdx, 1, 8);
   709→        if (val && parseBR(val) >= 10) ConsumoForaPonta = val;
   710→      }
   711→    }
   712→    // Strategy 3c: Garbled pdf-parse format where consumption is merged with KWH unit
   713→    // e.g., "1.346KWH0,61783822,70" → ConsumoPonta = 1.346
   714→    // e.g., "12.929KWH0,400305.171,60" → ConsumoForaPonta = 12.929
   715→    if (!ConsumoPonta || !ConsumoForaPonta) {
   716→      for (const line of lines) {
   717→        // Match "NNN.NNNKWH" pattern (integer BR number followed by KWH)
   718→        const garbledPonta = line.match(/^(\d{1,2}(?:\.\d{3})+)KWH/i);
   719→        if (garbledPonta && parseBR(garbledPonta[1]) >= 10) {
   720→          // Check if this line is near a "Consumo em kWh - Ponta" label or
   721→          // appears in the "Itens da Fatura" section
   722→          if (!ConsumoPonta) ConsumoPonta = garbledPonta[1];
   723→          continue;
   724→        }
   725→        const garbledFora = line.match(/^(\d{1,2}(?:\.\d{3})+)KWH/i);
   726→        if (garbledFora && parseBR(garbledFora[1]) >= 10) {
   727→          // If we already set Ponta from this strategy, this must be Fora Ponta
   728→          if (ConsumoPonta && !ConsumoForaPonta) ConsumoForaPonta = garbledFora[1];
   729→        }
   730→      }
   731→    }
   732→    // Strategy 3d: Garbled format with integer followed by KWH on same line
   733→    // Search all lines for "NNNKWH" or "N.NNNKWH" patterns after "Itens da Fatura"
   734→    if ((!ConsumoPonta || !ConsumoForaPonta) && isSplitTariff) {
   735→      const itensIdx = lines.findIndex(l => /Itens\s+da\s+Fatura/i.test(l));
   736→      if (itensIdx >= 0) {
   737→        const consumptionCandidates: string[] = [];
   738→        for (let i = itensIdx + 1; i < Math.min(itensIdx + 30, lines.length); i++) {
   739→          // Match "1.346KWH" or "12.929KWH" or "1346KWH"
   740→          const m = lines[i].match(/^(\d{1,2}(?:\.\d{3})+|\d{3,6})KWH/i);
   741→          if (m && parseBR(m[1]) >= 10) {
   742→            consumptionCandidates.push(m[1]);
   743→          }
   744→        }
   745→        if (consumptionCandidates.length >= 2) {
   746→          if (!ConsumoPonta) ConsumoPonta = consumptionCandidates[0];
   747→          if (!ConsumoForaPonta) ConsumoForaPonta = consumptionCandidates[1];
   748→        } else if (consumptionCandidates.length === 1) {
   749→          // Only one found — if Ponta is already set from another strategy, this is Fora Ponta
   750→          if (ConsumoPonta && !ConsumoForaPonta) ConsumoForaPonta = consumptionCandidates[0];
   751→          else if (!ConsumoPonta) ConsumoPonta = consumptionCandidates[0];
   752→        }
   753→      }
   754→    }
   755→    // Intermediária (Tarifa Branca / 3-period AZUL)
   756→    const interIdx = findLine(lines, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   757→    if (interIdx >= 0) ConsumoIntermediaria = findNumNearLine(lines, interIdx, 1, 8, false);
   758→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)(?!.*Intermedi)/i);
   759→    ValorConsumoIntermediaria = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   760→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   761→  } else {
   762→    for (const line of lines) {
   763→      // Only match "KWH" + space + number (e.g., "KWH 2829")
   764→      // Do NOT match "KWH1.198,96" which is a Verde/Azul Ponta-only value
   765→      const km = line.match(/^KWH\s+([\d.,]+)$/i);
   766→      if (km) { ConsumoTotal = km[1]; break; }
   767→    }
   768→    if (!ConsumoTotal) {
   769→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   770→      if (consIdx >= 0) {
   771→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   772→          const line = lines[i];
   773→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   774→          if (/[A-Za-z]{4,}/.test(line)) continue;
   775→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   776→          if (m) { ConsumoTotal = m[1]; break; }
   777→        }
   778→      }
   779→    }
   780→    // ELEKTRO/SANESUL pattern: "CONSUMO TE" / "CONSUMO TUSD" labels followed by kWh units and consumption value
   781→    // e.g., Line 22: "CONSUMO TE", Line 23: "CONSUMO TUSD", Line 25: "kWh", Line 28: "4.963"
   782→    if (!ConsumoTotal) {
   783→      for (let i = 0; i < lines.length - 3; i++) {
   784→        if (/^CONSUMO\s+(TE|TUSD)$/i.test(lines[i])) {
   785→          // Found CONSUMO TE/TUSD label — look for the first numeric value after kWh units
   786→          const afterLabels = lines.slice(i + 1, Math.min(i + 15, lines.length));
   787→          // Find the first numeric value that looks like a consumption number (e.g., "4.963" or "4963")
   788→          for (const al of afterLabels) {
   789→            // Skip unit labels like "kWh", "kW"
   790→            if (/^(kWh|kW)$/i.test(al)) continue;
   791→            // Match a number like "4.963" (with dot as thousands separator) or "4963"
   792→            const m = al.match(/^(\d{1,2}\.\d{3}|\d{3,6})$/);
   793→            if (m) {
   794→              ConsumoTotal = m[1];
   795→              break;
   796→            }
   797→          }
   798→          if (ConsumoTotal) break;
   799→        }
   800→      }
   801→    }
   802→    // ENERGISA DANF3E pattern: "Consumo kWh" label in history section, followed by
   803→    // tributos section headers ("Tributo", "Valor", "Alíquota", "Base de", "Cálc.(R$)"),
   804→    // then the actual consumption values as standalone BR-format money values.
   805→    // e.g., L119: "Consumo kWh" ... L127: "Tributo" ... L134: "39,00" L135: "136,00" ...
   806→    // The first 2-3 money values after "Cálc.(R$)" are the individual period consumptions.
   807→    if (!ConsumoTotal && isSplitTariff) {
   808→      const consumoKwhIdx = findLine(lines, /^Consumo\s+kWh$/i);
   809→      if (consumoKwhIdx >= 0) {
   810→        // Find "Cálc.(R$)" header after "Consumo kWh"
   811→        const calcIdx = lines.slice(consumoKwhIdx).findIndex(l => /C[aá]lc\.\(R\$\)/i.test(l));
   812→        if (calcIdx >= 0) {
   813→          const afterCalc = lines.slice(consumoKwhIdx + calcIdx + 1, Math.min(consumoKwhIdx + calcIdx + 15, lines.length));
   814→          const consumptionValues: string[] = [];
   815→          for (const line of afterCalc) {
   816→            // Match standalone BR-format money value (e.g., "39,00", "136,00", "987,00")
   817→            const m = line.match(/^(\d{1,3}(?:\.\d{3})*,\d{2})$/);
   818→            if (m) {
   819→              const num = parseBR(m[1]);
   820→              // Consumption values are typically between 1 and 200,000 kWh
   821→              if (num >= 1 && num <= 200000) {
   822→                consumptionValues.push(m[1]);
   823→                if (consumptionValues.length >= 3) break;
   824→              }
   825→            }
   826→          }
   827→          // Assign individual consumption values to periods
   828→          if (consumptionValues.length >= 2) {
   829→            // For Tarifa Branca (3 periods): Ponta, Intermediária, Fora Ponta
   830→            // For AZUL/VERDE (2 periods): Ponta, Fora Ponta
   831→            // The values appear in order: Ponta, Intermediária (if Branca), Fora Ponta
   832→            if (!ConsumoPonta) ConsumoPonta = consumptionValues[0];
   833→            if (consumptionValues.length >= 3) {
   834→              if (!ConsumoIntermediaria) ConsumoIntermediaria = consumptionValues[1];
   835→              if (!ConsumoForaPonta) ConsumoForaPonta = consumptionValues[2];
   836→            } else {
   837→              if (!ConsumoForaPonta) ConsumoForaPonta = consumptionValues[1];
   838→            }
   839→          }
   840→        }
   841→      }
   842→    }
   843→    // ENERGISA pattern: compressed "Ponta" line contains consumption kWh
   844→    // e.g., "Ponta12.771,0012.702,0040,002,500,000,002.829,002.829,00"
   845→    //   → numbers: [12771.00, 12702.00, 40.00, 2.50, 0.00, 2829.00, 2829.00]
   846→    //   → second-to-last = 2.829,00 = consumption kWh
   847→    // e.g., "Ponta23.484,0023.211,001,00273,00273,00"
   848→    //   → numbers: [23484.00, 23211.00, 1.00, 273.00, 273.00]
   849→    //   → second-to-last = 273,00 = consumption kWh
   850→    // ENERGISA Convencional: compressed "Ponta" line contains consumption kWh
   851→    // e.g., "Ponta12.771,0012.702,0040,002,500,000,002.829,002.829,00"
   852→    //   → numbers: [12771.00, 12702.00, 40.00, 2.50, 0.00, 2829.00, 2829.00]
   853→    //   → second-to-last = 2.829,00 = consumption kWh
   854→    // e.g., "Ponta23.484,0023.211,001,00273,00273,00"
   855→    //   → numbers: [23484.00, 23211.00, 1.00, 273.00, 273.00]
   856→    //   → second-to-last = 273,00 = consumption kWh
   857→    // Also matches "KWHPonta2.264,0039.862,00..." format
   858→    if (!ConsumoTotal) {
   859→      for (const line of lines) {
   860→        // Match "Ponta..." or "KWHPonta..." at start of line
   861→        const pontaMatch = line.match(/^(?:KWH)?Ponta([\d.,]+)/);
   862→        if (pontaMatch) {
   863→          // Extract all BR-format numbers (X.XXX,XX or XXX,XX) from after "Ponta"
   864→          const nums: string[] = [];
   865→          const pontaIdx = line.indexOf('Ponta');
   866→          const rest = line.substring(pontaIdx + 5); // skip "Ponta"
   867→          const numRegex = /(\d{1,2}(?:\.\d{3})*,\d{2})/g;
   868→          let nm;
   869→          while ((nm = numRegex.exec(rest)) !== null) {
   870→            nums.push(nm[1]);
   871→          }
   872→          if (nums.length >= 2) {
   873→            // "KWHPonta..." format: first number is consumption kWh
   874→            // "Ponta..." format: second-to-last number is consumption kWh
   875→            if (line.startsWith('KWH')) {
   876→              ConsumoTotal = nums[0];
   877→            } else {
   878→              ConsumoTotal = nums[nums.length - 2];
   879→            }
   880→          } else if (nums.length === 1) {
   881→            ConsumoTotal = nums[0];
   882→          }
   883→          break;
   884→        }
   885→      }
   886→    }
   887→    // ENERGISA Verde/Azul: extract consumption from "KWH"+value and compressed "Energia ativa em kWh" data
   888→    // Pattern A: "KWH1.198,96" → Ponta consumption = 1.198,96
   889→    // Pattern B: After "W9...Ponta/Fora Ponta" + "Energia ativa em kWh" → compressed data line
   890→    //   e.g., "401198,963105,09575,852" → after prefix "40", first number = 1198,96 → 1.198,96
   891→    //   We determine the prefix length by finding the known Ponta KWH value in the compressed data
   892→    if (!ConsumoTotal) {
   893→      const kwhValues: string[] = [];
   894→      // Pattern A: "KWH" immediately followed by a BR number (e.g., "KWH1.198,96")
   895→      let pontaKwhFromLabel = '';
   896→      for (const line of lines) {
   897→        const km = line.match(/^KWH(\d{1,2}(?:\.\d{3})*,\d{2})$/i);
   898→        if (km) { kwhValues.push(km[1]); pontaKwhFromLabel = km[1]; }
   899→      }
   900→      // Pattern B: Parse compressed data after "Energia ativa em kWh" for each period (Ponta/Fora Ponta)
   901→      // Find all (periodHeader + "Energia ativa em kWh" + compressedData) triples
   902→      const periodSections: { period: string; data: string }[] = [];
   903→      for (let i = 0; i < lines.length - 2; i++) {
   904→        if (/Ponta$/i.test(lines[i]) && !/Fora/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   905→          periodSections.push({ period: 'Ponta', data: lines[i + 2] });
   906→        }
   907→        if (/Fora\s*Ponta$/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   908→          periodSections.push({ period: 'ForaPonta', data: lines[i + 2] });
   909→        }
   910→      }
   911→      // If we have the Ponta KWH from the label, use it to determine prefix length
   912→      // Then apply same prefix length to Fora Ponta compressed data
   913→      if (pontaKwhFromLabel && periodSections.length >= 1) {
   914→        const pontaSection = periodSections.find(s => s.period === 'Ponta');
   915→        const foraSection = periodSections.find(s => s.period === 'ForaPonta');
   916→        if (pontaSection) {
   917→          // Find the Ponta KWH value (without dots) in the compressed data to determine prefix length
   918→          const pontaNoDot = pontaKwhFromLabel.replace(/\./g, '');
   919→          const prefixIdx = pontaSection.data.indexOf(pontaNoDot);
   920→          if (prefixIdx >= 0) {
   921→            const prefixLen = prefixIdx;
   922→            // Extract Fora Ponta consumption using same prefix length
   923→            if (foraSection && prefixLen < foraSection.data.length) {
   924→              const foraRest = foraSection.data.substring(prefixLen);
   925→              // Find the first number ending with ,XX (2 decimal places)
   926→              const commaPos = foraRest.indexOf(',');
   927→              if (commaPos >= 2 && commaPos <= 7) {
   928→                const foraNum = foraRest.substring(0, commaPos + 3); // include ,XX
   929→                // Add dots as thousands separator
   930→                const parts = foraNum.split(',');
   931→                const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   932→                const foraBR = intPart + ',' + parts[1];
   933→                if (!kwhValues.includes(foraBR)) kwhValues.push(foraBR);
   934→              }
   935→            }
   936→          }
   937→        }
   938→      }
   939→      // Also try: for each "Energia ativa em kWh" section, if we don't have the value yet,
   940→      // try to find it as a standalone BR number near the section
   941→      if (kwhValues.length === 0) {
   942→        for (let i = 0; i < lines.length - 1; i++) {
   943→          if (/^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i])) {
   944→            // Look for the consumption in the next line by trying different prefix lengths
   945→            const nextLine = lines[i + 1];
   946→            for (let prefix = 0; prefix < Math.min(4, nextLine.length); prefix++) {
   947→              const rest = nextLine.substring(prefix);
   948→              const commaPos = rest.indexOf(',');
   949→              if (commaPos >= 1 && commaPos <= 6) {
   950→                const numStr = rest.substring(0, commaPos + 3);
   951→                const parts = numStr.split(',');
   952→                if (parts.length === 2 && /^\d+$/.test(parts[0]) && /^\d{2}$/.test(parts[1])) {
   953→                  const val = parseInt(parts[0]);
   954→                  if (val >= 10 && val <= 999999) {
   955→                    const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   956→                    const brNum = intPart + ',' + parts[1];
   957→                    if (!kwhValues.includes(brNum)) kwhValues.push(brNum);
   958→                    break;
   959→                  }
   960→                }
   961→              }
   962→            }
   963→          }
   964→        }
   965→      }
   966→      if (kwhValues.length > 0) {
   967→        // Sum all period consumptions to get total
   968→        let totalKwh = 0;
   969→        for (const v of kwhValues) {
   970→          totalKwh += parseBR(v);
   971→        }
   972→        if (totalKwh > 0) ConsumoTotal = formatBR(totalKwh);
   973→      }
   974→    }
   975→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   976→  }
   977→
   978→  // ===== CONSUMO TOTAL: sum individual consumptions if not already set =====
   979→  // For split tariffs (AZUL/VERDE), ConsumoTotal may not have been extracted directly.
   980→  // Sum ConsumoPonta + ConsumoForaPonta + ConsumoIntermediaria to get the total.
   981→  if (!ConsumoTotal && (ConsumoPonta || ConsumoForaPonta || ConsumoIntermediaria)) {
   982→    const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
   983→    const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
   984→    const it = ConsumoIntermediaria ? parseBR(ConsumoIntermediaria) : 0;
   985→    const total = p + f + it;
   986→    if (total > 0) ConsumoTotal = formatBR(total);
   987→  }
   988→
   989→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   990→  let DemandaContratadaPonta = EMPTY;
   991→  let DemandaContratadaForaPonta = EMPTY;
   992→  let DemandaMedidaPonta = EMPTY;
   993→  let DemandaMedidaForaPonta = EMPTY;
   994→  let DemandaFaturadaPonta = EMPTY;
   995→  let DemandaFaturadaForaPonta = EMPTY;
   996→  let ValorDemandaPonta = EMPTY;
   997→  let ValorDemandaForaPonta = EMPTY;
   998→
   999→  if (isSplitTariff) {
  1000→    // ===== DEMANDA CONTRATADA EXTRACTION =====
  1001→    // Energisa PDFs show contracted demand in two locations:
  1002→    // 1) "Grandezas Contratadas" section: "Demanda ponta - kW  15,3" / "Demanda fora ponta - kW  30"
  1003→    // 2) Bottom line: "KW Ponta: 15,3  Fora Ponta: 30  TUSDG:"
  1004→    //
  1005→    // pdf-parse often garbles the column order, producing:
  1006→    //   "Demanda ponta - kW Demanda fora ponta - kW 30 15,3"  (values reversed)
  1007→    //   "30 Fora Ponta: 15,3 KW Ponta:"  (values before labels, or between labels)
  1008→    //
  1009→    // We try multiple strategies and use the first one that gives both values.
  1010→
  1011→    // Strategy 1: "Demanda ponta - kW <value>" on the SAME LINE (pdftotext-like format)
  1012→    const dpSameLine = flat.match(/Demanda\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
  1013→    const dfpSameLine = flat.match(/Demanda\s+fora\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
  1014→    if (dpSameLine && dfpSameLine) {
  1015→      DemandaContratadaPonta = dpSameLine[1].replace('.', ',');
  1016→      DemandaContratadaForaPonta = dfpSameLine[1].replace('.', ',');
  1017→    }
  1018→
  1019→    // Strategy 2: Detect pdf-parse column reordering in bottom line
  1020→    // When "Fora Ponta:" appears BEFORE "KW Ponta:" in the text, the values are
  1021→    // rearranged: "<fora_val> Fora Ponta: <ponta_val> KW Ponta:"
  1022→    // - Number BEFORE "Fora Ponta:" = Fora Ponta value
  1023→    // - Number BETWEEN "Fora Ponta:" and "KW Ponta:" = Ponta value
  1024→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
  1025→      const foraIdx = flat.search(/Fora\s+Ponta\s*:/i);
  1026→      const kwIdx = flat.search(/KW\s+Ponta\s*:/i);
  1027→      if (foraIdx >= 0 && kwIdx >= 0 && foraIdx < kwIdx) {
  1028→        // "Fora Ponta:" appears before "KW Ponta:" — this is the reordered pattern
  1029→        // Number between "Fora Ponta:" and "KW Ponta:" → Ponta value
  1030→        const between = flat.slice(foraIdx, kwIdx);
  1031→        const betweenMatch = between.match(/:\s*(\d+(?:[.,]\d+)?)/);
  1032→        if (betweenMatch && !DemandaContratadaPonta) {
  1033→          DemandaContratadaPonta = betweenMatch[1].replace('.', ',');
  1034→        }
  1035→        // Number before "Fora Ponta:" (within 25 chars) → Fora Ponta value
  1036→        const before = flat.slice(Math.max(0, foraIdx - 25), foraIdx);
  1037→        const beforeMatch = before.match(/(\d+(?:[.,]\d+)?)\s*$/);
  1038→        if (beforeMatch && !DemandaContratadaForaPonta) {
  1039→          DemandaContratadaForaPonta = beforeMatch[1].replace('.', ',');
  1040→        }
  1041→      } else if (kwIdx >= 0 && foraIdx >= 0 && kwIdx < foraIdx) {
  1042→        // Normal order: "KW Ponta:" appears before "Fora Ponta:" — values follow labels
  1043→        const kwAfter = flat.slice(kwIdx).match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
  1044→        const fpAfter = flat.slice(foraIdx).match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
  1045→        if (kwAfter && !DemandaContratadaPonta) DemandaContratadaPonta = kwAfter[1].replace('.', ',');
  1046→        if (fpAfter && !DemandaContratadaForaPonta) DemandaContratadaForaPonta = fpAfter[1].replace('.', ',');
  1047→      }
  1048→    }
  1049→
  1050→    // Strategy 3: "Demanda ponta/fora ponta - kW" labels with values on nearby lines
  1051→    // (pdf-parse puts labels and values on separate lines)
  1052→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
  1053→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
  1054→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
  1055→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
  1056→        // Collect numeric values from the lines after both labels
  1057→        const startLine = Math.max(dpLineIdx, dfpLineIdx) + 1;
  1058→        const numVal = /^(\d+(?:[.,]\d+)?)$/;
  1059→        const values: string[] = [];
  1060→        for (let i = startLine; i < Math.min(startLine + 5, lines.length); i++) {
  1061→          const m = lines[i].match(numVal);
  1062→          if (m) values.push(m[1].replace('.', ','));
  1063→          if (values.length >= 2) break;
  1064→        }
  1065→        if (values.length === 2) {
  1066→          // Use already-found values to disambiguate, or assume reverse order
  1067→          if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
  1068→            if (values[0] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[1];
  1069→            else if (values[1] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[0];
  1070→          } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
  1071→            if (values[0] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[1];
  1072→            else if (values[1] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[0];
  1073→          } else {
  1074→            // No disambiguation; assume pdf-parse reverse column order:
  1075→            // values[0] = Fora Ponta, values[1] = Ponta
  1076→            if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = values[0];
  1077→            if (!DemandaContratadaPonta) DemandaContratadaPonta = values[1];
  1078→          }
  1079→        }
  1080→      }
  1081→    }
  1082→
  1083→    // ===== DEMANDA TODOS OS PERÍODO (Horária Verde) =====
  1084→    // For Grupo A Horária Verde, the demand is the SAME for all periods.
  1085→    // The invoice may show "Demanda Todos os Período - kW <value>" or
  1086→    // just a single value between "Demanda ponta - kW" and "Demanda fora ponta - kW"
  1087→    // In this case, both DemandaContratadaPonta and DemandaContratadaForaPonta = same value.
  1088→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
  1089→      // Strategy 4a: "Demanda Todos os Período" explicit label
  1090→      const todosMatch = flat.match(/Demanda\s+Todos\s+(?:os\s+)?Per[ií]odo\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
  1091→      if (todosMatch) {
  1092→        const val = todosMatch[1].replace('.', ',');
  1093→        if (!DemandaContratadaPonta) DemandaContratadaPonta = val;
  1094→        if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = val;
  1095→      }
  1096→    }
  1097→
  1098→    // Strategy 4b: For VERDE tariff, if only one demand value found, assign to both
  1099→    // In Horária Verde, the demand is the same for Ponta and Fora Ponta
  1100→    const isVerde = /VERDE/i.test(ModalidadeTarifaria);
  1101→    if (isVerde) {
  1102→      if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
  1103→        DemandaContratadaPonta = DemandaContratadaForaPonta;
  1104→      } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
  1105→        DemandaContratadaForaPonta = DemandaContratadaPonta;
  1106→      }
  1107→    }
  1108→
  1109→    // Strategy 4c: Single value between "Demanda ponta - kW" and "Demanda fora ponta - kW"
  1110→    // When pdf-parse collapses both labels into adjacent lines with only one number between them
  1111→    if (!DemandaContratadaPonta && !DemandaContratadaForaPonta) {
  1112→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
  1113→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
  1114→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
  1115→        // Look for a single number between the two labels
  1116→        const start = Math.min(dpLineIdx, dfpLineIdx) + 1;
  1117→        const end = Math.max(dpLineIdx, dfpLineIdx);
  1118→        for (let i = start; i < end; i++) {
  1119→          const m = lines[i].match(/^(\d+(?:[.,]\d+)?)$/);
  1120→          if (m) {
  1121→            const val = m[1].replace('.', ',');
  1122→            DemandaContratadaPonta = val;
  1123→            DemandaContratadaForaPonta = val;
  1124→            break;
  1125→          }
  1126→        }
  1127→      }
  1128→    }
  1129→
  1130→    // Strategy 4d: "Demanda Todos" with value on next line
  1131→    if (!DemandaContratadaPonta && !DemandaContratadaForaPonta) {
  1132→      const todosLineIdx = lines.findIndex(l => /Demanda\s+Todos/i.test(l) && /Per[ií]odo/i.test(l));
  1133→      if (todosLineIdx >= 0) {
  1134→        // Check same line first
  1135→        const sameLine = lines[todosLineIdx].match(/(\d+(?:[.,]\d+)?)\s*$/);
  1136→        if (sameLine) {
  1137→          const val = sameLine[1].replace('.', ',');
  1138→          DemandaContratadaPonta = val;
  1139→          DemandaContratadaForaPonta = val;
  1140→        } else {
  1141→          // Check next line
  1142→          for (let i = todosLineIdx + 1; i < Math.min(todosLineIdx + 4, lines.length); i++) {
  1143→            const m = lines[i].match(/^(\d+(?:[.,]\d+)?)$/);
  1144→            if (m) {
  1145→              const val = m[1].replace('.', ',');
  1146→              DemandaContratadaPonta = val;
  1147→              DemandaContratadaForaPonta = val;
  1148→              break;
  1149→            }

  1513→          const qtd = parseBR(item.quantidade);
  1514→          if (qtd > 0) grupoB_consumoKWh_total += qtd;
  1515→        }
  1516→        if (item.valor) {
  1517→          const val = parseBR(item.valor);
  1518→          if (val > 0) grupoB_consumoValor_total += val;
  1519→        }
  1520→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
  1521→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
  1522→          if (item.quantidade) ConsumoTotal = item.quantidade;
  1523→          if (item.valor) ValorConsumoForaPonta = item.valor;
  1524→        }
  1525→        // BRANCA/AZUL Intermediária — dedicated field
  1526→        else if (/Intermedi[áa]ria/i.test(lbl)) {
  1527→          if (item.quantidade) ConsumoIntermediaria = item.quantidade;
  1528→          if (item.valor) ValorConsumoIntermediaria = item.valor;
  1529→        }
  1530→        // Split tariff (AZUL/VERDE)
  1531→        else if (isPonta) {
  1532→          if (item.quantidade) ConsumoPonta = item.quantidade;
  1533→          if (item.valor) ValorConsumoPonta = item.valor;
  1534→        } else if (isFora) {
  1535→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
  1536→          if (item.valor) ValorConsumoForaPonta = item.valor;
  1537→        }
  1538→      }
  1539→      // Demanda Medida
  1540→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
  1541→        if (!isFora) {
  1542→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
  1543→          if (item.valor) ValorDemandaPonta = item.valor;
  1544→        } else {
  1545→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
  1546→          if (item.valor) ValorDemandaForaPonta = item.valor;
  1547→        }
  1548→      }
  1549→      // Demanda Todos os Período (Horária Verde) — same demand for Ponta and Fora Ponta
  1550→      else if (/Demanda\s+Todos/i.test(lbl) && /Per[ií]odo/i.test(lbl)) {
  1551→        if (item.quantidade) {
  1552→          if (!DemandaContratadaPonta) DemandaContratadaPonta = item.quantidade;
  1553→          if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = item.quantidade;
  1554→        }
  1555→      }
  1556→      // Ultrapassagem
  1557→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
  1558→        if (!isFora) {
  1559→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
  1560→          if (item.valor) ValorUltrapassagemPonta = item.valor;
  1561→        } else {
  1562→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
  1563→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
  1564→        }
  1565→      }
  1566→      // Demanda Potência Não Consumida
  1567→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
  1568→        if (!isFora) {
  1569→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
  1570→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
  1571→        } else {
  1572→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
  1573→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
  1574→        }
  1575→      }
  1576→      // Reativa Excedente
  1577→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
  1578→        if (!isFora && /Ponta/i.test(lbl)) {
  1579→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
  1580→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1581→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
  1582→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
  1583→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
  1584→        } else if (!isFora && !/Ponta/i.test(lbl)) {
  1585→          // Generic reativa (Convencional)
  1586→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
  1587→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1588→        }
  1589→      }
  1590→      // CIP Municipal
  1591→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
  1592→        if (item.valor) cipFromTable = item.valor;
  1593→      }
  1594→      // GDI items
  1595→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
  1596→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
  1597→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
  1598→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
  1599→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
  1600→        // Multiple mUC entries: accumulate
  1601→        if (item.quantidade) {
  1602→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1603→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1604→            : item.quantidade;
  1605→        }
  1606→        if (item.valor) {
  1607→          tbl_mUC_Valor = tbl_mUC_Valor
  1608→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1609→            : item.valor;
  1610→        }
  1611→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
  1612→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
  1613→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
  1614→        // Accumulate as the same GDI type as the last full-label row
  1615→        if (gdiSectionType === 'mUC') {
  1616→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1617→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1618→            : item.quantidade;
  1619→          if (item.valor) {
  1620→            tbl_mUC_Valor = tbl_mUC_Valor
  1621→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1622→              : item.valor;
  1623→          }
  1624→        } else if (gdiSectionType === 'oUC') {
  1625→          tbl_oUC_Qtd = tbl_oUC_Qtd
  1626→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
  1627→            : item.quantidade;
  1628→          if (item.valor) {
  1629→            tbl_oUC_Valor = tbl_oUC_Valor
  1630→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
  1631→              : item.valor;
  1632→          }
  1633→        }
  1634→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
  1635→        // Non-GDI, non-truncated row ends the GDI section
  1636→        inGDISection = false;
  1637→      }
  1638→    }
  1639→
  1640→    // Override CIP from table
  1641→    if (cipFromTable) CIP = cipFromTable;
  1642→
  1643→    // Use table GDI values if found, otherwise keep text-based fallback
  1644→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
  1645→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
  1646→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
  1647→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
  1648→  }
  1649→
  1650→  // ===== CONSUMO TOTAL: recompute from individual consumptions after table processing =====
  1651→  // For split tariffs (AZUL/VERDE/BRANCA), ConsumoTotal is the sum of all periods.
  1652→  // Table extraction may have set ConsumoPonta/ForaPonta/Intermediaria with correct values.
  1653→  // Always recompute if we have individual values, to ensure ConsumoTotal is accurate.
  1654→  if (ConsumoPonta || ConsumoForaPonta || ConsumoIntermediaria) {
  1655→    const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1656→    const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1657→    const it = ConsumoIntermediaria ? parseBR(ConsumoIntermediaria) : 0;
  1658→    const total = p + f + it;
  1659→    if (total > 0) {
  1660→      // Only override if the computed total is larger than the current value,
  1661→      // or if ConsumoTotal is empty (not set by direct extraction)
  1662→      const currentTotal = ConsumoTotal ? parseBR(ConsumoTotal) : 0;
  1663→      if (!ConsumoTotal || total > currentTotal) {
  1664→        ConsumoTotal = formatBR(total);
  1665→      }
  1666→    }
  1667→  }
  1668→
  1669→  // ===== Grupo B: calculate dedicated fields =====
  1670→  // For Subgrupo B (B1, B2, B3, B4), fill ConsumoGrupoB_kWh and ConsumoGrupoB_Valor
  1671→  const isGrupoB = /^B/i.test(Subgrupo);
  1672→  if (isGrupoB) {
  1673→    // ConsumoGrupoB_kWh: total consumption in kWh
  1674→    if (grupoB_consumoKWh_total > 0) {
  1675→      ConsumoGrupoB_kWh = formatBR(grupoB_consumoKWh_total);
  1676→    } else if (ConsumoTotal) {
  1677→      // Fallback: use ConsumoTotal if no table accumulation
  1678→      ConsumoGrupoB_kWh = ConsumoTotal;
  1679→    } else {
  1680→      // Fallback: sum ConsumoPonta + ConsumoForaPonta
  1681→      const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1682→      const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1683→      if (p + f > 0) ConsumoGrupoB_kWh = formatBR(p + f);
  1684→    }
  1685→    // ConsumoGrupoB_Valor: total consumption value in R$
  1686→    if (grupoB_consumoValor_total > 0) {
  1687→      ConsumoGrupoB_Valor = formatBR(grupoB_consumoValor_total);
  1688→    } else {
  1689→      // Fallback: sum ValorConsumoPonta + ValorConsumoForaPonta
  1690→      const vp = ValorConsumoPonta ? parseBR(ValorConsumoPonta) : 0;
  1691→      const vf = ValorConsumoForaPonta ? parseBR(ValorConsumoForaPonta) : 0;
  1692→      if (vp + vf > 0) ConsumoGrupoB_Valor = formatBR(vp + vf);
  1693→    }
  1694→  }
  1695→
  1696→  return {
  1697→    UC: cleanUC, Cidade, Endereco, Tipo, Mercado, Concessionaria,
  1698→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
  1699→    ModalidadeTarifaria, Subgrupo,
  1700→    ConsumoPonta, ConsumoIntermediaria, ConsumoForaPonta, ConsumoTotal, ConsumoGrupoB_kWh, ConsumoGrupoB_Valor,
  1701→    DemandaContratadaPonta, DemandaContratadaForaPonta,
  1702→    DemandaMedidaPonta, DemandaMedidaForaPonta,
  1703→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
  1704→    UltrapassagemPonta, UltrapassagemForaPonta,
  1705→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
  1706→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
  1707→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
  1708→    ValorConsumoPonta, ValorConsumoIntermediaria, ValorConsumoForaPonta,
  1709→    ValorDemandaPonta, ValorDemandaForaPonta,
  1710→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
  1711→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
  1712→    PIS, COFINS, ICMS, GDI, CIP,
  1713→    EnergiaReativaTotal,
  1714→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
  1715→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
  1716→    SaldoAcumuladoGD,
  1717→    ValorTotal,
  1718→  };
  1719→}

  1721→// ===================== GDI EXTRACTION =====================
  1722→
  1723→function extractGDI(lines: string[], flat: string) {
  1724→  const EMPTY = '';
  1725→  let nOUC = 0;
  1726→  let nMUC = 0;
  1727→  let firstGDI = -1;
  1728→
  1729→  for (let i = 0; i < lines.length; i++) {
  1730→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
  1731→      nOUC++;
  1732→      if (firstGDI < 0) firstGDI = i;
  1733→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
  1734→      nMUC++;
  1735→      if (firstGDI < 0) firstGDI = i;
  1736→    }
  1737→  }
  1738→
  1739→  if (nOUC === 0 && nMUC === 0) {
  1740→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
  1741→  }
  1742→
  1743→  const chaveIdx = (() => {
  1744→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
  1745→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
  1746→    }
  1747→    return -1;
  1748→  })();
  1749→
  1750→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
  1751→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
  1752→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
  1753→      sectionEnd = i; break;
  1754→    }
  1755→  }
  1756→  if (chaveIdx >= 0) {
  1757→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
  1758→  }
  1759→
  1760→  const allQtds: number[] = [];
  1761→  if (chaveIdx >= 0) {
  1762→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
  1763→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1764→      if (m) allQtds.push(parseBR(m[1]));
  1765→    }
  1766→  }
  1767→  if (allQtds.length === 0) {
  1768→    for (let i = firstGDI; i < sectionEnd; i++) {
  1769→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1770→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
  1771→      if (allQtds.length >= nOUC + nMUC) break;
  1772→    }
  1773→  }
  1774→
  1775→  const allRVals: number[] = [];
  1776→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
  1777→  const totalNeeded = nOUC + nMUC;
  1778→
  1779→  for (let i = firstGDI; i < rSearchEnd; i++) {
  1780→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
  1781→    if (m && allRVals.length < totalNeeded) {
  1782→      allRVals.push(parseBR('-' + m[2]));
  1783→    }
  1784→  }
  1785→  if (allRVals.length < totalNeeded) {
  1786→    for (let i = firstGDI; i < rSearchEnd; i++) {
  1787→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
  1788→      if (m && allRVals.length < totalNeeded) {
  1789→        allRVals.push(parseBR(m[1]));
  1790→      }
  1791→    }
  1792→  }
  1793→
  1794→  const oUCQtds = allQtds.splice(0, nOUC);
  1795→  const oUCRVals = allRVals.splice(0, nOUC);
  1796→
  1797→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
  1798→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
  1799→
  1800→  return {
  1801→    oUC_Qtd: sumQtd(oUCQtds),
  1802→    oUC_Valor: sumR(oUCRVals),
  1803→    mUC_Qtd: sumQtd(allQtds),
  1804→    mUC_Valor: sumR(allRVals),
  1805→  };
  1806→}

  1808→// ===================== NUMBER FORMATTING =====================
  1809→
  1810→function fmtBRNumber(n: number): string {
  1811→  const abs = Math.abs(n);
  1812→  const intPart = Math.floor(abs).toString();
  1813→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
  1814→  let formatted = '';
  1815→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
  1816→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
  1817→    formatted = intPart[i] + formatted;
  1818→  }
  1819→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
  1820→}
  1821→
  1822→function fmtSumBR(a: string, b: string): string {
  1823→  return fmtBRNumber(parseBR(a) + parseBR(b));
  1824→}

  1828→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
  1829→  const idx = findLine(lines, labelPattern);
  1830→  if (idx < 0) return EMPTY;
  1831→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
  1832→    const line = lines[i];
  1833→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
  1834→    const m = line.match(/([\d.]+,\d{2})/);
  1835→    if (m) return m[1];
  1836→  }
  1837→  return EMPTY;
  1838→}

  1840→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
  1841→  const idx = findLine(lines, labelPattern);
  1842→  if (idx < 0) return EMPTY;
  1843→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
  1844→  const moneyVals = findAllMoney(region);
  1845→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
  1846→  return EMPTY;
  1847→}

  1849→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
  1850→  const idx = findLine(lines, labelPattern);
  1851→  if (idx >= 0) {
  1852→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
  1853→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1854→      if (m) return m[1];
  1855→    }
  1856→  }
  1857→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
  1858→  return flatMatch ? flatMatch[1] : EMPTY;
  1859→}
