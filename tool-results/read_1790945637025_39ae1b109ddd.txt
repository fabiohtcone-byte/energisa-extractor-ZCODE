    81→function parseBR(s: string): number {
    82→  const neg = s.startsWith('-');
    83→  const clean = neg ? s.slice(1) : s;
    84→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    85→}
    86→
    87→function formatBR(n: number): string {
    88→  if (n === 0) return '0,00';
    89→  const neg = n < 0;
    90→  const abs = Math.abs(n);
    91→  const rounded = Math.round(abs * 100) / 100;
    92→  const parts = rounded.toFixed(2).split('.');
    93→  const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    94→  const result = intPart + ',' + parts[1];
    95→  return neg ? '-' + result : result;
    96→}

    98→function findLine(lines: string[], pattern: RegExp): number {
    99→  return lines.findIndex(l => pattern.test(l));

   102→function findAllMoney(text: string): string[] {
   103→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
   104→  if (!matches) return [];
   105→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
   106→}

   108→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
   109→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
   110→    if (Math.abs(i - startIdx) > maxDist) break;
   111→    const line = lines[i];
   112→    if (mustBeOnlyNum) {
   113→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
   114→      if (m) return m[1];
   115→    } else {
   116→      const m = line.match(/([\d.]+,\d{2})/);
   117→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   118→    }
   119→  }
   120→  return EMPTY;
   121→}

   123→// ===================== MAIN EXTRACTOR =====================
   124→
   125→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   126→  const tableData = tableExtraction?.items;
   127→  const tributosFromTable = tableExtraction
   128→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   129→    : null;
   130→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   131→  const flat = lines.join(' ');
   132→
   133→  // ===== CONCESSIONÁRIA =====
   134→  // Priority: detect the ISSUING concessionária (not the customer).
   135→  // For electricity invoices (DANF3E / CONTA DE ENERGIA), ENERGISA/ELEKTRO is the issuer.
   136→  // For water/sewage invoices, SANESUL is the issuer.
   137→  const Concessionaria = (() => {
   138→    const u = flat.toUpperCase();
   139→    const isEnergyInvoice = /DANF3E|ENERGIA\s*EL[eÉ]TRICA|DISTR\.\s*DE\s*ENERGIA/i.test(u);
   140→    if (isEnergyInvoice) {
   141→      // For energy invoices, the issuing utility is ENERGISA/ELEKTRO/NEOENERGIA
   142→      if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   143→      if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   144→      if (/ENERGISA/i.test(u)) return 'ENERGISA';
   145→    }
   146→    // For non-energy invoices or when no energy indicator found
   147→    if (/SANESUL/i.test(u)) return 'SANESUL';
   148→    if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   149→    if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   150→    if (/ENERGISA/i.test(u)) return 'ENERGISA';
   151→    return EMPTY;
   152→  })();
   153→
   154→  // ===== TIPO =====
   155→  const Tipo = (() => {
   156→    const u = flat.toUpperCase();
   157→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   158→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   159→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   160→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   161→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   162→    if (/RURAL/.test(u)) return 'RURAL';
   163→    return EMPTY;
   164→  })();
   165→
   166→  // ===== MERCADO =====
   167→  const Mercado = (() => {
   168→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   169→    if (m) {
   170→      const val = m[1].trim().replace(/\s+/g, ' ');
   171→      if (val.length < 50) return val.toUpperCase();
   172→    }
   173→    return EMPTY;
   174→  })();
   175→
   176→  // ===== CIDADE =====
   177→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   178→  const Cidade = (() => {
   179→    for (const line of lines.slice(0, 60)) {
   180→      // Match: CityName [StateAbbr] (AG: XX)
   181→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   182→      if (m) {
   183→        const name = m[1].trim().toUpperCase();
   184→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   185→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   186→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   187→      }
   188→    }
   189→    return EMPTY;
   190→  })();
   191→
   192→  // ===== UC =====
   193→  const UC = (() => {
   194→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   195→    const isCnpjCpf = (s: string) => {
   196→      const raw = s.replace(/[.\-\/]/g, '');
   197→      return raw.length === 14 || raw.length === 11;
   198→    };
   199→
   200→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   201→    for (const line of lines) {
   202→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   203→      if (m) {
   204→        const num = m[1].replace(/^0+/, '') || '0';
   205→        if (num.length >= 3) return num;
   206→      }
   207→    }
   208→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   209→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   210→    if (slashCtx) {
   211→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   212→      if (num.length >= 3) return num;
   213→    }
   214→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   215→    // e.g., "1.409.407.051-47" → "140940705147"
   216→    for (const line of lines) {
   217→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   218→      if (m && !isCnpjCpf(m[1])) return m[1];
   219→    }
   220→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   221→    for (const line of lines) {
   222→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   223→      if (m) return m[1];
   224→    }
   225→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   226→    for (const line of lines) {
   227→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   228→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   229→    }
   230→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   231→    for (const line of lines) {
   232→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   233→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   234→    }
   235→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   236→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   237→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   238→    if (ucMat) return ucMat[1];
   239→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   240→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   241→    if (ucAddr) return ucAddr[1];
   242→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   243→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   244→    if (ucInst) return ucInst[1];
   245→    // Strategy 10: UC label with number — 4-13 digits
   246→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   247→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   248→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   249→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   250→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   251→    // Strategy 12: broad search for 3-group dotted format
   252→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   253→    if (broad3) return broad3[1];
   254→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   255→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   256→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   257→    return EMPTY;
   258→  })();
   259→
   260→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   261→  const cleanUC = UC.replace(/[.\-]/g, '');
   262→
   263→  // ===== ENDEREÇO =====
   264→  const Endereco = (() => {
   265→    // Address line typically follows the city (AG: XX) line and starts with RUA, AV, AVE, LOT, ESTRADA, etc.
   266→    // Examples from real PDFs:
   267→    //   "RUA SALDANHA DA GAMA, 896 - 21.004.65.029401 LAD RE RA - 422 - 79370000"
   268→    //   "LOT RECANTO NUARA, S/N - 24.416.10.154000 PAT 01/REL01 2289 - 79190000"
   269→    //   "AVE MATO GROSSO DO SUL, SN - 12.029.07.090000 COX EAT 006 2243 - 79400000"
   270→    const STREET_RE = /^(RUA|AV[. ]|AVE |AVENIDA|ESTRADA|LOT |RODOVIA|TRAVESSA|ALAMEDA|PCA[. ]|PRACA)/i;
   271→
   272→    // Clean an address line: keep street + number, strip internal codes, keep CEP
   273→    const extractCleanAddress = (line: string): string => {
   274→      // Split at the first " - " that precedes an internal code pattern (XX.XXX.XX.XXXXXX)
   275→      const beforeCodes = line.split(/\s*-\s*\d{2}\.\d{3}/)[0];
   276→      let addr = (beforeCodes || line).trim();
   277→      // Remove trailing dash
   278→      addr = addr.replace(/\s*-\s*$/, '').trim();
   279→      // Append CEP if found at end of original line
   280→      const cepMatch = line.match(/(\d{5}-?\d{3})\s*$/);
   281→      if (cepMatch) addr += ' - CEP ' + cepMatch[1];
   282→      return addr;
   283→    };
   284→
   285→    // Strategy 1: Find address line after city (AG: XX) pattern
   286→    for (let i = 0; i < Math.min(60, lines.length); i++) {
   287→      if (/\(\s*AG\s*:\s*\d+\s*\)/.test(lines[i])) {
   288→        // Check next 3 lines for an address pattern
   289→        for (let j = i + 1; j <= Math.min(i + 3, lines.length - 1); j++) {
   290→          if (STREET_RE.test(lines[j])) {
   291→            return extractCleanAddress(lines[j]);
   292→          }
   293→        }
   294→      }
   295→    }
   296→
   297→    // Strategy 2: Find address line near CNPJ/CPF line (appears before it)
   298→    for (let i = 0; i < lines.length; i++) {
   299→      if (/CNPJ\/CPF/i.test(lines[i]) && i > 0) {
   300→        if (STREET_RE.test(lines[i - 1])) {
   301→          return extractCleanAddress(lines[i - 1]);
   302→        }
   303→      }
   304→    }
   305→
   306→    // Strategy 3: Look for "ENDEREÇO" label followed by address
   307→    for (let i = 0; i < lines.length; i++) {
   308→      if (/^ENDERE[ÇC]O$/i.test(lines[i].trim()) && i + 1 < lines.length) {
   309→        const next = lines[i + 1];
   310→        const cleanAddr = next.replace(/\s*-\s*(SA[IÍ]DA|ENTRADA)\s+PARA\s+[^-]+/i, '')
   311→          .replace(/\s*-\s*[A-ZÀ-Ú]+\s*\/\s*[A-Z]{2}\s*-\s*CEP\s+\d{5}-\d{3}/i, '').trim();
   312→        if (cleanAddr.length > 5) return cleanAddr;
   313→      }
   314→    }
   315→
   316→    return EMPTY;
   317→  })();
   318→
   319→  // ===== MÊS REFERÊNCIA =====
   320→  const MesReferencia = (() => {
   321→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   322→    const numToMonth: Record<string, string> = {
   323→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   324→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   325→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   326→    };
   327→    const capitalize = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
   328→
   329→    // Helper: get Vencimento date components for sanity checks
   330→    const venMoneyMatch = flat.match(/(\d{2})\/(\d{2})\/(\d{4})\s*R\$/);
   331→    const venMonth = venMoneyMatch ? parseInt(venMoneyMatch[2], 10) : 0;
   332→    const venYear = venMoneyMatch ? parseInt(venMoneyMatch[3], 10) : 0;
   333→
   334→    // Helper: check if a month/year is consistent with the Vencimento date
   335→    // (reference month is typically 1-2 months before the vencimento month)
   336→    const isConsistentWithVencimento = (month: number, year: number): boolean => {
   337→      if (!venMonth || !venYear) return true; // no vencimento to compare
   338→      const venTotal = venYear * 12 + venMonth;
   339→      const refTotal = year * 12 + month;
   340→      const diff = venTotal - refTotal;
   341→      return diff >= 1 && diff <= 3; // reference is 1-3 months before vencimento
   342→    };
   343→
   344→    // Strategy 1a: "Mês / Ano" pattern BEFORE date + R$ on the SAME LINE (most reliable)
   345→    // Energisa format: "Fevereiro / 202621/03/2026R$ 538,49" or "Junho / 2026  21/07/2026  R$ 13.356,50"
   346→    for (const line of lines) {
   347→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})\\s*\\d{2}/\\d{2}/\\d{4}\\s*R\\$`, 'i'));
   348→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   349→    }
   350→    // Strategy 1b: date + R$ pattern BEFORE "Mês / Ano" on the SAME LINE
   351→    // pdf-parse reordering: "21/07/2026R$ 6.980,48Junho / 2026"
   352→    for (const line of lines) {
   353→      const m = line.match(new RegExp(`\\d{2}/\\d{2}/\\d{4}\\s*R\\$\\s*[\\d.,]+\\s*(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   354→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   355→    }
   356→    // Strategy 2: "Mês / Ano" pattern on the SAME LINE as VENCIMENTO or NOTA FISCAL
   357→    for (const line of lines) {
   358→      if (/VENCIMENTO|NOTA FISCAL|Nº FATURA/i.test(line)) {
   359→        const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   360→        if (m) return `${capitalize(m[1])} / ${m[2]}`;
   361→      }
   362→    }
   363→    // Strategy 3: "Mês / Ano" on a line that also contains a date DD/MM/YYYY (likely the billing line)
   364→    for (const line of lines) {
   365→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   366→      if (m && /\d{2}\/\d{2}\/\d{4}/.test(line)) {
   367→        const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   368→        const monthNum = monthIdx + 1;
   369→        if (isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) {
   370→          return `${capitalize(m[1])} / ${m[2]}`;
   371→        }
   372→      }
   373→    }
   374→    // Strategy 4: "Mês / Ano" anywhere in flat text — skip "Desde mês/ano", "Referência:" history, etc.
   375→    const allNameMatches = [...flat.matchAll(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'gi'))];
   376→    for (const m of allNameMatches) {
   377→      // Check context: skip if preceded by "Desde", "Referência:", "Ref", or other informative words
   378→      const start = Math.max(0, m.index! - 20);
   379→      const context = flat.slice(start, m.index! + m[0].length).toLowerCase();
   380→      if (/desde|até|a partir|inform|dispon|referência|ref\b/.test(context)) continue;
   381→      // Also validate against Vencimento date
   382→      const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   383→      const monthNum = monthIdx + 1;
   384→      if (!isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) continue;
   385→      return `${capitalize(m[1])} / ${m[2]}`;
   386→    }
   387→    // Strategy 5: Use Leitura Atual date to derive the reference month
   388→    // pdf-parse sometimes garbles text, so try multiple patterns
   389→    const leituraPatterns = [
   390→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*(\d{2})\s*\/\s*(\d{4})/i,
   391→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*(\d{2})\s*\/\s*(\d{4})/i,
   392→    ];
   393→    for (const pat of leituraPatterns) {
   394→      const lm = flat.match(pat);
   395→      if (lm && numToMonth[lm[1]]) {
   396→        const monthNum = parseInt(lm[1], 10);
   397→        const year = parseInt(lm[2], 10);
   398→        if (isConsistentWithVencimento(monthNum, year)) {
   399→          return `${numToMonth[lm[1]]} / ${lm[2]}`;
   400→        }
   401→      }
   402→    }
   403→    // Strategy 6: numeric MM/YYYY near Vencimento date — with consistency check
   404→    if (venMoneyMatch) {
   405→      const venIdx = flat.indexOf(venMoneyMatch[0]);
   406→      const region = flat.slice(Math.max(0, venIdx - 150), venIdx + 50);
   407→      const numMatches = [...region.matchAll(/(\d{2})\s*\/\s*(\d{4})/g)];
   408→      for (const nm of numMatches) {
   409→        if (numToMonth[nm[1]]) {
   410→          const monthNum = parseInt(nm[1], 10);
   411→          const year = parseInt(nm[2], 10);
   412→          if (isConsistentWithVencimento(monthNum, year)) {
   413→            return `${numToMonth[nm[1]]} / ${nm[2]}`;
   414→          }
   415→        }
   416→      }
   417→    }
   418→    // Strategy 7: first valid MM/YYYY that is a month and consistent with Vencimento
   419→    const allNumMatches = [...flat.matchAll(/(\d{2})\s*[\/]\s*(\d{4})/g)];
   420→    for (const nm of allNumMatches) {
   421→      if (numToMonth[nm[1]]) {
   422→        const monthNum = parseInt(nm[1], 10);
   423→        const year = parseInt(nm[2], 10);
   424→        if (isConsistentWithVencimento(monthNum, year)) {
   425→          return `${numToMonth[nm[1]]} / ${nm[2]}`;
   426→        }
   427→      }
   428→    }
   429→    return EMPTY;
   430→  })();
   431→
   432→  // ===== ANO LEITURA =====
   433→  const AnoLeitura = (() => {
   434→    // Try multiple patterns for Leitura Atual date (pdf-parse can garble text)
   435→    const leituraYearPatterns = [
   436→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i,
   437→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*\d{2}\s*\/\s*(\d{4})/i,
   438→    ];
   439→    for (const pat of leituraYearPatterns) {
   440→      const m = flat.match(pat);
   441→      if (m) return m[1];
   442→    }
   443→    // Fallback: extract year from MesReferencia
   444→    if (MesReferencia) {
   445→      const ym = MesReferencia.match(/(\d{4})/);
   446→      if (ym) return ym[1];
   447→    }
   448→    return EMPTY;
   449→  })();
   450→
   451→  // ===== VENCIMENTO =====
   452→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   453→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   454→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   455→  const Vencimento = (() => {
   456→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   457→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   458→    if (venMoney) return venMoney[1];
   459→
   460→    // Strategy 2: R$ + money + date on same line (reversed order)
   461→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   462→    if (moneyVen) return moneyVen[1];
   463→
   464→    // Strategy 3: VENCIMENTO label nearby (original approach)
   465→    const venIdx = findLine(lines, /VENCIMENTO/i);
   466→    if (venIdx >= 0) {
   467→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   468→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   469→        if (m) return m[1];
   470→      }
   471→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   472→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   473→        if (m) return m[1];
   474→      }
   475→    }
   476→
   477→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   478→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   479→    if (apresMatch) return apresMatch[1];
   480→
   481→    return EMPTY;
   482→  })();
   483→
   484→  // ===== NOTA FISCAL =====
   485→  const NotaFiscal = (() => {
   486→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   487→    return m ? m[1] : EMPTY;
   488→  })();
   489→
   490→  // ===== MODALIDADE TARIFÁRIA =====
   491→  const ModalidadeTarifaria = (() => {
   492→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   493→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   494→    if (/AZUL/i.test(flat)) return 'AZUL';
   495→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   496→    if (/VERDE/i.test(flat)) return 'VERDE';
   497→    return EMPTY;
   498→  })();
   499→
   500→  // ===== SUBGRUPO =====
   501→  const Subgrupo = (() => {
   502→    // Look for "/A4" or "/B3" etc. in the classification line
   503→    // Must be a letter + single digit, right after "/"
   504→    const m = flat.match(/\/\s*([AB]\d)\s/);
   505→    if (m) return m[1].toUpperCase();
   506→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   507→    if (classIdx >= 0) {
   508→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   509→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   510→      if (m2) return m2[1].toUpperCase();
   511→    }
   512→    return EMPTY;
   513→  })();
   514→
   515→  const isSplitTariff = (/AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat))
   516→    || /Consumo\s+em\s+kWh\s*-\s*Intermedi/i.test(flat);
   517→
   518→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   519→  let ConsumoPonta = EMPTY;
   520→  let ConsumoIntermediaria = EMPTY;
   521→  let ConsumoForaPonta = EMPTY;
   522→  let ConsumoTotal = EMPTY;
   523→  let ConsumoGrupoB_kWh = EMPTY;
   524→  let ConsumoGrupoB_Valor = EMPTY;
   525→  let ValorConsumoPonta = EMPTY;
   526→  let ValorConsumoIntermediaria = EMPTY;
   527→  let ValorConsumoForaPonta = EMPTY;
   528→  // Accumulators for Grupo B consumption
   529→  let grupoB_consumoKWh_total = 0;
   530→  let grupoB_consumoValor_total = 0;
   531→
   532→  if (isSplitTariff) {
   533→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   534→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   535→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   536→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   537→    // Intermediária (Tarifa Branca / 3-period AZUL)
   538→    const interIdx = findLine(lines, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   539→    if (interIdx >= 0) ConsumoIntermediaria = findNumNearLine(lines, interIdx, 1, 8, false);
   540→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)(?!.*Intermedi)/i);
   541→    ValorConsumoIntermediaria = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   542→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   543→
   544→    // Fallback: KWH + value pattern for split tariff (e.g., "KWH1.052,34")
   545→    if (!ConsumoPonta) {
   546→      for (const line of lines) {
   547→        const km = line.match(/^KWH(\d{1,2}(?:\.\d{3})*,\d{2})$/i);
   548→        if (km) { ConsumoPonta = km[1]; break; }
   549→      }
   550→    }
   551→  } else {
   552→    for (const line of lines) {
   553→      // Only match "KWH" + space + number (e.g., "KWH 2829")
   554→      // Do NOT match "KWH1.198,96" which is a Verde/Azul Ponta-only value
   555→      const km = line.match(/^KWH\s+([\d.,]+)$/i);
   556→      if (km) { ConsumoTotal = km[1]; break; }
   557→    }
   558→    if (!ConsumoTotal) {
   559→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   560→      if (consIdx >= 0) {
   561→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   562→          const line = lines[i];
   563→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   564→          if (/[A-Za-z]{4,}/.test(line)) continue;
   565→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   566→          if (m) { ConsumoTotal = m[1]; break; }
   567→        }
   568→      }
   569→    }
   570→    // ELEKTRO/SANESUL pattern: "CONSUMO TE" / "CONSUMO TUSD" labels followed by kWh units and consumption value
   571→    // e.g., Line 22: "CONSUMO TE", Line 23: "CONSUMO TUSD", Line 25: "kWh", Line 28: "4.963"
   572→    if (!ConsumoTotal) {
   573→      for (let i = 0; i < lines.length - 3; i++) {
   574→        if (/^CONSUMO\s+(TE|TUSD)$/i.test(lines[i])) {
   575→          // Found CONSUMO TE/TUSD label — look for the first numeric value after kWh units
   576→          const afterLabels = lines.slice(i + 1, Math.min(i + 15, lines.length));
   577→          // Find the first numeric value that looks like a consumption number (e.g., "4.963" or "4963")
   578→          for (const al of afterLabels) {
   579→            // Skip unit labels like "kWh", "kW"
   580→            if (/^(kWh|kW)$/i.test(al)) continue;
   581→            // Match a number like "4.963" (with dot as thousands separator) or "4963"
   582→            const m = al.match(/^(\d{1,2}\.\d{3}|\d{3,6})$/);
   583→            if (m) {
   584→              ConsumoTotal = m[1];
   585→              break;
   586→            }
   587→          }
   588→          if (ConsumoTotal) break;
   589→        }
   590→      }
   591→    }
   592→    // ENERGISA DANF3E pattern: "Consumo kWh" label in history section, followed by
   593→    // tributos section headers ("Tributo", "Valor", "Alíquota", "Base de", "Cálc.(R$)"),
   594→    // then the actual consumption values as standalone BR-format money values.
   595→    // e.g., L119: "Consumo kWh" ... L127: "Tributo" ... L134: "39,00" L135: "136,00" ...
   596→    // The first 2-3 money values after "Cálc.(R$)" are the individual period consumptions.
   597→    if (!ConsumoTotal && isSplitTariff) {
   598→      const consumoKwhIdx = findLine(lines, /^Consumo\s+kWh$/i);
   599→      if (consumoKwhIdx >= 0) {
   600→        // Find "Cálc.(R$)" header after "Consumo kWh"
   601→        const calcIdx = lines.slice(consumoKwhIdx).findIndex(l => /C[aá]lc\.\(R\$\)/i.test(l));
   602→        if (calcIdx >= 0) {
   603→          const afterCalc = lines.slice(consumoKwhIdx + calcIdx + 1, Math.min(consumoKwhIdx + calcIdx + 15, lines.length));
   604→          const consumptionValues: string[] = [];
   605→          for (const line of afterCalc) {
   606→            // Match standalone BR-format money value (e.g., "39,00", "136,00", "987,00")
   607→            const m = line.match(/^(\d{1,3}(?:\.\d{3})*,\d{2})$/);
   608→            if (m) {
   609→              const num = parseBR(m[1]);
   610→              // Consumption values are typically between 1 and 200,000 kWh
   611→              if (num >= 1 && num <= 200000) {
   612→                consumptionValues.push(m[1]);
   613→                if (consumptionValues.length >= 3) break;
   614→              }
   615→            }
   616→          }
   617→          // Assign individual consumption values to periods
   618→          if (consumptionValues.length >= 2) {
   619→            // For Tarifa Branca (3 periods): Ponta, Intermediária, Fora Ponta
   620→            // For AZUL/VERDE (2 periods): Ponta, Fora Ponta
   621→            // The values appear in order: Ponta, Intermediária (if Branca), Fora Ponta
   622→            if (!ConsumoPonta) ConsumoPonta = consumptionValues[0];
   623→            if (consumptionValues.length >= 3) {
   624→              if (!ConsumoIntermediaria) ConsumoIntermediaria = consumptionValues[1];
   625→              if (!ConsumoForaPonta) ConsumoForaPonta = consumptionValues[2];
   626→            } else {
   627→              if (!ConsumoForaPonta) ConsumoForaPonta = consumptionValues[1];
   628→            }
   629→          }
   630→        }
   631→      }
   632→    }
   633→    // ENERGISA pattern: compressed "Ponta" line contains consumption kWh
   634→    // e.g., "Ponta12.771,0012.702,0040,002,500,000,002.829,002.829,00"
   635→    //   → numbers: [12771.00, 12702.00, 40.00, 2.50, 0.00, 2829.00, 2829.00]
   636→    //   → second-to-last = 2.829,00 = consumption kWh
   637→    // e.g., "Ponta23.484,0023.211,001,00273,00273,00"
   638→    //   → numbers: [23484.00, 23211.00, 1.00, 273.00, 273.00]
   639→    //   → second-to-last = 273,00 = consumption kWh
   640→    // ENERGISA Convencional: compressed "Ponta" line contains consumption kWh
   641→    // e.g., "Ponta12.771,0012.702,0040,002,500,000,002.829,002.829,00"
   642→    //   → numbers: [12771.00, 12702.00, 40.00, 2.50, 0.00, 2829.00, 2829.00]
   643→    //   → second-to-last = 2.829,00 = consumption kWh
   644→    // e.g., "Ponta23.484,0023.211,001,00273,00273,00"
   645→    //   → numbers: [23484.00, 23211.00, 1.00, 273.00, 273.00]
   646→    //   → second-to-last = 273,00 = consumption kWh
   647→    // Also matches "KWHPonta2.264,0039.862,00..." format
   648→    if (!ConsumoTotal) {
   649→      for (const line of lines) {
   650→        // Match "Ponta..." or "KWHPonta..." at start of line
   651→        const pontaMatch = line.match(/^(?:KWH)?Ponta([\d.,]+)/);
   652→        if (pontaMatch) {
   653→          // Extract all BR-format numbers (X.XXX,XX or XXX,XX) from after "Ponta"
   654→          const nums: string[] = [];
   655→          const pontaIdx = line.indexOf('Ponta');
   656→          const rest = line.substring(pontaIdx + 5); // skip "Ponta"
   657→          const numRegex = /(\d{1,2}(?:\.\d{3})*,\d{2})/g;
   658→          let nm;
   659→          while ((nm = numRegex.exec(rest)) !== null) {
   660→            nums.push(nm[1]);
   661→          }
   662→          if (nums.length >= 2) {
   663→            // "KWHPonta..." format: first number is consumption kWh
   664→            // "Ponta..." format: second-to-last number is consumption kWh
   665→            if (line.startsWith('KWH')) {
   666→              ConsumoTotal = nums[0];
   667→            } else {
   668→              ConsumoTotal = nums[nums.length - 2];
   669→            }
   670→          } else if (nums.length === 1) {
   671→            ConsumoTotal = nums[0];
   672→          }
   673→          break;
   674→        }
   675→      }
   676→    }
   677→    // ENERGISA Verde/Azul: extract consumption from "KWH"+value and compressed "Energia ativa em kWh" data
   678→    // Pattern A: "KWH1.198,96" → Ponta consumption = 1.198,96
   679→    // Pattern B: After "W9...Ponta/Fora Ponta" + "Energia ativa em kWh" → compressed data line
   680→    //   e.g., "401198,963105,09575,852" → after prefix "40", first number = 1198,96 → 1.198,96
   681→    //   We determine the prefix length by finding the known Ponta KWH value in the compressed data
   682→    if (!ConsumoTotal) {
   683→      const kwhValues: string[] = [];
   684→      // Pattern A: "KWH" immediately followed by a BR number (e.g., "KWH1.198,96")
   685→      let pontaKwhFromLabel = '';
   686→      for (const line of lines) {
   687→        const km = line.match(/^KWH(\d{1,2}(?:\.\d{3})*,\d{2})$/i);
   688→        if (km) { kwhValues.push(km[1]); pontaKwhFromLabel = km[1]; }
   689→      }
   690→      // Pattern B: Parse compressed data after "Energia ativa em kWh" for each period (Ponta/Fora Ponta)
   691→      // Find all (periodHeader + "Energia ativa em kWh" + compressedData) triples
   692→      const periodSections: { period: string; data: string }[] = [];
   693→      for (let i = 0; i < lines.length - 2; i++) {
   694→        if (/Ponta$/i.test(lines[i]) && !/Fora/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   695→          periodSections.push({ period: 'Ponta', data: lines[i + 2] });
   696→        }
   697→        if (/Fora\s*Ponta$/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   698→          periodSections.push({ period: 'ForaPonta', data: lines[i + 2] });
   699→        }
   700→      }
   701→      // If we have the Ponta KWH from the label, use it to determine prefix length
   702→      // Then apply same prefix length to Fora Ponta compressed data
   703→      if (pontaKwhFromLabel && periodSections.length >= 1) {
   704→        const pontaSection = periodSections.find(s => s.period === 'Ponta');
   705→        const foraSection = periodSections.find(s => s.period === 'ForaPonta');
   706→        if (pontaSection) {
   707→          // Find the Ponta KWH value (without dots) in the compressed data to determine prefix length
   708→          const pontaNoDot = pontaKwhFromLabel.replace(/\./g, '');
   709→          const prefixIdx = pontaSection.data.indexOf(pontaNoDot);
   710→          if (prefixIdx >= 0) {
   711→            const prefixLen = prefixIdx;
   712→            // Extract Fora Ponta consumption using same prefix length
   713→            if (foraSection && prefixLen < foraSection.data.length) {
   714→              const foraRest = foraSection.data.substring(prefixLen);
   715→              // Find the first number ending with ,XX (2 decimal places)
   716→              const commaPos = foraRest.indexOf(',');
   717→              if (commaPos >= 2 && commaPos <= 7) {
   718→                const foraNum = foraRest.substring(0, commaPos + 3); // include ,XX
   719→                // Add dots as thousands separator
   720→                const parts = foraNum.split(',');
   721→                const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   722→                const foraBR = intPart + ',' + parts[1];
   723→                if (!kwhValues.includes(foraBR)) kwhValues.push(foraBR);
   724→              }
   725→            }
   726→          }
   727→        }
   728→      }
   729→      // Also try: for each "Energia ativa em kWh" section, if we don't have the value yet,
   730→      // try to find it as a standalone BR number near the section
   731→      if (kwhValues.length === 0) {
   732→        for (let i = 0; i < lines.length - 1; i++) {
   733→          if (/^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i])) {
   734→            // Look for the consumption in the next line by trying different prefix lengths
   735→            const nextLine = lines[i + 1];
   736→            for (let prefix = 0; prefix < Math.min(4, nextLine.length); prefix++) {
   737→              const rest = nextLine.substring(prefix);
   738→              const commaPos = rest.indexOf(',');
   739→              if (commaPos >= 1 && commaPos <= 6) {
   740→                const numStr = rest.substring(0, commaPos + 3);
   741→                const parts = numStr.split(',');
   742→                if (parts.length === 2 && /^\d+$/.test(parts[0]) && /^\d{2}$/.test(parts[1])) {
   743→                  const val = parseInt(parts[0]);
   744→                  if (val >= 10 && val <= 999999) {
   745→                    const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   746→                    const brNum = intPart + ',' + parts[1];
   747→                    if (!kwhValues.includes(brNum)) kwhValues.push(brNum);
   748→                    break;
   749→                  }
   750→                }
   751→              }
   752→            }
   753→          }
   754→        }
   755→      }
   756→      if (kwhValues.length > 0) {
   757→        // Sum all period consumptions to get total
   758→        let totalKwh = 0;
   759→        for (const v of kwhValues) {
   760→          totalKwh += parseBR(v);
   761→        }
   762→        if (totalKwh > 0) ConsumoTotal = formatBR(totalKwh);
   763→      }
   764→    }
   765→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   766→  }
   767→
   768→  // ===== CONSUMO TOTAL: sum individual consumptions if not already set =====
   769→  // For split tariffs (AZUL/VERDE), ConsumoTotal may not have been extracted directly.
   770→  // Sum ConsumoPonta + ConsumoForaPonta + ConsumoIntermediaria to get the total.
   771→  if (!ConsumoTotal && (ConsumoPonta || ConsumoForaPonta || ConsumoIntermediaria)) {
   772→    const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
   773→    const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
   774→    const it = ConsumoIntermediaria ? parseBR(ConsumoIntermediaria) : 0;
   775→    const total = p + f + it;
   776→    if (total > 0) ConsumoTotal = formatBR(total);
   777→  }
   778→
   779→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   780→  let DemandaContratadaPonta = EMPTY;
   781→  let DemandaContratadaForaPonta = EMPTY;
   782→  let DemandaMedidaPonta = EMPTY;
   783→  let DemandaMedidaForaPonta = EMPTY;
   784→  let DemandaFaturadaPonta = EMPTY;
   785→  let DemandaFaturadaForaPonta = EMPTY;
   786→  let ValorDemandaPonta = EMPTY;
   787→  let ValorDemandaForaPonta = EMPTY;
   788→
   789→  if (isSplitTariff) {
   790→    // ===== DEMANDA CONTRATADA EXTRACTION =====
   791→    // Energisa PDFs show contracted demand in two locations:
   792→    // 1) "Grandezas Contratadas" section: "Demanda ponta - kW  15,3" / "Demanda fora ponta - kW  30"
   793→    // 2) Bottom line: "KW Ponta: 15,3  Fora Ponta: 30  TUSDG:"
   794→    //
   795→    // pdf-parse often garbles the column order, producing:
   796→    //   "Demanda ponta - kW Demanda fora ponta - kW 30 15,3"  (values reversed)
   797→    //   "30 Fora Ponta: 15,3 KW Ponta:"  (values before labels, or between labels)
   798→    //
   799→    // We try multiple strategies and use the first one that gives both values.
   800→
   801→    // Strategy 1: "Demanda ponta - kW <value>" on the SAME LINE (pdftotext-like format)
   802→    const dpSameLine = flat.match(/Demanda\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   803→    const dfpSameLine = flat.match(/Demanda\s+fora\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   804→    if (dpSameLine && dfpSameLine) {
   805→      DemandaContratadaPonta = dpSameLine[1].replace('.', ',');
   806→      DemandaContratadaForaPonta = dfpSameLine[1].replace('.', ',');
   807→    }
   808→
   809→    // Strategy 2: Detect pdf-parse column reordering in bottom line
   810→    // When "Fora Ponta:" appears BEFORE "KW Ponta:" in the text, the values are
   811→    // rearranged: "<fora_val> Fora Ponta: <ponta_val> KW Ponta:"
   812→    // - Number BEFORE "Fora Ponta:" = Fora Ponta value
   813→    // - Number BETWEEN "Fora Ponta:" and "KW Ponta:" = Ponta value
   814→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   815→      const foraIdx = flat.search(/Fora\s+Ponta\s*:/i);
   816→      const kwIdx = flat.search(/KW\s+Ponta\s*:/i);
   817→      if (foraIdx >= 0 && kwIdx >= 0 && foraIdx < kwIdx) {
   818→        // "Fora Ponta:" appears before "KW Ponta:" — this is the reordered pattern
   819→        // Number between "Fora Ponta:" and "KW Ponta:" → Ponta value
   820→        const between = flat.slice(foraIdx, kwIdx);
   821→        const betweenMatch = between.match(/:\s*(\d+(?:[.,]\d+)?)/);
   822→        if (betweenMatch && !DemandaContratadaPonta) {
   823→          DemandaContratadaPonta = betweenMatch[1].replace('.', ',');
   824→        }
   825→        // Number before "Fora Ponta:" (within 25 chars) → Fora Ponta value
   826→        const before = flat.slice(Math.max(0, foraIdx - 25), foraIdx);
   827→        const beforeMatch = before.match(/(\d+(?:[.,]\d+)?)\s*$/);
   828→        if (beforeMatch && !DemandaContratadaForaPonta) {
   829→          DemandaContratadaForaPonta = beforeMatch[1].replace('.', ',');
   830→        }
   831→      } else if (kwIdx >= 0 && foraIdx >= 0 && kwIdx < foraIdx) {
   832→        // Normal order: "KW Ponta:" appears before "Fora Ponta:" — values follow labels
   833→        const kwAfter = flat.slice(kwIdx).match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   834→        const fpAfter = flat.slice(foraIdx).match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   835→        if (kwAfter && !DemandaContratadaPonta) DemandaContratadaPonta = kwAfter[1].replace('.', ',');
   836→        if (fpAfter && !DemandaContratadaForaPonta) DemandaContratadaForaPonta = fpAfter[1].replace('.', ',');
   837→      }
   838→    }
   839→
   840→    // Strategy 3: "Demanda ponta/fora ponta - kW" labels with values on nearby lines
   841→    // (pdf-parse puts labels and values on separate lines)
   842→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   843→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
   844→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
   845→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
   846→        // Collect numeric values from the lines after both labels
   847→        const startLine = Math.max(dpLineIdx, dfpLineIdx) + 1;
   848→        const numVal = /^(\d+(?:[.,]\d+)?)$/;
   849→        const values: string[] = [];
   850→        for (let i = startLine; i < Math.min(startLine + 5, lines.length); i++) {
   851→          const m = lines[i].match(numVal);
   852→          if (m) values.push(m[1].replace('.', ','));
   853→          if (values.length >= 2) break;
   854→        }
   855→        if (values.length === 2) {
   856→          // Use already-found values to disambiguate, or assume reverse order
   857→          if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
   858→            if (values[0] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[1];
   859→            else if (values[1] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[0];
   860→          } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
   861→            if (values[0] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[1];
   862→            else if (values[1] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[0];
   863→          } else {
   864→            // No disambiguation; assume pdf-parse reverse column order:
   865→            // values[0] = Fora Ponta, values[1] = Ponta
   866→            if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = values[0];
   867→            if (!DemandaContratadaPonta) DemandaContratadaPonta = values[1];
   868→          }
   869→        }
   870→      }
   871→    }
   872→
   873→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   874→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   875→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   876→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   877→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   878→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   879→  }
   880→
   881→  // ===== ULTRAPASSAGEM =====
   882→  let UltrapassagemPonta = EMPTY;
   883→  let UltrapassagemForaPonta = EMPTY;
   884→  let ValorUltrapassagemPonta = EMPTY;
   885→  let ValorUltrapassagemForaPonta = EMPTY;
   886→  // ===== DEMANDA NÃO CONSUMIDA =====
   887→  let DemandaNaoConsumidaPonta = EMPTY;
   888→  let DemandaNaoConsumidaForaPonta = EMPTY;
   889→  let ValorNaoConsumidaPonta = EMPTY;
   890→  let ValorNaoConsumidaForaPonta = EMPTY;
   891→  if (isSplitTariff) {
   892→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   893→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   894→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   895→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   896→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   897→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   898→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   899→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   900→  }
   901→
   902→  // ===== REATIVA EXCEDENTE =====
   903→  let ReativaExcedentePonta = EMPTY;
   904→  let ReativaExcedenteForaPonta = EMPTY;
   905→  let EnergiaReativaTotal = EMPTY;
   906→  let ValorReativaExcedentePonta = EMPTY;
   907→  let ValorReativaExcedenteForaPonta = EMPTY;
   908→  if (isSplitTariff) {
   909→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   910→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   911→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   912→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   913→  } else {
   914→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   915→  }
   916→
   917→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   918→  // Priority: table extraction > text-based extraction
   919→  let PIS = EMPTY;
   920→  let COFINS = EMPTY;
   921→  let ICMS = EMPTY;
   922→
   923→  // Use tributos from table extraction if available (most reliable)
   924→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   925→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   926→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   927→
   928→  // Fallback: ELEKTRO/SANESUL pattern — ICMS, COFINS, PIS on consecutive lines,
   929→  // followed by base values, aliquotas (%), then actual tributo values.
   930→  // Example:
   931→  //   L52: ICMS
   932→  //   L53: COFINS
   933→  //   L54: PIS
   934→  //   L55: 5087,88   ← Base de cálculo ICMS
   935→  //   L56: 4222,95   ← Base de cálculo COFINS
   936→  //   L57: 4222,95   ← Base de cálculo PIS
   937→  //   L58: 17%       ← Alíquota ICMS
   938→  //   L59: 3,48%     ← Alíquota COFINS
   939→  //   L60: 0,75%     ← Alíquota PIS
   940→  //   L61: 864,93    ← Valor ICMS ✓
   941→  //   L62: 146,96    ← Valor COFINS ✓
   942→  //   L63: 31,67     ← Valor PIS ✓
   943→  if (!PIS || !COFINS || !ICMS) {
   944→    for (let i = 0; i < lines.length - 2; i++) {
   945→      if (/^ICMS$/i.test(lines[i]) && /^COFINS$/i.test(lines[i + 1]) && /^PIS$/i.test(lines[i + 2])) {
   946→        // Found the ICMS/COFINS/PIS label block
   947→        // Look for 3 money values after the labels (base values), then skip aliquotas, then get 3 more money values (actual values)
   948→        const afterLabels = lines.slice(i + 3, Math.min(i + 20, lines.length));
   949→        const moneyValues: string[] = [];
   950→        for (const line of afterLabels) {
   951→          const m = line.match(/^(\d[\d.]*,\d{2})$/);
   952→          if (m) moneyValues.push(m[1]);
   953→          if (moneyValues.length >= 6) break;
   954→        }
   955→        // If we have at least 6 money values, the last 3 are the actual tributo values
   956→        // (first 3 = bases, then 3 = actual values)
   957→        if (moneyValues.length >= 6) {
   958→          if (!ICMS) ICMS = moneyValues[3]; // 4th value = ICMS valor
   959→          if (!COFINS) COFINS = moneyValues[4]; // 5th value = COFINS valor
   960→          if (!PIS) PIS = moneyValues[5]; // 6th value = PIS valor
   961→        } else if (moneyValues.length >= 3) {
   962→          // If only 3 values found, they might be the base values
   963→          // Check if there are percentage values between the base and actual values
   964→          const afterBases = afterLabels.slice(3);
   965→          const actualValues: string[] = [];
   966→          for (const line of afterBases) {
   967→            if (/^\d+([.,]\d+)?%$/.test(line)) continue; // skip aliquotas
   968→            const m = line.match(/^(\d[\d.]*,\d{2})$/);
   969→            if (m) actualValues.push(m[1]);
   970→            if (actualValues.length >= 3) break;
   971→          }
   972→          if (actualValues.length >= 3) {
   973→            if (!ICMS) ICMS = actualValues[0];
   974→            if (!COFINS) COFINS = actualValues[1];
   975→            if (!PIS) PIS = actualValues[2];
   976→          }
   977→        }
   978→        break; // Only process the first match
   979→      }
   980→    }
   981→  }
   982→
   983→  // Fallback: ENERGISA DANF3E pattern — "PIS / ICMS / COFINS" on consecutive lines,
   984→  // with PIS and COFINS total values above, and ICMS total embedded in the line before.
   985→  // Example from 1069.PDF:
   986→  //   L124: 18,47        ← PIS total ✓
   987→  //   L125: 85,06        ← COFINS total ✓
   988→  //   L126: 0,6757       ← (alíquota PIS)
   989→  //   L127: 3,1122       ← (alíquota COFINS)
   990→  //   L128: 17,00        ← (alíquota ICMS)
   991→  //   L129: 2.733,23     ← (base ICMS)
   992→  //   L130: 2.733,23     ← (base ICMS)
   993→  //   L131: 3.293,07559,82 ← "baseICMS,ICMS_total" concatenated
   994→  //   L132: PIS
   995→  //   L133: ICMS
   996→  //   L134: COFINS
   997→  if (!PIS || !COFINS || !ICMS) {
   998→    for (let i = 0; i < lines.length - 2; i++) {
   999→      if (/^PIS$/i.test(lines[i]) && /^ICMS$/i.test(lines[i + 1]) && /^COFINS$/i.test(lines[i + 2])) {
  1000→        // Look backwards from the block. The PIS/COFINS values appear BEFORE the aliquotas.
  1001→        // Pattern: <pis_val> <cofins_val> <aliq_pis> <aliq_cofins> <aliq_icms> <base_icms> <base_icms> <concat_base_icms,icms_total>
  1002→        // The aliquotas look like "0,6757" or "3,1122" or "17,00"
  1003→        // Find the first aliquota-looking value going backwards, then PIS/COFINS are the 2 money values before it
  1004→        let firstAliquotIdx = -1;
  1005→        for (let j = i - 1; j >= Math.max(0, i - 12); j--) {
  1006→          if (/^\d+,\d{2,4}$/.test(lines[j]) && !/^\d+,\d{2}$/.test(lines[j])) {
  1007→            // This looks like an aliquota (3+ decimal places) but NOT a money value (2 decimal places)
  1008→            firstAliquotIdx = j;
  1009→            break;
  1010→          }
  1011→        }
  1012→        if (firstAliquotIdx >= 0) {
  1013→          // PIS and COFINS are the 2 money values before the first aliquota
  1014→          const pisCofinsValues: string[] = [];
  1015→          for (let j = firstAliquotIdx - 1; j >= Math.max(0, firstAliquotIdx - 5); j--) {
  1016→            const m = lines[j].match(/^(\d[\d.]*,\d{2})$/);
  1017→            if (m) {
  1018→              pisCofinsValues.unshift(m[1]);
  1019→              if (pisCofinsValues.length >= 2) break;
  1020→            }
  1021→          }
  1022→          if (pisCofinsValues.length >= 2) {
  1023→            if (!PIS) PIS = pisCofinsValues[0];
  1024→            if (!COFINS) COFINS = pisCofinsValues[1];
  1025→          }
  1026→        }
  1027→        // ICMS total is in the concatenated line just before the block (e.g., "3.293,07559,82")
  1028→        // The ICMS total is the LAST number in this concatenated string
  1029→        if (!ICMS) {
  1030→          for (let j = i - 1; j >= Math.max(0, i - 4); j--) {
  1031→            // Look for a line with two concatenated money values (no space between them)
  1032→            const catMatch = lines[j].match(/(\d[\d.]*,\d{2})(\d[\d.]*,\d{2})$/);
  1033→            if (catMatch) {
  1034→              ICMS = catMatch[2];
  1035→              break;
  1036→            }
  1037→          }
  1038→        }
  1039→        break; // Only process the first match
  1040→      }
  1041→    }
  1042→  }
  1043→
  1044→  // Fallback: text-based extraction for any missing tributo
  1045→  if (!PIS || !COFINS || !ICMS) {
  1046→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
  1047→    if (pisIdx >= 0) {
  1048→      const nums: string[] = [];
  1049→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
  1050→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
  1051→        if (lineNums) {
  1052→          for (const n of lineNums) {
  1053→            nums.push(n);
  1054→            if (nums.length >= 3) break;
  1055→          }
  1056→        }
  1057→      }
  1058→      if (nums.length >= 3) {
  1059→        if (!PIS) PIS = nums[0];
  1060→        if (!COFINS) COFINS = nums[1];
  1061→        if (!ICMS) ICMS = nums[2];
  1062→      }
  1063→    }
  1064→    if (!PIS && !COFINS) {
  1065→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
  1066→      if (combinedIdx >= 0) {
  1067→        const nums: string[] = [];
  1068→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
  1069→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1070→          if (m) nums.push(m[1]);
  1071→        }
  1072→        if (nums.length >= 3) {
  1073→          if (!PIS) PIS = nums[0];
  1074→          if (!COFINS) COFINS = nums[1];
  1075→          if (!ICMS) ICMS = nums[2];
  1076→        }
  1077→      }
  1078→    }
  1079→    if (!PIS || !COFINS) {
  1080→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
  1081→      if (flatMatch) {
  1082→        if (!PIS) PIS = flatMatch[1];
  1083→        if (!COFINS) COFINS = flatMatch[2];
  1084→      }
  1085→    }
  1086→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
  1087→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
  1088→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
  1089→  }
  1090→
  1091→  // ===== GDI =====
  1092→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
  1093→
  1094→  // ===== CIP (text-based) =====
  1095→  const CIP_text = (() => {
  1096→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
  1097→    if (cipIdx >= 0) {
  1098→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
  1099→      const money = findAllMoney(region);
  1100→      if (money.length > 0) return money[money.length - 1];
  1101→    }
  1102→    return findTributoValue(lines, flat, /\bCIP\b/i);
  1103→  })();
  1104→
  1105→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
  1106→  let {
  1107→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
  1108→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
  1109→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
  1110→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
  1111→  } = extractGDI(lines, flat);
  1112→
  1113→  // ===== SALDO ACUMULADO GD =====
  1114→  const SaldoAcumuladoGD = (() => {
  1115→    const m = flat.match(/Saldo\s*Acumulado\s*:\s*([\d.]+,\d{2})/i);
  1116→    if (m) return m[1];
  1117→    // Fallback: integer format like "14.100"
  1118→    const m2 = flat.match(/Saldo\s*Acumulado\s*:\s*(\d[\d.]*)/i);
  1119→    if (m2) return m2[1];
  1120→    return EMPTY;
  1121→  })();
  1122→
  1123→  // ===== VALOR TOTAL =====
  1124→  // Best pattern: R$ value on same line as vencimento date + mes referência
  1125→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
  1126→  const ValorTotal = (() => {
  1127→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
  1128→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
  1129→    if (venTotal) return venTotal[1];
  1130→
  1131→    // Strategy 2: R$ + value + mes referência on same line
  1132→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
  1133→    if (totalMes) return totalMes[1];
  1134→
  1135→    // Strategy 3: "TOTAL A PAGAR" with R$
  1136→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
  1137→    if (totalIdx >= 0) {
  1138→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
  1139→      if (m1) return m1[1];
  1140→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
  1141→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
  1142→        if (m2) return m2[1];
  1143→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1144→        if (m3) return m3[1];
  1145→      }
  1146→    }
  1147→
  1148→    // Strategy 4: Last resort - largest R$ value
  1149→    const lastMoney = findAllMoney(flat);
  1150→    if (lastMoney.length > 0) {
  1151→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
  1152→      return lastMoney[0];
  1153→    }
  1154→    return EMPTY;
  1155→  })();
  1156→
  1157→  // ===== TABLE DATA OVERRIDES =====
  1158→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
  1159→  let CIP = CIP_text;
  1160→
  1161→  if (tableData && tableData.length > 0) {
  1162→    let cipFromTable = EMPTY;
  1163→    // Reset GDI accumulators for clean table-based extraction
  1164→    let tbl_oUC_Qtd = '' as string;
  1165→    let tbl_oUC_Valor = '' as string;
  1166→    let tbl_mUC_Qtd = '' as string;
  1167→    let tbl_mUC_Valor = '' as string;
  1168→    // Track GDI section in table for rows with truncated/missing labels
  1169→    let inGDISection = false;
  1170→    let gdiSectionType = '' as string; // 'oUC' or 'mUC'
  1171→    let lastGDILabel = '' as string;
  1172→
  1173→    for (const item of tableData) {
  1174→      const lbl = item.label;
  1175→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
  1176→      const isPonta = /Ponta/i.test(lbl) && !isFora;
  1177→
  1178→      // Consumo - check Convencional (no Ponta/Fora) first, then split
  1179→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
  1180→        // Accumulate for Grupo B (regardless of Ponta/Fora/Intermediária)
  1181→        if (item.quantidade) {
  1182→          const qtd = parseBR(item.quantidade);
  1183→          if (qtd > 0) grupoB_consumoKWh_total += qtd;
  1184→        }
  1185→        if (item.valor) {
  1186→          const val = parseBR(item.valor);
  1187→          if (val > 0) grupoB_consumoValor_total += val;
  1188→        }
  1189→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
  1190→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
  1191→          if (item.quantidade) ConsumoTotal = item.quantidade;
  1192→          if (item.valor) ValorConsumoForaPonta = item.valor;
  1193→        }
  1194→        // BRANCA/AZUL Intermediária — dedicated field
  1195→        else if (/Intermedi[áa]ria/i.test(lbl)) {
  1196→          if (item.quantidade) ConsumoIntermediaria = item.quantidade;
  1197→          if (item.valor) ValorConsumoIntermediaria = item.valor;
  1198→        }
  1199→        // Split tariff (AZUL/VERDE)
  1200→        else if (isPonta) {
  1201→          if (item.quantidade) ConsumoPonta = item.quantidade;
  1202→          if (item.valor) ValorConsumoPonta = item.valor;
  1203→        } else if (isFora) {
  1204→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
  1205→          if (item.valor) ValorConsumoForaPonta = item.valor;
  1206→        }
  1207→      }
  1208→      // Demanda Medida
  1209→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
  1210→        if (!isFora) {
  1211→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
  1212→          if (item.valor) ValorDemandaPonta = item.valor;
  1213→        } else {
  1214→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
  1215→          if (item.valor) ValorDemandaForaPonta = item.valor;
  1216→        }
  1217→      }
  1218→      // Ultrapassagem
  1219→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
  1220→        if (!isFora) {
  1221→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
  1222→          if (item.valor) ValorUltrapassagemPonta = item.valor;
  1223→        } else {
  1224→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
  1225→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
  1226→        }
  1227→      }
  1228→      // Demanda Potência Não Consumida
  1229→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
  1230→        if (!isFora) {
  1231→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
  1232→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
  1233→        } else {
  1234→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
  1235→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
  1236→        }
  1237→      }
  1238→      // Reativa Excedente
  1239→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
  1240→        if (!isFora && /Ponta/i.test(lbl)) {
  1241→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
  1242→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1243→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
  1244→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
  1245→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
  1246→        } else if (!isFora && !/Ponta/i.test(lbl)) {
  1247→          // Generic reativa (Convencional)
  1248→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
  1249→          if (item.valor) ValorReativaExcedentePonta = item.valor;
  1250→        }
  1251→      }
  1252→      // CIP Municipal
  1253→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
  1254→        if (item.valor) cipFromTable = item.valor;
  1255→      }
  1256→      // GDI items
  1257→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
  1258→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
  1259→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
  1260→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
  1261→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
  1262→        // Multiple mUC entries: accumulate
  1263→        if (item.quantidade) {
  1264→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1265→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1266→            : item.quantidade;
  1267→        }
  1268→        if (item.valor) {
  1269→          tbl_mUC_Valor = tbl_mUC_Valor
  1270→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1271→            : item.valor;
  1272→        }
  1273→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
  1274→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
  1275→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
  1276→        // Accumulate as the same GDI type as the last full-label row
  1277→        if (gdiSectionType === 'mUC') {
  1278→          tbl_mUC_Qtd = tbl_mUC_Qtd
  1279→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
  1280→            : item.quantidade;
  1281→          if (item.valor) {
  1282→            tbl_mUC_Valor = tbl_mUC_Valor
  1283→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
  1284→              : item.valor;
  1285→          }
  1286→        } else if (gdiSectionType === 'oUC') {
  1287→          tbl_oUC_Qtd = tbl_oUC_Qtd
  1288→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
  1289→            : item.quantidade;
  1290→          if (item.valor) {
  1291→            tbl_oUC_Valor = tbl_oUC_Valor
  1292→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
  1293→              : item.valor;
  1294→          }
  1295→        }
  1296→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
  1297→        // Non-GDI, non-truncated row ends the GDI section
  1298→        inGDISection = false;
  1299→      }
  1300→    }
  1301→
  1302→    // Override CIP from table
  1303→    if (cipFromTable) CIP = cipFromTable;
  1304→
  1305→    // Use table GDI values if found, otherwise keep text-based fallback
  1306→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
  1307→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
  1308→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
  1309→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
  1310→  }
  1311→
  1312→  // ===== CONSUMO TOTAL: recompute from individual consumptions after table processing =====
  1313→  // For split tariffs (AZUL/VERDE/BRANCA), ConsumoTotal is the sum of all periods.
  1314→  // Table extraction may have set ConsumoPonta/ForaPonta/Intermediaria with correct values.
  1315→  // Always recompute if we have individual values, to ensure ConsumoTotal is accurate.
  1316→  if (ConsumoPonta || ConsumoForaPonta || ConsumoIntermediaria) {
  1317→    const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1318→    const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1319→    const it = ConsumoIntermediaria ? parseBR(ConsumoIntermediaria) : 0;
  1320→    const total = p + f + it;
  1321→    if (total > 0) {
  1322→      // Only override if the computed total is larger than the current value,
  1323→      // or if ConsumoTotal is empty (not set by direct extraction)
  1324→      const currentTotal = ConsumoTotal ? parseBR(ConsumoTotal) : 0;
  1325→      if (!ConsumoTotal || total > currentTotal) {
  1326→        ConsumoTotal = formatBR(total);
  1327→      }
  1328→    }
  1329→  }
  1330→
  1331→  // ===== Grupo B: calculate dedicated fields =====
  1332→  // For Subgrupo B (B1, B2, B3, B4), fill ConsumoGrupoB_kWh and ConsumoGrupoB_Valor
  1333→  const isGrupoB = /^B/i.test(Subgrupo);
  1334→  if (isGrupoB) {
  1335→    // ConsumoGrupoB_kWh: total consumption in kWh
  1336→    if (grupoB_consumoKWh_total > 0) {
  1337→      ConsumoGrupoB_kWh = formatBR(grupoB_consumoKWh_total);
  1338→    } else if (ConsumoTotal) {
  1339→      // Fallback: use ConsumoTotal if no table accumulation
  1340→      ConsumoGrupoB_kWh = ConsumoTotal;
  1341→    } else {
  1342→      // Fallback: sum ConsumoPonta + ConsumoForaPonta
  1343→      const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1344→      const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1345→      if (p + f > 0) ConsumoGrupoB_kWh = formatBR(p + f);
  1346→    }
  1347→    // ConsumoGrupoB_Valor: total consumption value in R$
  1348→    if (grupoB_consumoValor_total > 0) {
  1349→      ConsumoGrupoB_Valor = formatBR(grupoB_consumoValor_total);
  1350→    } else {
  1351→      // Fallback: sum ValorConsumoPonta + ValorConsumoForaPonta
  1352→      const vp = ValorConsumoPonta ? parseBR(ValorConsumoPonta) : 0;
  1353→      const vf = ValorConsumoForaPonta ? parseBR(ValorConsumoForaPonta) : 0;
  1354→      if (vp + vf > 0) ConsumoGrupoB_Valor = formatBR(vp + vf);
  1355→    }
  1356→  }
  1357→
  1358→  return {
  1359→    UC: cleanUC, Cidade, Endereco, Tipo, Mercado, Concessionaria,
  1360→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
  1361→    ModalidadeTarifaria, Subgrupo,
  1362→    ConsumoPonta, ConsumoIntermediaria, ConsumoForaPonta, ConsumoTotal, ConsumoGrupoB_kWh, ConsumoGrupoB_Valor,
  1363→    DemandaContratadaPonta, DemandaContratadaForaPonta,
  1364→    DemandaMedidaPonta, DemandaMedidaForaPonta,
  1365→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
  1366→    UltrapassagemPonta, UltrapassagemForaPonta,
  1367→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
  1368→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
  1369→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
  1370→    ValorConsumoPonta, ValorConsumoIntermediaria, ValorConsumoForaPonta,
  1371→    ValorDemandaPonta, ValorDemandaForaPonta,
  1372→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
  1373→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
  1374→    PIS, COFINS, ICMS, GDI, CIP,
  1375→    EnergiaReativaTotal,
  1376→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
  1377→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
  1378→    SaldoAcumuladoGD,
  1379→    ValorTotal,
  1380→  };
  1381→}

  1385→function extractGDI(lines: string[], flat: string) {
  1386→  const EMPTY = '';
  1387→  let nOUC = 0;
  1388→  let nMUC = 0;
  1389→  let firstGDI = -1;
  1390→
  1391→  for (let i = 0; i < lines.length; i++) {
  1392→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
  1393→      nOUC++;
  1394→      if (firstGDI < 0) firstGDI = i;
  1395→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
  1396→      nMUC++;
  1397→      if (firstGDI < 0) firstGDI = i;
  1398→    }
  1399→  }
  1400→
  1401→  if (nOUC === 0 && nMUC === 0) {
  1402→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
  1403→  }
  1404→
  1405→  const chaveIdx = (() => {
  1406→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
  1407→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
  1408→    }
  1409→    return -1;
  1410→  })();
  1411→
  1412→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
  1413→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
  1414→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
  1415→      sectionEnd = i; break;
  1416→    }
  1417→  }
  1418→  if (chaveIdx >= 0) {
  1419→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
  1420→  }
  1421→
  1422→  const allQtds: number[] = [];
  1423→  if (chaveIdx >= 0) {
  1424→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
  1425→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1426→      if (m) allQtds.push(parseBR(m[1]));
  1427→    }
  1428→  }
  1429→  if (allQtds.length === 0) {
  1430→    for (let i = firstGDI; i < sectionEnd; i++) {
  1431→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1432→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
  1433→      if (allQtds.length >= nOUC + nMUC) break;
  1434→    }
  1435→  }
  1436→
  1437→  const allRVals: number[] = [];
  1438→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
  1439→  const totalNeeded = nOUC + nMUC;
  1440→
  1441→  for (let i = firstGDI; i < rSearchEnd; i++) {
  1442→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
  1443→    if (m && allRVals.length < totalNeeded) {
  1444→      allRVals.push(parseBR('-' + m[2]));
  1445→    }
  1446→  }
  1447→  if (allRVals.length < totalNeeded) {
  1448→    for (let i = firstGDI; i < rSearchEnd; i++) {
  1449→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
  1450→      if (m && allRVals.length < totalNeeded) {
  1451→        allRVals.push(parseBR(m[1]));
  1452→      }
  1453→    }
  1454→  }
  1455→
  1456→  const oUCQtds = allQtds.splice(0, nOUC);
  1457→  const oUCRVals = allRVals.splice(0, nOUC);
  1458→
  1459→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
  1460→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
  1461→
  1462→  return {
  1463→    oUC_Qtd: sumQtd(oUCQtds),
  1464→    oUC_Valor: sumR(oUCRVals),
  1465→    mUC_Qtd: sumQtd(allQtds),
  1466→    mUC_Valor: sumR(allRVals),
  1467→  };
  1468→}

  1470→// ===================== NUMBER FORMATTING =====================

  1472→function fmtBRNumber(n: number): string {
  1473→  const abs = Math.abs(n);
  1474→  const intPart = Math.floor(abs).toString();
  1475→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
  1476→  let formatted = '';
  1477→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
  1478→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
  1479→    formatted = intPart[i] + formatted;
  1480→  }
  1481→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
  1482→}
  1483→
  1484→function fmtSumBR(a: string, b: string): string {
  1485→  return fmtBRNumber(parseBR(a) + parseBR(b));
  1486→}

  1490→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
  1491→  const idx = findLine(lines, labelPattern);
  1492→  if (idx < 0) return EMPTY;
  1493→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
  1494→    const line = lines[i];
  1495→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
  1496→    const m = line.match(/([\d.]+,\d{2})/);
  1497→    if (m) return m[1];
  1498→  }
  1499→  return EMPTY;
  1500→}

  1502→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
  1503→  const idx = findLine(lines, labelPattern);
  1504→  if (idx < 0) return EMPTY;
  1505→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
  1506→  const moneyVals = findAllMoney(region);
  1507→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
  1508→  return EMPTY;
  1509→}

  1511→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
  1512→  const idx = findLine(lines, labelPattern);
  1513→  if (idx >= 0) {
  1514→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
  1515→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1516→      if (m) return m[1];
  1517→    }
  1518→  }
  1519→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
  1520→  return flatMatch ? flatMatch[1] : EMPTY;
  1521→}
