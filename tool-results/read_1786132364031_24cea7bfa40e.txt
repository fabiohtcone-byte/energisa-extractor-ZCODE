     1→export interface EnergisaInvoiceData {
     2→  UC: string;
     3→  Cidade: string;
     4→  Tipo: string;
     5→  Mercado: string;
     6→  Concessionaria: string;
     7→  MesReferencia: string;
     8→  AnoLeitura: string;
     9→  Vencimento: string;
    10→  NotaFiscal: string;
    11→  ModalidadeTarifaria: string;
    12→  Subgrupo: string;
    13→  ConsumoPonta: string;
    14→  ConsumoForaPonta: string;
    15→  ConsumoTotal: string;
    16→  DemandaContratadaPonta: string;
    17→  DemandaContratadaForaPonta: string;
    18→  DemandaMedidaPonta: string;
    19→  DemandaMedidaForaPonta: string;
    20→  DemandaFaturadaPonta: string;
    21→  DemandaFaturadaForaPonta: string;
    22→  UltrapassagemPonta: string;
    23→  UltrapassagemForaPonta: string;
    24→  ReativaExcedentePonta: string;
    25→  ReativaExcedenteForaPonta: string;
    26→  ValorConsumoPonta: string;
    27→  ValorConsumoForaPonta: string;
    28→  ValorDemandaPonta: string;
    29→  ValorDemandaForaPonta: string;
    30→  ValorUltrapassagemPonta: string;
    31→  ValorUltrapassagemForaPonta: string;
    32→  ValorReativaExcedentePonta: string;
    33→  ValorReativaExcedenteForaPonta: string;
    34→  PIS: string;
    35→  COFINS: string;
    36→  ICMS: string;
    37→  GDI: string;
    38→  CIP: string;
    39→  EnergiaReativaTotal: string;
    40→  EnergiaAtvInjetadaGDIoUC_Qtd: string;
    41→  EnergiaAtvInjetadaGDIoUC_Valor: string;
    42→  EnergiaAtvInjetadaGDImUC_Qtd: string;
    43→  EnergiaAtvInjetadaGDImUC_Valor: string;
    44→  ValorTotal: string;
    45→}
    46→
    47→const EMPTY = '';
    48→
    49→// ===================== HELPERS =====================
    50→
    51→// Parse Brazilian number format: "1.135,00" → 1135.00, "-110,49" → -110.49
    52→function parseBR(s: string): number {
    53→  const neg = s.startsWith('-');
    54→  const clean = neg ? s.slice(1) : s;
    55→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    56→}
    57→
    58→function findLine(lines: string[], pattern: RegExp): number {
    59→  return lines.findIndex(l => pattern.test(l));
    60→}
    61→
    62→function findAllMoney(text: string): string[] {
    63→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
    64→  if (!matches) return [];
    65→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
    66→}
    67→
    68→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    69→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    70→    if (Math.abs(i - startIdx) > maxDist) break;
    71→    const line = lines[i];
    72→    if (mustBeOnlyNum) {
    73→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    74→      if (m) return m[1];
    75→    } else {
    76→      const m = line.match(/([\d.]+,\d{2})/);
    77→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
    78→    }
    79→  }
    80→  return EMPTY;
    81→}
    82→
    83→// ===================== MAIN EXTRACTOR =====================
    84→
    85→export function extractEnergisaData(rawText: string): EnergisaInvoiceData {
    86→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
    87→  const flat = lines.join(' ');
    88→
    89→  // ===== CONCESSIONÁRIA =====
    90→  // Full company name on its own line near the top
    91→  const Concessionaria = (() => {
    92→    for (const line of lines.slice(0, 20)) {
    93→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
    94→        return line.replace(/\s+/g, ' ');
    95→      }
    96→    }
    97→    return EMPTY;
    98→  })();
    99→
   100→  // ===== TIPO =====
   101→  const Tipo = (() => {
   102→    const u = flat.toUpperCase();
   103→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   104→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   105→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   106→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   107→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   108→    if (/RURAL/.test(u)) return 'RURAL';
   109→    return EMPTY;
   110→  })();
   111→
   112→  // ===== MERCADO =====
   113→  // Look for classification line: "SERVIÇO PÚBLICO / ÁGUA, ESGOTO E SANEAMENTO"
   114→  const Mercado = (() => {
   115→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   116→    if (m) {
   117→      const val = m[1].trim().replace(/\s+/g, ' ');
   118→      if (val.length < 50) return val.toUpperCase();
   119→    }
   120→    return EMPTY;
   121→  })();
   122→
   123→  // ===== CIDADE =====
   124→  // Pattern: "DOURADOS (AG: 10)"
   125→  const Cidade = (() => {
   126→    for (const line of lines.slice(0, 60)) {
   127→      const m = line.match(/([A-ZÀ-Ú]{4,25})\s*\(\s*AG\s*:\s*\d+\s*\)/);
   128→      if (m) {
   129→        const name = m[1].toUpperCase();
   130→        // Skip company names
   131→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA'];
   132→        if (!skip.includes(name)) return name;
   133→      }
   134→    }
   135→    return EMPTY;
   136→  })();
   137→
   138→  // ===== UC =====
   139→  // Pattern: dotted format X.XXX.XXX-XX on its own line or with "KWH" prefix
   140→  const UC = (() => {
   141→    // Strategy 1: line with only UC (e.g., "3.562.051-59", "43.210.051-68", "131.752.051-07")
   142→    for (const line of lines) {
   143→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   144→      if (m) return m[1];
   145→    }
   146→    // Strategy 2: UC embedded in short line
   147→    for (const line of lines) {
   148→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   149→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   150→    }
   151→    // Strategy 3: broad search
   152→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   153→    return broad ? broad[1] : EMPTY;
   154→  })();
   155→
   156→  // ===== MÊS REFERÊNCIA =====
   157→  // Pattern: "Junho / 2026" or "JUNHO / 2026"
   158→  const MesReferencia = (() => {
   159→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   160→    const numToMonth: Record<string, string> = {
   161→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   162→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   163→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   164→    };
   165→    // Named month pattern
   166→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   167→    if (nameMatch) {
   168→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   169→      return `${month} / ${nameMatch[2]}`;
   170→    }
   171→    // Numeric month pattern (e.g., "06/2026")
   172→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   173→    if (numMatch && numToMonth[numMatch[1]]) {
   174→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   175→    }
   176→    return EMPTY;
   177→  })();
   178→
   179→  // ===== ANO LEITURA =====
   180→  const AnoLeitura = (() => {
   181→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   182→    if (m) return m[1];
   183→    if (MesReferencia) {
   184→      const ym = MesReferencia.match(/(\d{4})/);
   185→      if (ym) return ym[1];
   186→    }
   187→    return EMPTY;
   188→  })();
   189→
   190→  // ===== VENCIMENTO =====
   191→  // ENERGISA PDFs often concatenate "VENCIMENTO" with other labels like "MATRÍCULA"
   192→  // The date may appear BEFORE or AFTER the label
   193→  const Vencimento = (() => {
   194→    // Strategy 1: find line containing VENCIMENTO (even concatenated), check nearby for date
   195→    const venIdx = findLine(lines, /VENCIMENTO/i);
   196→    if (venIdx >= 0) {
   197→      // Check lines BEFORE the label (ENERGISA often puts date before)
   198→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   199→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   200→        if (m) return m[1];
   201→      }
   202→      // Check lines AFTER the label
   203→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   204→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   205→        if (m) return m[1];
   206→      }
   207→      // Check same line
   208→      const sameLine = lines[venIdx].match(/(\d{2}\/\d{2}\/\d{4})/);
   209→      if (sameLine) return sameLine[1];
   210→    }
   211→    // Strategy 2: "TOTAL A PAGAR" area - date often nearby
   212→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   213→    if (totalIdx >= 0) {
   214→      for (let i = totalIdx - 3; i <= Math.min(totalIdx + 3, lines.length - 1); i++) {
   215→        if (i < 0) continue;
   216→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   217→        if (m) return m[1];
   218→      }
   219→    }
   220→    return EMPTY;
   221→  })();
   222→
   223→  // ===== NOTA FISCAL =====
   224→  const NotaFiscal = (() => {
   225→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   226→    return m ? m[1] : EMPTY;
   227→  })();
   228→
   229→  // ===== MODALIDADE TARIFÁRIA =====
   230→  const ModalidadeTarifaria = (() => {
   231→    // In the classification line: "MTC-CONVENCIONAL BAIXA TENSÃO / B3"
   232→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   233→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   234→    if (/AZUL/i.test(flat)) return 'AZUL';
   235→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   236→    if (/VERDE/i.test(flat)) return 'VERDE';
   237→    return EMPTY;
   238→  })();
   239→
   240→  // ===== SUBGRUPO =====
   241→  // Pattern: "/ B3" in classification line
   242→  const Subgrupo = (() => {
   243→    const m = flat.match(/\/(\s*[AB]\d?\s*)(?:\s|\/|$)/);
   244→    if (m) return m[1].trim().toUpperCase();
   245→    // Fallback: search in classification area
   246→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   247→    if (classIdx >= 0) {
   248→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   249→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   250→      if (m2) return m2[1].toUpperCase();
   251→    }
   252→    return EMPTY;
   253→  })();
   254→
   255→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   256→
   257→  // ===== CONSUMO =====
   258→  let ConsumoPonta = EMPTY;
   259→  let ConsumoForaPonta = EMPTY;
   260→  let ConsumoTotal = EMPTY;
   261→  let ValorConsumoPonta = EMPTY;
   262→  let ValorConsumoForaPonta = EMPTY;
   263→
   264→  if (isSplitTariff) {
   265→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   266→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   267→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   268→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   269→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   270→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   271→  } else {
   272→    // Convencional: look for "KWH1.044,00" or "KWH 1.044,00"
   273→    for (const line of lines) {
   274→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   275→      if (km) { ConsumoTotal = km[1]; break; }
   276→    }
   277→    // Fallback: "CONSUMO FATURADO" area
   278→    if (!ConsumoTotal) {
   279→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   280→      if (consIdx >= 0) {
   281→        // Look for numbers near CONSUMO FATURADO, but skip the month reference line
   282→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   283→          const line = lines[i];
   284→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   285→          if (/[A-Za-z]{4,}/.test(line)) continue;
   286→          const m = line.match(/^([\d.]+,\d{2})$/);
   287→          if (m) { ConsumoTotal = m[1]; break; }
   288→        }
   289→      }
   290→    }
   291→    // R$ value for consumo (Convencional)
   292→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   293→  }
   294→
   295→  // ===== DEMANDA (only split tariffs) =====
   296→  let DemandaContratadaPonta = EMPTY;
   297→  let DemandaContratadaForaPonta = EMPTY;
   298→  let DemandaMedidaPonta = EMPTY;
   299→  let DemandaMedidaForaPonta = EMPTY;
   300→  let DemandaFaturadaPonta = EMPTY;
   301→  let DemandaFaturadaForaPonta = EMPTY;
   302→  let ValorDemandaPonta = EMPTY;
   303→  let ValorDemandaForaPonta = EMPTY;
   304→  if (isSplitTariff) {
   305→    DemandaContratadaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Ponta(?![^\n]*Fora)/i);
   306→    DemandaContratadaForaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Fora\s*Ponta/i);
   307→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?![^\n]*Fora)/i);
   308→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   309→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?![^\n]*Fora)/i) || DemandaMedidaPonta;
   310→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   311→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?![^\n]*Fora)/i);
   312→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   313→  }
   314→
   315→  // ===== ULTRAPASSAGEM (only split tariffs) =====
   316→  let UltrapassagemPonta = EMPTY;
   317→  let UltrapassagemForaPonta = EMPTY;
   318→  let ValorUltrapassagemPonta = EMPTY;
   319→  let ValorUltrapassagemForaPonta = EMPTY;
   320→  if (isSplitTariff) {
   321→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   322→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   323→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   324→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   325→  }
   326→
   327→  // ===== REATIVA EXCEDENTE =====
   328→  let ReativaExcedentePonta = EMPTY;
   329→  let ReativaExcedenteForaPonta = EMPTY;
   330→  let EnergiaReativaTotal = EMPTY;
   331→  let ValorReativaExcedentePonta = EMPTY;
   332→  let ValorReativaExcedenteForaPonta = EMPTY;
   333→  if (isSplitTariff) {
   334→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   335→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   336→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   337→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   338→  } else {
   339→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   340→  }
   341→
   342→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   343→  // ENERGISA format (from real PDF):
   344→  // Line N: "PIS/"  Line N+1: "COFINS (R$)"  Line N+2: "18,47 85,09 236,20"
   345→  // Or on the same flat section: "PIS/COFINS (R$) 18,47 85,09 236,20"
   346→  let PIS = EMPTY;
   347→  let COFINS = EMPTY;
   348→  let ICMS = EMPTY;
   349→
   350→  (() => {
   351→    // Strategy 1: Find "PIS/" line (the label is split)
   352→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   353→    if (pisIdx >= 0) {
   354→      // Next line might be "COFINS (R$)" or values
   355→      // Check next few lines for the 3 consecutive numbers
   356→      const nums: string[] = [];
   357→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   358→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   359→        if (lineNums) {
   360→          for (const n of lineNums) {
   361→            nums.push(n);
   362→            if (nums.length >= 3) break;
   363→          }
   364→        }
   365→      }
   366→      if (nums.length >= 3) {
   367→        PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   368→      }
   369→    }
   370→    // Strategy 2: "PIS/COFINS" on same line
   371→    if (!PIS && !COFINS) {
   372→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   373→      if (combinedIdx >= 0) {
   374→        const nums: string[] = [];
   375→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   376→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   377→          if (m) nums.push(m[1]);
   378→        }
   379→        if (nums.length >= 3) {
   380→          PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   381→        }
   382→      }
   383→    }
   384→    // Strategy 3: flat text search
   385→    if (!PIS || !COFINS) {
   386→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   387→      if (flatMatch) {
   388→        if (!PIS) PIS = flatMatch[1];
   389→        if (!COFINS) COFINS = flatMatch[2];
   390→      }
   391→    }
   392→    // Strategy 4: Individual fallback
   393→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   394→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   395→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%)/i);
   396→  })();
   397→
   398→  // ===== GDI =====
   399→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   400→
   401→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   402→  // ENERGISA GDI tabular layout: quantities appear as standalone numbers
   403→  // after "chave de acesso:", R$ values are negative numbers before it.
   404→  // When multiple mUC entries exist, they must be SUMMED.
   405→  const {
   406→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   407→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   408→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   409→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   410→  } = (() => {
   411→    // Count how many oUC and mUC label lines exist
   412→    let nOUC = 0;
   413→    let nMUC = 0;
   414→    let firstGDI = -1;
   415→    for (let i = 0; i < lines.length; i++) {
   416→      if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   417→        nOUC++;
   418→        if (firstGDI < 0) firstGDI = i;
   419→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   420→        nMUC++;
   421→        if (firstGDI < 0) firstGDI = i;
   422→      }
   423→    }
   424→    if (nOUC === 0 && nMUC === 0) return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   425→
   426→    // Find "chave de acesso:" as section anchor (search up to 100 lines ahead)
   427→    const chaveIdx = (() => {
   428→      for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   429→        if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   430→      }
   431→      return -1;
   432→    })();
   433→
   434→    // Find section end boundary (search up to 120 lines ahead)
   435→    let sectionEnd = Math.min(firstGDI + 100, lines.length);
   436→    for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   437→      if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) { sectionEnd = i; break; }
   438→    }
   439→    // Ensure sectionEnd extends past chaveIdx so quantities can be found
   440→    if (chaveIdx >= 0) {
   441→      sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   442→    }
   443→
   444→    // --- Quantities (kWh) after "chave de acesso:" ---
   445→    const allQtds: number[] = [];
   446→    if (chaveIdx >= 0) {
   447→      for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   448→        const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   449→        if (m) allQtds.push(parseBR(m[1]));
   450→      }
   451→    }
   452→    // Fallback: large standalone numbers in GDI section
   453→    if (allQtds.length === 0) {
   454→      for (let i = firstGDI; i < sectionEnd; i++) {
   455→        const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   456→        if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   457→        if (allQtds.length >= nOUC + nMUC) break;
   458→      }
   459→    }
   460→
   461→    // --- R$ Values (embedded negative numbers before "chave de acesso:") ---
   462→    // ENERGISA PDFs put GDI R$ values embedded in composite lines like
   463→    // "1,083930-17,34" (alíquota + credit value). Standalone negatives like
   464→    // "-1,56" are tax adjustments, NOT the GDI credit amounts.
   465→    const allRVals: number[] = [];
   466→    const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   467→    const totalNeeded = nOUC + nMUC;
   468→
   469→    // Strategy 1: Embedded negatives — lines ending with a negative number
   470→    // but containing additional content before it (e.g. "1,083930-17,34")
   471→    for (let i = firstGDI; i < rSearchEnd; i++) {
   472→      const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   473→      if (m && allRVals.length < totalNeeded) {
   474→        allRVals.push(parseBR('-' + m[2]));
   475→      }
   476→    }
   477→    // Strategy 2 fallback: standalone negative numbers (for PDFs without embedded format)
   478→    if (allRVals.length < totalNeeded) {
   479→      for (let i = firstGDI; i < rSearchEnd; i++) {
   480→        const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   481→        if (m && allRVals.length < totalNeeded) {
   482→          allRVals.push(parseBR(m[1]));
   483→        }
   484→      }
   485→    }
   486→
   487→    // Split: first nOUC values → oUC, next nMUC → mUC (sum each group)
   488→    const oUCQtds = allQtds.splice(0, nOUC);
   489→    const oUCRVals = allRVals.splice(0, nOUC);
   490→    // Remaining are mUC
   491→    const mUCQtds = allQtds; // already spliced
   492→    const mUCRVals = allRVals;
   493→
   494→    const fmt = (n: number) => n.toFixed(2).replace('.', ',');
   495→    const fmtInt = (n: number) => {
   496→      const abs = Math.abs(n);
   497→      const intPart = Math.floor(abs).toString();
   498→      const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   499→      let formatted = '';
   500→      for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   501→        if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   502→        formatted = intPart[i] + formatted;
   503→      }
   504→      return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   505→    };
   506→
   507→    const sumQtd = (arr: number[]) => arr.length > 0 ? fmtInt(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   508→    const sumR = (arr: number[]) => arr.length > 0 ? fmtInt(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   509→
   510→    return {
   511→      oUC_Qtd: sumQtd(oUCQtds),
   512→      oUC_Valor: sumR(oUCRVals),
   513→      mUC_Qtd: sumQtd(mUCQtds),
   514→      mUC_Valor: sumR(mUCRVals),
   515→    };
   516→  })();
   517→
   518→  // ===== CIP =====
   519→  const CIP = (() => {
   520→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   521→    if (cipIdx >= 0) {
   522→      // Look on same line and nearby for R$ value
   523→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   524→      const money = findAllMoney(region);
   525→      if (money.length > 0) return money[money.length - 1];
   526→    }
   527→    return findTributoValue(lines, flat, /\bCIP\b/i);
   528→  })();
   529→
   530→  // ===== VALOR TOTAL =====
   531→  const ValorTotal = (() => {
   532→    // Strategy 1: "TOTAL A PAGAR" with R$ on same or next line
   533→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   534→    if (totalIdx >= 0) {
   535→      // Check same line
   536→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   537→      if (m1) return m1[1];
   538→      // Check next lines
   539→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   540→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   541→        if (m2) return m2[1];
   542→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   543→        if (m3) return m3[1];
   544→      }
   545→    }
   546→    // Strategy 2: flat search
   547→    const flatMatch = flat.match(/TOTAL\s*A\s+PAGAR[^\d]*?R\$\s*([\d.]+,\d{2})/i);
   548→    if (flatMatch) return flatMatch[1];
   549→    // Strategy 3: "R$ X.XXX,XX" near end of document
   550→    const lastMoney = findAllMoney(flat);
   551→    if (lastMoney.length > 0) {
   552→      // The total is typically one of the largest values
   553→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   554→      return lastMoney[0];
   555→    }
   556→    return EMPTY;
   557→  })();
   558→
   559→  return {
   560→    UC, Cidade, Tipo, Mercado, Concessionaria,
   561→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   562→    ModalidadeTarifaria, Subgrupo,
   563→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   564→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   565→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   566→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   567→    UltrapassagemPonta, UltrapassagemForaPonta,
   568→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   569→    ValorConsumoPonta, ValorConsumoForaPonta,
   570→    ValorDemandaPonta, ValorDemandaForaPonta,
   571→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   572→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   573→    PIS, COFINS, ICMS, GDI, CIP,
   574→    EnergiaReativaTotal,
   575→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   576→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   577→    ValorTotal,
   578→  };
   579→}

   581→// ===================== ITEM HELPERS =====================
   582→
   583→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   584→  const idx = findLine(lines, labelPattern);
   585→  if (idx < 0) return EMPTY;
   586→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   587→    const line = lines[i];
   588→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   589→    const m = line.match(/([\d.]+,\d{2})/);
   590→    if (m) return m[1];
   591→  }
   592→  return EMPTY;
   593→}
   594→
   595→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   596→  const idx = findLine(lines, labelPattern);
   597→  if (idx < 0) return EMPTY;
   598→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   599→  const moneyVals = findAllMoney(region);
   600→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   601→  return EMPTY;
   602→}
   603→
   604→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   605→  const idx = findLine(lines, labelPattern);
   606→  if (idx >= 0) {
   607→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   608→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   609→      if (m) return m[1];
   610→    }
   611→  }
   612→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   613→  return flatMatch ? flatMatch[1] : EMPTY;
   614→}
