     1→'use client'
     2→
     3→import { useState, useCallback, useRef } from 'react';
     4→import { toast } from 'sonner';
     5→import { Button } from '@/components/ui/button';
     6→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Progress } from '@/components/ui/progress';
     9→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    10→import {
    11→  Table,
    12→  TableBody,
    13→  TableCell,
    14→  TableHead,
    15→  TableHeader,
    16→  TableRow,
    17→} from '@/components/ui/table';
    18→import {
    19→  Upload,
    20→  FileText,
    21→  Download,
    22→  Trash2,
    23→  CheckCircle2,
    24→  AlertCircle,
    25→  Zap,
    26→  BarChart3,
    27→  FileSpreadsheet,
    28→  Bug,
    29→  Loader2,
    30→} from 'lucide-react';
    31→import { FIELD_LABELS, FIELD_CATEGORIES, type EnergisaInvoiceData } from '@/lib/energisa-fields';
    32→
    33→interface FileProgress {
    34→  fileName: string;
    35→  status: 'pending' | 'processing' | 'done' | 'error';
    36→  result?: ExtractionResult;
    37→  errorMsg?: string;
    38→}
    39→
    40→interface ExtractionResult {
    41→  fileName: string;
    42→  data: EnergisaInvoiceData;
    43→  stats: {
    44→    extractedCount: number;
    45→    totalFields: number;
    46→    extractionRate: number;
    47→    pageCount: number;
    48→  };
    49→  rawText?: string;
    50→}
    51→
    52→export default function Home() {
    53→  const [results, setResults] = useState<ExtractionResult[]>([]);
    54→  const [isProcessing, setIsProcessing] = useState(false);
    55→  const [isDragging, setIsDragging] = useState(false);
    56→  const [activeTab, setActiveTab] = useState('upload');
    57→  const [debugMode, setDebugMode] = useState(false);
    58→  const [fileProgress, setFileProgress] = useState<FileProgress[]>([]);
    59→  const fileInputRef = useRef<HTMLInputElement>(null);
    60→
    61→  const processFiles = useCallback(async (files: FileList | File[]) => {
    62→    const pdfFiles = Array.from(files).filter((f) => f.type === 'application/pdf');
    63→    if (pdfFiles.length === 0) {
    64→      toast.error('Selecione ao menos um arquivo PDF.');
    65→      return;
    66→    }
    67→
    68→    setIsProcessing(true);
    69→    const newResults: ExtractionResult[] = [];
    70→
    71→    // Initialize progress for all files
    72→    const initialProgress: FileProgress[] = pdfFiles.map((f) => ({
    73→      fileName: f.name,
    74→      status: 'pending',
    75→    }));
    76→    setFileProgress(initialProgress);
    77→
    78→    // Process files sequentially with per-file progress
    79→    for (let i = 0; i < pdfFiles.length; i++) {
    80→      const file = pdfFiles[i];
    81→
    82→      // Mark current file as processing
    83→      setFileProgress((prev) =>
    84→        prev.map((p, idx) =>
    85→          idx === i ? { ...p, status: 'processing' as const } : p
    86→        )
    87→      );
    88→
    89→      try {
    90→        const formData = new FormData();
    91→        formData.append('file', file);
    92→        const response = await fetch('/api/extract-pdf', {
    93→          method: 'POST',
    94→          body: formData,
    95→        });
    96→        const result = await response.json();
    97→
    98→        if (!response.ok) {
    99→          setFileProgress((prev) =>
   100→            prev.map((p, idx) =>
   101→              idx === i ? { ...p, status: 'error' as const, errorMsg: result.error } : p
   102→            )
   103→          );
   104→          toast.error(`${file.name}: ${result.error}`);
   105→        } else {
   106→          const extractionResult: ExtractionResult = {
   107→            fileName: file.name,
   108→            data: result.data,
   109→            stats: result.stats,
   110→            rawText: result.rawText,
   111→          };
   112→          newResults.push(extractionResult);
   113→          setFileProgress((prev) =>
   114→            prev.map((p, idx) =>
   115→              idx === i ? { ...p, status: 'done' as const, result: extractionResult } : p
   116→            )
   117→          );
   118→          toast.success(`${file.name}: ${result.stats.extractedCount}/${result.stats.totalFields} campos (${result.stats.extractionRate}%)`);
   119→        }
   120→      } catch {
   121→        setFileProgress((prev) =>
   122→          prev.map((p, idx) =>
   123→            idx === i ? { ...p, status: 'error' as const, errorMsg: 'Erro ao processar' } : p
   124→          )
   125→        );
   126→        toast.error(`${file.name}: Erro ao processar.`);
   127→      }
   128→    }
   129→
   130→    setResults((prev) => {
   131→      const updated = [...newResults];
   132→      for (const existing of prev) {
   133→        const existingUC = existing.data.UC.replace(/[^0-9]/g, '');
   134→        const alreadyAdded = updated.some(
   135→          (r) => r.data.UC.replace(/[^0-9]/g, '') === existingUC && existingUC !== ''
   136→        );
   137→        if (!alreadyAdded) updated.push(existing);
   138→      }
   139→      for (const nr of newResults) {
   140→        const nrUC = nr.data.UC.replace(/[^0-9]/g, '');
   141→        if (nrUC && prev.some((r) => r.data.UC.replace(/[^0-9]/g, '') === nrUC)) {
   142→          toast.info(`UC ${nr.data.UC} atualizada com nova extração.`);
   143→        }
   144→      }
   145→      return updated;
   146→    });
   147→    setIsProcessing(false);
   148→    setFileProgress([]);
   149→
   150→    if (newResults.length > 0) {
   151→      setActiveTab('results');
   152→    }
   153→  }, []);
   154→  const handleDrop = useCallback(
   155→    (e: React.DragEvent) => {
   156→      e.preventDefault();
   157→      setIsDragging(false);
   158→      if (e.dataTransfer.files.length > 0) {
   159→        processFiles(e.dataTransfer.files);
   160→      }
   161→    },
   162→    [processFiles]
   163→  );
   164→
   165→  const handleDragOver = useCallback((e: React.DragEvent) => {
   166→    e.preventDefault();
   167→    setIsDragging(true);
   168→  }, []);
   169→
   170→  const handleDragLeave = useCallback(() => {
   171→    setIsDragging(false);
   172→  }, []);
   173→
   174→  const handleFileChange = useCallback(
   175→    (e: React.ChangeEvent<HTMLInputElement>) => {
   176→      if (e.target.files && e.target.files.length > 0) {
   177→        processFiles(e.target.files);
   178→        e.target.value = '';
   179→      }
   180→    },
   181→    [processFiles]
   182→  );
   183→
   184→  const clearAll = useCallback(() => {
   185→    setResults([]);
   186→    setActiveTab('upload');
   187→    toast.info('Todas as faturas foram removidas.');
   188→  }, []);
   189→
   190→  const removeResult = useCallback((index: number) => {
   191→    setResults((prev) => prev.filter((_, i) => i !== index));
   192→    toast.info('Fatura removida.');
   193→  }, []);
   194→
   195→  const exportToExcel = useCallback(async () => {
   196→    if (results.length === 0) {
   197→      toast.error('Nenhuma fatura para exportar.');
   198→      return;
   199→    }
   200→
   201→    try {
   202→      const response = await fetch('/api/export-excel', {
   203→        method: 'POST',
   204→        headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ invoices: results.map((r) => ({ ...r.data, _fileName: r.fileName })) }),
   206→      });
   207→
   208→      if (!response.ok) {
   209→        toast.error('Erro ao gerar Excel.');
   210→        return;
   211→      }
   212→
   213→      const blob = await response.blob();
   214→      const url = URL.createObjectURL(blob);
   215→      const a = document.createElement('a');
   216→      a.href = url;
   217→      a.download = 'faturas_energisa.xlsx';
   218→      document.body.appendChild(a);
   219→      a.click();
   220→      document.body.removeChild(a);
   221→      URL.revokeObjectURL(url);
   222→
   223→      toast.success('Excel exportado! Colunas em amarelo são para preenchimento manual.');
   224→    } catch {
   225→      toast.error('Erro ao exportar.');
   226→    }
   227→  }, [results]);
   228→
   229→  return (
   230→    <div className="min-h-screen flex flex-col bg-gradient-to-b from-slate-50 to-slate-100">
   231→      {/* Header */}
   232→      <header className="border-b bg-white shadow-sm sticky top-0 z-50">
   233→        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-4 flex items-center justify-between">
   234→          <div className="flex items-center gap-3">
   235→            <div className="p-2 rounded-lg bg-emerald-100 text-emerald-700">
   236→              <Zap className="h-6 w-6" />
   237→            </div>
   238→            <div>
   239→              <h1 className="text-lg sm:text-xl font-bold text-slate-900">Extração de Faturas ENERGISA</h1>
   240→              <p className="text-xs sm:text-sm text-slate-500">Extraia 40+ campos automaticamente de faturas de energia</p>
   241→            </div>
   242→          </div>
   243→          <div className="flex items-center gap-2">
   244→            {results.length > 0 && (
   245→              <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 hidden sm:flex gap-1">
   246→                <FileText className="h-3 w-3" />
   247→                {results.length} fatura{results.length !== 1 ? 's' : ''}
   248→              </Badge>
   249→            )}
   250→            {results.length > 0 && (
   251→              <Button variant="outline" size="sm" onClick={exportToExcel} className="gap-1.5">
   252→                <FileSpreadsheet className="h-4 w-4" />
   253→                <span className="hidden sm:inline">Exportar Excel</span>
   254→              </Button>
   255→            )}
   256→          </div>
   257→        </div>
   258→      </header>
   259→
   260→      {/* Main */}
   261→      <main className="flex-1 max-w-7xl mx-auto w-full px-4 sm:px-6 py-6">
   262→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   263→          <TabsList className="mb-6">
   264→            <TabsTrigger value="upload" className="gap-1.5">
   265→              <Upload className="h-4 w-4" />
   266→              Upload
   267→            </TabsTrigger>
   268→            <TabsTrigger value="results" disabled={results.length === 0} className="gap-1.5">
   269→              <BarChart3 className="h-4 w-4" />
   270→              Resultados ({results.length})
   271→            </TabsTrigger>
   272→          </TabsList>
   273→
   274→          {/* Upload Tab */}
   275→          <TabsContent value="upload">
   276→            <Card>
   277→              <CardHeader>
   278→                <CardTitle className="flex items-center gap-2">
   279→                  <Upload className="h-5 w-5" />
   280→                  Enviar Faturas em PDF
   281→                </CardTitle>
   282→                <CardDescription>
   283→                  Arraste e solte faturas ENERGISA em PDF ou clique para selecionar. Múltiplos arquivos permitidos.
   284→                </CardDescription>
   285→              </CardHeader>
   286→              <CardContent>
   287→                <div
   288→                  onDrop={handleDrop}
   289→                  onDragOver={handleDragOver}
   290→                  onDragLeave={handleDragLeave}
   291→                  onClick={() => fileInputRef.current?.click()}
   292→                  className={`
   293→                    border-2 border-dashed rounded-xl p-8 sm:p-12 text-center cursor-pointer transition-all duration-200
   294→                    ${isDragging ? 'border-emerald-500 bg-emerald-50 scale-[1.01]' : 'border-slate-300 bg-slate-50 hover:border-emerald-400 hover:bg-emerald-50/50'}
   295→                    ${isProcessing ? 'pointer-events-none opacity-60' : ''}
   296→                  `}
   297→                >
   298→                  <input
   299→                    ref={fileInputRef}
   300→                    type="file"
   301→                    accept="application/pdf"
   302→                    multiple
   303→                    onChange={handleFileChange}
   304→                    className="hidden"
   305→                  />
   306→                  <div className="flex flex-col items-center gap-3">
   307→                    {!isProcessing && (
   308→                      <div className="p-3 rounded-full bg-emerald-100 text-emerald-600">
   309→                        <FileText className="h-8 w-8" />
   310→                      </div>
   311→                    )}
   312→                    {!isProcessing && (
   313→                      <div>
   314→                        <p className="text-base font-semibold text-slate-700">
   315→                          Arraste faturas PDF aqui
   316→                        </p>
   317→                        <p className="text-sm text-slate-500 mt-1">
   318→                          ou clique para selecionar os arquivos
   319→                        </p>
   320→                      </div>
   321→                    )}
   322→                    {/* Debug toggle - only when not processing */}
   323→                    {!isProcessing && (
   324→                      <label className="flex items-center gap-2 mt-4 cursor-pointer select-none">
   325→                        <input
   326→                          type="checkbox"
   327→                          checked={debugMode}
   328→                          onChange={(e) => setDebugMode(e.target.checked)}
   329→                          className="rounded border-slate-300 text-emerald-600 focus:ring-emerald-500"
   330→                        />
   331→                        <Bug className="h-4 w-4 text-slate-500" />
   332→                        <span className="text-xs text-slate-500">Modo Debug (mostra texto bruto do PDF)</span>
   333→                      </label>
   334→                    )}
   335→                  </div>
   336→
   337→                  {/* Processing progress panel */}
   338→                  {isProcessing && fileProgress.length > 0 && (
   339→                    <div className="w-full max-w-md mx-auto text-left">
   340→                      {/* Overall progress bar */}
   341→                      <div className="mb-4">
   342→                        <div className="flex items-center justify-between mb-2">
   343→                          <span className="text-sm font-semibold text-slate-700">
   344→                            Processando faturas
   345→                          </span>
   346→                          <span className="text-sm font-medium text-emerald-700">
   347→                            {fileProgress.filter((p) => p.status === 'done' || p.status === 'error').length}/{fileProgress.length}
   348→                          </span>
   349→                        </div>
   350→                        <Progress
   351→                          value={(fileProgress.filter((p) => p.status === 'done' || p.status === 'error').length / fileProgress.length) * 100}
   352→                          className="h-2.5 [&>div]:bg-emerald-500"
   353→                        />
   354→                      </div>
   355→
   356→                      {/* Per-file list */}
   357→                      <div className="space-y-2 max-h-64 overflow-y-auto pr-1">
   358→                        {fileProgress.map((fp, idx) => (
   359→                          <div
   360→                            key={idx}
   361→                            className={`flex items-center gap-3 px-3 py-2 rounded-lg text-sm transition-all ${
   362→                              fp.status === 'processing'
   363→                                ? 'bg-emerald-50 border border-emerald-200'
   364→                                : fp.status === 'done'
   365→                                  ? 'bg-slate-50'
   366→                                  : fp.status === 'error'
   367→                                    ? 'bg-red-50 border border-red-100'
   368→                                    : 'bg-slate-50/50'
   369→                            }`}
   370→                          >
   371→                            {/* Status icon */}
   372→                            <div className="flex-shrink-0">
   373→                              {fp.status === 'pending' && (
   374→                                <div className="h-5 w-5 rounded-full border-2 border-slate-300" />
   375→                              )}
   376→                              {fp.status === 'processing' && (
   377→                                <Loader2 className="h-5 w-5 text-emerald-600 animate-spin" />
   378→                              )}
   379→                              {fp.status === 'done' && (
   380→                                <CheckCircle2 className="h-5 w-5 text-emerald-500" />
   381→                              )}
   382→                              {fp.status === 'error' && (
   383→                                <AlertCircle className="h-5 w-5 text-red-500" />
   384→                              )}
   385→                            </div>
   386→
   387→                            {/* File name + stats */}
   388→                            <div className="flex-1 min-w-0">
   389→                              <p className={`truncate text-sm ${
   390→                                fp.status === 'processing' ? 'font-medium text-emerald-800' :
   391→                                fp.status === 'error' ? 'text-red-700' :
   392→                                fp.status === 'done' ? 'text-slate-700' : 'text-slate-400'
   393→                              }`}>
   394→                                {fp.fileName}
   395→                              </p>
   396→                              {fp.status === 'processing' && (
   397→                                <p className="text-xs text-emerald-600 mt-0.5">Extraindo dados...</p>
   398→                              )}
   399→                              {fp.status === 'done' && fp.result && (
   400→                                <p className="text-xs text-slate-500 mt-0.5">
   401→                                  {fp.result.stats.extractedCount}/{fp.result.stats.totalFields} campos ({fp.result.stats.extractionRate}%) — {fp.result.data.UC || 'UC não encontrada'}
   402→                                </p>
   403→                              )}
   404→                              {fp.status === 'error' && (
   405→                                <p className="text-xs text-red-500 mt-0.5">{fp.errorMsg}</p>
   406→                              )}
   407→                            </div>
   408→
   409→                            {/* Extraction rate badge for done */}
   410→                            {fp.status === 'done' && fp.result && (
   411→                              <span className={`flex-shrink-0 text-xs font-medium px-2 py-0.5 rounded-full ${
   412→                                fp.result.stats.extractionRate >= 70
   413→                                  ? 'bg-emerald-100 text-emerald-700'
   414→                                  : fp.result.stats.extractionRate >= 40
   415→                                    ? 'bg-amber-100 text-amber-700'
   416→                                    : 'bg-red-100 text-red-700'
   417→                              }`}>
   418→                                {fp.result.stats.extractionRate}%
   419→                              </span>
   420→                            )}
   421→                          </div>
   422→                        ))}
   423→                      </div>
   424→                    </div>
   425→                  )}
   426→                </div>
   427→
   428→                {results.length > 0 && (
   429→                  <div className="mt-6 flex items-center justify-between">
   430→                    <p className="text-sm text-slate-500">
   431→                      {results.length} fatura{results.length !== 1 ? 's' : ''} processada{results.length !== 1 ? 's' : ''}
   432→                    </p>
   433→                    <div className="flex gap-2">
   434→                      <Button variant="outline" size="sm" onClick={clearAll} className="gap-1.5 text-destructive hover:text-destructive">
   435→                        <Trash2 className="h-4 w-4" />
   436→                        Limpar Tudo
   437→                      </Button>
   438→                      <Button size="sm" onClick={exportToExcel} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   439→                        <Download className="h-4 w-4" />
   440→                        Exportar Excel
   441→                      </Button>
   442→                    </div>
   443→                  </div>
   444→                )}
   445→              </CardContent>
   446→            </Card>
   447→
   448→            {/* Info Cards */}
   449→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mt-6">
   450→              <Card>
   451→                <CardContent className="p-4">
   452→                  <div className="flex items-center gap-3">
   453→                    <div className="p-2 rounded-lg bg-blue-100 text-blue-700">
   454→                      <FileText className="h-5 w-5" />
   455→                    </div>
   456→                    <div>
   457→                      <p className="text-sm font-medium text-slate-700">40+ Campos</p>
   458→                      <p className="text-xs text-slate-500">UC, consumo, demanda, tributos e mais</p>
   459→                    </div>
   460→                  </div>
   461→                </CardContent>
   462→              </Card>
   463→              <Card>
   464→                <CardContent className="p-4">
   465→                  <div className="flex items-center gap-3">
   466→                    <div className="p-2 rounded-lg bg-amber-100 text-amber-700">
   467→                      <FileSpreadsheet className="h-5 w-5" />
   468→                    </div>
   469→                    <div>
   470→                      <p className="text-sm font-medium text-slate-700">Exportar Excel</p>
   471→                      <p className="text-xs text-slate-500">Gerência e LOCINS para preenchimento manual</p>
   472→                    </div>
   473→                  </div>
   474→                </CardContent>
   475→              </Card>
   476→              <Card>
   477→                <CardContent className="p-4">
   478→                  <div className="flex items-center gap-3">
   479→                    <div className="p-2 rounded-lg bg-emerald-100 text-emerald-700">
   480→                      <Zap className="h-5 w-5" />
   481→                    </div>
   482→                    <div>
   483→                      <p className="text-sm font-medium text-slate-700">Múltiplos Modelos</p>
   484→                      <p className="text-xs text-slate-500">Compatível com diferentes formatos ENERGISA</p>
   485→                    </div>
   486→                  </div>
   487→                </CardContent>
   488→              </Card>
   489→            </div>
   490→          </TabsContent>
   491→
   492→          {/* Results Tab */}
   493→          <TabsContent value="results">
   494→            {results.length === 0 ? (
   495→              <Card>
   496→                <CardContent className="p-12 text-center">
   497→                  <FileText className="h-12 w-12 text-slate-300 mx-auto mb-3" />
   498→                  <p className="text-slate-500">Nenhuma fatura processada ainda.</p>
   499→                </CardContent>
   500→              </Card>
   501→            ) : (
   502→              <div className="space-y-4">
   503→                {results.map((result, idx) => (
   504→                  <>
   505→                  <Card key={`${result.fileName}-${idx}`}>
   506→                    <CardHeader className="pb-3">
   507→                      <div className="flex items-center justify-between flex-wrap gap-2">
   508→                        <div className="flex items-center gap-3">
   509→                          <div className="p-2 rounded-lg bg-slate-100">
   510→                            <FileText className="h-5 w-5 text-slate-600" />
   511→                          </div>
   512→                          <div>
   513→                            <CardTitle className="text-base">{result.fileName}</CardTitle>
   514→                            <CardDescription className="text-xs">
   515→                              {result.stats.pageCount} página{result.stats.pageCount !== 1 ? 's' : ''} | {result.stats.extractedCount}/{result.stats.totalFields} campos extraídos
   516→                            </CardDescription>
   517→                          </div>
   518→                        </div>
   519→                        <div className="flex items-center gap-3">
   520→                          <div className="flex items-center gap-2 w-40">
   521→                            <Progress
   522→                              value={result.stats.extractionRate}
   523→                              className={`h-2 ${
   524→                                result.stats.extractionRate >= 70
   525→                                  ? '[&>div]:bg-emerald-500'
   526→                                  : result.stats.extractionRate >= 40
   527→                                    ? '[&>div]:bg-amber-500'
   528→                                    : '[&>div]:bg-red-500'
   529→                              }`}
   530→                            />
   531→                            <span className="text-xs font-medium text-slate-600 w-10 text-right">
   532→                              {result.stats.extractionRate}%
   533→                            </span>
   534→                          </div>
   535→                          <Badge
   536→                            variant={
   537→                              result.stats.extractionRate >= 70
   538→                                ? 'default'
   539→                                : result.stats.extractionRate >= 40
   540→                                  ? 'secondary'
   541→                                  : 'destructive'
   542→                            }
