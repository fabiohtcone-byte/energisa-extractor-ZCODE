    51→    51→// Parse Brazilian number format: "1.135,00" → 1135.00, "-110,49" → -110.49
    52→    52→function parseBR(s: string): number {
    53→    53→  const neg = s.startsWith('-');
    54→    54→  const clean = neg ? s.slice(1) : s;
    55→    55→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    56→    56→}
    57→    57→
    58→    58→function findLine(lines: string[], pattern: RegExp): number {
    59→    59→  return lines.findIndex(l => pattern.test(l));
    60→    60→}
    61→    61→
    62→    62→function findAllMoney(text: string): string[] {
    63→    63→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
    64→    64→  if (!matches) return [];
    65→    65→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
    66→    66→}
    67→    67→
    68→    68→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    69→    69→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    70→    70→    if (Math.abs(i - startIdx) > maxDist) break;
    71→    71→    const line = lines[i];
    72→    72→    if (mustBeOnlyNum) {
    73→    73→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    74→    74→      if (m) return m[1];
    75→    75→    } else {
    76→    76→      const m = line.match(/([\d.]+,\d{2})/);
    77→    77→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
    78→    78→    }
    79→    79→  }
    80→    80→  return EMPTY;
    81→    81→}

    83→    83→// ===================== MAIN EXTRACTOR =====================
    84→    84→
    85→    85→export function extractEnergisaData(rawText: string): EnergisaInvoiceData {
    86→    86→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
    87→    87→  const flat = lines.join(' ');
    88→    88→
    89→    89→  // ===== CONCESSIONÁRIA =====
    90→    90→  // Full company name on its own line near the top
    91→    91→  const Concessionaria = (() => {
    92→    92→    for (const line of lines.slice(0, 20)) {
    93→    93→      if (/^ENERGISA/i.test(line) && line.length > 15 && !/\d{2}\s*\/\s*\d{2}/.test(line)) {
    94→    94→        return line.replace(/\s+/g, ' ');
    95→    95→      }
    96→    96→    }
    97→    97→    return EMPTY;
    98→    98→  })();
    99→    99→
   100→   100→  // ===== TIPO =====
   101→   101→  const Tipo = (() => {
   102→   102→    const u = flat.toUpperCase();
   103→   103→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   104→   104→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   105→   105→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   106→   106→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   107→   107→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   108→   108→    if (/RURAL/.test(u)) return 'RURAL';
   109→   109→    return EMPTY;
   110→   110→  })();
   111→   111→
   112→   112→  // ===== MERCADO =====
   113→   113→  // Look for classification line: "SERVIÇO PÚBLICO / ÁGUA, ESGOTO E SANEAMENTO"
   114→   114→  const Mercado = (() => {
   115→   115→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   116→   116→    if (m) {
   117→   117→      const val = m[1].trim().replace(/\s+/g, ' ');
   118→   118→      if (val.length < 50) return val.toUpperCase();
   119→   119→    }
   120→   120→    return EMPTY;
   121→   121→  })();
   122→   122→
   123→   123→  // ===== CIDADE =====
   124→   124→  // Pattern: "DOURADOS (AG: 10)"
   125→   125→  const Cidade = (() => {
   126→   126→    for (const line of lines.slice(0, 60)) {
   127→   127→      const m = line.match(/([A-ZÀ-Ú]{4,25})\s*\(\s*AG\s*:\s*\d+\s*\)/);
   128→   128→      if (m) {
   129→   129→        const name = m[1].toUpperCase();
   130→   130→        // Skip company names
   131→   131→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA'];
   132→   132→        if (!skip.includes(name)) return name;
   133→   133→      }
   134→   134→    }
   135→   135→    return EMPTY;
   136→   136→  })();
   137→   137→
   138→   138→  // ===== UC =====
   139→   139→  // Pattern: dotted format X.XXX.XXX-XX on its own line or with "KWH" prefix
   140→   140→  const UC = (() => {
   141→   141→    // Strategy 1: line with only UC (e.g., "3.562.051-59", "43.210.051-68", "131.752.051-07")
   142→   142→    for (const line of lines) {
   143→   143→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   144→   144→      if (m) return m[1];
   145→   145→    }
   146→   146→    // Strategy 2: UC embedded in short line
   147→   147→    for (const line of lines) {
   148→   148→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   149→   149→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   150→   150→    }
   151→   151→    // Strategy 3: broad search
   152→   152→    const broad = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   153→   153→    return broad ? broad[1] : EMPTY;
   154→   154→  })();
   155→   155→
   156→   156→  // ===== MÊS REFERÊNCIA =====
   157→   157→  // Pattern: "Junho / 2026" or "JUNHO / 2026"
   158→   158→  const MesReferencia = (() => {
   159→   159→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   160→   160→    const numToMonth: Record<string, string> = {
   161→   161→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   162→   162→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   163→   163→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   164→   164→    };
   165→   165→    // Named month pattern
   166→   166→    const nameMatch = flat.match(new RegExp(`(${meses})\\s*[\\/\\-]\\s*(\\d{4})`, 'i'));
   167→   167→    if (nameMatch) {
   168→   168→      const month = nameMatch[1].charAt(0).toUpperCase() + nameMatch[1].slice(1).toLowerCase();
   169→   169→      return `${month} / ${nameMatch[2]}`;
   170→   170→    }
   171→   171→    // Numeric month pattern (e.g., "06/2026")
   172→   172→    const numMatch = flat.match(/(\d{2})\s*[\\/]\s*(\d{4})/);
   173→   173→    if (numMatch && numToMonth[numMatch[1]]) {
   174→   174→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   175→   175→    }
   176→   176→    return EMPTY;
   177→   177→  })();
   178→   178→
   179→   179→  // ===== ANO LEITURA =====
   180→   180→  const AnoLeitura = (() => {
   181→   181→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   182→   182→    if (m) return m[1];
   183→   183→    if (MesReferencia) {
   184→   184→      const ym = MesReferencia.match(/(\d{4})/);
   185→   185→      if (ym) return ym[1];
   186→   186→    }
   187→   187→    return EMPTY;
   188→   188→  })();
   189→   189→
   190→   190→  // ===== VENCIMENTO =====
   191→   191→  // ENERGISA PDFs often concatenate "VENCIMENTO" with other labels like "MATRÍCULA"
   192→   192→  // The date may appear BEFORE or AFTER the label
   193→   193→  const Vencimento = (() => {
   194→   194→    // Strategy 1: find line containing VENCIMENTO (even concatenated), check nearby for date
   195→   195→    const venIdx = findLine(lines, /VENCIMENTO/i);
   196→   196→    if (venIdx >= 0) {
   197→   197→      // Check lines BEFORE the label (ENERGISA often puts date before)
   198→   198→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   199→   199→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   200→   200→        if (m) return m[1];
   201→   201→      }
   202→   202→      // Check lines AFTER the label
   203→   203→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   204→   204→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   205→   205→        if (m) return m[1];
   206→   206→      }
   207→   207→      // Check same line
   208→   208→      const sameLine = lines[venIdx].match(/(\d{2}\/\d{2}\/\d{4})/);
   209→   209→      if (sameLine) return sameLine[1];
   210→   210→    }
   211→   211→    // Strategy 2: "TOTAL A PAGAR" area - date often nearby
   212→   212→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   213→   213→    if (totalIdx >= 0) {
   214→   214→      for (let i = totalIdx - 3; i <= Math.min(totalIdx + 3, lines.length - 1); i++) {
   215→   215→        if (i < 0) continue;
   216→   216→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   217→   217→        if (m) return m[1];
   218→   218→      }
   219→   219→    }
   220→   220→    return EMPTY;
   221→   221→  })();
   222→   222→
   223→   223→  // ===== NOTA FISCAL =====
   224→   224→  const NotaFiscal = (() => {
   225→   225→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   226→   226→    return m ? m[1] : EMPTY;
   227→   227→  })();
   228→   228→
   229→   229→  // ===== MODALIDADE TARIFÁRIA =====
   230→   230→  const ModalidadeTarifaria = (() => {
   231→   231→    // In the classification line: "MTC-CONVENCIONAL BAIXA TENSÃO / B3"
   232→   232→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   233→   233→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   234→   234→    if (/AZUL/i.test(flat)) return 'AZUL';
   235→   235→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   236→   236→    if (/VERDE/i.test(flat)) return 'VERDE';
   237→   237→    return EMPTY;
   238→   238→  })();
   239→   239→
   240→   240→  // ===== SUBGRUPO =====
   241→   241→  // Pattern: "/ B3" in classification line
   242→   242→  const Subgrupo = (() => {
   243→   243→    const m = flat.match(/\/(\s*[AB]\d?\s*)(?:\s|\/|$)/);
   244→   244→    if (m) return m[1].trim().toUpperCase();
   245→   245→    // Fallback: search in classification area
   246→   246→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   247→   247→    if (classIdx >= 0) {
   248→   248→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   249→   249→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   250→   250→      if (m2) return m2[1].toUpperCase();
   251→   251→    }
   252→   252→    return EMPTY;
   253→   253→  })();
   254→   254→
   255→   255→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   256→   256→
   257→   257→  // ===== CONSUMO =====
   258→   258→  let ConsumoPonta = EMPTY;
   259→   259→  let ConsumoForaPonta = EMPTY;
   260→   260→  let ConsumoTotal = EMPTY;
   261→   261→  let ValorConsumoPonta = EMPTY;
   262→   262→  let ValorConsumoForaPonta = EMPTY;
   263→   263→
   264→   264→  if (isSplitTariff) {
   265→   265→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   266→   266→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   267→   267→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   268→   268→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   269→   269→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   270→   270→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   271→   271→  } else {
   272→   272→    // Convencional: look for "KWH1.044,00" or "KWH 1.044,00"
   273→   273→    for (const line of lines) {
   274→   274→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   275→   275→      if (km) { ConsumoTotal = km[1]; break; }
   276→   276→    }
   277→   277→    // Fallback: "CONSUMO FATURADO" area
   278→   278→    if (!ConsumoTotal) {
   279→   279→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   280→   280→      if (consIdx >= 0) {
   281→   281→        // Look for numbers near CONSUMO FATURADO, but skip the month reference line
   282→   282→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   283→   283→          const line = lines[i];
   284→   284→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   285→   285→          if (/[A-Za-z]{4,}/.test(line)) continue;
   286→   286→          const m = line.match(/^([\d.]+,\d{2})$/);
   287→   287→          if (m) { ConsumoTotal = m[1]; break; }
   288→   288→        }
   289→   289→      }
   290→   290→    }
   291→   291→    // R$ value for consumo (Convencional)
   292→   292→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   293→   293→  }
   294→   294→
   295→   295→  // ===== DEMANDA (only split tariffs) =====
   296→   296→  let DemandaContratadaPonta = EMPTY;
   297→   297→  let DemandaContratadaForaPonta = EMPTY;
   298→   298→  let DemandaMedidaPonta = EMPTY;
   299→   299→  let DemandaMedidaForaPonta = EMPTY;
   300→   300→  let DemandaFaturadaPonta = EMPTY;
   301→   301→  let DemandaFaturadaForaPonta = EMPTY;
   302→   302→  let ValorDemandaPonta = EMPTY;
   303→   303→  let ValorDemandaForaPonta = EMPTY;
   304→   304→  if (isSplitTariff) {
   305→   305→    DemandaContratadaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Ponta(?![^\n]*Fora)/i);
   306→   306→    DemandaContratadaForaPonta = findQtyNearLine(lines, /Demanda\s+Contrat[ae]da.*Fora\s*Ponta/i);
   307→   307→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?![^\n]*Fora)/i);
   308→   308→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   309→   309→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?![^\n]*Fora)/i) || DemandaMedidaPonta;
   310→   310→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   311→   311→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?![^\n]*Fora)/i);
   312→   312→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   313→   313→  }
   314→   314→
   315→   315→  // ===== ULTRAPASSAGEM (only split tariffs) =====
   316→   316→  let UltrapassagemPonta = EMPTY;
   317→   317→  let UltrapassagemForaPonta = EMPTY;
   318→   318→  let ValorUltrapassagemPonta = EMPTY;
   319→   319→  let ValorUltrapassagemForaPonta = EMPTY;
   320→   320→  if (isSplitTariff) {
   321→   321→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   322→   322→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   323→   323→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?![^\n]*Fora)/i);
   324→   324→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   325→   325→  }
   326→   326→
   327→   327→  // ===== REATIVA EXCEDENTE =====
   328→   328→  let ReativaExcedentePonta = EMPTY;
   329→   329→  let ReativaExcedenteForaPonta = EMPTY;
   330→   330→  let EnergiaReativaTotal = EMPTY;
   331→   331→  let ValorReativaExcedentePonta = EMPTY;
   332→   332→  let ValorReativaExcedenteForaPonta = EMPTY;
   333→   333→  if (isSplitTariff) {
   334→   334→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   335→   335→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   336→   336→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?![^\n]*Fora)/i);
   337→   337→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   338→   338→  } else {
   339→   339→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   340→   340→  }
   341→   341→
   342→   342→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   343→   343→  // ENERGISA format (from real PDF):
   344→   344→  // Line N: "PIS/"  Line N+1: "COFINS (R$)"  Line N+2: "18,47 85,09 236,20"
   345→   345→  // Or on the same flat section: "PIS/COFINS (R$) 18,47 85,09 236,20"
   346→   346→  let PIS = EMPTY;
   347→   347→  let COFINS = EMPTY;
   348→   348→  let ICMS = EMPTY;
   349→   349→
   350→   350→  (() => {
   351→   351→    // Strategy 1: Find "PIS/" line (the label is split)
   352→   352→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   353→   353→    if (pisIdx >= 0) {
   354→   354→      // Next line might be "COFINS (R$)" or values
   355→   355→      // Check next few lines for the 3 consecutive numbers
   356→   356→      const nums: string[] = [];
   357→   357→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   358→   358→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   359→   359→        if (lineNums) {
   360→   360→          for (const n of lineNums) {
   361→   361→            nums.push(n);
   362→   362→            if (nums.length >= 3) break;
   363→   363→          }
   364→   364→        }
   365→   365→      }
   366→   366→      if (nums.length >= 3) {
   367→   367→        PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   368→   368→      }
   369→   369→    }
   370→   370→    // Strategy 2: "PIS/COFINS" on same line
   371→   371→    if (!PIS && !COFINS) {
   372→   372→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   373→   373→      if (combinedIdx >= 0) {
   374→   374→        const nums: string[] = [];
   375→   375→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   376→   376→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   377→   377→          if (m) nums.push(m[1]);
   378→   378→        }
   379→   379→        if (nums.length >= 3) {
   380→   380→          PIS = nums[0]; COFINS = nums[1]; ICMS = nums[2];
   381→   381→        }
   382→   382→      }
   383→   383→    }
   384→   384→    // Strategy 3: flat text search
   385→   385→    if (!PIS || !COFINS) {
   386→   386→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   387→   387→      if (flatMatch) {
   388→   388→        if (!PIS) PIS = flatMatch[1];
   389→   389→        if (!COFINS) COFINS = flatMatch[2];
   390→   390→      }
   391→   391→    }
   392→   392→    // Strategy 4: Individual fallback
   393→   393→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   394→   394→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   395→   395→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%)/i);
   396→   396→  })();
   397→   397→
   398→   398→  // ===== GDI =====
   399→   399→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   400→   400→
   401→   401→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   402→   402→  // ENERGISA GDI tabular layout: quantities appear as standalone numbers
   403→   403→  // after "chave de acesso:", R$ values are negative numbers before it.
   404→   404→  // When multiple mUC entries exist, they must be SUMMED.
   405→   405→  const {
   406→   406→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   407→   407→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   408→   408→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   409→   409→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   410→   410→  } = (() => {
   411→   411→    // Count how many oUC and mUC label lines exist
   412→   412→    let nOUC = 0;
   413→   413→    let nMUC = 0;
   414→   414→    let firstGDI = -1;
   415→   415→    for (let i = 0; i < lines.length; i++) {
   416→   416→      if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   417→   417→        nOUC++;
   418→   418→        if (firstGDI < 0) firstGDI = i;
   419→   419→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   420→   420→        nMUC++;
   421→   421→        if (firstGDI < 0) firstGDI = i;
   422→   422→      }
   423→   423→    }
   424→   424→    if (nOUC === 0 && nMUC === 0) return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   425→   425→
   426→   426→    // Find "chave de acesso:" as section anchor
   427→   427→    const chaveIdx = (() => {
   428→   428→      for (let i = firstGDI; i < Math.min(firstGDI + 40, lines.length); i++) {
   429→   429→        if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   430→   430→      }
   431→   431→      return -1;
   432→   432→    })();
   433→   433→
   434→   434→    // Find section end boundary
   435→   435→    let sectionEnd = Math.min(firstGDI + 45, lines.length);
   436→   436→    for (let i = firstGDI; i < Math.min(firstGDI + 55, lines.length); i++) {
   437→   437→      if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) { sectionEnd = i; break; }
   438→   438→    }
   439→   439→
   440→   440→    // --- Quantities (kWh) after "chave de acesso:" ---
   441→   441→    const allQtds: number[] = [];
   442→   442→    if (chaveIdx >= 0) {
   443→   443→      for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 10, sectionEnd); i++) {
   444→   444→        const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   445→   445→        if (m) allQtds.push(parseBR(m[1]));
   446→   446→      }
   447→   447→    }
   448→   448→    // Fallback: large standalone numbers in GDI section
   449→   449→    if (allQtds.length === 0) {
   450→   450→      for (let i = firstGDI; i < sectionEnd; i++) {
   451→   451→        const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   452→   452→        if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   453→   453→        if (allQtds.length >= nOUC + nMUC) break;
   454→   454→      }
   455→   455→    }
   456→   456→
   457→   457→    // --- R$ Values (negative numbers before "chave de acesso:") ---
   458→   458→    const allRVals: number[] = [];
   459→   459→    const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   460→   460→    for (let i = firstGDI; i < rSearchEnd; i++) {
   461→   461→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   462→   462→      if (m) allRVals.push(parseBR(m[1]));
   463→   463→    }
   464→   464→
   465→   465→    // Split: first nOUC values → oUC, next nMUC → mUC (sum each group)
   466→   466→    const oUCQtds = allQtds.splice(0, nOUC);
   467→   467→    const oUCRVals = allRVals.splice(0, nOUC);
   468→   468→    // Remaining are mUC
   469→   469→    const mUCQtds = allQtds; // already spliced
   470→   470→    const mUCRVals = allRVals;
   471→   471→
   472→   472→    const fmt = (n: number) => n.toFixed(2).replace('.', ',');
   473→   473→    const fmtInt = (n: number) => {
   474→   474→      const abs = Math.abs(n);
   475→   475→      const intPart = Math.floor(abs).toString();
   476→   476→      const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   477→   477→      let formatted = '';
   478→   478→      for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   479→   479→        if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   480→   480→        formatted = intPart[i] + formatted;
   481→   481→      }
   482→   482→      return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   483→   483→    };
   484→   484→
   485→   485→    const sumQtd = (arr: number[]) => arr.length > 0 ? fmtInt(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   486→   486→    const sumR = (arr: number[]) => arr.length > 0 ? fmtInt(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   487→   487→
   488→   488→    return {
   489→   489→      oUC_Qtd: sumQtd(oUCQtds),
   490→   490→      oUC_Valor: sumR(oUCRVals),
   491→   491→      mUC_Qtd: sumQtd(mUCQtds),
   492→   492→      mUC_Valor: sumR(mUCRVals),
   493→   493→    };
   494→   494→  })();
   495→   495→
   496→   496→  // ===== CIP =====
   497→   497→  const CIP = (() => {
   498→   498→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   499→   499→    if (cipIdx >= 0) {
   500→   500→      // Look on same line and nearby for R$ value
   501→   501→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   502→   502→      const money = findAllMoney(region);
   503→   503→      if (money.length > 0) return money[money.length - 1];
   504→   504→    }
   505→   505→    return findTributoValue(lines, flat, /\bCIP\b/i);
   506→   506→  })();
   507→   507→
   508→   508→  // ===== VALOR TOTAL =====
   509→   509→  const ValorTotal = (() => {
   510→   510→    // Strategy 1: "TOTAL A PAGAR" with R$ on same or next line
   511→   511→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   512→   512→    if (totalIdx >= 0) {
   513→   513→      // Check same line
   514→   514→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   515→   515→      if (m1) return m1[1];
   516→   516→      // Check next lines
   517→   517→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   518→   518→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   519→   519→        if (m2) return m2[1];
   520→   520→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   521→   521→        if (m3) return m3[1];
   522→   522→      }
   523→   523→    }
   524→   524→    // Strategy 2: flat search
   525→   525→    const flatMatch = flat.match(/TOTAL\s*A\s+PAGAR[^\d]*?R\$\s*([\d.]+,\d{2})/i);
   526→   526→    if (flatMatch) return flatMatch[1];
   527→   527→    // Strategy 3: "R$ X.XXX,XX" near end of document
   528→   528→    const lastMoney = findAllMoney(flat);
   529→   529→    if (lastMoney.length > 0) {
   530→   530→      // The total is typically one of the largest values
   531→   531→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   532→   532→      return lastMoney[0];
   533→   533→    }
   534→   534→    return EMPTY;
   535→   535→  })();
   536→   536→
   537→   537→  return {
   538→   538→    UC, Cidade, Tipo, Mercado, Concessionaria,
   539→   539→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   540→   540→    ModalidadeTarifaria, Subgrupo,
   541→   541→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   542→   542→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   543→   543→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   544→   544→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   545→   545→    UltrapassagemPonta, UltrapassagemForaPonta,
   546→   546→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   547→   547→    ValorConsumoPonta, ValorConsumoForaPonta,
   548→   548→    ValorDemandaPonta, ValorDemandaForaPonta,
   549→   549→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   550→   550→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   551→   551→    PIS, COFINS, ICMS, GDI, CIP,
   552→   552→    EnergiaReativaTotal,
   553→   553→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   554→   554→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   555→   555→    ValorTotal,
   556→   556→  };
   557→   557→}

   561→   561→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   562→   562→  const idx = findLine(lines, labelPattern);
   563→   563→  if (idx < 0) return EMPTY;
   564→   564→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   565→   565→    const line = lines[i];
   566→   566→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   567→   567→    const m = line.match(/([\d.]+,\d{2})/);
   568→   568→    if (m) return m[1];
   569→   569→  }
   570→   570→  return EMPTY;
   571→   571→}
   572→   572→
   573→   573→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   574→   574→  const idx = findLine(lines, labelPattern);
   575→   575→  if (idx < 0) return EMPTY;
   576→   576→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   577→   577→  const moneyVals = findAllMoney(region);
   578→   578→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   579→   579→  return EMPTY;
   580→   580→}
   581→   581→
   582→   582→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   583→   583→  const idx = findLine(lines, labelPattern);
   584→   584→  if (idx >= 0) {
   585→   585→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   586→   586→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   587→   587→      if (m) return m[1];
   588→   588→    }
   589→   589→  }
   590→   590→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   591→   591→  return flatMatch ? flatMatch[1] : EMPTY;
   592→   592→}
