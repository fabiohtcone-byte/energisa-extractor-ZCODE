    92→    92→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
    93→    93→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
    94→    94→    if (Math.abs(i - startIdx) > maxDist) break;
    95→    95→    const line = lines[i];
    96→    96→    if (mustBeOnlyNum) {
    97→    97→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
    98→    98→      if (m) return m[1];
    99→    99→    } else {
   100→   100→      const m = line.match(/([\d.]+,\d{2})/);
   101→   101→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   102→   102→    }
   103→   103→  }
   104→   104→  return EMPTY;

   109→   109→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   110→   110→  const tableData = tableExtraction?.items;
   111→   111→  const tributosFromTable = tableExtraction
   112→   112→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   113→   113→    : null;
   114→   114→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   115→   115→  const flat = lines.join(' ');
   116→   116→
   117→   117→  // ===== CONCESSIONÁRIA =====
   118→   118→  const Concessionaria = /ENERGISA/i.test(flat) ? 'ENERGISA' : EMPTY;
   119→   119→
   120→   120→  // ===== TIPO =====
   121→   121→  const Tipo = (() => {
   122→   122→    const u = flat.toUpperCase();
   123→   123→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   124→   124→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   125→   125→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   126→   126→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   127→   127→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   128→   128→    if (/RURAL/.test(u)) return 'RURAL';
   129→   129→    return EMPTY;
   130→   130→  })();
   131→   131→
   132→   132→  // ===== MERCADO =====
   133→   133→  const Mercado = (() => {
   134→   134→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   135→   135→    if (m) {
   136→   136→      const val = m[1].trim().replace(/\s+/g, ' ');
   137→   137→      if (val.length < 50) return val.toUpperCase();
   138→   138→    }
   139→   139→    return EMPTY;
   140→   140→  })();
   141→   141→
   142→   142→  // ===== CIDADE =====
   143→   143→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   144→   144→  const Cidade = (() => {
   145→   145→    for (const line of lines.slice(0, 60)) {
   146→   146→      // Match: CityName [StateAbbr] (AG: XX)
   147→   147→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   148→   148→      if (m) {
   149→   149→        const name = m[1].trim().toUpperCase();
   150→   150→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   151→   151→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   152→   152→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   153→   153→      }
   154→   154→    }
   155→   155→    return EMPTY;
   156→   156→  })();
   157→   157→
   158→   158→  // ===== UC =====
   159→   159→  const UC = (() => {
   160→   160→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   161→   161→    const isCnpjCpf = (s: string) => {
   162→   162→      const raw = s.replace(/[.\-\/]/g, '');
   163→   163→      return raw.length === 14 || raw.length === 11;
   164→   164→    };
   165→   165→
   166→   166→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   167→   167→    for (const line of lines) {
   168→   168→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   169→   169→      if (m) {
   170→   170→        const num = m[1].replace(/^0+/, '') || '0';
   171→   171→        if (num.length >= 3) return num;
   172→   172→      }
   173→   173→    }
   174→   174→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   175→   175→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   176→   176→    if (slashCtx) {
   177→   177→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   178→   178→      if (num.length >= 3) return num;
   179→   179→    }
   180→   180→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   181→   181→    // e.g., "1.409.407.051-47" → "140940705147"
   182→   182→    for (const line of lines) {
   183→   183→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   184→   184→      if (m && !isCnpjCpf(m[1])) return m[1];
   185→   185→    }
   186→   186→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   187→   187→    for (const line of lines) {
   188→   188→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   189→   189→      if (m) return m[1];
   190→   190→    }
   191→   191→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   192→   192→    for (const line of lines) {
   193→   193→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   194→   194→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   195→   195→    }
   196→   196→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   197→   197→    for (const line of lines) {
   198→   198→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   199→   199→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   200→   200→    }
   201→   201→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   202→   202→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   203→   203→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   204→   204→    if (ucMat) return ucMat[1];
   205→   205→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   206→   206→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   207→   207→    if (ucAddr) return ucAddr[1];
   208→   208→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   209→   209→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   210→   210→    if (ucInst) return ucInst[1];
   211→   211→    // Strategy 10: UC label with number — 4-13 digits
   212→   212→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   213→   213→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   214→   214→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   215→   215→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   216→   216→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   217→   217→    // Strategy 12: broad search for 3-group dotted format
   218→   218→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   219→   219→    if (broad3) return broad3[1];
   220→   220→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   221→   221→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   222→   222→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   223→   223→    return EMPTY;
   224→   224→  })();
   225→   225→
   226→   226→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   227→   227→  const cleanUC = UC.replace(/[.\-]/g, '');
   228→   228→
   229→   229→  // ===== MÊS REFERÊNCIA =====
   230→   230→  const MesReferencia = (() => {
   231→   231→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   232→   232→    const numToMonth: Record<string, string> = {
   233→   233→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   234→   234→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   235→   235→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   236→   236→    };
   237→   237→    const capitalize = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
   238→   238→
   239→   239→    // Strategy 1: "Mês / Ano" pattern on the SAME LINE as a date + R$ value (most reliable)
   240→   240→    // Energisa format: "Fevereiro / 202621/03/2026R$ 538,49" or "Junho / 2026  21/07/2026  R$ 13.356,50"
   241→   241→    for (const line of lines) {
   242→   242→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})\\s*\\d{2}/\\d{2}/\\d{4}\\s*R\\$`, 'i'));
   243→   243→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   244→   244→    }
   245→   245→    // Strategy 2: "Mês / Ano" pattern on the SAME LINE as VENCIMENTO or NOTA FISCAL
   246→   246→    for (const line of lines) {
   247→   247→      if (/VENCIMENTO|NOTA FISCAL|Nº FATURA/i.test(line)) {
   248→   248→        const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   249→   249→        if (m) return `${capitalize(m[1])} / ${m[2]}`;
   250→   250→      }
   251→   251→    }
   252→   252→    // Strategy 3: "Mês / Ano" on a line that also contains a date DD/MM/YYYY (likely the billing line)
   253→   253→    for (const line of lines) {
   254→   254→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   255→   255→      if (m && /\d{2}\/\d{2}\/\d{4}/.test(line)) {
   256→   256→        return `${capitalize(m[1])} / ${m[2]}`;
   257→   257→      }
   258→   258→    }
   259→   259→    // Strategy 4: "Mês / Ano" anywhere in flat text — but skip "Desde mês/ano" (informative text)
   260→   260→    const allNameMatches = [...flat.matchAll(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'gi'))];
   261→   261→    for (const m of allNameMatches) {
   262→   262→      // Check context: skip if preceded by "Desde" or other informative words
   263→   263→      const start = Math.max(0, m.index! - 15);
   264→   264→      const context = flat.slice(start, m.index! + m[0].length).toLowerCase();
   265→   265→      if (/desde|até|a partir|inform|dispon/.test(context)) continue;
   266→   266→      return `${capitalize(m[1])} / ${m[2]}`;
   267→   267→    }
   268→   268→    // Strategy 5: numeric MM/YYYY — prefer the one near Vencimento date
   269→   269→    const venDate = flat.match(/(\d{2})\/(\d{2})\/(\d{4})\s*R\$/);
   270→   270→    if (venDate) {
   271→   271→      const venYear = venDate[3];
   272→   272→      // Look for MM/YYYY near the vencimento
   273→   273→      const venIdx = flat.indexOf(venDate[0]);
   274→   274→      const region = flat.slice(Math.max(0, venIdx - 100), venIdx + 50);
   275→   275→      const numM = region.match(/(\d{2})\s*\/\s*(\d{4})/);
   276→   276→      if (numM && numToMonth[numM[1]] && numM[2] === venYear) {
   277→   277→        return `${numToMonth[numM[1]]} / ${numM[2]}`;
   278→   278→      }
   279→   279→    }
   280→   280→    // Strategy 6: first valid MM/YYYY that is a month
   281→   281→    const numMatch = flat.match(/(\d{2})\s*[\/]\s*(\d{4})/);
   282→   282→    if (numMatch && numToMonth[numMatch[1]]) {
   283→   283→      return `${numToMonth[numMatch[1]]} / ${numMatch[2]}`;
   284→   284→    }
   285→   285→    return EMPTY;
   286→   286→  })();
   287→   287→
   288→   288→  // ===== ANO LEITURA =====
   289→   289→  const AnoLeitura = (() => {
   290→   290→    const m = flat.match(/Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i);
   291→   291→    if (m) return m[1];
   292→   292→    if (MesReferencia) {
   293→   293→      const ym = MesReferencia.match(/(\d{4})/);
   294→   294→      if (ym) return ym[1];
   295→   295→    }
   296→   296→    return EMPTY;
   297→   297→  })();
   298→   298→
   299→   299→  // ===== VENCIMENTO =====
   300→   300→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   301→   301→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   302→   302→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   303→   303→  const Vencimento = (() => {
   304→   304→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   305→   305→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   306→   306→    if (venMoney) return venMoney[1];
   307→   307→
   308→   308→    // Strategy 2: R$ + money + date on same line (reversed order)
   309→   309→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   310→   310→    if (moneyVen) return moneyVen[1];
   311→   311→
   312→   312→    // Strategy 3: VENCIMENTO label nearby (original approach)
   313→   313→    const venIdx = findLine(lines, /VENCIMENTO/i);
   314→   314→    if (venIdx >= 0) {
   315→   315→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   316→   316→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   317→   317→        if (m) return m[1];
   318→   318→      }
   319→   319→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   320→   320→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   321→   321→        if (m) return m[1];
   322→   322→      }
   323→   323→    }
   324→   324→
   325→   325→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   326→   326→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   327→   327→    if (apresMatch) return apresMatch[1];
   328→   328→
   329→   329→    return EMPTY;
   330→   330→  })();
   331→   331→
   332→   332→  // ===== NOTA FISCAL =====
   333→   333→  const NotaFiscal = (() => {
   334→   334→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   335→   335→    return m ? m[1] : EMPTY;
   336→   336→  })();
   337→   337→
   338→   338→  // ===== MODALIDADE TARIFÁRIA =====
   339→   339→  const ModalidadeTarifaria = (() => {
   340→   340→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   341→   341→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   342→   342→    if (/AZUL/i.test(flat)) return 'AZUL';
   343→   343→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   344→   344→    if (/VERDE/i.test(flat)) return 'VERDE';
   345→   345→    return EMPTY;
   346→   346→  })();
   347→   347→
   348→   348→  // ===== SUBGRUPO =====
   349→   349→  const Subgrupo = (() => {
   350→   350→    // Look for "/A4" or "/B3" etc. in the classification line
   351→   351→    // Must be a letter + single digit, right after "/"
   352→   352→    const m = flat.match(/\/\s*([AB]\d)\s/);
   353→   353→    if (m) return m[1].toUpperCase();
   354→   354→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   355→   355→    if (classIdx >= 0) {

   555→   555→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);

   629→   629→          if (item.valor) ValorDemandaPonta = item.valor;

   674→   674→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
   675→   675→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
   676→   676→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
   677→   677→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
   678→   678→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
   679→   679→        // Multiple mUC entries: accumulate
   680→   680→        if (item.quantidade) {
   681→   681→          tbl_mUC_Qtd = tbl_mUC_Qtd
   682→   682→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   683→   683→            : item.quantidade;
   684→   684→        }
   685→   685→        if (item.valor) {
   686→   686→          tbl_mUC_Valor = tbl_mUC_Valor
   687→   687→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
   688→   688→            : item.valor;
   689→   689→        }
   690→   690→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
   691→   691→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
   692→   692→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
   693→   693→        // Accumulate as the same GDI type as the last full-label row
   694→   694→        if (gdiSectionType === 'mUC') {
   695→   695→          tbl_mUC_Qtd = tbl_mUC_Qtd
   696→   696→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   697→   697→            : item.quantidade;
   698→   698→          if (item.valor) {
   699→   699→            tbl_mUC_Valor = tbl_mUC_Valor
   700→   700→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
   701→   701→              : item.valor;
   702→   702→          }
   703→   703→        } else if (gdiSectionType === 'oUC') {
   704→   704→          tbl_oUC_Qtd = tbl_oUC_Qtd
   705→   705→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
   706→   706→            : item.quantidade;
   707→   707→          if (item.valor) {
   708→   708→            tbl_oUC_Valor = tbl_oUC_Valor
   709→   709→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
   710→   710→              : item.valor;
   711→   711→          }
   712→   712→        }
   713→   713→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
   714→   714→        // Non-GDI, non-truncated row ends the GDI section
   715→   715→        inGDISection = false;
   716→   716→      }
   717→   717→    }
   718→   718→
   719→   719→    // Override CIP from table
   720→   720→    if (cipFromTable) CIP = cipFromTable;
   721→   721→
   722→   722→    // Use table GDI values if found, otherwise keep text-based fallback
   723→   723→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
   724→   724→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
   725→   725→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
   726→   726→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
   727→   727→  }
   728→   728→
   729→   729→  return {
   730→   730→    UC: cleanUC, Cidade, Tipo, Mercado, Concessionaria,
   731→   731→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
   732→   732→    ModalidadeTarifaria, Subgrupo,
   733→   733→    ConsumoPonta, ConsumoForaPonta, ConsumoTotal,
   734→   734→    DemandaContratadaPonta, DemandaContratadaForaPonta,
   735→   735→    DemandaMedidaPonta, DemandaMedidaForaPonta,
   736→   736→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
   737→   737→    UltrapassagemPonta, UltrapassagemForaPonta,
   738→   738→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
   739→   739→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
   740→   740→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
   741→   741→    ValorConsumoPonta, ValorConsumoForaPonta,
   742→   742→    ValorDemandaPonta, ValorDemandaForaPonta,
   743→   743→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
   744→   744→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
   745→   745→    PIS, COFINS, ICMS, GDI, CIP,
   746→   746→    EnergiaReativaTotal,
   747→   747→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
   748→   748→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
   749→   749→    SaldoAcumuladoGD,
   750→   750→    ValorTotal,
   751→   751→  };

   754→   754→// ===================== GDI EXTRACTION =====================
   755→   755→
   756→   756→function extractGDI(lines: string[], flat: string) {
   757→   757→  const EMPTY = '';
   758→   758→  let nOUC = 0;
   759→   759→  let nMUC = 0;
   760→   760→  let firstGDI = -1;
   761→   761→
   762→   762→  for (let i = 0; i < lines.length; i++) {
   763→   763→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
   764→   764→      nOUC++;
   765→   765→      if (firstGDI < 0) firstGDI = i;
   766→   766→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
   767→   767→      nMUC++;
   768→   768→      if (firstGDI < 0) firstGDI = i;
   769→   769→    }
   770→   770→  }
   771→   771→
   772→   772→  if (nOUC === 0 && nMUC === 0) {
   773→   773→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
   774→   774→  }
   775→   775→
   776→   776→  const chaveIdx = (() => {
   777→   777→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
   778→   778→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
   779→   779→    }
   780→   780→    return -1;
   781→   781→  })();
   782→   782→
   783→   783→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
   784→   784→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
   785→   785→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
   786→   786→      sectionEnd = i; break;
   787→   787→    }
   788→   788→  }
   789→   789→  if (chaveIdx >= 0) {
   790→   790→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
   791→   791→  }
   792→   792→
   793→   793→  const allQtds: number[] = [];
   794→   794→  if (chaveIdx >= 0) {
   795→   795→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
   796→   796→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   797→   797→      if (m) allQtds.push(parseBR(m[1]));
   798→   798→    }
   799→   799→  }
   800→   800→  if (allQtds.length === 0) {
   801→   801→    for (let i = firstGDI; i < sectionEnd; i++) {
   802→   802→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   803→   803→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
   804→   804→      if (allQtds.length >= nOUC + nMUC) break;
   805→   805→    }
   806→   806→  }
   807→   807→
   808→   808→  const allRVals: number[] = [];
   809→   809→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
   810→   810→  const totalNeeded = nOUC + nMUC;
   811→   811→
   812→   812→  for (let i = firstGDI; i < rSearchEnd; i++) {
   813→   813→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
   814→   814→    if (m && allRVals.length < totalNeeded) {
   815→   815→      allRVals.push(parseBR('-' + m[2]));
   816→   816→    }
   817→   817→  }
   818→   818→  if (allRVals.length < totalNeeded) {
   819→   819→    for (let i = firstGDI; i < rSearchEnd; i++) {
   820→   820→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
   821→   821→      if (m && allRVals.length < totalNeeded) {
   822→   822→        allRVals.push(parseBR(m[1]));
   823→   823→      }
   824→   824→    }
   825→   825→  }
   826→   826→
   827→   827→  const oUCQtds = allQtds.splice(0, nOUC);
   828→   828→  const oUCRVals = allRVals.splice(0, nOUC);
   829→   829→
   830→   830→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
   831→   831→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
   832→   832→
   833→   833→  return {
   834→   834→    oUC_Qtd: sumQtd(oUCQtds),
   835→   835→    oUC_Valor: sumR(oUCRVals),
   836→   836→    mUC_Qtd: sumQtd(allQtds),
   837→   837→    mUC_Valor: sumR(allRVals),
   838→   838→  };
   839→   839→}

   841→   841→// ===================== NUMBER FORMATTING =====================
   842→   842→
   843→   843→function fmtBRNumber(n: number): string {
   844→   844→  const abs = Math.abs(n);
   845→   845→  const intPart = Math.floor(abs).toString();
   846→   846→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
   847→   847→  let formatted = '';
   848→   848→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
   849→   849→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
   850→   850→    formatted = intPart[i] + formatted;
   851→   851→  }
   852→   852→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
   853→   853→}
   854→   854→
   855→   855→function fmtSumBR(a: string, b: string): string {
   856→   856→  return fmtBRNumber(parseBR(a) + parseBR(b));
   857→   857→}

   861→   861→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
   862→   862→  const idx = findLine(lines, labelPattern);
   863→   863→  if (idx < 0) return EMPTY;
   864→   864→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
   865→   865→    const line = lines[i];
   866→   866→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
   867→   867→    const m = line.match(/([\d.]+,\d{2})/);
   868→   868→    if (m) return m[1];
   869→   869→  }
   870→   870→  return EMPTY;
   871→   871→}

   873→   873→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
   874→   874→  const idx = findLine(lines, labelPattern);
   875→   875→  if (idx < 0) return EMPTY;
   876→   876→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
   877→   877→  const moneyVals = findAllMoney(region);
   878→   878→  if (moneyVals.length > 0) return moneyVals[moneyVals.length - 1];
   879→   879→  return EMPTY;
   880→   880→}

   882→   882→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
   883→   883→  const idx = findLine(lines, labelPattern);
   884→   884→  if (idx >= 0) {
   885→   885→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
   886→   886→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   887→   887→      if (m) return m[1];
   888→   888→    }
   889→   889→  }
   890→   890→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
   891→   891→  return flatMatch ? flatMatch[1] : EMPTY;
   892→   892→}
