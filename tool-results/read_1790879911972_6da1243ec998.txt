    81→function parseBR(s: string): number {
    82→  const neg = s.startsWith('-');
    83→  const clean = neg ? s.slice(1) : s;
    84→  return parseFloat(clean.replace('.', '').replace(',', '.')) * (neg ? -1 : 1);
    85→}
    86→
    87→function formatBR(n: number): string {
    88→  if (n === 0) return '0,00';
    89→  const neg = n < 0;
    90→  const abs = Math.abs(n);
    91→  const rounded = Math.round(abs * 100) / 100;
    92→  const parts = rounded.toFixed(2).split('.');
    93→  const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    94→  const result = intPart + ',' + parts[1];
    95→  return neg ? '-' + result : result;
    96→}

    98→function findLine(lines: string[], pattern: RegExp): number {
    99→  return lines.findIndex(l => pattern.test(l));

   102→function findAllMoney(text: string): string[] {
   103→  const matches = text.match(/R\$\s*([\d.]+,\d{2})/g);
   104→  if (!matches) return [];
   105→  return matches.map(m => m.match(/([\d.]+,\d{2})/)?.[1] || '').filter(Boolean);
   106→}

   108→function findNumNearLine(lines: string[], startIdx: number, direction: 1 | -1, maxDist: number, mustBeOnlyNum: boolean): string {
   109→  for (let i = startIdx + direction; i >= 0 && i < lines.length; i += direction) {
   110→    if (Math.abs(i - startIdx) > maxDist) break;
   111→    const line = lines[i];
   112→    if (mustBeOnlyNum) {
   113→      const m = line.match(/^(\d[\d.]+,\d{2})$/);
   114→      if (m) return m[1];
   115→    } else {
   116→      const m = line.match(/([\d.]+,\d{2})/);
   117→      if (m && !/^[A-Za-z]/.test(line)) return m[1];
   118→    }
   119→  }
   120→  return EMPTY;
   121→}

   123→// ===================== MAIN EXTRACTOR =====================

   125→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   126→  const tableData = tableExtraction?.items;
   127→  const tributosFromTable = tableExtraction
   128→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   129→    : null;
   130→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   131→  const flat = lines.join(' ');
   132→
   133→  // ===== CONCESSIONÁRIA =====
   134→  const Concessionaria = /ENERGISA/i.test(flat) ? 'ENERGISA' : EMPTY;
   135→
   136→  // ===== TIPO =====
   137→  const Tipo = (() => {
   138→    const u = flat.toUpperCase();
   139→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   140→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   141→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   142→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   143→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   144→    if (/RURAL/.test(u)) return 'RURAL';
   145→    return EMPTY;
   146→  })();
   147→
   148→  // ===== MERCADO =====
   149→  const Mercado = (() => {
   150→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   151→    if (m) {
   152→      const val = m[1].trim().replace(/\s+/g, ' ');
   153→      if (val.length < 50) return val.toUpperCase();
   154→    }
   155→    return EMPTY;
   156→  })();
   157→
   158→  // ===== CIDADE =====
   159→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   160→  const Cidade = (() => {
   161→    for (const line of lines.slice(0, 60)) {
   162→      // Match: CityName [StateAbbr] (AG: XX)
   163→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   164→      if (m) {
   165→        const name = m[1].trim().toUpperCase();
   166→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   167→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   168→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   169→      }
   170→    }
   171→    return EMPTY;
   172→  })();
   173→
   174→  // ===== UC =====
   175→  const UC = (() => {
   176→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   177→    const isCnpjCpf = (s: string) => {
   178→      const raw = s.replace(/[.\-\/]/g, '');
   179→      return raw.length === 14 || raw.length === 11;
   180→    };
   181→
   182→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   183→    for (const line of lines) {
   184→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   185→      if (m) {
   186→        const num = m[1].replace(/^0+/, '') || '0';
   187→        if (num.length >= 3) return num;
   188→      }
   189→    }
   190→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   191→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   192→    if (slashCtx) {
   193→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   194→      if (num.length >= 3) return num;
   195→    }
   196→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   197→    // e.g., "1.409.407.051-47" → "140940705147"
   198→    for (const line of lines) {
   199→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   200→      if (m && !isCnpjCpf(m[1])) return m[1];
   201→    }
   202→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   203→    for (const line of lines) {
   204→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   205→      if (m) return m[1];
   206→    }
   207→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   208→    for (const line of lines) {
   209→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   210→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   211→    }
   212→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   213→    for (const line of lines) {
   214→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   215→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   216→    }
   217→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   218→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   219→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   220→    if (ucMat) return ucMat[1];
   221→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   222→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   223→    if (ucAddr) return ucAddr[1];
   224→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   225→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   226→    if (ucInst) return ucInst[1];
   227→    // Strategy 10: UC label with number — 4-13 digits
   228→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   229→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   230→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   231→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   232→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   233→    // Strategy 12: broad search for 3-group dotted format
   234→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   235→    if (broad3) return broad3[1];
   236→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   237→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   238→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   239→    return EMPTY;
   240→  })();
   241→
   242→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   243→  const cleanUC = UC.replace(/[.\-]/g, '');
   244→
   245→  // ===== ENDEREÇO =====
   246→  const Endereco = (() => {
   247→    // Address line typically follows the city (AG: XX) line and starts with RUA, AV, AVE, LOT, ESTRADA, etc.
   248→    // Examples from real PDFs:
   249→    //   "RUA SALDANHA DA GAMA, 896 - 21.004.65.029401 LAD RE RA - 422 - 79370000"
   250→    //   "LOT RECANTO NUARA, S/N - 24.416.10.154000 PAT 01/REL01 2289 - 79190000"
   251→    //   "AVE MATO GROSSO DO SUL, SN - 12.029.07.090000 COX EAT 006 2243 - 79400000"
   252→    const STREET_RE = /^(RUA|AV[. ]|AVE |AVENIDA|ESTRADA|LOT |RODOVIA|TRAVESSA|ALAMEDA|PCA[. ]|PRACA)/i;
   253→
   254→    // Clean an address line: keep street + number, strip internal codes, keep CEP
   255→    const extractCleanAddress = (line: string): string => {
   256→      // Split at the first " - " that precedes an internal code pattern (XX.XXX.XX.XXXXXX)
   257→      const beforeCodes = line.split(/\s*-\s*\d{2}\.\d{3}/)[0];
   258→      let addr = (beforeCodes || line).trim();
   259→      // Remove trailing dash
   260→      addr = addr.replace(/\s*-\s*$/, '').trim();
   261→      // Append CEP if found at end of original line
   262→      const cepMatch = line.match(/(\d{5}-?\d{3})\s*$/);
   263→      if (cepMatch) addr += ' - CEP ' + cepMatch[1];
   264→      return addr;
   265→    };
   266→
   267→    // Strategy 1: Find address line after city (AG: XX) pattern
   268→    for (let i = 0; i < Math.min(60, lines.length); i++) {
   269→      if (/\(\s*AG\s*:\s*\d+\s*\)/.test(lines[i])) {
   270→        // Check next 3 lines for an address pattern
   271→        for (let j = i + 1; j <= Math.min(i + 3, lines.length - 1); j++) {
   272→          if (STREET_RE.test(lines[j])) {
   273→            return extractCleanAddress(lines[j]);
   274→          }
   275→        }
   276→      }
   277→    }
   278→
   279→    // Strategy 2: Find address line near CNPJ/CPF line (appears before it)
   280→    for (let i = 0; i < lines.length; i++) {
   281→      if (/CNPJ\/CPF/i.test(lines[i]) && i > 0) {
   282→        if (STREET_RE.test(lines[i - 1])) {
   283→          return extractCleanAddress(lines[i - 1]);
   284→        }
   285→      }
   286→    }
   287→
   288→    // Strategy 3: Look for "ENDEREÇO" label followed by address
   289→    for (let i = 0; i < lines.length; i++) {
   290→      if (/^ENDERE[ÇC]O$/i.test(lines[i].trim()) && i + 1 < lines.length) {
   291→        const next = lines[i + 1];
   292→        const cleanAddr = next.replace(/\s*-\s*(SA[IÍ]DA|ENTRADA)\s+PARA\s+[^-]+/i, '')
   293→          .replace(/\s*-\s*[A-ZÀ-Ú]+\s*\/\s*[A-Z]{2}\s*-\s*CEP\s+\d{5}-\d{3}/i, '').trim();
   294→        if (cleanAddr.length > 5) return cleanAddr;
   295→      }
   296→    }
   297→
   298→    return EMPTY;
   299→  })();
   300→
   301→  // ===== MÊS REFERÊNCIA =====
   302→  const MesReferencia = (() => {
   303→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   304→    const numToMonth: Record<string, string> = {
   305→      '01': 'Janeiro', '02': 'Fevereiro', '03': 'Março', '04': 'Abril',
   306→      '05': 'Maio', '06': 'Junho', '07': 'Julho', '08': 'Agosto',
   307→      '09': 'Setembro', '10': 'Outubro', '11': 'Novembro', '12': 'Dezembro',
   308→    };
   309→    const capitalize = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
   310→
   311→    // Helper: get Vencimento date components for sanity checks
   312→    const venMoneyMatch = flat.match(/(\d{2})\/(\d{2})\/(\d{4})\s*R\$/);
   313→    const venMonth = venMoneyMatch ? parseInt(venMoneyMatch[2], 10) : 0;
   314→    const venYear = venMoneyMatch ? parseInt(venMoneyMatch[3], 10) : 0;
   315→
   316→    // Helper: check if a month/year is consistent with the Vencimento date
   317→    // (reference month is typically 1-2 months before the vencimento month)
   318→    const isConsistentWithVencimento = (month: number, year: number): boolean => {
   319→      if (!venMonth || !venYear) return true; // no vencimento to compare
   320→      const venTotal = venYear * 12 + venMonth;
   321→      const refTotal = year * 12 + month;
   322→      const diff = venTotal - refTotal;
   323→      return diff >= 1 && diff <= 3; // reference is 1-3 months before vencimento
   324→    };
   325→
   326→    // Strategy 1a: "Mês / Ano" pattern BEFORE date + R$ on the SAME LINE (most reliable)
   327→    // Energisa format: "Fevereiro / 202621/03/2026R$ 538,49" or "Junho / 2026  21/07/2026  R$ 13.356,50"
   328→    for (const line of lines) {
   329→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})\\s*\\d{2}/\\d{2}/\\d{4}\\s*R\\$`, 'i'));
   330→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   331→    }
   332→    // Strategy 1b: date + R$ pattern BEFORE "Mês / Ano" on the SAME LINE
   333→    // pdf-parse reordering: "21/07/2026R$ 6.980,48Junho / 2026"
   334→    for (const line of lines) {
   335→      const m = line.match(new RegExp(`\\d{2}/\\d{2}/\\d{4}\\s*R\\$\\s*[\\d.,]+\\s*(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   336→      if (m) return `${capitalize(m[1])} / ${m[2]}`;
   337→    }
   338→    // Strategy 2: "Mês / Ano" pattern on the SAME LINE as VENCIMENTO or NOTA FISCAL
   339→    for (const line of lines) {
   340→      if (/VENCIMENTO|NOTA FISCAL|Nº FATURA/i.test(line)) {
   341→        const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   342→        if (m) return `${capitalize(m[1])} / ${m[2]}`;
   343→      }
   344→    }
   345→    // Strategy 3: "Mês / Ano" on a line that also contains a date DD/MM/YYYY (likely the billing line)
   346→    for (const line of lines) {
   347→      const m = line.match(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'i'));
   348→      if (m && /\d{2}\/\d{2}\/\d{4}/.test(line)) {
   349→        const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   350→        const monthNum = monthIdx + 1;
   351→        if (isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) {
   352→          return `${capitalize(m[1])} / ${m[2]}`;
   353→        }
   354→      }
   355→    }
   356→    // Strategy 4: "Mês / Ano" anywhere in flat text — skip "Desde mês/ano", "Referência:" history, etc.
   357→    const allNameMatches = [...flat.matchAll(new RegExp(`(${meses})\\s*/\\s*(\\d{4})`, 'gi'))];
   358→    for (const m of allNameMatches) {
   359→      // Check context: skip if preceded by "Desde", "Referência:", "Ref", or other informative words
   360→      const start = Math.max(0, m.index! - 20);
   361→      const context = flat.slice(start, m.index! + m[0].length).toLowerCase();
   362→      if (/desde|até|a partir|inform|dispon|referência|ref\b/.test(context)) continue;
   363→      // Also validate against Vencimento date
   364→      const monthIdx = Object.values(numToMonth).indexOf(capitalize(m[1]));
   365→      const monthNum = monthIdx + 1;
   366→      if (!isConsistentWithVencimento(monthNum, parseInt(m[2], 10))) continue;
   367→      return `${capitalize(m[1])} / ${m[2]}`;
   368→    }
   369→    // Strategy 5: Use Leitura Atual date to derive the reference month
   370→    // pdf-parse sometimes garbles text, so try multiple patterns
   371→    const leituraPatterns = [
   372→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*(\d{2})\s*\/\s*(\d{4})/i,
   373→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*(\d{2})\s*\/\s*(\d{4})/i,
   374→    ];
   375→    for (const pat of leituraPatterns) {
   376→      const lm = flat.match(pat);
   377→      if (lm && numToMonth[lm[1]]) {
   378→        const monthNum = parseInt(lm[1], 10);
   379→        const year = parseInt(lm[2], 10);
   380→        if (isConsistentWithVencimento(monthNum, year)) {
   381→          return `${numToMonth[lm[1]]} / ${lm[2]}`;
   382→        }
   383→      }
   384→    }
   385→    // Strategy 6: numeric MM/YYYY near Vencimento date — with consistency check
   386→    if (venMoneyMatch) {
   387→      const venIdx = flat.indexOf(venMoneyMatch[0]);
   388→      const region = flat.slice(Math.max(0, venIdx - 150), venIdx + 50);
   389→      const numMatches = [...region.matchAll(/(\d{2})\s*\/\s*(\d{4})/g)];
   390→      for (const nm of numMatches) {
   391→        if (numToMonth[nm[1]]) {
   392→          const monthNum = parseInt(nm[1], 10);
   393→          const year = parseInt(nm[2], 10);
   394→          if (isConsistentWithVencimento(monthNum, year)) {
   395→            return `${numToMonth[nm[1]]} / ${nm[2]}`;
   396→          }
   397→        }
   398→      }
   399→    }
   400→    // Strategy 7: first valid MM/YYYY that is a month and consistent with Vencimento
   401→    const allNumMatches = [...flat.matchAll(/(\d{2})\s*[\/]\s*(\d{4})/g)];
   402→    for (const nm of allNumMatches) {
   403→      if (numToMonth[nm[1]]) {
   404→        const monthNum = parseInt(nm[1], 10);
   405→        const year = parseInt(nm[2], 10);
   406→        if (isConsistentWithVencimento(monthNum, year)) {
   407→          return `${numToMonth[nm[1]]} / ${nm[2]}`;
   408→        }
   409→      }
   410→    }
   411→    return EMPTY;
   412→  })();
   413→
   414→  // ===== ANO LEITURA =====
   415→  const AnoLeitura = (() => {
   416→    // Try multiple patterns for Leitura Atual date (pdf-parse can garble text)
   417→    const leituraYearPatterns = [
   418→      /Leitura\s+Atual\s*:\s*\d{2}\s*\/\s*\d{2}\s*\/\s*(\d{4})/i,
   419→      /Leitura\s+Atual\s*:?\s*\d{2,4}\/?\s*\d{2}\s*\/\s*(\d{4})/i,
   420→    ];
   421→    for (const pat of leituraYearPatterns) {
   422→      const m = flat.match(pat);
   423→      if (m) return m[1];
   424→    }
   425→    // Fallback: extract year from MesReferencia
   426→    if (MesReferencia) {
   427→      const ym = MesReferencia.match(/(\d{4})/);
   428→      if (ym) return ym[1];
   429→    }
   430→    return EMPTY;
   431→  })();
   432→
   433→  // ===== VENCIMENTO =====
   434→  // Best pattern: DD/MM/YYYY immediately followed by R$ (date + total on same line)
   435→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   436→  // Also handles: "Junho / 202621/07/2026R$ 1.635,98"
   437→  const Vencimento = (() => {
   438→    // Strategy 1: date + R$ on same line (most reliable for Energisa MS)
   439→    const venMoney = flat.match(/(\d{2}\/\d{2}\/\d{4})\s*R\$\s*[\d.]+,\d{2}/);
   440→    if (venMoney) return venMoney[1];
   441→
   442→    // Strategy 2: R$ + money + date on same line (reversed order)
   443→    const moneyVen = flat.match(/R\$\s*[\d.]+,\d{2}\s*(\d{2}\/\d{2}\/\d{4})/);
   444→    if (moneyVen) return moneyVen[1];
   445→
   446→    // Strategy 3: VENCIMENTO label nearby (original approach)
   447→    const venIdx = findLine(lines, /VENCIMENTO/i);
   448→    if (venIdx >= 0) {
   449→      for (let i = venIdx - 1; i >= Math.max(0, venIdx - 5); i--) {
   450→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   451→        if (m) return m[1];
   452→      }
   453→      for (let i = venIdx + 1; i < Math.min(venIdx + 3, lines.length); i++) {
   454→        const m = lines[i].match(/(\d{2}\/\d{2}\/\d{4})/);
   455→        if (m) return m[1];
   456→      }
   457→    }
   458→
   459→    // Strategy 4: date in "Data de Apresentação:" line (next business day, close enough)
   460→    const apresMatch = flat.match(/Data de Apresentação:\s*(\d{2}\/\d{2}\/\d{4})/i);
   461→    if (apresMatch) return apresMatch[1];
   462→
   463→    return EMPTY;
   464→  })();
   465→
   466→  // ===== NOTA FISCAL =====
   467→  const NotaFiscal = (() => {
   468→    const m = flat.match(/NOTA\s+FISCAL\s*N[º°]?\s*[:\-]?\s*(\d[\d.]*)/i);
   469→    return m ? m[1] : EMPTY;
   470→  })();
   471→
   472→  // ===== MODALIDADE TARIFÁRIA =====
   473→  const ModalidadeTarifaria = (() => {
   474→    if (/CONVENCIONAL/i.test(flat)) return 'CONVENCIONAL';
   475→    if (/AZUL/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO AZUL';
   476→    if (/AZUL/i.test(flat)) return 'AZUL';
   477→    if (/VERDE/i.test(flat) && /HOR[ÁA]RIO/i.test(flat)) return 'HORÁRIO VERDE';
   478→    if (/VERDE/i.test(flat)) return 'VERDE';
   479→    return EMPTY;
   480→  })();
   481→
   482→  // ===== SUBGRUPO =====
   483→  const Subgrupo = (() => {
   484→    // Look for "/A4" or "/B3" etc. in the classification line
   485→    // Must be a letter + single digit, right after "/"
   486→    const m = flat.match(/\/\s*([AB]\d)\s/);
   487→    if (m) return m[1].toUpperCase();
   488→    const classIdx = findLine(lines, /Classifica[çc][aã]o/i);
   489→    if (classIdx >= 0) {
   490→      const region = lines.slice(classIdx, Math.min(classIdx + 3, lines.length)).join(' ');
   491→      const m2 = region.match(/\b([AB][1-4]|AS|BS)\b/);
   492→      if (m2) return m2[1].toUpperCase();
   493→    }
   494→    return EMPTY;
   495→  })();
   496→
   497→  const isSplitTariff = /AZUL|VERDE/i.test(flat) && !/CONVENCIONAL/i.test(flat);
   498→
   499→  // ===== CONSUMO (text-based, will be overridden by table data if available) =====
   500→  let ConsumoPonta = EMPTY;
   501→  let ConsumoIntermediaria = EMPTY;
   502→  let ConsumoForaPonta = EMPTY;
   503→  let ConsumoTotal = EMPTY;
   504→  let ConsumoGrupoB_kWh = EMPTY;
   505→  let ConsumoGrupoB_Valor = EMPTY;
   506→  let ValorConsumoPonta = EMPTY;
   507→  let ValorConsumoIntermediaria = EMPTY;
   508→  let ValorConsumoForaPonta = EMPTY;
   509→  // Accumulators for Grupo B consumption
   510→  let grupoB_consumoKWh_total = 0;
   511→  let grupoB_consumoValor_total = 0;
   512→
   513→  if (isSplitTariff) {
   514→    const pontaIdx = findLine(lines, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)/i);
   515→    const fpIdx = findLine(lines, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   516→    if (pontaIdx >= 0) ConsumoPonta = findNumNearLine(lines, pontaIdx, 1, 8, false);
   517→    if (fpIdx >= 0) ConsumoForaPonta = findNumNearLine(lines, fpIdx, 1, 8, false);
   518→    // Intermediária (Tarifa Branca / 3-period AZUL)
   519→    const interIdx = findLine(lines, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   520→    if (interIdx >= 0) ConsumoIntermediaria = findNumNearLine(lines, interIdx, 1, 8, false);
   521→    ValorConsumoPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Ponta(?!.*Fora)(?!.*Intermedi)/i);
   522→    ValorConsumoIntermediaria = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Intermedi[áa]ria/i);
   523→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh.*Fora\s*Ponta/i);
   524→  } else {
   525→    for (const line of lines) {
   526→      const km = line.match(/^KWH\s*([\d.,]+)$/i);
   527→      if (km) { ConsumoTotal = km[1]; break; }
   528→    }
   529→    if (!ConsumoTotal) {
   530→      const consIdx = findLine(lines, /CONSUMO\s+FATURADO/i);
   531→      if (consIdx >= 0) {
   532→        for (let i = consIdx + 1; i < Math.min(consIdx + 15, lines.length); i++) {
   533→          const line = lines[i];
   534→          if (/^(JAN|FEV|MAR|ABR|MAI|JUN|JUL|AGO|SET|OUT|NOV|DEZ)/i.test(line)) continue;
   535→          if (/[A-Za-z]{4,}/.test(line)) continue;
   536→          const m = line.match(/^(\d[\d.]+,\d{2})$/);
   537→          if (m) { ConsumoTotal = m[1]; break; }
   538→        }
   539→      }
   540→    }
   541→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   542→  }
   543→
   544→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   545→  let DemandaContratadaPonta = EMPTY;
   546→  let DemandaContratadaForaPonta = EMPTY;
   547→  let DemandaMedidaPonta = EMPTY;
   548→  let DemandaMedidaForaPonta = EMPTY;
   549→  let DemandaFaturadaPonta = EMPTY;
   550→  let DemandaFaturadaForaPonta = EMPTY;
   551→  let ValorDemandaPonta = EMPTY;
   552→  let ValorDemandaForaPonta = EMPTY;
   553→
   554→  if (isSplitTariff) {
   555→    // ===== DEMANDA CONTRATADA EXTRACTION =====
   556→    // Energisa PDFs show contracted demand in two locations:
   557→    // 1) "Grandezas Contratadas" section: "Demanda ponta - kW  15,3" / "Demanda fora ponta - kW  30"
   558→    // 2) Bottom line: "KW Ponta: 15,3  Fora Ponta: 30  TUSDG:"
   559→    //
   560→    // pdf-parse often garbles the column order, producing:
   561→    //   "Demanda ponta - kW Demanda fora ponta - kW 30 15,3"  (values reversed)
   562→    //   "30 Fora Ponta: 15,3 KW Ponta:"  (values before labels, or between labels)
   563→    //
   564→    // We try multiple strategies and use the first one that gives both values.
   565→
   566→    // Strategy 1: "Demanda ponta - kW <value>" on the SAME LINE (pdftotext-like format)
   567→    const dpSameLine = flat.match(/Demanda\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   568→    const dfpSameLine = flat.match(/Demanda\s+fora\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   569→    if (dpSameLine && dfpSameLine) {
   570→      DemandaContratadaPonta = dpSameLine[1].replace('.', ',');
   571→      DemandaContratadaForaPonta = dfpSameLine[1].replace('.', ',');
   572→    }
   573→
   574→    // Strategy 2: Detect pdf-parse column reordering in bottom line
   575→    // When "Fora Ponta:" appears BEFORE "KW Ponta:" in the text, the values are
   576→    // rearranged: "<fora_val> Fora Ponta: <ponta_val> KW Ponta:"
   577→    // - Number BEFORE "Fora Ponta:" = Fora Ponta value
   578→    // - Number BETWEEN "Fora Ponta:" and "KW Ponta:" = Ponta value
   579→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   580→      const foraIdx = flat.search(/Fora\s+Ponta\s*:/i);
   581→      const kwIdx = flat.search(/KW\s+Ponta\s*:/i);
   582→      if (foraIdx >= 0 && kwIdx >= 0 && foraIdx < kwIdx) {
   583→        // "Fora Ponta:" appears before "KW Ponta:" — this is the reordered pattern
   584→        // Number between "Fora Ponta:" and "KW Ponta:" → Ponta value
   585→        const between = flat.slice(foraIdx, kwIdx);
   586→        const betweenMatch = between.match(/:\s*(\d+(?:[.,]\d+)?)/);
   587→        if (betweenMatch && !DemandaContratadaPonta) {
   588→          DemandaContratadaPonta = betweenMatch[1].replace('.', ',');
   589→        }
   590→        // Number before "Fora Ponta:" (within 25 chars) → Fora Ponta value
   591→        const before = flat.slice(Math.max(0, foraIdx - 25), foraIdx);
   592→        const beforeMatch = before.match(/(\d+(?:[.,]\d+)?)\s*$/);
   593→        if (beforeMatch && !DemandaContratadaForaPonta) {
   594→          DemandaContratadaForaPonta = beforeMatch[1].replace('.', ',');
   595→        }
   596→      } else if (kwIdx >= 0 && foraIdx >= 0 && kwIdx < foraIdx) {
   597→        // Normal order: "KW Ponta:" appears before "Fora Ponta:" — values follow labels
   598→        const kwAfter = flat.slice(kwIdx).match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   599→        const fpAfter = flat.slice(foraIdx).match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   600→        if (kwAfter && !DemandaContratadaPonta) DemandaContratadaPonta = kwAfter[1].replace('.', ',');
   601→        if (fpAfter && !DemandaContratadaForaPonta) DemandaContratadaForaPonta = fpAfter[1].replace('.', ',');
   602→      }
   603→    }
   604→
   605→    // Strategy 3: "Demanda ponta/fora ponta - kW" labels with values on nearby lines
   606→    // (pdf-parse puts labels and values on separate lines)
   607→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   608→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
   609→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
   610→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
   611→        // Collect numeric values from the lines after both labels
   612→        const startLine = Math.max(dpLineIdx, dfpLineIdx) + 1;
   613→        const numVal = /^(\d+(?:[.,]\d+)?)$/;
   614→        const values: string[] = [];
   615→        for (let i = startLine; i < Math.min(startLine + 5, lines.length); i++) {
   616→          const m = lines[i].match(numVal);
   617→          if (m) values.push(m[1].replace('.', ','));
   618→          if (values.length >= 2) break;
   619→        }
   620→        if (values.length === 2) {
   621→          // Use already-found values to disambiguate, or assume reverse order
   622→          if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
   623→            if (values[0] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[1];
   624→            else if (values[1] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[0];
   625→          } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
   626→            if (values[0] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[1];
   627→            else if (values[1] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[0];
   628→          } else {
   629→            // No disambiguation; assume pdf-parse reverse column order:
   630→            // values[0] = Fora Ponta, values[1] = Ponta
   631→            if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = values[0];
   632→            if (!DemandaContratadaPonta) DemandaContratadaPonta = values[1];
   633→          }
   634→        }
   635→      }
   636→    }
   637→
   638→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   639→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   640→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   641→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   642→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   643→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   644→  }
   645→
   646→  // ===== ULTRAPASSAGEM =====
   647→  let UltrapassagemPonta = EMPTY;
   648→  let UltrapassagemForaPonta = EMPTY;
   649→  let ValorUltrapassagemPonta = EMPTY;
   650→  let ValorUltrapassagemForaPonta = EMPTY;
   651→  // ===== DEMANDA NÃO CONSUMIDA =====
   652→  let DemandaNaoConsumidaPonta = EMPTY;
   653→  let DemandaNaoConsumidaForaPonta = EMPTY;
   654→  let ValorNaoConsumidaPonta = EMPTY;
   655→  let ValorNaoConsumidaForaPonta = EMPTY;
   656→  if (isSplitTariff) {
   657→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   658→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   659→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   660→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   661→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   662→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   663→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   664→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   665→  }
   666→
   667→  // ===== REATIVA EXCEDENTE =====
   668→  let ReativaExcedentePonta = EMPTY;
   669→  let ReativaExcedenteForaPonta = EMPTY;
   670→  let EnergiaReativaTotal = EMPTY;
   671→  let ValorReativaExcedentePonta = EMPTY;
   672→  let ValorReativaExcedenteForaPonta = EMPTY;
   673→  if (isSplitTariff) {
   674→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   675→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   676→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   677→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   678→  } else {
   679→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   680→  }
   681→
   682→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   683→  // Priority: table extraction > text-based extraction
   684→  let PIS = EMPTY;
   685→  let COFINS = EMPTY;
   686→  let ICMS = EMPTY;
   687→
   688→  // Use tributos from table extraction if available (most reliable)
   689→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   690→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   691→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   692→
   693→  // Fallback: ELEKTRO/SANESUL pattern — ICMS, COFINS, PIS on consecutive lines,
   694→  // followed by base values, aliquotas (%), then actual tributo values.
   695→  // Example:
   696→  //   L52: ICMS
   697→  //   L53: COFINS
   698→  //   L54: PIS
   699→  //   L55: 5087,88   ← Base de cálculo ICMS
   700→  //   L56: 4222,95   ← Base de cálculo COFINS
   701→  //   L57: 4222,95   ← Base de cálculo PIS
   702→  //   L58: 17%       ← Alíquota ICMS
   703→  //   L59: 3,48%     ← Alíquota COFINS
   704→  //   L60: 0,75%     ← Alíquota PIS
   705→  //   L61: 864,93    ← Valor ICMS ✓
   706→  //   L62: 146,96    ← Valor COFINS ✓
   707→  //   L63: 31,67     ← Valor PIS ✓
   708→  if (!PIS || !COFINS || !ICMS) {
   709→    for (let i = 0; i < lines.length - 2; i++) {
   710→      if (/^ICMS$/i.test(lines[i]) && /^COFINS$/i.test(lines[i + 1]) && /^PIS$/i.test(lines[i + 2])) {
   711→        // Found the ICMS/COFINS/PIS label block
   712→        // Look for 3 money values after the labels (base values), then skip aliquotas, then get 3 more money values (actual values)
   713→        const afterLabels = lines.slice(i + 3, Math.min(i + 20, lines.length));
   714→        const moneyValues: string[] = [];
   715→        for (const line of afterLabels) {
   716→          const m = line.match(/^(\d[\d.]*,\d{2})$/);
   717→          if (m) moneyValues.push(m[1]);
   718→          if (moneyValues.length >= 6) break;
   719→        }
   720→        // If we have at least 6 money values, the last 3 are the actual tributo values
   721→        // (first 3 = bases, then 3 = actual values)
   722→        if (moneyValues.length >= 6) {
   723→          if (!ICMS) ICMS = moneyValues[3]; // 4th value = ICMS valor
   724→          if (!COFINS) COFINS = moneyValues[4]; // 5th value = COFINS valor
   725→          if (!PIS) PIS = moneyValues[5]; // 6th value = PIS valor
   726→        } else if (moneyValues.length >= 3) {
   727→          // If only 3 values found, they might be the base values
   728→          // Check if there are percentage values between the base and actual values
   729→          const afterBases = afterLabels.slice(3);
   730→          const actualValues: string[] = [];
   731→          for (const line of afterBases) {
   732→            if (/^\d+([.,]\d+)?%$/.test(line)) continue; // skip aliquotas
   733→            const m = line.match(/^(\d[\d.]*,\d{2})$/);
   734→            if (m) actualValues.push(m[1]);
   735→            if (actualValues.length >= 3) break;
   736→          }
   737→          if (actualValues.length >= 3) {
   738→            if (!ICMS) ICMS = actualValues[0];
   739→            if (!COFINS) COFINS = actualValues[1];
   740→            if (!PIS) PIS = actualValues[2];
   741→          }
   742→        }
   743→        break; // Only process the first match
   744→      }
   745→    }
   746→  }
   747→
   748→  // Fallback: text-based extraction for any missing tributo
   749→  if (!PIS || !COFINS || !ICMS) {
   750→    const pisIdx = findLine(lines, /^PIS\s*\/\s*$/i);
   751→    if (pisIdx >= 0) {
   752→      const nums: string[] = [];
   753→      for (let i = pisIdx + 1; i < Math.min(pisIdx + 10, lines.length) && nums.length < 3; i++) {
   754→        const lineNums = lines[i].match(/(\d[\d.]*,\d{2})/g);
   755→        if (lineNums) {
   756→          for (const n of lineNums) {
   757→            nums.push(n);
   758→            if (nums.length >= 3) break;
   759→          }
   760→        }
   761→      }
   762→      if (nums.length >= 3) {
   763→        if (!PIS) PIS = nums[0];
   764→        if (!COFINS) COFINS = nums[1];
   765→        if (!ICMS) ICMS = nums[2];
   766→      }
   767→    }
   768→    if (!PIS && !COFINS) {
   769→      const combinedIdx = findLine(lines, /PIS\/\s*COFINS/i);
   770→      if (combinedIdx >= 0) {
   771→        const nums: string[] = [];
   772→        for (let i = combinedIdx + 1; i < Math.min(combinedIdx + 6, lines.length) && nums.length < 3; i++) {
   773→          const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
   774→          if (m) nums.push(m[1]);
   775→        }
   776→        if (nums.length >= 3) {
   777→          if (!PIS) PIS = nums[0];
   778→          if (!COFINS) COFINS = nums[1];
   779→          if (!ICMS) ICMS = nums[2];
   780→        }
   781→      }
   782→    }
   783→    if (!PIS || !COFINS) {
   784→      const flatMatch = flat.match(/PIS\s*\/\s*COFINS[^\d]*?(\d[\d.]*,\d{2})[^\d]*?(\d[\d.]*,\d{2})/i);
   785→      if (flatMatch) {
   786→        if (!PIS) PIS = flatMatch[1];
   787→        if (!COFINS) COFINS = flatMatch[2];
   788→      }
   789→    }
   790→    if (!PIS) PIS = findTributoValue(lines, flat, /\bPIS\b(?!\/)/i);
   791→    if (!COFINS) COFINS = findTributoValue(lines, flat, /\bCOFINS\b/i);
   792→    if (!ICMS) ICMS = findTributoValue(lines, flat, /\bICMS\b(?!\s*%|\s*\()/i);
   793→  }
   794→
   795→  // ===== GDI =====
   796→  const GDI = findTributoValue(lines, flat, /\bGDI\b/i);
   797→
   798→  // ===== CIP (text-based) =====
   799→  const CIP_text = (() => {
   800→    const cipIdx = findLine(lines, /CIP\s*MUNICIPAL/i);
   801→    if (cipIdx >= 0) {
   802→      const region = lines.slice(Math.max(0, cipIdx - 1), Math.min(cipIdx + 5, lines.length)).join(' ');
   803→      const money = findAllMoney(region);
   804→      if (money.length > 0) return money[money.length - 1];
   805→    }
   806→    return findTributoValue(lines, flat, /\bCIP\b/i);
   807→  })();
   808→
   809→  // ===== ENERGIA ATIVA INJETADA GDI oUC / mUC =====
   810→  let {
   811→    oUC_Qtd: EnergiaAtvInjetadaGDIoUC_Qtd,
   812→    oUC_Valor: EnergiaAtvInjetadaGDIoUC_Valor,
   813→    mUC_Qtd: EnergiaAtvInjetadaGDImUC_Qtd,
   814→    mUC_Valor: EnergiaAtvInjetadaGDImUC_Valor,
   815→  } = extractGDI(lines, flat);
   816→
   817→  // ===== SALDO ACUMULADO GD =====
   818→  const SaldoAcumuladoGD = (() => {
   819→    const m = flat.match(/Saldo\s*Acumulado\s*:\s*([\d.]+,\d{2})/i);
   820→    if (m) return m[1];
   821→    // Fallback: integer format like "14.100"
   822→    const m2 = flat.match(/Saldo\s*Acumulado\s*:\s*(\d[\d.]*)/i);
   823→    if (m2) return m2[1];
   824→    return EMPTY;
   825→  })();
   826→
   827→  // ===== VALOR TOTAL =====
   828→  // Best pattern: R$ value on same line as vencimento date + mes referência
   829→  // e.g., "21/07/2026R$ 13.356,50Junho / 2026"
   830→  const ValorTotal = (() => {
   831→    // Strategy 1: R$ on same line as date and mes referência (most reliable)
   832→    const venTotal = flat.match(/\d{2}\/\d{2}\/\d{4}\s*R\$\s*([\d.]+,\d{2})/);
   833→    if (venTotal) return venTotal[1];
   834→
   835→    // Strategy 2: R$ + value + mes referência on same line
   836→    const totalMes = flat.match(/R\$\s*([\d.]+,\d{2})\s*(?:Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro)/i);
   837→    if (totalMes) return totalMes[1];
   838→
   839→    // Strategy 3: "TOTAL A PAGAR" with R$
   840→    const totalIdx = findLine(lines, /TOTAL\s+A\s+PAGAR/i);
   841→    if (totalIdx >= 0) {
   842→      const m1 = lines[totalIdx].match(/R\$\s*([\d.]+,\d{2})/);
   843→      if (m1) return m1[1];
   844→      for (let i = totalIdx + 1; i < Math.min(totalIdx + 3, lines.length); i++) {
   845→        const m2 = lines[i].match(/R\$\s*([\d.]+,\d{2})/);
   846→        if (m2) return m2[1];
   847→        const m3 = lines[i].match(/^(\d[\d.]+,\d{2})$/);
   848→        if (m3) return m3[1];
   849→      }
   850→    }
   851→
   852→    // Strategy 4: Last resort - largest R$ value
   853→    const lastMoney = findAllMoney(flat);
   854→    if (lastMoney.length > 0) {
   855→      lastMoney.sort((a, b) => parseFloat(b.replace('.', '').replace(',', '.')) - parseFloat(a.replace('.', '').replace(',', '.')));
   856→      return lastMoney[0];
   857→    }
   858→    return EMPTY;
   859→  })();
   860→
   861→  // ===== TABLE DATA OVERRIDES =====
   862→  // Table data from pdfjs position-based extraction is much more reliable for tabular data
   863→  let CIP = CIP_text;
   864→
   865→  if (tableData && tableData.length > 0) {
   866→    let cipFromTable = EMPTY;
   867→    // Reset GDI accumulators for clean table-based extraction
   868→    let tbl_oUC_Qtd = '' as string;
   869→    let tbl_oUC_Valor = '' as string;
   870→    let tbl_mUC_Qtd = '' as string;
   871→    let tbl_mUC_Valor = '' as string;
   872→    // Track GDI section in table for rows with truncated/missing labels
   873→    let inGDISection = false;
   874→    let gdiSectionType = '' as string; // 'oUC' or 'mUC'
   875→    let lastGDILabel = '' as string;
   876→
   877→    for (const item of tableData) {
   878→      const lbl = item.label;
   879→      const isFora = /Fora/i.test(lbl) || /F[\s.]*Ponta/i.test(lbl);
   880→      const isPonta = /Ponta/i.test(lbl) && !isFora;
   881→
   882→      // Consumo - check Convencional (no Ponta/Fora) first, then split
   883→      if (/Consumo\s+em\s+kWh/i.test(lbl) && !/Reativ/i.test(lbl) && !/Injetada/i.test(lbl)) {
   884→        // Accumulate for Grupo B (regardless of Ponta/Fora/Intermediária)
   885→        if (item.quantidade) {
   886→          const qtd = parseBR(item.quantidade);
   887→          if (qtd > 0) grupoB_consumoKWh_total += qtd;
   888→        }
   889→        if (item.valor) {
   890→          const val = parseBR(item.valor);
   891→          if (val > 0) grupoB_consumoValor_total += val;
   892→        }
   893→        // Convencional: label is exactly "Consumo em kWh" without Ponta/Fora
   894→        if (!/Ponta/i.test(lbl) && !/Fora/i.test(lbl) && !/Intermedi/i.test(lbl)) {
   895→          if (item.quantidade) ConsumoTotal = item.quantidade;
   896→          if (item.valor) ValorConsumoForaPonta = item.valor;
   897→        }
   898→        // BRANCA/AZUL Intermediária — dedicated field
   899→        else if (/Intermedi[áa]ria/i.test(lbl)) {
   900→          if (item.quantidade) ConsumoIntermediaria = item.quantidade;
   901→          if (item.valor) ValorConsumoIntermediaria = item.valor;
   902→        }
   903→        // Split tariff (AZUL/VERDE)
   904→        else if (isPonta) {
   905→          if (item.quantidade) ConsumoPonta = item.quantidade;
   906→          if (item.valor) ValorConsumoPonta = item.valor;
   907→        } else if (isFora) {
   908→          if (item.quantidade) ConsumoForaPonta = item.quantidade;
   909→          if (item.valor) ValorConsumoForaPonta = item.valor;
   910→        }
   911→      }
   912→      // Demanda Medida
   913→      else if (/Demanda.*Medida/i.test(lbl) && !/Ultrapassagem/i.test(lbl) && !/Não.*Consumida/i.test(lbl)) {
   914→        if (!isFora) {
   915→          if (item.quantidade) DemandaMedidaPonta = item.quantidade;
   916→          if (item.valor) ValorDemandaPonta = item.valor;
   917→        } else {
   918→          if (item.quantidade) DemandaMedidaForaPonta = item.quantidade;
   919→          if (item.valor) ValorDemandaForaPonta = item.valor;
   920→        }
   921→      }
   922→      // Ultrapassagem
   923→      else if (/(?:Ultrapassagem|Ultrap)/i.test(lbl) && /Demanda/i.test(lbl) && !/Não\s*Consumida/i.test(lbl)) {
   924→        if (!isFora) {
   925→          if (item.quantidade) UltrapassagemPonta = item.quantidade;
   926→          if (item.valor) ValorUltrapassagemPonta = item.valor;
   927→        } else {
   928→          if (item.quantidade) UltrapassagemForaPonta = item.quantidade;
   929→          if (item.valor) ValorUltrapassagemForaPonta = item.valor;
   930→        }
   931→      }
   932→      // Demanda Potência Não Consumida
   933→      else if (/Não\s*Consumida/i.test(lbl) && /Demanda/i.test(lbl)) {
   934→        if (!isFora) {
   935→          if (item.quantidade) DemandaNaoConsumidaPonta = item.quantidade;
   936→          if (item.valor) ValorNaoConsumidaPonta = item.valor;
   937→        } else {
   938→          if (item.quantidade) DemandaNaoConsumidaForaPonta = item.quantidade;
   939→          if (item.valor) ValorNaoConsumidaForaPonta = item.valor;
   940→        }
   941→      }
   942→      // Reativa Excedente
   943→      else if (/Reactiv?a\s*Excedente/i.test(lbl) || /Energia\s*Reativa\s*Exced/i.test(lbl)) {
   944→        if (!isFora && /Ponta/i.test(lbl)) {
   945→          if (item.quantidade) ReativaExcedentePonta = item.quantidade;
   946→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   947→        } else if (isFora || /F[\s.]*Ponta/i.test(lbl)) {
   948→          if (item.quantidade) ReativaExcedenteForaPonta = item.quantidade;
   949→          if (item.valor) ValorReativaExcedenteForaPonta = item.valor;
   950→        } else if (!isFora && !/Ponta/i.test(lbl)) {
   951→          // Generic reativa (Convencional)
   952→          if (item.quantidade) EnergiaReativaTotal = item.quantidade;
   953→          if (item.valor) ValorReativaExcedentePonta = item.valor;
   954→        }
   955→      }
   956→      // CIP Municipal
   957→      else if (/CIP\s*MUNICIPAL/i.test(lbl) || /^CONT\.IL\.PUB/i.test(lbl)) {
   958→        if (item.valor) cipFromTable = item.valor;
   959→      }
   960→      // GDI items
   961→      else if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lbl)) {
   962→        if (item.quantidade && !tbl_oUC_Qtd) tbl_oUC_Qtd = item.quantidade;
   963→        if (item.valor && !tbl_oUC_Valor) tbl_oUC_Valor = item.valor;
   964→        inGDISection = true; gdiSectionType = 'oUC'; lastGDILabel = lbl;
   965→      } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lbl)) {
   966→        // Multiple mUC entries: accumulate
   967→        if (item.quantidade) {
   968→          tbl_mUC_Qtd = tbl_mUC_Qtd
   969→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   970→            : item.quantidade;
   971→        }
   972→        if (item.valor) {
   973→          tbl_mUC_Valor = tbl_mUC_Valor
   974→            ? fmtSumBR(tbl_mUC_Valor, item.valor)
   975→            : item.valor;
   976→        }
   977→        inGDISection = true; gdiSectionType = 'mUC'; lastGDILabel = lbl;
   978→      } else if (inGDISection && (/^K?WH$/i.test(lbl.trim()) || lbl.trim() === '') && item.quantidade) {
   979→        // Row with truncated/missing label inside GDI section (e.g. just "KWH" or empty)
   980→        // Accumulate as the same GDI type as the last full-label row
   981→        if (gdiSectionType === 'mUC') {
   982→          tbl_mUC_Qtd = tbl_mUC_Qtd
   983→            ? fmtSumBR(tbl_mUC_Qtd, item.quantidade)
   984→            : item.quantidade;
   985→          if (item.valor) {
   986→            tbl_mUC_Valor = tbl_mUC_Valor
   987→              ? fmtSumBR(tbl_mUC_Valor, item.valor)
   988→              : item.valor;
   989→          }
   990→        } else if (gdiSectionType === 'oUC') {
   991→          tbl_oUC_Qtd = tbl_oUC_Qtd
   992→            ? fmtSumBR(tbl_oUC_Qtd, item.quantidade)
   993→            : item.quantidade;
   994→          if (item.valor) {
   995→            tbl_oUC_Valor = tbl_oUC_Valor
   996→              ? fmtSumBR(tbl_oUC_Valor, item.valor)
   997→              : item.valor;
   998→          }
   999→        }
  1000→      } else if (lbl.trim() !== '' && !/^K?WH$/i.test(lbl.trim())) {
  1001→        // Non-GDI, non-truncated row ends the GDI section
  1002→        inGDISection = false;
  1003→      }
  1004→    }
  1005→
  1006→    // Override CIP from table
  1007→    if (cipFromTable) CIP = cipFromTable;
  1008→
  1009→    // Use table GDI values if found, otherwise keep text-based fallback
  1010→    if (tbl_oUC_Qtd) EnergiaAtvInjetadaGDIoUC_Qtd = tbl_oUC_Qtd;
  1011→    if (tbl_oUC_Valor) EnergiaAtvInjetadaGDIoUC_Valor = tbl_oUC_Valor;
  1012→    if (tbl_mUC_Qtd) EnergiaAtvInjetadaGDImUC_Qtd = tbl_mUC_Qtd;
  1013→    if (tbl_mUC_Valor) EnergiaAtvInjetadaGDImUC_Valor = tbl_mUC_Valor;
  1014→  }
  1015→
  1016→  // ===== Grupo B: calculate dedicated fields =====
  1017→  // For Subgrupo B (B1, B2, B3, B4), fill ConsumoGrupoB_kWh and ConsumoGrupoB_Valor
  1018→  const isGrupoB = /^B/i.test(Subgrupo);
  1019→  if (isGrupoB) {
  1020→    // ConsumoGrupoB_kWh: total consumption in kWh
  1021→    if (grupoB_consumoKWh_total > 0) {
  1022→      ConsumoGrupoB_kWh = formatBR(grupoB_consumoKWh_total);
  1023→    } else if (ConsumoTotal) {
  1024→      // Fallback: use ConsumoTotal if no table accumulation
  1025→      ConsumoGrupoB_kWh = ConsumoTotal;
  1026→    } else {
  1027→      // Fallback: sum ConsumoPonta + ConsumoForaPonta
  1028→      const p = ConsumoPonta ? parseBR(ConsumoPonta) : 0;
  1029→      const f = ConsumoForaPonta ? parseBR(ConsumoForaPonta) : 0;
  1030→      if (p + f > 0) ConsumoGrupoB_kWh = formatBR(p + f);
  1031→    }
  1032→    // ConsumoGrupoB_Valor: total consumption value in R$
  1033→    if (grupoB_consumoValor_total > 0) {
  1034→      ConsumoGrupoB_Valor = formatBR(grupoB_consumoValor_total);
  1035→    } else {
  1036→      // Fallback: sum ValorConsumoPonta + ValorConsumoForaPonta
  1037→      const vp = ValorConsumoPonta ? parseBR(ValorConsumoPonta) : 0;
  1038→      const vf = ValorConsumoForaPonta ? parseBR(ValorConsumoForaPonta) : 0;
  1039→      if (vp + vf > 0) ConsumoGrupoB_Valor = formatBR(vp + vf);
  1040→    }
  1041→  }
  1042→
  1043→  return {
  1044→    UC: cleanUC, Cidade, Endereco, Tipo, Mercado, Concessionaria,
  1045→    MesReferencia, AnoLeitura, Vencimento, NotaFiscal,
  1046→    ModalidadeTarifaria, Subgrupo,
  1047→    ConsumoPonta, ConsumoIntermediaria, ConsumoForaPonta, ConsumoTotal, ConsumoGrupoB_kWh, ConsumoGrupoB_Valor,
  1048→    DemandaContratadaPonta, DemandaContratadaForaPonta,
  1049→    DemandaMedidaPonta, DemandaMedidaForaPonta,
  1050→    DemandaFaturadaPonta, DemandaFaturadaForaPonta,
  1051→    UltrapassagemPonta, UltrapassagemForaPonta,
  1052→    DemandaNaoConsumidaPonta, DemandaNaoConsumidaForaPonta,
  1053→    ValorNaoConsumidaPonta, ValorNaoConsumidaForaPonta,
  1054→    ReativaExcedentePonta, ReativaExcedenteForaPonta,
  1055→    ValorConsumoPonta, ValorConsumoIntermediaria, ValorConsumoForaPonta,
  1056→    ValorDemandaPonta, ValorDemandaForaPonta,
  1057→    ValorUltrapassagemPonta, ValorUltrapassagemForaPonta,
  1058→    ValorReativaExcedentePonta, ValorReativaExcedenteForaPonta,
  1059→    PIS, COFINS, ICMS, GDI, CIP,
  1060→    EnergiaReativaTotal,
  1061→    EnergiaAtvInjetadaGDIoUC_Qtd, EnergiaAtvInjetadaGDIoUC_Valor,
  1062→    EnergiaAtvInjetadaGDImUC_Qtd, EnergiaAtvInjetadaGDImUC_Valor,
  1063→    SaldoAcumuladoGD,
  1064→    ValorTotal,
  1065→  };
  1066→}

  1068→// ===================== GDI EXTRACTION =====================
  1069→
  1070→function extractGDI(lines: string[], flat: string) {
  1071→  const EMPTY = '';
  1072→  let nOUC = 0;
  1073→  let nMUC = 0;
  1074→  let firstGDI = -1;
  1075→
  1076→  for (let i = 0; i < lines.length; i++) {
  1077→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
  1078→      nOUC++;
  1079→      if (firstGDI < 0) firstGDI = i;
  1080→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
  1081→      nMUC++;
  1082→      if (firstGDI < 0) firstGDI = i;
  1083→    }
  1084→  }
  1085→
  1086→  if (nOUC === 0 && nMUC === 0) {
  1087→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
  1088→  }
  1089→
  1090→  const chaveIdx = (() => {
  1091→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
  1092→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
  1093→    }
  1094→    return -1;
  1095→  })();
  1096→
  1097→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
  1098→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
  1099→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
  1100→      sectionEnd = i; break;
  1101→    }
  1102→  }
  1103→  if (chaveIdx >= 0) {
  1104→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
  1105→  }
  1106→
  1107→  const allQtds: number[] = [];
  1108→  if (chaveIdx >= 0) {
  1109→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
  1110→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1111→      if (m) allQtds.push(parseBR(m[1]));
  1112→    }
  1113→  }
  1114→  if (allQtds.length === 0) {
  1115→    for (let i = firstGDI; i < sectionEnd; i++) {
  1116→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1117→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
  1118→      if (allQtds.length >= nOUC + nMUC) break;
  1119→    }
  1120→  }
  1121→
  1122→  const allRVals: number[] = [];
  1123→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
  1124→  const totalNeeded = nOUC + nMUC;
  1125→
  1126→  for (let i = firstGDI; i < rSearchEnd; i++) {
  1127→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
  1128→    if (m && allRVals.length < totalNeeded) {
  1129→      allRVals.push(parseBR('-' + m[2]));
  1130→    }
  1131→  }
  1132→  if (allRVals.length < totalNeeded) {
  1133→    for (let i = firstGDI; i < rSearchEnd; i++) {
  1134→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
  1135→      if (m && allRVals.length < totalNeeded) {
  1136→        allRVals.push(parseBR(m[1]));
  1137→      }
  1138→    }
  1139→  }
  1140→
  1141→  const oUCQtds = allQtds.splice(0, nOUC);
  1142→  const oUCRVals = allRVals.splice(0, nOUC);
  1143→
  1144→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
  1145→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
  1146→
  1147→  return {
  1148→    oUC_Qtd: sumQtd(oUCQtds),
  1149→    oUC_Valor: sumR(oUCRVals),
  1150→    mUC_Qtd: sumQtd(allQtds),
  1151→    mUC_Valor: sumR(allRVals),
  1152→  };
  1153→}
  1154→
  1155→// ===================== NUMBER FORMATTING =====================
  1156→
  1157→function fmtBRNumber(n: number): string {
  1158→  const abs = Math.abs(n);
  1159→  const intPart = Math.floor(abs).toString();
  1160→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
  1161→  let formatted = '';
  1162→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
  1163→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
  1164→    formatted = intPart[i] + formatted;
  1165→  }

  1175→function findQtyNearLine(lines: string[], labelPattern: RegExp): string {
  1176→  const idx = findLine(lines, labelPattern);
  1177→  if (idx < 0) return EMPTY;
  1178→  for (let i = idx + 1; i < Math.min(idx + 10, lines.length); i++) {
  1179→    const line = lines[i];
  1180→    if (/[A-Za-z]{4,}/.test(line) && !/kWh|kW|R\$|tarifa/i.test(line)) continue;
  1181→    const m = line.match(/([\d.]+,\d{2})/);
  1182→    if (m) return m[1];
  1183→  }
  1184→  return EMPTY;
  1185→}

  1187→function findRValueNearLine(lines: string[], flat: string, labelPattern: RegExp): string {
  1188→  const idx = findLine(lines, labelPattern);

  1190→  const region = lines.slice(idx, Math.min(idx + 12, lines.length)).join(' ');
  1191→  const moneyVals = findAllMoney(region);

  1196→function findTributoValue(lines: string[], flat: string, labelPattern: RegExp): string {
  1197→  const idx = findLine(lines, labelPattern);
  1198→  if (idx >= 0) {
  1199→    for (let i = idx + 1; i < Math.min(idx + 6, lines.length); i++) {
  1200→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1201→      if (m) return m[1];
  1202→    }
  1203→  }
  1204→  const flatMatch = flat.match(new RegExp(labelPattern.source + '[^\\d]*?([\\d.]+,\\d{2})', 'i'));
  1205→  return flatMatch ? flatMatch[1] : EMPTY;
  1206→}
