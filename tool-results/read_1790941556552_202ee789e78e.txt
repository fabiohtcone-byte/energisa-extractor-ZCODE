   125→   125→   125→export function extractEnergisaData(rawText: string, tableExtraction?: TableExtraction): EnergisaInvoiceData {
   126→   126→   126→  const tableData = tableExtraction?.items;
   127→   127→   127→  const tributosFromTable = tableExtraction
   128→   128→   128→    ? { pis: tableExtraction.pisTotal, cofins: tableExtraction.cofinsTotal, icms: tableExtraction.icmsTotal }
   129→   129→   129→    : null;
   130→   130→   130→  const lines = rawText.split(/[\r\n]+/).map(l => l.trim()).filter(l => l.length > 0);
   131→   131→   131→  const flat = lines.join(' ');
   132→   132→   132→
   133→   133→   133→  // ===== CONCESSIONÁRIA =====
   134→   134→   134→  // Priority: detect the ISSUING concessionária (not the customer).
   135→   135→   135→  // For electricity invoices (DANF3E / CONTA DE ENERGIA), ENERGISA/ELEKTRO is the issuer.
   136→   136→   136→  // For water/sewage invoices, SANESUL is the issuer.
   137→   137→   137→  const Concessionaria = (() => {
   138→   138→   138→    const u = flat.toUpperCase();
   139→   139→   139→    const isEnergyInvoice = /DANF3E|ENERGIA\s*EL[eÉ]TRICA|DISTR\.\s*DE\s*ENERGIA/i.test(u);
   140→   140→   140→    if (isEnergyInvoice) {
   141→   141→   141→      // For energy invoices, the issuing utility is ENERGISA/ELEKTRO/NEOENERGIA
   142→   142→   142→      if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   143→   143→   143→      if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   144→   144→   144→      if (/ENERGISA/i.test(u)) return 'ENERGISA';
   145→   145→   145→    }
   146→   146→   146→    // For non-energy invoices or when no energy indicator found
   147→   147→   147→    if (/SANESUL/i.test(u)) return 'SANESUL';
   148→   148→   148→    if (/ELEKTRO/i.test(u)) return 'ELEKTRO';
   149→   149→   149→    if (/NEOENERGIA/i.test(u)) return 'NEOENERGIA';
   150→   150→   150→    if (/ENERGISA/i.test(u)) return 'ENERGISA';
   151→   151→   151→    return EMPTY;
   152→   152→   152→  })();
   153→   153→   153→
   154→   154→   154→  // ===== TIPO =====
   155→   155→   155→  const Tipo = (() => {
   156→   156→   156→    const u = flat.toUpperCase();
   157→   157→   157→    if (/SERVI[ÇC]O\s+P[ÚU]BLICO/.test(u)) return 'SERVIÇO PÚBLICO';
   158→   158→   158→    if (/PODER\s+P[ÚU]BLICO/.test(u)) return 'PODER PÚBLICO';
   159→   159→   159→    if (/RESIDENCIAL/.test(u)) return 'RESIDENCIAL';
   160→   160→   160→    if (/COMERCIAL/.test(u)) return 'COMERCIAL';
   161→   161→   161→    if (/INDUSTRIAL/.test(u)) return 'INDUSTRIAL';
   162→   162→   162→    if (/RURAL/.test(u)) return 'RURAL';
   163→   163→   163→    return EMPTY;
   164→   164→   164→  })();
   165→   165→   165→
   166→   166→   166→  // ===== MERCADO =====
   167→   167→   167→  const Mercado = (() => {
   168→   168→   168→    const m = flat.match(/SERVI[ÇC]O\s+P[ÚU]BLICO\s*\/\s*([A-ZÀ-Ú][A-ZÀ-Úa-zà-ú,\s]{3,40}?)(?:\s*TENSÃO|\s*\d|\s*$)/i);
   169→   169→   169→    if (m) {
   170→   170→   170→      const val = m[1].trim().replace(/\s+/g, ' ');
   171→   171→   171→      if (val.length < 50) return val.toUpperCase();
   172→   172→   172→    }
   173→   173→   173→    return EMPTY;
   174→   174→   174→  })();
   175→   175→   175→
   176→   176→   176→  // ===== CIDADE =====
   177→   177→   177→  // Pattern: "DOURADOS (AG: 10)" or "ANASTACIO MS (AG: 37)" or "RIO VERDE DE MATO GROSSO (AG: 37)"
   178→   178→   178→  const Cidade = (() => {
   179→   179→   179→    for (const line of lines.slice(0, 60)) {
   180→   180→   180→      // Match: CityName [StateAbbr] (AG: XX)
   181→   181→   181→      const m = line.match(/([A-ZÀ-Ú][A-ZÀ-Ú\s]{3,30}?)\s+(?:MS|SP|PR|MT|GO|MG|RS)?\s*\(\s*AG\s*:\s*\d+\s*\)/i);
   182→   182→   182→      if (m) {
   183→   183→   183→        const name = m[1].trim().toUpperCase();
   184→   184→   184→        const skip = ['SANESUL', 'ENERGISA', 'DISTRIBUIDORA', 'SA', 'EMPRESA DE SANEAMENTO',
   185→   185→   185→          'RUA ', 'AV ', 'DOM.', 'ENDEREÇO', 'PAGADOR'];
   186→   186→   186→        if (name.length >= 4 && name.length <= 30 && !skip.some(s => name.startsWith(s))) return name;
   187→   187→   187→      }
   188→   188→   188→    }
   189→   189→   189→    return EMPTY;
   190→   190→   190→  })();
   191→   191→   191→
   192→   192→   192→  // ===== UC =====
   193→   193→   193→  const UC = (() => {
   194→   194→   194→    // Helper: check if a number is likely a CNPJ (14 digits raw) or CPF (11 digits raw)
   195→   195→   195→    const isCnpjCpf = (s: string) => {
   196→   196→   196→      const raw = s.replace(/[.\-\/]/g, '');
   197→   197→   197→      return raw.length === 14 || raw.length === 11;
   198→   198→   198→    };
   199→   199→   199→
   200→   200→   200→    // Strategy 1: Energisa slash format "NN/NNNNN-N" (e.g., "10/33836-8") — most reliable
   201→   201→   201→    for (const line of lines) {
   202→   202→   202→      const m = line.match(/^(?:\d{1,3}\/)?(\d{4,8})-\d$/);
   203→   203→   203→      if (m) {
   204→   204→   204→        const num = m[1].replace(/^0+/, '') || '0';
   205→   205→   205→        if (num.length >= 3) return num;
   206→   206→   206→      }
   207→   207→   207→    }
   208→   208→   208→    // Strategy 2: Slash format within a line near "Código:" (e.g., "Código: 10/33836-8")
   209→   209→   209→    const slashCtx = flat.match(/Código\s*:\s*(\d{1,3})\/(\d{4,8})-\d/i);
   210→   210→   210→    if (slashCtx) {
   211→   211→   211→      const num = slashCtx[2].replace(/^0+/, '') || '0';
   212→   212→   212→      if (num.length >= 3) return num;
   213→   213→   213→    }
   214→   214→   214→    // Strategy 3: 4-group dotted format X.XXX.XXX.XXX-XX on its own line (12-digit UC)
   215→   215→   215→    // e.g., "1.409.407.051-47" → "140940705147"
   216→   216→   216→    for (const line of lines) {
   217→   217→   217→      const m = line.match(/^(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})$/);
   218→   218→   218→      if (m && !isCnpjCpf(m[1])) return m[1];
   219→   219→   219→    }
   220→   220→   220→    // Strategy 4: 3-group dotted format X.XXX.XXX-XX on its own line
   221→   221→   221→    for (const line of lines) {
   222→   222→   222→      const m = line.match(/^(\d{1,3}\.\d{3}\.\d{3}-\d{2})$/);
   223→   223→   223→      if (m) return m[1];
   224→   224→   224→    }
   225→   225→   225→    // Strategy 5: 4-group dotted format on a short line (not CNPJ/CPF/CEP)
   226→   226→   226→    for (const line of lines) {
   227→   227→   227→      const m = line.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   228→   228→   228→      if (m && line.length < 50 && !/CNPJ|CPF|CEP/i.test(line) && !isCnpjCpf(m[1])) return m[1];
   229→   229→   229→    }
   230→   230→   230→    // Strategy 6: 3-group dotted format on a short line (not CNPJ/CPF/CEP)
   231→   231→   231→    for (const line of lines) {
   232→   232→   232→      const m = line.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   233→   233→   233→      if (m && line.length < 40 && !/CNPJ|CPF|CEP/i.test(line)) return m[1];
   234→   234→   234→    }
   235→   235→   235→    // Strategy 7: MATRÍCULA field — extended to support up to 13 digits
   236→   236→   236→    // e.g., "MATRÍCULA:0140940705147-2025-08-6"
   237→   237→   237→    const ucMat = flat.match(/MATR[IÍ]CULA\s*:\s*0*(\d{4,13})(?:-\d{4}-\d{2}-\d)?/i);
   238→   238→   238→    if (ucMat) return ucMat[1];
   239→   239→   239→    // Strategy 8: 10-13 digit UC near address lines (Energisa MS format)
   240→   240→   240→    const ucAddr = flat.match(/(?:RUA|AV|ESTRADA)[^\d]*(\d{10,13})\b/);
   241→   241→   241→    if (ucAddr) return ucAddr[1];
   242→   242→   242→    // Strategy 9: standalone or near "Instalação:" — 7-13 digits
   243→   243→   243→    const ucInst = flat.match(/Instalação\s*:?\s*(\d{7,13})/i);
   244→   244→   244→    if (ucInst) return ucInst[1];
   245→   245→   245→    // Strategy 10: UC label with number — 4-13 digits
   246→   246→   246→    const ucLabel = flat.match(/(?:UC\s*:?|Unid\.?\s*Cons\.?)[^\d]*(\d{4,13})/i);
   247→   247→   247→    if (ucLabel) return ucLabel[1].replace(/^0+/, '') || ucLabel[1];
   248→   248→   248→    // Strategy 11: broad search for 4-group dotted format (before 3-group to avoid partial match)
   249→   249→   249→    const broad4 = flat.match(/(\d{1,2}\.\d{3}\.\d{3}\.\d{3}-\d{1,2})/);
   250→   250→   250→    if (broad4 && !isCnpjCpf(broad4[1])) return broad4[1];
   251→   251→   251→    // Strategy 12: broad search for 3-group dotted format
   252→   252→   252→    const broad3 = flat.match(/(\d{1,3}\.\d{3}\.\d{3}-\d{2})/);
   253→   253→   253→    if (broad3) return broad3[1];
   254→   254→   254→    // Strategy 13: standalone 10-13 digit number (not CNPJ/CPF) — as last resort
   255→   255→   255→    const bareLong = flat.match(/\b(\d{10,13})\b/);
   256→   256→   256→    if (bareLong && !isCnpjCpf(bareLong[1])) return bareLong[1];
   257→   257→   257→    return EMPTY;
   258→   258→   258→  })();
   259→   259→   259→
   260→   260→   260→  // Clean UC: remove dots and dashes (e.g., "1.409.407.051-47" → "140940705147")
   261→   261→   261→  const cleanUC = UC.replace(/[.\-]/g, '');
   262→   262→   262→
   263→   263→   263→  // ===== ENDEREÇO =====
   264→   264→   264→  const Endereco = (() => {
   265→   265→   265→    // Address line typically follows the city (AG: XX) line and starts with RUA, AV, AVE, LOT, ESTRADA, etc.
   266→   266→   266→    // Examples from real PDFs:
   267→   267→   267→    //   "RUA SALDANHA DA GAMA, 896 - 21.004.65.029401 LAD RE RA - 422 - 79370000"
   268→   268→   268→    //   "LOT RECANTO NUARA, S/N - 24.416.10.154000 PAT 01/REL01 2289 - 79190000"
   269→   269→   269→    //   "AVE MATO GROSSO DO SUL, SN - 12.029.07.090000 COX EAT 006 2243 - 79400000"
   270→   270→   270→    const STREET_RE = /^(RUA|AV[. ]|AVE |AVENIDA|ESTRADA|LOT |RODOVIA|TRAVESSA|ALAMEDA|PCA[. ]|PRACA)/i;
   271→   271→   271→
   272→   272→   272→    // Clean an address line: keep street + number, strip internal codes, keep CEP
   273→   273→   273→    const extractCleanAddress = (line: string): string => {
   274→   274→   274→      // Split at the first " - " that precedes an internal code pattern (XX.XXX.XX.XXXXXX)
   275→   275→   275→      const beforeCodes = line.split(/\s*-\s*\d{2}\.\d{3}/)[0];
   276→   276→   276→      let addr = (beforeCodes || line).trim();
   277→   277→   277→      // Remove trailing dash
   278→   278→   278→      addr = addr.replace(/\s*-\s*$/, '').trim();
   279→   279→   279→      // Append CEP if found at end of original line
   280→   280→   280→      const cepMatch = line.match(/(\d{5}-?\d{3})\s*$/);
   281→   281→   281→      if (cepMatch) addr += ' - CEP ' + cepMatch[1];
   282→   282→   282→      return addr;
   283→   283→   283→    };
   284→   284→   284→
   285→   285→   285→    // Strategy 1: Find address line after city (AG: XX) pattern
   286→   286→   286→    for (let i = 0; i < Math.min(60, lines.length); i++) {
   287→   287→   287→      if (/\(\s*AG\s*:\s*\d+\s*\)/.test(lines[i])) {
   288→   288→   288→        // Check next 3 lines for an address pattern
   289→   289→   289→        for (let j = i + 1; j <= Math.min(i + 3, lines.length - 1); j++) {
   290→   290→   290→          if (STREET_RE.test(lines[j])) {
   291→   291→   291→            return extractCleanAddress(lines[j]);
   292→   292→   292→          }
   293→   293→   293→        }
   294→   294→   294→      }
   295→   295→   295→    }
   296→   296→   296→
   297→   297→   297→    // Strategy 2: Find address line near CNPJ/CPF line (appears before it)
   298→   298→   298→    for (let i = 0; i < lines.length; i++) {
   299→   299→   299→      if (/CNPJ\/CPF/i.test(lines[i]) && i > 0) {
   300→   300→   300→        if (STREET_RE.test(lines[i - 1])) {
   301→   301→   301→          return extractCleanAddress(lines[i - 1]);
   302→   302→   302→        }
   303→   303→   303→      }
   304→   304→   304→    }
   305→   305→   305→
   306→   306→   306→    // Strategy 3: Look for "ENDEREÇO" label followed by address
   307→   307→   307→    for (let i = 0; i < lines.length; i++) {
   308→   308→   308→      if (/^ENDERE[ÇC]O$/i.test(lines[i].trim()) && i + 1 < lines.length) {
   309→   309→   309→        const next = lines[i + 1];
   310→   310→   310→        const cleanAddr = next.replace(/\s*-\s*(SA[IÍ]DA|ENTRADA)\s+PARA\s+[^-]+/i, '')
   311→   311→   311→          .replace(/\s*-\s*[A-ZÀ-Ú]+\s*\/\s*[A-Z]{2}\s*-\s*CEP\s+\d{5}-\d{3}/i, '').trim();
   312→   312→   312→        if (cleanAddr.length > 5) return cleanAddr;
   313→   313→   313→      }
   314→   314→   314→    }
   315→   315→   315→
   316→   316→   316→    return EMPTY;
   317→   317→   317→  })();
   318→   318→   318→
   319→   319→   319→  // ===== MÊS REFERÊNCIA =====
   320→   320→   320→  const MesReferencia = (() => {
   321→   321→   321→    const meses = 'Janeiro|Fevereiro|Mar[çc]o|Abril|Maio|Junho|Julho|Agosto|Setembro|Outubro|Novembro|Dezembro';
   322→   322→   322→    const numToMonth: Record<string, string> = {

   414→   414→   414→          }
   415→   415→   415→        }
   416→   416→   416→      }
   417→   417→   417→    }
   418→   418→   418→    // Strategy 7: first valid MM/YYYY that is a month and consistent with Vencimento

   424→   424→   424→        if (isConsistentWithVencimento(monthNum, year)) {

   426→   426→   426→        }
   427→   427→   427→      }

   608→   608→   608→          let nm;
   609→   609→   609→          while ((nm = numRegex.exec(rest)) !== null) {
   610→   610→   610→            nums.push(nm[1]);
   611→   611→   611→          }
   612→   612→   612→          if (nums.length >= 2) {
   613→   613→   613→            // "KWHPonta..." format: first number is consumption kWh
   614→   614→   614→            // "Ponta..." format: second-to-last number is consumption kWh
   615→   615→   615→            if (line.startsWith('KWH')) {
   616→   616→   616→              ConsumoTotal = nums[0];
   617→   617→   617→            } else {
   618→   618→   618→              ConsumoTotal = nums[nums.length - 2];
   619→   619→   619→            }
   620→   620→   620→          } else if (nums.length === 1) {
   621→   621→   621→            ConsumoTotal = nums[0];
   622→   622→   622→          }
   623→   623→   623→          break;
   624→   624→   624→        }
   625→   625→   625→      }
   626→   626→   626→    }
   627→   627→   627→    // ENERGISA Verde/Azul: extract consumption from "KWH"+value and compressed "Energia ativa em kWh" data
   628→   628→   628→    // Pattern A: "KWH1.198,96" → Ponta consumption = 1.198,96
   629→   629→   629→    // Pattern B: After "W9...Ponta/Fora Ponta" + "Energia ativa em kWh" → compressed data line
   630→   630→   630→    //   e.g., "401198,963105,09575,852" → after prefix "40", first number = 1198,96 → 1.198,96
   631→   631→   631→    //   We determine the prefix length by finding the known Ponta KWH value in the compressed data
   632→   632→   632→    if (!ConsumoTotal) {
   633→   633→   633→      const kwhValues: string[] = [];
   634→   634→   634→      // Pattern A: "KWH" immediately followed by a BR number (e.g., "KWH1.198,96")
   635→   635→   635→      let pontaKwhFromLabel = '';
   636→   636→   636→      for (const line of lines) {
   637→   637→   637→        const km = line.match(/^KWH(\d{1,2}(?:\.\d{3})*,\d{2})$/i);
   638→   638→   638→        if (km) { kwhValues.push(km[1]); pontaKwhFromLabel = km[1]; }
   639→   639→   639→      }
   640→   640→   640→      // Pattern B: Parse compressed data after "Energia ativa em kWh" for each period (Ponta/Fora Ponta)
   641→   641→   641→      // Find all (periodHeader + "Energia ativa em kWh" + compressedData) triples
   642→   642→   642→      const periodSections: { period: string; data: string }[] = [];
   643→   643→   643→      for (let i = 0; i < lines.length - 2; i++) {
   644→   644→   644→        if (/Ponta$/i.test(lines[i]) && !/Fora/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   645→   645→   645→          periodSections.push({ period: 'Ponta', data: lines[i + 2] });
   646→   646→   646→        }
   647→   647→   647→        if (/Fora\s*Ponta$/i.test(lines[i]) && /^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i + 1])) {
   648→   648→   648→          periodSections.push({ period: 'ForaPonta', data: lines[i + 2] });
   649→   649→   649→        }
   650→   650→   650→      }
   651→   651→   651→      // If we have the Ponta KWH from the label, use it to determine prefix length
   652→   652→   652→      // Then apply same prefix length to Fora Ponta compressed data
   653→   653→   653→      if (pontaKwhFromLabel && periodSections.length >= 1) {
   654→   654→   654→        const pontaSection = periodSections.find(s => s.period === 'Ponta');
   655→   655→   655→        const foraSection = periodSections.find(s => s.period === 'ForaPonta');
   656→   656→   656→        if (pontaSection) {
   657→   657→   657→          // Find the Ponta KWH value (without dots) in the compressed data to determine prefix length
   658→   658→   658→          const pontaNoDot = pontaKwhFromLabel.replace(/\./g, '');
   659→   659→   659→          const prefixIdx = pontaSection.data.indexOf(pontaNoDot);
   660→   660→   660→          if (prefixIdx >= 0) {
   661→   661→   661→            const prefixLen = prefixIdx;
   662→   662→   662→            // Extract Fora Ponta consumption using same prefix length
   663→   663→   663→            if (foraSection && prefixLen < foraSection.data.length) {
   664→   664→   664→              const foraRest = foraSection.data.substring(prefixLen);
   665→   665→   665→              // Find the first number ending with ,XX (2 decimal places)
   666→   666→   666→              const commaPos = foraRest.indexOf(',');
   667→   667→   667→              if (commaPos >= 2 && commaPos <= 7) {
   668→   668→   668→                const foraNum = foraRest.substring(0, commaPos + 3); // include ,XX
   669→   669→   669→                // Add dots as thousands separator
   670→   670→   670→                const parts = foraNum.split(',');
   671→   671→   671→                const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   672→   672→   672→                const foraBR = intPart + ',' + parts[1];
   673→   673→   673→                if (!kwhValues.includes(foraBR)) kwhValues.push(foraBR);
   674→   674→   674→              }
   675→   675→   675→            }
   676→   676→   676→          }
   677→   677→   677→        }
   678→   678→   678→      }
   679→   679→   679→      // Also try: for each "Energia ativa em kWh" section, if we don't have the value yet,
   680→   680→   680→      // try to find it as a standalone BR number near the section
   681→   681→   681→      if (kwhValues.length === 0) {
   682→   682→   682→        for (let i = 0; i < lines.length - 1; i++) {
   683→   683→   683→          if (/^Energia\s+ativa\s+em\s+kWh$/i.test(lines[i])) {
   684→   684→   684→            // Look for the consumption in the next line by trying different prefix lengths
   685→   685→   685→            const nextLine = lines[i + 1];
   686→   686→   686→            for (let prefix = 0; prefix < Math.min(4, nextLine.length); prefix++) {
   687→   687→   687→              const rest = nextLine.substring(prefix);
   688→   688→   688→              const commaPos = rest.indexOf(',');
   689→   689→   689→              if (commaPos >= 1 && commaPos <= 6) {
   690→   690→   690→                const numStr = rest.substring(0, commaPos + 3);
   691→   691→   691→                const parts = numStr.split(',');
   692→   692→   692→                if (parts.length === 2 && /^\d+$/.test(parts[0]) && /^\d{2}$/.test(parts[1])) {
   693→   693→   693→                  const val = parseInt(parts[0]);
   694→   694→   694→                  if (val >= 10 && val <= 999999) {
   695→   695→   695→                    const intPart = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.');
   696→   696→   696→                    const brNum = intPart + ',' + parts[1];
   697→   697→   697→                    if (!kwhValues.includes(brNum)) kwhValues.push(brNum);
   698→   698→   698→                    break;
   699→   699→   699→                  }
   700→   700→   700→                }
   701→   701→   701→              }
   702→   702→   702→            }
   703→   703→   703→          }
   704→   704→   704→        }
   705→   705→   705→      }
   706→   706→   706→      if (kwhValues.length > 0) {
   707→   707→   707→        // Sum all period consumptions to get total
   708→   708→   708→        let totalKwh = 0;
   709→   709→   709→        for (const v of kwhValues) {
   710→   710→   710→          totalKwh += parseBR(v);
   711→   711→   711→        }
   712→   712→   712→        if (totalKwh > 0) ConsumoTotal = formatBR(totalKwh);
   713→   713→   713→      }
   714→   714→   714→    }
   715→   715→   715→    ValorConsumoForaPonta = findRValueNearLine(lines, flat, /Consumo\s+em\s+kWh(?!.*Ponta)/i);
   716→   716→   716→  }
   717→   717→   717→
   718→   718→   718→  // ===== DEMANDA (text-based, will be overridden by table data if available) =====
   719→   719→   719→  let DemandaContratadaPonta = EMPTY;
   720→   720→   720→  let DemandaContratadaForaPonta = EMPTY;
   721→   721→   721→  let DemandaMedidaPonta = EMPTY;
   722→   722→   722→  let DemandaMedidaForaPonta = EMPTY;
   723→   723→   723→  let DemandaFaturadaPonta = EMPTY;
   724→   724→   724→  let DemandaFaturadaForaPonta = EMPTY;
   725→   725→   725→  let ValorDemandaPonta = EMPTY;
   726→   726→   726→  let ValorDemandaForaPonta = EMPTY;
   727→   727→   727→
   728→   728→   728→  if (isSplitTariff) {
   729→   729→   729→    // ===== DEMANDA CONTRATADA EXTRACTION =====
   730→   730→   730→    // Energisa PDFs show contracted demand in two locations:
   731→   731→   731→    // 1) "Grandezas Contratadas" section: "Demanda ponta - kW  15,3" / "Demanda fora ponta - kW  30"
   732→   732→   732→    // 2) Bottom line: "KW Ponta: 15,3  Fora Ponta: 30  TUSDG:"
   733→   733→   733→    //
   734→   734→   734→    // pdf-parse often garbles the column order, producing:
   735→   735→   735→    //   "Demanda ponta - kW Demanda fora ponta - kW 30 15,3"  (values reversed)
   736→   736→   736→    //   "30 Fora Ponta: 15,3 KW Ponta:"  (values before labels, or between labels)
   737→   737→   737→    //
   738→   738→   738→    // We try multiple strategies and use the first one that gives both values.
   739→   739→   739→
   740→   740→   740→    // Strategy 1: "Demanda ponta - kW <value>" on the SAME LINE (pdftotext-like format)
   741→   741→   741→    const dpSameLine = flat.match(/Demanda\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   742→   742→   742→    const dfpSameLine = flat.match(/Demanda\s+fora\s+ponta\s*-\s*kW\s+(\d+(?:[.,]\d+)?)/i);
   743→   743→   743→    if (dpSameLine && dfpSameLine) {
   744→   744→   744→      DemandaContratadaPonta = dpSameLine[1].replace('.', ',');
   745→   745→   745→      DemandaContratadaForaPonta = dfpSameLine[1].replace('.', ',');
   746→   746→   746→    }
   747→   747→   747→
   748→   748→   748→    // Strategy 2: Detect pdf-parse column reordering in bottom line
   749→   749→   749→    // When "Fora Ponta:" appears BEFORE "KW Ponta:" in the text, the values are
   750→   750→   750→    // rearranged: "<fora_val> Fora Ponta: <ponta_val> KW Ponta:"
   751→   751→   751→    // - Number BEFORE "Fora Ponta:" = Fora Ponta value
   752→   752→   752→    // - Number BETWEEN "Fora Ponta:" and "KW Ponta:" = Ponta value
   753→   753→   753→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   754→   754→   754→      const foraIdx = flat.search(/Fora\s+Ponta\s*:/i);
   755→   755→   755→      const kwIdx = flat.search(/KW\s+Ponta\s*:/i);
   756→   756→   756→      if (foraIdx >= 0 && kwIdx >= 0 && foraIdx < kwIdx) {
   757→   757→   757→        // "Fora Ponta:" appears before "KW Ponta:" — this is the reordered pattern
   758→   758→   758→        // Number between "Fora Ponta:" and "KW Ponta:" → Ponta value
   759→   759→   759→        const between = flat.slice(foraIdx, kwIdx);
   760→   760→   760→        const betweenMatch = between.match(/:\s*(\d+(?:[.,]\d+)?)/);
   761→   761→   761→        if (betweenMatch && !DemandaContratadaPonta) {
   762→   762→   762→          DemandaContratadaPonta = betweenMatch[1].replace('.', ',');
   763→   763→   763→        }
   764→   764→   764→        // Number before "Fora Ponta:" (within 25 chars) → Fora Ponta value
   765→   765→   765→        const before = flat.slice(Math.max(0, foraIdx - 25), foraIdx);
   766→   766→   766→        const beforeMatch = before.match(/(\d+(?:[.,]\d+)?)\s*$/);
   767→   767→   767→        if (beforeMatch && !DemandaContratadaForaPonta) {
   768→   768→   768→          DemandaContratadaForaPonta = beforeMatch[1].replace('.', ',');
   769→   769→   769→        }
   770→   770→   770→      } else if (kwIdx >= 0 && foraIdx >= 0 && kwIdx < foraIdx) {
   771→   771→   771→        // Normal order: "KW Ponta:" appears before "Fora Ponta:" — values follow labels
   772→   772→   772→        const kwAfter = flat.slice(kwIdx).match(/KW\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   773→   773→   773→        const fpAfter = flat.slice(foraIdx).match(/Fora\s+Ponta\s*:\s*(\d+(?:[.,]\d+)?)/i);
   774→   774→   774→        if (kwAfter && !DemandaContratadaPonta) DemandaContratadaPonta = kwAfter[1].replace('.', ',');
   775→   775→   775→        if (fpAfter && !DemandaContratadaForaPonta) DemandaContratadaForaPonta = fpAfter[1].replace('.', ',');
   776→   776→   776→      }
   777→   777→   777→    }
   778→   778→   778→
   779→   779→   779→    // Strategy 3: "Demanda ponta/fora ponta - kW" labels with values on nearby lines
   780→   780→   780→    // (pdf-parse puts labels and values on separate lines)
   781→   781→   781→    if (!DemandaContratadaPonta || !DemandaContratadaForaPonta) {
   782→   782→   782→      const dpLineIdx = lines.findIndex(l => /Demanda\s+ponta\s*-\s*kW/i.test(l));
   783→   783→   783→      const dfpLineIdx = lines.findIndex(l => /Demanda\s+fora\s+ponta\s*-\s*kW/i.test(l));
   784→   784→   784→      if (dpLineIdx >= 0 && dfpLineIdx >= 0) {
   785→   785→   785→        // Collect numeric values from the lines after both labels
   786→   786→   786→        const startLine = Math.max(dpLineIdx, dfpLineIdx) + 1;
   787→   787→   787→        const numVal = /^(\d+(?:[.,]\d+)?)$/;
   788→   788→   788→        const values: string[] = [];
   789→   789→   789→        for (let i = startLine; i < Math.min(startLine + 5, lines.length); i++) {
   790→   790→   790→          const m = lines[i].match(numVal);
   791→   791→   791→          if (m) values.push(m[1].replace('.', ','));
   792→   792→   792→          if (values.length >= 2) break;
   793→   793→   793→        }
   794→   794→   794→        if (values.length === 2) {
   795→   795→   795→          // Use already-found values to disambiguate, or assume reverse order
   796→   796→   796→          if (DemandaContratadaForaPonta && !DemandaContratadaPonta) {
   797→   797→   797→            if (values[0] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[1];
   798→   798→   798→            else if (values[1] === DemandaContratadaForaPonta) DemandaContratadaPonta = values[0];
   799→   799→   799→          } else if (DemandaContratadaPonta && !DemandaContratadaForaPonta) {
   800→   800→   800→            if (values[0] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[1];
   801→   801→   801→            else if (values[1] === DemandaContratadaPonta) DemandaContratadaForaPonta = values[0];
   802→   802→   802→          } else {
   803→   803→   803→            // No disambiguation; assume pdf-parse reverse column order:
   804→   804→   804→            // values[0] = Fora Ponta, values[1] = Ponta
   805→   805→   805→            if (!DemandaContratadaForaPonta) DemandaContratadaForaPonta = values[0];
   806→   806→   806→            if (!DemandaContratadaPonta) DemandaContratadaPonta = values[1];
   807→   807→   807→          }
   808→   808→   808→        }
   809→   809→   809→      }
   810→   810→   810→    }
   811→   811→   811→
   812→   812→   812→    DemandaMedidaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Ponta(?!.*Fora)/i);
   813→   813→   813→    DemandaMedidaForaPonta = findQtyNearLine(lines, /Demanda\s+Medida.*Fora\s*Ponta/i);
   814→   814→   814→    DemandaFaturadaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Ponta(?!.*Fora)/i) || DemandaMedidaPonta;
   815→   815→   815→    DemandaFaturadaForaPonta = findQtyNearLine(lines, /Demanda\s+Fatur[ae]da.*Fora\s*Ponta/i) || DemandaMedidaForaPonta;
   816→   816→   816→    ValorDemandaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Ponta(?!.*Fora)/i);
   817→   817→   817→    ValorDemandaForaPonta = findRValueNearLine(lines, flat, /Demanda\s+(?:Medida|Fatur[ae]da).*Fora\s*Ponta/i);
   818→   818→   818→  }
   819→   819→   819→
   820→   820→   820→  // ===== ULTRAPASSAGEM =====
   821→   821→   821→  let UltrapassagemPonta = EMPTY;
   822→   822→   822→  let UltrapassagemForaPonta = EMPTY;
   823→   823→   823→  let ValorUltrapassagemPonta = EMPTY;
   824→   824→   824→  let ValorUltrapassagemForaPonta = EMPTY;
   825→   825→   825→  // ===== DEMANDA NÃO CONSUMIDA =====
   826→   826→   826→  let DemandaNaoConsumidaPonta = EMPTY;
   827→   827→   827→  let DemandaNaoConsumidaForaPonta = EMPTY;
   828→   828→   828→  let ValorNaoConsumidaPonta = EMPTY;
   829→   829→   829→  let ValorNaoConsumidaForaPonta = EMPTY;
   830→   830→   830→  if (isSplitTariff) {
   831→   831→   831→    UltrapassagemPonta = findQtyNearLine(lines, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   832→   832→   832→    UltrapassagemForaPonta = findQtyNearLine(lines, /Ultrapassagem.*Fora\s*Ponta/i);
   833→   833→   833→    ValorUltrapassagemPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Ponta(?!.*Fora)/i);
   834→   834→   834→    ValorUltrapassagemForaPonta = findRValueNearLine(lines, flat, /Ultrapassagem.*Fora\s*Ponta/i);
   835→   835→   835→    DemandaNaoConsumidaPonta = findQtyNearLine(lines, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   836→   836→   836→    DemandaNaoConsumidaForaPonta = findQtyNearLine(lines, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   837→   837→   837→    ValorNaoConsumidaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*Ponta(?!.*(?:Fora|F\s*Ponta))/i);
   838→   838→   838→    ValorNaoConsumidaForaPonta = findRValueNearLine(lines, flat, /Não\s*Consumida.*(?:Fora|F)\s*Ponta/i);
   839→   839→   839→  }
   840→   840→   840→
   841→   841→   841→  // ===== REATIVA EXCEDENTE =====
   842→   842→   842→  let ReativaExcedentePonta = EMPTY;
   843→   843→   843→  let ReativaExcedenteForaPonta = EMPTY;
   844→   844→   844→  let EnergiaReativaTotal = EMPTY;
   845→   845→   845→  let ValorReativaExcedentePonta = EMPTY;
   846→   846→   846→  let ValorReativaExcedenteForaPonta = EMPTY;
   847→   847→   847→  if (isSplitTariff) {
   848→   848→   848→    ReativaExcedentePonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   849→   849→   849→    ReativaExcedenteForaPonta = findQtyNearLine(lines, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   850→   850→   850→    ValorReativaExcedentePonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Ponta(?!.*Fora)/i);
   851→   851→   851→    ValorReativaExcedenteForaPonta = findRValueNearLine(lines, flat, /Reactiv?a\s+Excedente.*Fora\s*Ponta/i);
   852→   852→   852→  } else {
   853→   853→   853→    EnergiaReativaTotal = findQtyNearLine(lines, /Reactiv?a\s+Excedente/i);
   854→   854→   854→  }
   855→   855→   855→
   856→   856→   856→  // ===== TRIBUTOS: PIS, COFINS, ICMS =====
   857→   857→   857→  // Priority: table extraction > text-based extraction
   858→   858→   858→  let PIS = EMPTY;
   859→   859→   859→  let COFINS = EMPTY;
   860→   860→   860→  let ICMS = EMPTY;
   861→   861→   861→
   862→   862→   862→  // Use tributos from table extraction if available (most reliable)
   863→   863→   863→  if (tributosFromTable?.pis) PIS = tributosFromTable.pis;
   864→   864→   864→  if (tributosFromTable?.cofins) COFINS = tributosFromTable.cofins;
   865→   865→   865→  if (tributosFromTable?.icms) ICMS = tributosFromTable.icms;
   866→   866→   866→
   867→   867→   867→  // Fallback: ELEKTRO/SANESUL pattern — ICMS, COFINS, PIS on consecutive lines,
   868→   868→   868→  // followed by base values, aliquotas (%), then actual tributo values.
   869→   869→   869→  // Example:
   870→   870→   870→  //   L52: ICMS
   871→   871→   871→  //   L53: COFINS
   872→   872→   872→  //   L54: PIS
   873→   873→   873→  //   L55: 5087,88   ← Base de cálculo ICMS
   874→   874→   874→  //   L56: 4222,95   ← Base de cálculo COFINS
   875→   875→   875→  //   L57: 4222,95   ← Base de cálculo PIS
   876→   876→   876→  //   L58: 17%       ← Alíquota ICMS
   877→   877→   877→  //   L59: 3,48%     ← Alíquota COFINS
   878→   878→   878→  //   L60: 0,75%     ← Alíquota PIS
   879→   879→   879→  //   L61: 864,93    ← Valor ICMS ✓
   880→   880→   880→  //   L62: 146,96    ← Valor COFINS ✓
   881→   881→   881→  //   L63: 31,67     ← Valor PIS ✓
   882→   882→   882→  if (!PIS || !COFINS || !ICMS) {
   883→   883→   883→    for (let i = 0; i < lines.length - 2; i++) {
   884→   884→   884→      if (/^ICMS$/i.test(lines[i]) && /^COFINS$/i.test(lines[i + 1]) && /^PIS$/i.test(lines[i + 2])) {

  1305→  1305→  1305→function extractGDI(lines: string[], flat: string) {
  1306→  1306→  1306→  const EMPTY = '';
  1307→  1307→  1307→  let nOUC = 0;
  1308→  1308→  1308→  let nMUC = 0;
  1309→  1309→  1309→  let firstGDI = -1;
  1310→  1310→  1310→
  1311→  1311→  1311→  for (let i = 0; i < lines.length; i++) {
  1312→  1312→  1312→    if (/Energia\s*Atv\s*Injetada\s*GDI\s*oUC/i.test(lines[i])) {
  1313→  1313→  1313→      nOUC++;
  1314→  1314→  1314→      if (firstGDI < 0) firstGDI = i;
  1315→  1315→  1315→    } else if (/Energia\s*Atv\s*Injetada\s*GDI\s*mUC/i.test(lines[i])) {
  1316→  1316→  1316→      nMUC++;
  1317→  1317→  1317→      if (firstGDI < 0) firstGDI = i;
  1318→  1318→  1318→    }
  1319→  1319→  1319→  }
  1320→  1320→  1320→
  1321→  1321→  1321→  if (nOUC === 0 && nMUC === 0) {
  1322→  1322→  1322→    return { oUC_Qtd: EMPTY, oUC_Valor: EMPTY, mUC_Qtd: EMPTY, mUC_Valor: EMPTY };
  1323→  1323→  1323→  }
  1324→  1324→  1324→
  1325→  1325→  1325→  const chaveIdx = (() => {
  1326→  1326→  1326→    for (let i = firstGDI; i < Math.min(firstGDI + 100, lines.length); i++) {
  1327→  1327→  1327→      if (/^chave de acesso:\s*$/i.test(lines[i])) return i;
  1328→  1328→  1328→    }
  1329→  1329→  1329→    return -1;
  1330→  1330→  1330→  })();
  1331→  1331→  1331→
  1332→  1332→  1332→  let sectionEnd = Math.min(firstGDI + 100, lines.length);
  1333→  1333→  1333→  for (let i = firstGDI; i < Math.min(firstGDI + 120, lines.length); i++) {
  1334→  1334→  1334→    if (/^(Declara[cç]|EMITIDO EM|NOTA FISCAL N|Importante:)/i.test(lines[i])) {
  1335→  1335→  1335→      sectionEnd = i; break;
  1336→  1336→  1336→    }
  1337→  1337→  1337→  }
  1338→  1338→  1338→  if (chaveIdx >= 0) {
  1339→  1339→  1339→    sectionEnd = Math.min(Math.max(sectionEnd, chaveIdx + 20), lines.length);
  1340→  1340→  1340→  }
  1341→  1341→  1341→
  1342→  1342→  1342→  const allQtds: number[] = [];
  1343→  1343→  1343→  if (chaveIdx >= 0) {
  1344→  1344→  1344→    for (let i = chaveIdx + 1; i < Math.min(chaveIdx + 20, sectionEnd); i++) {
  1345→  1345→  1345→      const m = lines[i].match(/^(\d[\d.]*,\d{2})$/);
  1346→  1346→  1346→      if (m) allQtds.push(parseBR(m[1]));
  1347→  1347→  1347→    }
  1348→  1348→  1348→  }
  1349→  1349→  1349→  if (allQtds.length === 0) {
  1350→  1350→  1350→    for (let i = firstGDI; i < sectionEnd; i++) {
  1351→  1351→  1351→      const m = lines[i].match(/^(\d[\d.]+,\d{2})$/);
  1352→  1352→  1352→      if (m && parseBR(m[1]) >= 100) allQtds.push(parseBR(m[1]));
  1353→  1353→  1353→      if (allQtds.length >= nOUC + nMUC) break;
  1354→  1354→  1354→    }
  1355→  1355→  1355→  }
  1356→  1356→  1356→
  1357→  1357→  1357→  const allRVals: number[] = [];
  1358→  1358→  1358→  const rSearchEnd = chaveIdx >= 0 ? chaveIdx : sectionEnd;
  1359→  1359→  1359→  const totalNeeded = nOUC + nMUC;
  1360→  1360→  1360→
  1361→  1361→  1361→  for (let i = firstGDI; i < rSearchEnd; i++) {
  1362→  1362→  1362→    const m = lines[i].match(/^(.+?)-((?:\d{1,3}\.)*\d+,\d{2})$/);
  1363→  1363→  1363→    if (m && allRVals.length < totalNeeded) {
  1364→  1364→  1364→      allRVals.push(parseBR('-' + m[2]));
  1365→  1365→  1365→    }
  1366→  1366→  1366→  }
  1367→  1367→  1367→  if (allRVals.length < totalNeeded) {
  1368→  1368→  1368→    for (let i = firstGDI; i < rSearchEnd; i++) {
  1369→  1369→  1369→      const m = lines[i].match(/^(-\d[\d.]*,\d{2})$/);
  1370→  1370→  1370→      if (m && allRVals.length < totalNeeded) {
  1371→  1371→  1371→        allRVals.push(parseBR(m[1]));
  1372→  1372→  1372→      }
  1373→  1373→  1373→    }
  1374→  1374→  1374→  }
  1375→  1375→  1375→
  1376→  1376→  1376→  const oUCQtds = allQtds.splice(0, nOUC);
  1377→  1377→  1377→  const oUCRVals = allRVals.splice(0, nOUC);
  1378→  1378→  1378→
  1379→  1379→  1379→  const sumQtd = (arr: number[]) => arr.length > 0 ? fmtBRNumber(arr.reduce((a, b) => a + b, 0)) : EMPTY;
  1380→  1380→  1380→  const sumR = (arr: number[]) => arr.length > 0 ? fmtBRNumber(Math.abs(arr.reduce((a, b) => a + b, 0))) : EMPTY;
  1381→  1381→  1381→
  1382→  1382→  1382→  return {
  1383→  1383→  1383→    oUC_Qtd: sumQtd(oUCQtds),
  1384→  1384→  1384→    oUC_Valor: sumR(oUCRVals),
  1385→  1385→  1385→    mUC_Qtd: sumQtd(allQtds),
  1386→  1386→  1386→    mUC_Valor: sumR(allRVals),
  1387→  1387→  1387→  };
  1388→  1388→  1388→}

  1392→  1392→  1392→function fmtBRNumber(n: number): string {
  1393→  1393→  1393→  const abs = Math.abs(n);
  1394→  1394→  1394→  const intPart = Math.floor(abs).toString();
  1395→  1395→  1395→  const decPart = Math.round((abs - Math.floor(abs)) * 100).toString().padStart(2, '0');
  1396→  1396→  1396→  let formatted = '';
  1397→  1397→  1397→  for (let i = intPart.length - 1, c = 0; i >= 0; i--, c++) {
  1398→  1398→  1398→    if (c > 0 && c % 3 === 0) formatted = '.' + formatted;
  1399→  1399→  1399→    formatted = intPart[i] + formatted;
  1400→  1400→  1400→  }
  1401→  1401→  1401→  return (n < 0 ? '-' : '') + formatted + ',' + decPart;
  1402→  1402→  1402→}

  1405→  1405→  1405→  return fmtBRNumber(parseBR(a) + parseBR(b));
