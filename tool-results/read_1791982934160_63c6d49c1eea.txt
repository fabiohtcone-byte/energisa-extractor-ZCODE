    88→/* ── Helpers ── */
    89→function shortenLabel(label: string): string {
    90→  return label
    91→    .replace(/^Demanda (Contratada|Medida|Faturada) /, '')
    92→    .replace(/^Demanda Potência Não Consumida - /, 'Não Consumida ')
    93→    .replace(/^Valor (Consumo|Demanda|Ultrapassagem|Reativa Excedente|Não Consumida) /, '')
    94→    .replace(/^Energia Atv Injetada GDI /, '')
    95→    .replace(/^Consumo em kWh \(Grupo B\)/, 'Consumo kWh')
    96→    .replace(/^Consumo \(R\$\) \(Grupo B\)/, 'Consumo R$')
    97→    .replace(/^Ultrapassagem /, '')
    98→    .replace(/^Reativa Excedente /, '')
    99→    .replace(/^Energia Reativa Total/, 'Total')
   100→    .replace(/^Saldo Acumulado GD/, 'Saldo GD')
   101→    .replace(/\s*\(kWh\)\s*/, '')
   102→    .replace(/\s*\(kW\)\s*/, '')
   103→    .replace(/\s*\(R\$\)\s*/, '')
   104→    .trim();
   105→}

   128→
   129→    try {
   130→      for (let i = 0; i < pdfFiles.length; i++) {
   131→        const file = pdfFiles[i];
   132→        setFileProgress((p) => p.map((x, idx) => idx === i ? { ...x, status: 'processing' as const } : x));
   133→        try {
   134→          const fd = new FormData(); fd.append('file', file);

   147→            toast.error(`${file.name}: ${json.error}`);
   148→          } else {
   149→            const er: ExtractionResult = { fileName: file.name, data: json.data, stats: json.stats, rawText: json.rawText };
   150→            newResults.push(er);
   151→            setFileProgress((p) => p.map((x, idx) => idx === i ? { ...x, status: 'done' as const, result: er } : x));
   152→            toast.success(`${file.name}: ${json.stats.extractedCount}/${json.stats.totalFields} campos (${json.stats.extractionRate}%)`);
   153→          }

   158→          toast.error(`${file.name}: ${msg}.`);
   159→        }
   160→      }

   165→          const uc = e.data.UC.replace(/\D/g, '');

   167→        }
   168→        for (const n of newResults) {
   169→          const uc = n.data.UC.replace(/\D/g, '');
   170→          if (uc && prev.some((r) => r.data.UC.replace(/\D/g, '') === uc)) toast.info(`UC ${n.data.UC} atualizada.`);
   171→        }
   172→        return updated;
   173→      });
   174→      if (newResults.length > 0) setActiveTab('results');
   175→    } finally {
   176→      // ALWAYS reset processing state, even if something unexpected happens
   177→      setIsProcessing(false);
   178→      setFileProgress([]);
   179→    }
   180→  }, []);
   181→
   182→  const handleDrop = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(false); if (e.dataTransfer.files.length > 0) processFiles(e.dataTransfer.files); }, [processFiles]);
   183→  const handleDragOver = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(true); }, []);
   184→  const handleDragLeave = useCallback(() => setIsDragging(false), []);
   185→  const handleFileChange = useCallback((e: React.ChangeEvent<HTMLInputElement>) => { if (e.target.files?.length) { processFiles(e.target.files); e.target.value = ''; } }, [processFiles]);
